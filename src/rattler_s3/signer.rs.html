1//! Glue between the credentials of the AWS SDK and the request signer that
2//! `opendal` uses for S3.
3
4use std::time::{SystemTime, UNIX_EPOCH};
5
6use aws_credential_types::provider::{ProvideCredentials, SharedCredentialsProvider};
7use opendal::{
8    Configurator,
9    services::{S3, S3Config},
10};
11use reqsign_core::{ProvideCredentialChain, time::Timestamp};
12
13use crate::{S3AddressingStyle, S3CredentialSource};
14
15impl S3CredentialSource {
16    /// Create an `opendal` S3 builder for `bucket`, rooted at `root`.
17    ///
18    /// Requests are signed with the credentials of this source. `opendal` asks
19    /// for a new set whenever the ones it holds are about to expire, so an
20    /// operator built this way keeps working past the lifetime of the temporary
21    /// credentials of AWS SSO, an assumed role or the instance metadata service.
22    pub fn opendal_builder(&self, bucket: &str, root: &str) -> S3 {
23        let mut config = S3Config::default();
24        config.bucket = bucket.to_string();
25        config.root = Some(root.to_string());
26        config.region = Some(self.region.clone());
27        config.endpoint = Some(self.endpoint_url.to_string());
28        config.enable_virtual_host_style = self.addressing_style == S3AddressingStyle::VirtualHost;
29        // The settings are resolved already, so don't let `opendal` read the AWS
30        // configuration a second time.
31        config.disable_config_load = true;
32
33        config.into_builder().credential_provider_chain(
34            ProvideCredentialChain::new()
35                .push(AwsSdkCredentialProvider(self.credentials_provider.clone())),
36        )
37    }
38}

40/// Hands the credentials of the AWS SDK to the request signer of `opendal`.
41///
42/// Every call reaches the AWS SDK, which renews whatever it needs to produce a
43/// fresh set: it re-reads the credential file, re-assumes the role, re-queries
44/// the instance metadata, or exchanges the refresh token of an `sso_session` for
45/// a new SSO access token, depending on how it is configured.
46#[derive(Debug)]
47struct AwsSdkCredentialProvider(SharedCredentialsProvider);
48
49impl reqsign_core::ProvideCredential for AwsSdkCredentialProvider {
50    type Credential = reqsign_aws_v4::Credential;
51
52    async fn provide_credential(
53        &self,
54        _ctx: &reqsign_core::Context,
55    ) -> reqsign_core::Result<Option<Self::Credential>> {
56        let credentials = self.0.provide_credentials().await.map_err(|error| {
57            reqsign_core::Error::unexpected("failed to resolve the AWS credentials")
58                .with_source(error)
59        })?;
60
61        Ok(Some(reqsign_aws_v4::Credential {
62            access_key_id: credentials.access_key_id().to_string(),
63            secret_access_key: credentials.secret_access_key().to_string(),
64            session_token: credentials.session_token().map(ToString::to_string),
65            // Credentials without an expiry are long-lived, and `opendal` keeps
66            // using those for as long as the process runs.
67            expires_in: credentials.expiry().map(to_timestamp).transpose()?,
68        }))
69    }
70}
71
72fn to_timestamp(time: SystemTime) -> reqsign_core::Result<Timestamp> {
73    let since_epoch = time.duration_since(UNIX_EPOCH).map_err(|error| {
74        reqsign_core::Error::unexpected("AWS credentials expire before the Unix epoch")
75            .with_source(error)
76    })?;
77
78    Timestamp::from_millisecond(since_epoch.as_millis() as i64)
79}

81#[cfg(test)]
82mod tests {
83    use std::time::Duration;
84
85    use aws_credential_types::Credentials;
86    use reqsign_core::{ProvideCredential, SigningCredential};
87
88    use super::*;
89
90    #[tokio::test]
91    async fn temporary_credentials_carry_their_expiry_to_the_signer() {
92        let expiry = SystemTime::now() + Duration::from_secs(12 * 60 * 60);
93        let provider = SharedCredentialsProvider::new(Credentials::new(
94            "ASIAEXAMPLE",
95            "secret",
96            Some("session-token".to_string()),
97            Some(expiry),
98            "test",
99        ));
100
101        let credential = AwsSdkCredentialProvider(provider)
102            .provide_credential(&reqsign_core::Context::new())
103            .await
104            .expect("the provider should hand out credentials")
105            .expect("the provider should hand out credentials");
106
107        assert_eq!(credential.access_key_id, "ASIAEXAMPLE");
108        assert_eq!(credential.secret_access_key, "secret");
109        assert_eq!(credential.session_token.as_deref(), Some("session-token"));
110        // The signer asks for new credentials once the expiry is near, so it has
111        // to see it.
112        assert!(credential.is_valid());
113        assert!(!credential.is_valid_at(to_timestamp(expiry).unwrap()));
114    }
115
116    #[tokio::test]
117    async fn long_lived_credentials_never_expire() {
118        let provider = SharedCredentialsProvider::new(Credentials::new(
119            "AKIAEXAMPLE",
120            "secret",
121            None,
122            None,
123            "test",
124        ));
125
126        let credential = AwsSdkCredentialProvider(provider)
127            .provide_credential(&reqsign_core::Context::new())
128            .await
129            .expect("the provider should hand out credentials")
130            .expect("the provider should hand out credentials");
131
132        assert_eq!(credential.session_token, None);
133        assert_eq!(credential.expires_in, None);
134        assert!(credential.is_valid());
135    }
