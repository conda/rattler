1//! `rattler_config` is the shared configuration model of the rattler
2//! ecosystem: the same `config.toml` keys are understood by pixi,
3//! rattler-build, rattler-index and any other tool built on rattler.
4//!
5//! # Extending the configuration
6//!
7//! Tools plug their own keys into the shared model by defining an
8//! *extension* struct and using it as the type parameter of
9//! [`config::ConfigBase`]. Extension keys live at the top level of the same
10//! TOML document, next to the shared keys:
11//!
12//! ```rust
13//! use rattler_config::config::{Config, ConfigBase, MergeError};
14//! use serde::{Deserialize, Serialize};
15//!
16//! /// Keys that only make sense for *my* tool.
17//! #[derive(Debug, Default, Clone, PartialEq, Serialize, Deserialize)]
18//! #[serde(rename_all = "kebab-case")]
19//! pub struct MyToolConfig {
20//!     #[serde(default, skip_serializing_if = "Option::is_none")]
21//!     pub fancy_output: Option<bool>,
22//! }
23//!
24//! impl Config for MyToolConfig {
25//!     fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
26//!         Ok(Self {
27//!             fancy_output: other.fancy_output.or(self.fancy_output),
28//!         })
29//!     }
30//!
31//!     // `validate` and `keys` have sensible defaults, override as needed.
32//!     fn keys(&self) -> Vec<String> {
33//!         vec!["fancy-output".to_string()]
34//!     }
35//! }
36//!
37//! pub type MyConfig = ConfigBase<MyToolConfig>;
38//!
39//! let (config, unused_keys) = MyConfig::from_toml_str(
40//!     r#"
41//!     default-channels = ["conda-forge"]
42//!     fancy-output = true
43//!     definitely-a-typo = 1
44//!     "#,
45//! )
46//! .unwrap();
47//! assert_eq!(config.default_channels.as_ref().map(Vec::len), Some(1));
48//! assert_eq!(config.extensions.fancy_output, Some(true));
49//! assert!(unused_keys.contains("definitely-a-typo"));
50//! ```
51//!
52//! Extensions must not use `#[serde(deny_unknown_fields)]`: every extension
53//! is deserialized from the full configuration document and has to tolerate
54//! the shared keys (and the keys of sibling tools).
55//!
56//! # Loading
57//!
58//! [`config::ConfigBase::load_from_files`] merges a list of files in order
59//! (later files win) and validates the result.
60//! [`config::ConfigBase::load_from_default_locations`] does the same for the
61//! conventional locations described in [`locations`], which is how tools
62//! share one configuration: e.g. rattler-build can load
63//! `&["pixi", "rattler-build"]` to layer its own configuration on top of
64//! pixi's.
65//!
66//! # Editing
67//!
68//! With the `edit` feature (enabled by default), [`config::ConfigBase::set`]
69//! sets or unsets any key — including extension keys — by its dotted TOML
70//! path, and [`config::ConfigBase::save`] writes the configuration back to
71//! disk.

92    #[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, Default)]
93    pub struct TestExtension {
94        #[serde(default)]
95        pub custom_field: Option<String>,
96        #[serde(default)]
97        pub numeric_field: Option<u32>,
98        #[serde(default)]
99        pub bool_field: Option<bool>,
100        #[serde(default)]
101        pub array_field: Vec<String>,
102        #[serde(default)]
103        pub nested: HashMap<String, String>,
104    }
105
106    impl Config for TestExtension {
107        fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
108            Ok(Self {
109                custom_field: other.custom_field.clone().or(self.custom_field),
110                numeric_field: other.numeric_field.or(self.numeric_field),
111                bool_field: other.bool_field.or(self.bool_field),
112                array_field: if other.array_field.is_empty() {
113                    self.array_field
114                } else {
115                    other.array_field.clone()
116                },
117                nested: self
118                    .nested
119                    .into_iter()
120                    .chain(other.nested.iter().map(|(k, v)| (k.clone(), v.clone())))
121                    .collect(),
122            })
123        }
124
125        fn validate(&self) -> Result<(), ValidationError> {
126            if let Some(numeric) = self.numeric_field
127                && numeric > 100
128            {
129                return Err(ValidationError::InvalidValue(
130                    "numeric_field".to_string(),
131                    "must be <= 100".to_string(),
132                ));
133            }
134            Ok(())
135        }
136
137        fn keys(&self) -> Vec<String> {
138            vec![
139                "custom_field".to_string(),
140                "numeric_field".to_string(),
141                "bool_field".to_string(),
142                "array_field".to_string(),
143                "nested".to_string(),
144            ]
145        }
146    }

367    fn test_config_merge() {
368        let config1 = TestConfig {
369            common: CommonConfig {
370                default_channels: Some(vec!["conda-forge".parse().unwrap()]),
371                tls_no_verify: Some(false),
372                tls_root_certs: Some(TlsRootCerts::Webpki),
373                ..Default::default()
374            },
375            extensions: TestExtension {
376                custom_field: Some("original".to_string()),
377                numeric_field: Some(42),
378                ..Default::default()
379            },
380            ..Default::default()
381        };
382
383        let config2 = TestConfig {
384            common: CommonConfig {
385                default_channels: Some(vec!["bioconda".parse().unwrap()]),
386                authentication_override_file: Some(PathBuf::from("/new/auth")),
387                tls_root_certs: Some(TlsRootCerts::System),
388                ..Default::default()
389            },
390            extensions: TestExtension {
391                custom_field: Some("updated".to_string()),
392                bool_field: Some(true),
393                array_field: vec!["item1".to_string(), "item2".to_string()],
394                ..Default::default()
395            },
396            ..Default::default()
397        };
398
399        let merged = config1.merge_config(&config2).unwrap();
400
401        // The second config should take priority for overlapping fields
402        assert_eq!(
403            merged.default_channels.as_ref().map(std::vec::Vec::len),
404            Some(1)
405        );
406        assert_eq!(
407            merged.default_channels.as_ref().map(|v| v[0].to_string()),
408            Some("bioconda".to_string())
409        );
410        assert_eq!(
411            merged.authentication_override_file,
412            Some(PathBuf::from("/new/auth"))
413        );
414        assert_eq!(merged.tls_no_verify, Some(false)); // from config1
415        assert_eq!(merged.tls_root_certs, Some(TlsRootCerts::System));
416
417        // Extension fields should be merged properly
418        assert_eq!(merged.extensions.custom_field, Some("updated".to_string())); // from config2
419        assert_eq!(merged.extensions.numeric_field, Some(42)); // from config1
420        assert_eq!(merged.extensions.bool_field, Some(true)); // from config2
421        assert_eq!(merged.extensions.array_field.len(), 2); // from config2
422    }

425    fn test_from_toml_str_reports_unused_keys() {
426        let toml = r#"
427            default-channels = ["conda-forge"]
428            custom_field = "consumed by the extension"
429            definitely-a-typo = true
430
431            [concurrency]
432            solves = 3
433            bogus = 1
434
435            [repodata-config]
436            disable-bzip2 = true
437            disable-jlap = true
438        "#;
439
440        let (config, unused) = TestConfig::from_toml_str(toml).unwrap();
441
442        // Values consumed by either the common config or the extension.
443        assert_eq!(config.default_channels.as_ref().map(Vec::len), Some(1));
444        assert_eq!(
445            config.extensions.custom_field.as_deref(),
446            Some("consumed by the extension")
447        );
448        assert_eq!(config.concurrency.solves, 3);
449        assert_eq!(config.repodata_config.default.disable_bzip2, Some(true));
450
451        // Keys nobody understands are reported, at full depth.
452        assert!(unused.contains("definitely-a-typo"));
453        assert!(unused.contains("concurrency.bogus"));
454        assert!(unused.contains("repodata-config.disable-jlap"));
455
456        // Known keys are not reported.
457        assert!(!unused.iter().any(|k| k.starts_with("default-channels")));
458        assert!(!unused.contains("custom_field"));
459    }

904    #[test]
905    fn test_merge_multiple_configs() {
906        let base_config = TestConfig {
907            common: CommonConfig {
908                default_channels: Some(vec!["defaults".parse().unwrap()]),
909                tls_no_verify: Some(false),
910                ..Default::default()
911            },
912            ..Default::default()
913        };
914
915        let user_config = TestConfig {
916            common: CommonConfig {
917                default_channels: Some(vec!["conda-forge".parse().unwrap()]),
918                authentication_override_file: Some(PathBuf::from("/home/user/.conda-auth")),
919                ..Default::default()
920            },
921            extensions: TestExtension {
922                custom_field: Some("user-value".to_string()),
923                ..Default::default()
924            },
925            ..Default::default()
926        };
927
928        let project_config = TestConfig {
929            common: CommonConfig {
930                concurrency: crate::config::concurrency::ConcurrencyConfig {
931                    solves: 4,
932                    downloads: 8,
933                },
934                ..Default::default()
935            },
936            extensions: TestExtension {
937                numeric_field: Some(50),
938                bool_field: Some(true),
939                ..Default::default()
940            },
941            ..Default::default()
942        };
943
944        // Merge configs in order: base -> user -> project
945        let merged = base_config
946            .merge_config(&user_config)
947            .unwrap()
948            .merge_config(&project_config)
949            .unwrap();
950
951        // Project config takes priority for overlapping fields
952        assert_eq!(merged.concurrency.solves, 4);
953        assert_eq!(merged.concurrency.downloads, 8);
954
955        // User config values are preserved where not overridden
956        assert_eq!(
957            merged.default_channels.as_ref().map(std::vec::Vec::len),
958            Some(1)
959        );
960        assert_eq!(
961            merged.authentication_override_file,
962            Some(PathBuf::from("/home/user/.conda-auth"))
963        );
964
965        // Base config values are preserved where not overridden
966        assert_eq!(merged.tls_no_verify, Some(false));
967
968        // Extension values are merged properly
969        assert_eq!(
970            merged.extensions.custom_field,
971            Some("user-value".to_string())
972        );
973        assert_eq!(merged.extensions.numeric_field, Some(50));
974        assert_eq!(merged.extensions.bool_field, Some(true));
975    }
