1//! The shared configuration model for rattler-based tools.
2//!
3//! The entry point is [`ConfigBase<T>`]: the set of configuration keys that
4//! every rattler-based tool (pixi, rattler-build, rattler-index, …)
5//! understands, plus a tool-specific *extension* `T` whose keys live at the
6//! top level of the same TOML document.
7//!
8//! See the crate-level documentation for a full example of writing an
9//! extension.

86/// A fragment of configuration: either a nested section (like
87/// [`ConcurrencyConfig`]) or a tool-specific extension plugged into
88/// [`ConfigBase`].
89///
90/// Implementors only *have* to provide [`Config::merge_config`]; everything
91/// else has a sensible default. Extensions must **not** use
92/// `#[serde(deny_unknown_fields)]`: an extension is deserialized from the
93/// full configuration document, so it has to tolerate the keys handled by
94/// [`CommonConfig`] (and report them as ignored, which the derive does by
95/// default).
96pub trait Config:
97    Serialize + DeserializeOwned + std::fmt::Debug + Clone + PartialEq + Default
98{
99    /// Merge another configuration (file) into this one.
100    /// Note: the "other" configuration takes priority over the current one.
101    fn merge_config(self, other: &Self) -> Result<Self, MergeError>;
102
103    /// Validate the configuration. Called after all configuration files have
104    /// been merged.
105    fn validate(&self) -> Result<(), ValidationError> {
106        Ok(())
107    }
108
109    /// Returns true if the configuration equals its default value.
110    fn is_default(&self) -> bool {
111        self == &Self::default()
112    }
113
114    /// The dotted TOML key paths understood by this fragment, used to build
115    /// "supported keys" listings for error messages of `set`.
116    ///
117    /// Keys are relative to the fragment itself; [`ConfigBase`] adds the
118    /// proper prefixes for nested sections.
119    fn keys(&self) -> Vec<String> {
120        Vec::new()
121    }
122}

284/// Prefix every key in `keys` with `prefix.`.
285fn prefixed_keys(prefix: &str, keys: Vec<String>) -> Vec<String> {
286    keys.into_iter().map(|k| format!("{prefix}.{k}")).collect()
287}

289/// Formats the path of an ignored key as a dotted key, e.g. `table.key`.
290/// Unlike the `Display` of the path, this leaves out
291/// the `?` that marks a value inside an `Option`.
292fn key_path(path: &serde_ignored::Path<'_>) -> String {
293    use serde_ignored::Path;
294
295    let (parent, segment) = match path {
296        Path::Root => return String::new(),
297        Path::Seq { parent, index } => (parent, index.to_string()),
298        Path::Map { parent, key } => (parent, key.clone()),
299        Path::Some { parent }
300        | Path::NewtypeStruct { parent }
301        | Path::NewtypeVariant { parent } => return key_path(parent),
302    };
303    match key_path(parent) {
304        parent if parent.is_empty() => segment,
305        parent => format!("{parent}.{segment}"),
306    }
307}
308
309/// Is `key` equal to, or nested inside, one of the `ignored` paths?
310fn covered_by(key: &str, ignored: &BTreeSet<String>) -> bool {
311    ignored.iter().any(|path| {
312        key == path
313            || key
314                .strip_prefix(path.as_str())
315                .is_some_and(|rest| rest.starts_with('.'))
316    })
317}

319impl Config for CommonConfig {
320    /// Merge another configuration (file) into this one.
321    /// Note: the "other" configuration takes priority over the current one.
322    fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
323        Ok(Self {
324            default_channels: other
325                .default_channels
326                .as_ref()
327                .or(self.default_channels.as_ref())
328                .cloned(),
329            exclude_newer: other
330                .exclude_newer
331                .as_ref()
332                .or(self.exclude_newer.as_ref())
333                .cloned(),
334            authentication_override_file: other
335                .authentication_override_file
336                .as_ref()
337                .or(self.authentication_override_file.as_ref())
338                .cloned(),
339            tls_no_verify: other.tls_no_verify.or(self.tls_no_verify),
340            tls_root_certs: other.tls_root_certs.or(self.tls_root_certs),
341            mirrors: self
342                .mirrors
343                .iter()
344                .chain(other.mirrors.iter())
345                .map(|(k, v)| (k.clone(), v.clone()))
346                .collect(),
347            build: self.build.merge_config(&other.build)?,
348            // Currently this is always the default so it doesn't matter which one we take.
349            channel_config: self.channel_config,
350            repodata_config: self.repodata_config.merge_config(&other.repodata_config)?,
351            concurrency: self.concurrency.merge_config(&other.concurrency)?,
352            proxy_config: self.proxy_config.merge_config(&other.proxy_config)?,
353            s3_options: self.s3_options.merge_config(&other.s3_options)?,
354            index_config: self.index_config.merge_config(&other.index_config)?,
355            run_post_link_scripts: other
356                .run_post_link_scripts
357                .clone()
358                .or(self.run_post_link_scripts),
359            allow_symbolic_links: other.allow_symbolic_links.or(self.allow_symbolic_links),
360            allow_hard_links: other.allow_hard_links.or(self.allow_hard_links),
361            allow_ref_links: other.allow_ref_links.or(self.allow_ref_links),
362            virtual_package_detectors: self
363                .virtual_package_detectors
364                .merge_config(&other.virtual_package_detectors)?,
365        })
366    }
367
368    fn validate(&self) -> Result<(), ValidationError> {
369        if let Some(exclude_newer) = &self.exclude_newer {
370            exclude_newer.validate().map_err(|err| {
371                ValidationError::InvalidValue("exclude-newer".to_string(), err.to_string())
372            })?;
373        }
374        self.build.validate()?;
375        self.repodata_config.validate()?;
376        self.concurrency.validate()?;
377        self.proxy_config.validate()?;
378        self.s3_options.validate()?;
379        self.index_config.validate()?;
380        self.virtual_package_detectors.validate()?;
381        Ok(())
382    }
383
384    fn keys(&self) -> Vec<String> {
385        let mut keys = vec![
386            "default-channels".to_string(),
387            "exclude-newer".to_string(),
388            "authentication-override-file".to_string(),
389            "tls-no-verify".to_string(),
390            "tls-root-certs".to_string(),
391            "mirrors".to_string(),
392            "run-post-link-scripts".to_string(),
393            "allow-symbolic-links".to_string(),
394            "allow-hard-links".to_string(),
395            "allow-ref-links".to_string(),
396            "s3-options".to_string(),
397            "index-config".to_string(),
398        ];
399        keys.extend(prefixed_keys("build", self.build.keys()));
400        keys.extend(prefixed_keys(
401            "repodata-config",
402            self.repodata_config.keys(),
403        ));
404        keys.extend(prefixed_keys("concurrency", self.concurrency.keys()));
405        keys.extend(prefixed_keys("proxy-config", self.proxy_config.keys()));
406        keys.extend(prefixed_keys("s3-options", self.s3_options.keys()));
407        keys.extend(prefixed_keys(
408            "virtual-package-detectors",
409            self.virtual_package_detectors.keys(),
410        ));
411        keys
412    }
413}

415/// The complete configuration of a rattler-based tool: the
416/// [common keys](CommonConfig) plus a tool-specific extension `T` whose keys
417/// live at the top level of the same TOML document.
418///
419/// `ConfigBase` dereferences to [`CommonConfig`], so the shared keys can be
420/// accessed directly: `config.default_channels`, `config.concurrency`, ….
421#[derive(Debug, Default, Clone, PartialEq, Serialize, Deserialize)]
422pub struct ConfigBase<T = NoExtension> {
423    /// The configuration keys shared by all rattler-based tools.
424    #[serde(flatten)]
425    pub common: CommonConfig,
426
427    /// Tool-specific configuration keys.
428    #[serde(flatten)]
429    pub extensions: T,
430
431    /// The configuration files this configuration was loaded from, in load
432    /// order (lowest precedence first).
433    #[serde(skip)]
434    pub loaded_from: Vec<PathBuf>,
435}
436
437impl<T> Deref for ConfigBase<T> {
438    type Target = CommonConfig;
439
440    fn deref(&self) -> &Self::Target {
441        &self.common
442    }
443}
444
445impl<T> DerefMut for ConfigBase<T> {
446    fn deref_mut(&mut self) -> &mut Self::Target {
447        &mut self.common
448    }
449}

451impl<T> Config for ConfigBase<T>
452where
453    T: Config,
454{
455    /// Merge another configuration (file) into this one.
456    /// Note: the "other" configuration takes priority over the current one.
457    fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
458        Ok(Self {
459            common: self.common.merge_config(&other.common)?,
460            extensions: self.extensions.merge_config(&other.extensions)?,
461            loaded_from: self
462                .loaded_from
463                .iter()
464                .chain(&other.loaded_from)
465                .cloned()
466                .collect(),
467        })
468    }
469
470    fn validate(&self) -> Result<(), ValidationError> {
471        self.common.validate()?;
472        self.extensions.validate()
473    }
474
475    /// Gather all the keys of the configuration, including the extension
476    /// keys.
477    fn keys(&self) -> Vec<String> {
478        let mut keys = self.common.keys();
479        keys.extend(self.extensions.keys());
480        keys
481    }
482}

484impl<T> ConfigBase<T>
485where
486    T: Config,
487{
488    /// Parse a configuration from a TOML string.
489    ///
490    /// Returns the parsed configuration together with the set of keys that
491    /// neither [`CommonConfig`] nor the extension `T` recognized. Callers
492    /// should surface these to the user as warnings (they are typos or
493    /// keys of other tools).
494    pub fn from_toml_str(input: &str) -> Result<(Self, BTreeSet<String>), toml::de::Error> {
495        // The document is deserialized twice: once into the common
496        // configuration and once into the extension. Each pass records the
497        // keys it did not recognize; only keys unknown to *both* passes are
498        // truly unused. This is also why `CommonConfig` must not contain
499        // `#[serde(flatten)]` fields: flattening swallows unknown keys
500        // before `serde_ignored` can see them.
501        let mut unknown_to_common = BTreeSet::new();
502        let common: CommonConfig = serde_ignored::deserialize(
503            toml::de::Deserializer::parse(input)?,
504            |path: serde_ignored::Path<'_>| {
505                unknown_to_common.insert(key_path(&path));
506            },
507        )?;
508
509        let mut unknown_to_extension = BTreeSet::new();
510        let extensions: T = serde_ignored::deserialize(
511            toml::de::Deserializer::parse(input)?,
512            |path: serde_ignored::Path<'_>| {
513                unknown_to_extension.insert(key_path(&path));
514            },
515        )?;
516
517        // A key is truly unused when neither pass consumed it. The two
518        // passes may record it at different depths (the extension ignores
519        // an entire table that the common configuration partially
520        // consumed, or vice versa), so match against ancestors too and
521        // keep the most specific path.
522        let unused = unknown_to_common
523            .iter()
524            .filter(|key| covered_by(key, &unknown_to_extension))
525            .chain(
526                unknown_to_extension
527                    .iter()
528                    .filter(|key| covered_by(key, &unknown_to_common)),
529            )
530            .cloned()
531            .collect::<BTreeSet<String>>();
532        // Drop entries whose descendants are also reported: the deeper
533        // path is the actionable one.
534        let unused = unused
535            .iter()
536            .filter(|key| {
537                !unused.iter().any(|other| {
538                    other
539                        .strip_prefix(key.as_str())
540                        .is_some_and(|rest| rest.starts_with('.'))
541                })
542            })
543            .cloned()
544            .collect();
545
546        Ok((
547            Self {
548                common,
549                extensions,
550                loaded_from: Vec::new(),
551            },
552            unused,
553        ))
554    }
555
556    /// Parse a *shared* configuration file from a TOML string.
557    ///
558    /// Shared files (see [`crate::locations::ConfigLayer::Shared`]) may only
559    /// contain the keys shared by all rattler-based tools: the document is
560    /// deserialized into [`CommonConfig`] alone and the extension is left at
561    /// its default. Returns the parsed configuration together with the set
562    /// of keys [`CommonConfig`] did not recognize, including extension keys
563    /// the tool itself would understand, so that a shared file means the
564    /// same thing to every tool reading it.
565    pub fn from_toml_str_shared(input: &str) -> Result<(Self, BTreeSet<String>), toml::de::Error> {
566        let mut unknown = BTreeSet::new();
567        let common: CommonConfig = serde_ignored::deserialize(
568            toml::de::Deserializer::parse(input)?,
569            |path: serde_ignored::Path<'_>| {
570                unknown.insert(key_path(&path));
571            },
572        )?;
573
574        Ok((
575            Self {
576                common,
577                extensions: T::default(),
578                loaded_from: Vec::new(),
579            },
580            unknown,
581        ))
582    }
583
584    /// Parse the file at `path` according to its `layer` and merge it into
585    /// `self`, warning about ignored keys.
586    fn merge_from_path(self, path: &Path, layer: ConfigLayer) -> Result<Self, LoadError> {
587        let content = fs_err::read_to_string(path)?;
588        let (mut other, unused) = match layer {
589            ConfigLayer::Shared => Self::from_toml_str_shared(&content)?,
590            ConfigLayer::Tool => Self::from_toml_str(&content)?,
591        };
592        for key in &unused {
593            match layer {
594                ConfigLayer::Shared => tracing::warn!(
595                    "Ignoring configuration key `{key}` in {}: not a key shared by all rattler-based tools",
596                    path.display()
597                ),
598                ConfigLayer::Tool => tracing::warn!(
599                    "Ignoring unknown configuration key `{key}` in {}",
600                    path.display()
601                ),
602            }
603        }
604        other.loaded_from.push(path.to_path_buf());
605        self.merge_config(&other)
606            .map_err(|e| LoadError::MergeError(e, path.to_path_buf()))
607    }
608
609    /// Load the configuration by merging all the given files, in order:
610    /// later files take precedence over earlier ones. Every file is parsed
611    /// as a tool file (common keys plus extension keys); unrecognized keys
612    /// are reported as `tracing` warnings. The merged configuration is
613    /// validated before it is returned.
614    ///
615    /// Missing files result in an error; callers that search default
616    /// locations should filter for existing files first (see
617    /// [`ConfigBase::load_from_default_locations`]).
618    pub fn load_from_files<I, P>(paths: I) -> Result<Self, LoadError>
619    where
620        I: IntoIterator<Item = P>,
621        P: AsRef<Path>,
622    {
623        Self::load_from_locations(paths.into_iter().map(|path| ConfigLocation {
624            path: path.as_ref().to_path_buf(),
625            layer: ConfigLayer::Tool,
626        }))
627    }
628
629    /// Load the configuration by merging all the given locations, in order:
630    /// later locations take precedence over earlier ones. Each file is
631    /// parsed according to its layer: shared files accept only the common
632    /// keys (see [`ConfigBase::from_toml_str_shared`]), tool files also
633    /// accept the extension keys. Unrecognized keys are reported as
634    /// `tracing` warnings; the merged configuration is validated before it
635    /// is returned.
636    pub fn load_from_locations<I>(locations: I) -> Result<Self, LoadError>
637    where
638        I: IntoIterator<Item = ConfigLocation>,
639    {
640        let mut config = Self::default();
641
642        for location in locations {
643            config = config.merge_from_path(&location.path, location.layer)?;
644        }
645
646        config.validate()?;
647        Ok(config)
648    }
649
650    /// Load the configuration from the default locations of the given tool
651    /// (e.g. `"rattler-build"`), skipping files that do not exist: the
652    /// shared `rattler` configuration layered with the tool's own files.
653    /// See [`crate::locations::config_search_paths`] for the exact search
654    /// order.
655    pub fn load_from_default_locations(tool: &str) -> Result<Self, LoadError> {
656        Self::load_from_locations(
657            crate::locations::config_search_paths(tool)
658                .into_iter()
659                .filter(|location| location.path.is_file()),
660        )
661    }
662}
