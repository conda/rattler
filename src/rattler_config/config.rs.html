1//! The shared configuration model for rattler-based tools.
2//!
3//! The entry point is [`ConfigBase<T>`]: the set of configuration keys that
4//! every rattler-based tool (pixi, rattler-build, rattler-index, …)
5//! understands, plus a tool-specific *extension* `T` whose keys live at the
6//! top level of the same TOML document.
7//!
8//! See the crate-level documentation for a full example of writing an
9//! extension.

42pub enum ValidationError {

82/// A fragment of configuration: either a nested section (like
83/// [`ConcurrencyConfig`]) or a tool-specific extension plugged into
84/// [`ConfigBase`].
85///
86/// Implementors only *have* to provide [`Config::merge_config`]; everything
87/// else has a sensible default. Extensions must **not** use
88/// `#[serde(deny_unknown_fields)]`: an extension is deserialized from the
89/// full configuration document, so it has to tolerate the keys handled by
90/// [`CommonConfig`] (and report them as ignored, which the derive does by
91/// default).
92pub trait Config:
93    Serialize + DeserializeOwned + std::fmt::Debug + Clone + PartialEq + Default
94{
95    /// Merge another configuration (file) into this one.
96    /// Note: the "other" configuration takes priority over the current one.
97    fn merge_config(self, other: &Self) -> Result<Self, MergeError>;
98
99    /// Validate the configuration. Called after all configuration files have
100    /// been merged.
101    fn validate(&self) -> Result<(), ValidationError> {
102        Ok(())
103    }
104
105    /// Returns true if the configuration equals its default value.
106    fn is_default(&self) -> bool {
107        self == &Self::default()
108    }
109
110    /// The dotted TOML key paths understood by this fragment, used to build
111    /// "supported keys" listings for error messages of `set`.
112    ///
113    /// Keys are relative to the fragment itself; [`ConfigBase`] adds the
114    /// proper prefixes for nested sections.
115    fn keys(&self) -> Vec<String> {
116        Vec::new()
117    }
118}

132/// The configuration keys shared by all rattler-based tools.
133///
134/// This struct deliberately contains no `#[serde(flatten)]` fields: unknown
135/// top-level keys must surface through `serde_ignored` so that
136/// [`ConfigBase::from_toml_str`] can report typos and tool-specific keys
137/// reliably (`flatten` swallows unknown keys before `serde_ignored` can see
138/// them).
139#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
140#[serde(rename_all = "kebab-case")]
141pub struct CommonConfig {
142    #[serde(default)]
143    #[serde(alias = "default_channels")] // BREAK: remove to stop supporting snake_case alias
144    #[serde(skip_serializing_if = "Option::is_none")]
145    pub default_channels: Option<Vec<NamedChannelOrUrl>>,
146
147    /// Path to the file containing the authentication token.
148    #[serde(default)]
149    #[serde(alias = "authentication_override_file")] // BREAK: remove to stop supporting snake_case alias
150    #[serde(skip_serializing_if = "Option::is_none")]
151    pub authentication_override_file: Option<PathBuf>,
152
153    /// If set to true, the HTTPS client will not verify TLS server
154    /// certificates.
155    #[serde(default)]
156    #[serde(alias = "tls_no_verify")] // BREAK: remove to stop supporting snake_case alias
157    #[serde(skip_serializing_if = "Option::is_none")]
158    pub tls_no_verify: Option<bool>,
159
160    /// Which TLS root certificates to use (webpki vs system).
161    /// Accepts legacy spellings `"native"` and `"all"` as aliases for
162    /// `"system"`. See [`tls::TlsRootCerts`] for details.
163    #[serde(default)]
164    #[serde(alias = "tls_root_certs")] // BREAK: remove to stop supporting snake_case alias
165    #[serde(skip_serializing_if = "Option::is_none")]
166    pub tls_root_certs: Option<tls::TlsRootCerts>,
167
168    #[serde(default)]
169    #[serde(skip_serializing_if = "IndexMap::is_empty")]
170    pub mirrors: IndexMap<Url, Vec<Url>>,
171
172    #[serde(default, skip_serializing_if = "BuildConfig::is_default")]
173    pub build: BuildConfig,
174
175    #[serde(skip, default = "default_channel_config")]
176    pub channel_config: ChannelConfig,
177
178    /// Configuration for repodata fetching.
179    #[serde(alias = "repodata_config")] // BREAK: remove to stop supporting snake_case alias
180    #[serde(default, skip_serializing_if = "RepodataConfig::is_empty")]
181    pub repodata_config: RepodataConfig,
182
183    /// Configuration for the concurrency of rattler.
184    #[serde(default)]
185    #[serde(skip_serializing_if = "ConcurrencyConfig::is_default")]
186    pub concurrency: ConcurrencyConfig,
187
188    /// Https/Http proxy configuration.
189    #[serde(default)]
190    #[serde(skip_serializing_if = "ProxyConfig::is_default")]
191    pub proxy_config: ProxyConfig,
192
193    /// Configuration for S3.
194    #[serde(default)]
195    #[serde(skip_serializing_if = "S3OptionsMap::is_default")]
196    pub s3_options: S3OptionsMap,
197
198    /// Per-channel configuration for `rattler-index`.
199    #[serde(default, skip_serializing_if = "IndexConfig::is_empty")]
200    pub index_config: IndexConfig,
201
202    /// Run the post link scripts
203    #[serde(default)]
204    #[serde(skip_serializing_if = "Option::is_none")]
205    pub run_post_link_scripts: Option<RunPostLinkScripts>,
206
207    /// If set to false, symbolic links will not be used during package
208    /// installation.
209    #[serde(default)]
210    #[serde(alias = "allow_symbolic_links")] // BREAK: remove to stop supporting snake_case alias
211    #[serde(skip_serializing_if = "Option::is_none")]
212    pub allow_symbolic_links: Option<bool>,
213
214    /// If set to false, hard links will not be used during package
215    /// installation.
216    #[serde(default)]
217    #[serde(alias = "allow_hard_links")] // BREAK: remove to stop supporting snake_case alias
218    #[serde(skip_serializing_if = "Option::is_none")]
219    pub allow_hard_links: Option<bool>,
220
221    /// If set to false, ref links (copy-on-write) will not be used during
222    /// package installation.
223    #[serde(default)]
224    #[serde(alias = "allow_ref_links")] // BREAK: remove to stop supporting snake_case alias
225    #[serde(skip_serializing_if = "Option::is_none")]
226    pub allow_ref_links: Option<bool>,
227    // Missing in rattler but should be available in pixi:
228    //   experimental
229    //   shell
230    //   pinning_strategy
231    //   detached_environments
232    //   pypi_config
233    //
234    // Deprecated fields:
235    //   change_ps1
236    //   force_activate
237}

264/// Prefix every key in `keys` with `prefix.`.
265fn prefixed_keys(prefix: &str, keys: Vec<String>) -> Vec<String> {
266    keys.into_iter().map(|k| format!("{prefix}.{k}")).collect()
267}

269/// Is `key` equal to, or nested inside, one of the `ignored` paths?
270fn covered_by(key: &str, ignored: &BTreeSet<String>) -> bool {
271    ignored.iter().any(|path| {
272        key == path
273            || key
274                .strip_prefix(path.as_str())
275                .is_some_and(|rest| rest.starts_with('.'))
276    })
277}

279impl Config for CommonConfig {
280    /// Merge another configuration (file) into this one.
281    /// Note: the "other" configuration takes priority over the current one.
282    fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
283        Ok(Self {
284            default_channels: other
285                .default_channels
286                .as_ref()
287                .or(self.default_channels.as_ref())
288                .cloned(),
289            authentication_override_file: other
290                .authentication_override_file
291                .as_ref()
292                .or(self.authentication_override_file.as_ref())
293                .cloned(),
294            tls_no_verify: other.tls_no_verify.or(self.tls_no_verify),
295            tls_root_certs: other.tls_root_certs.or(self.tls_root_certs),
296            mirrors: self
297                .mirrors
298                .iter()
299                .chain(other.mirrors.iter())
300                .map(|(k, v)| (k.clone(), v.clone()))
301                .collect(),
302            build: self.build.merge_config(&other.build)?,
303            // Currently this is always the default so it doesn't matter which one we take.
304            channel_config: self.channel_config,
305            repodata_config: self.repodata_config.merge_config(&other.repodata_config)?,
306            concurrency: self.concurrency.merge_config(&other.concurrency)?,
307            proxy_config: self.proxy_config.merge_config(&other.proxy_config)?,
308            s3_options: self.s3_options.merge_config(&other.s3_options)?,
309            index_config: self.index_config.merge_config(&other.index_config)?,
310            run_post_link_scripts: other
311                .run_post_link_scripts
312                .clone()
313                .or(self.run_post_link_scripts),
314            allow_symbolic_links: other.allow_symbolic_links.or(self.allow_symbolic_links),
315            allow_hard_links: other.allow_hard_links.or(self.allow_hard_links),
316            allow_ref_links: other.allow_ref_links.or(self.allow_ref_links),
317        })
318    }
319
320    fn validate(&self) -> Result<(), ValidationError> {
321        self.build.validate()?;
322        self.repodata_config.validate()?;
323        self.concurrency.validate()?;
324        self.proxy_config.validate()?;
325        self.s3_options.validate()?;
326        self.index_config.validate()?;
327        Ok(())
328    }
329
330    fn keys(&self) -> Vec<String> {
331        let mut keys = vec![
332            "default-channels".to_string(),
333            "authentication-override-file".to_string(),
334            "tls-no-verify".to_string(),
335            "tls-root-certs".to_string(),
336            "mirrors".to_string(),
337            "run-post-link-scripts".to_string(),
338            "allow-symbolic-links".to_string(),
339            "allow-hard-links".to_string(),
340            "allow-ref-links".to_string(),
341            "s3-options".to_string(),
342            "index-config".to_string(),
343        ];
344        keys.extend(prefixed_keys("build", self.build.keys()));
345        keys.extend(prefixed_keys(
346            "repodata-config",
347            self.repodata_config.keys(),
348        ));
349        keys.extend(prefixed_keys("concurrency", self.concurrency.keys()));
350        keys.extend(prefixed_keys("proxy-config", self.proxy_config.keys()));
351        keys.extend(prefixed_keys("s3-options", self.s3_options.keys()));
352        keys
353    }
354}

356/// The complete configuration of a rattler-based tool: the
357/// [common keys](CommonConfig) plus a tool-specific extension `T` whose keys
358/// live at the top level of the same TOML document.
359///
360/// `ConfigBase` dereferences to [`CommonConfig`], so the shared keys can be
361/// accessed directly: `config.default_channels`, `config.concurrency`, ….
362#[derive(Debug, Default, Clone, PartialEq, Serialize, Deserialize)]
363pub struct ConfigBase<T = NoExtension> {
364    /// The configuration keys shared by all rattler-based tools.
365    #[serde(flatten)]
366    pub common: CommonConfig,
367
368    /// Tool-specific configuration keys.
369    #[serde(flatten)]
370    pub extensions: T,
371
372    /// The configuration files this configuration was loaded from, in load
373    /// order (lowest precedence first).
374    #[serde(skip)]
375    pub loaded_from: Vec<PathBuf>,
376}
377
378impl<T> Deref for ConfigBase<T> {
379    type Target = CommonConfig;
380
381    fn deref(&self) -> &Self::Target {
382        &self.common
383    }
384}
385
386impl<T> DerefMut for ConfigBase<T> {
387    fn deref_mut(&mut self) -> &mut Self::Target {
388        &mut self.common
389    }
390}
391
392impl<T> Config for ConfigBase<T>
393where
394    T: Config,
395{
396    /// Merge another configuration (file) into this one.
397    /// Note: the "other" configuration takes priority over the current one.
398    fn merge_config(self, other: &Self) -> Result<Self, MergeError> {
399        Ok(Self {
400            common: self.common.merge_config(&other.common)?,
401            extensions: self.extensions.merge_config(&other.extensions)?,
402            loaded_from: self
403                .loaded_from
404                .iter()
405                .chain(&other.loaded_from)
406                .cloned()
407                .collect(),
408        })
409    }
410
411    fn validate(&self) -> Result<(), ValidationError> {
412        self.common.validate()?;
413        self.extensions.validate()
414    }
415
416    /// Gather all the keys of the configuration, including the extension
417    /// keys.
418    fn keys(&self) -> Vec<String> {
419        let mut keys = self.common.keys();
420        keys.extend(self.extensions.keys());
421        keys
422    }
423}

425impl<T> ConfigBase<T>
426where
427    T: Config,
428{
429    /// Parse a configuration from a TOML string.
430    ///
431    /// Returns the parsed configuration together with the set of keys that
432    /// neither [`CommonConfig`] nor the extension `T` recognized. Callers
433    /// should surface these to the user as warnings (they are typos or
434    /// keys of other tools).
435    pub fn from_toml_str(input: &str) -> Result<(Self, BTreeSet<String>), toml::de::Error> {
436        // The document is deserialized twice: once into the common
437        // configuration and once into the extension. Each pass records the
438        // keys it did not recognize; only keys unknown to *both* passes are
439        // truly unused. This is also why `CommonConfig` must not contain
440        // `#[serde(flatten)]` fields: flattening swallows unknown keys
441        // before `serde_ignored` can see them.
442        let mut unknown_to_common = BTreeSet::new();
443        let common: CommonConfig = serde_ignored::deserialize(
444            toml::de::Deserializer::parse(input)?,
445            |path: serde_ignored::Path<'_>| {
446                unknown_to_common.insert(path.to_string());
447            },
448        )?;
449
450        let mut unknown_to_extension = BTreeSet::new();
451        let extensions: T = serde_ignored::deserialize(
452            toml::de::Deserializer::parse(input)?,
453            |path: serde_ignored::Path<'_>| {
454                unknown_to_extension.insert(path.to_string());
455            },
456        )?;
457
458        // A key is truly unused when neither pass consumed it. The two
459        // passes may record it at different depths (the extension ignores
460        // an entire table that the common configuration partially
461        // consumed, or vice versa), so match against ancestors too and
462        // keep the most specific path.
463        let unused = unknown_to_common
464            .iter()
465            .filter(|key| covered_by(key, &unknown_to_extension))
466            .chain(
467                unknown_to_extension
468                    .iter()
469                    .filter(|key| covered_by(key, &unknown_to_common)),
470            )
471            .cloned()
472            .collect::<BTreeSet<String>>();
473        // Drop entries whose descendants are also reported: the deeper
474        // path is the actionable one.
475        let unused = unused
476            .iter()
477            .filter(|key| {
478                !unused.iter().any(|other| {
479                    other
480                        .strip_prefix(key.as_str())
481                        .is_some_and(|rest| rest.starts_with('.'))
482                })
483            })
484            .cloned()
485            .collect();
486
487        Ok((
488            Self {
489                common,
490                extensions,
491                loaded_from: Vec::new(),
492            },
493            unused,
494        ))
495    }
496
497    /// Parse a *shared* configuration file from a TOML string.
498    ///
499    /// Shared files (see [`crate::locations::ConfigLayer::Shared`]) may only
500    /// contain the keys shared by all rattler-based tools: the document is
501    /// deserialized into [`CommonConfig`] alone and the extension is left at
502    /// its default. Returns the parsed configuration together with the set
503    /// of keys [`CommonConfig`] did not recognize, including extension keys
504    /// the tool itself would understand, so that a shared file means the
505    /// same thing to every tool reading it.
506    pub fn from_toml_str_shared(input: &str) -> Result<(Self, BTreeSet<String>), toml::de::Error> {
507        let mut unknown = BTreeSet::new();
508        let common: CommonConfig = serde_ignored::deserialize(
509            toml::de::Deserializer::parse(input)?,
510            |path: serde_ignored::Path<'_>| {
511                unknown.insert(path.to_string());
512            },
513        )?;
514
515        Ok((
516            Self {
517                common,
518                extensions: T::default(),
519                loaded_from: Vec::new(),
520            },
521            unknown,
522        ))
523    }
524
525    /// Parse the file at `path` according to its `layer` and merge it into
526    /// `self`, warning about ignored keys.
527    fn merge_from_path(self, path: &Path, layer: ConfigLayer) -> Result<Self, LoadError> {
528        let content = fs_err::read_to_string(path)?;
529        let (mut other, unused) = match layer {
530            ConfigLayer::Shared => Self::from_toml_str_shared(&content)?,
531            ConfigLayer::Tool => Self::from_toml_str(&content)?,
532        };
533        for key in &unused {
534            match layer {
535                ConfigLayer::Shared => tracing::warn!(
536                    "Ignoring configuration key `{key}` in {}: not a key shared by all rattler-based tools",
537                    path.display()
538                ),
539                ConfigLayer::Tool => tracing::warn!(
540                    "Ignoring unknown configuration key `{key}` in {}",
541                    path.display()
542                ),
543            }
544        }
545        other.loaded_from.push(path.to_path_buf());
546        self.merge_config(&other)
547            .map_err(|e| LoadError::MergeError(e, path.to_path_buf()))
548    }
549
550    /// Load the configuration by merging all the given files, in order:
551    /// later files take precedence over earlier ones. Every file is parsed
552    /// as a tool file (common keys plus extension keys); unrecognized keys
553    /// are reported as `tracing` warnings. The merged configuration is
554    /// validated before it is returned.
555    ///
556    /// Missing files result in an error; callers that search default
557    /// locations should filter for existing files first (see
558    /// [`ConfigBase::load_from_default_locations`]).
559    pub fn load_from_files<I, P>(paths: I) -> Result<Self, LoadError>
560    where
561        I: IntoIterator<Item = P>,
562        P: AsRef<Path>,
563    {
564        Self::load_from_locations(paths.into_iter().map(|path| ConfigLocation {
565            path: path.as_ref().to_path_buf(),
566            layer: ConfigLayer::Tool,
567        }))
568    }
569
570    /// Load the configuration by merging all the given locations, in order:
571    /// later locations take precedence over earlier ones. Each file is
572    /// parsed according to its layer: shared files accept only the common
573    /// keys (see [`ConfigBase::from_toml_str_shared`]), tool files also
574    /// accept the extension keys. Unrecognized keys are reported as
575    /// `tracing` warnings; the merged configuration is validated before it
576    /// is returned.
577    pub fn load_from_locations<I>(locations: I) -> Result<Self, LoadError>
578    where
579        I: IntoIterator<Item = ConfigLocation>,
580    {
581        let mut config = Self::default();
582
583        for location in locations {
584            config = config.merge_from_path(&location.path, location.layer)?;
585        }
586
587        config.validate()?;
588        Ok(config)
589    }
590
591    /// Load the configuration from the default locations of the given tool
592    /// (e.g. `"rattler-build"`), skipping files that do not exist: the
593    /// shared `rattler` configuration layered with the tool's own files.
594    /// See [`crate::locations::config_search_paths`] for the exact search
595    /// order.
596    pub fn load_from_default_locations(tool: &str) -> Result<Self, LoadError> {
597        Self::load_from_locations(
598            crate::locations::config_search_paths(tool)
599                .into_iter()
600                .filter(|location| location.path.is_file()),
601        )
602    }
