18/// Given the contents of a file, copies it to the `destination` and in the process replace any
19/// binary c-style string that contains the text `prefix_placeholder` with a binary compatible
20/// c-string where the `prefix_placeholder` text is replaced with the `target_prefix` text.
21///
22/// The length of the input will match the output.
23///
24/// The placeholder is replaced under every encoding defined by the draft CEP, so paths stored as
25/// wide strings are patched as well. The NUL terminator of a c-string is the zero code unit of its
26/// encoding: one zero byte for UTF-8, two for UTF-16 and four for UTF-32.
27///
28/// This function replaces binary c-style strings. If you want to simply find-and-replace text in a
29/// file instead use the [`super::copy_and_replace_textual_placeholder`] function.
30pub fn copy_and_replace_cstring_placeholder(
31    source_bytes: &[u8],
32    destination: impl Write,
33    prefix_placeholder: &str,
34    target_prefix: &str,
35) -> Result<(), std::io::Error> {
36    let prefixes = EncodedPrefix::all(prefix_placeholder, target_prefix);
37
38    let patches = find_cstring_patches(source_bytes, &prefixes);
39    // Only the encodings that actually occur in the file constrain the target prefix: an encoding
40    // whose replacement would not fit but that the file does not use is irrelevant.
41    reject_growing_prefix(patches.iter().map(|patch| patch.prefix))?;
42
43    write_patched_cstrings(destination, source_bytes, &patches)
44}

46/// Finds every c-string that contains a placeholder occurrence, under every encoding, ordered by
47/// position in the file.
48///
49/// Candidates of different encodings can describe overlapping c-strings, because the encodings of
50/// an ASCII placeholder are byte-shifted variants of one another: UTF-16-BE of `/pfx` is UTF-16-LE
51/// of `/pfx` shifted by one byte, so a genuine little-endian occurrence preceded by a zero byte
52/// always produces a spurious big-endian candidate one byte earlier (and the other way around).
53/// A spurious candidate is misaligned with the string it sits in, so its terminator scan often
54/// finds no zero code unit at all and its c-string would swallow the rest of the file.
55///
56/// Overlaps are therefore resolved by preferring, in order: UTF-8 (whose encoding cannot be a
57/// shifted variant of another one), a candidate whose offset is code-unit aligned in the file
58/// (which a compiler-emitted wide string is and its shifted twin is not), a c-string with a real
59/// terminator over one that runs to end-of-file, and finally the earlier offset. A candidate that
60/// loses leaves its placeholder in place, which is what rattler did before it replaced wide
61/// strings at all; writing another encoding's bytes over the string instead would corrupt it.
62///
63/// A candidate collects every occurrence up to its terminator, so a later occurrence at the same
64/// alignment inside that range is skipped instead of starting a candidate of its own: it would
65/// lose to the earlier one anyway, and building it rescans the c-string, which is quadratic for
66/// long c-strings with many occurrences.
67fn find_cstring_patches<'a>(
68    source_bytes: &[u8],
69    prefixes: &'a [EncodedPrefix],
70) -> Vec<CStringPatch<'a>> {
71    let mut candidates = Vec::new();
72    for prefix in prefixes {
73        let placeholder = prefix.placeholder.as_slice();
74        let unit = prefix.code_unit_size();
75        // Per alignment class (offset modulo the code unit size), the end of the last candidate.
76        let mut covered_until = vec![0; unit];
77        let mut search_from = 0;
78        while let Some(found) = memchr::memmem::find(&source_bytes[search_from..], placeholder) {
79            let first = search_from + found;
80            let after_first = first + placeholder.len();
81            // Resume right after this occurrence, not after its c-string: an occurrence of the
82            // same encoding at a different alignment can start before this terminator.
83            search_from = after_first;
84            let alignment = first % unit;
85            if first < covered_until[alignment] {
86                continue;
87            }
88
89            let nul_pos = cstring_end(source_bytes, after_first, unit);
90
91            // Collect the remaining occurrences in the same c-string: they share its terminator
92            // and the padding that keeps the file length unchanged.
93            let end = nul_pos.unwrap_or(source_bytes.len());
94            let mut offsets = vec![first];
95            let mut next = after_first;
96            while let Some(found) = memchr::memmem::find(&source_bytes[next..end], placeholder) {
97                offsets.push(next + found);
98                next += found + placeholder.len();
99            }
100            covered_until[alignment] = end;
101
102            let rank = match (&prefix.encoding, alignment == 0, nul_pos.is_some()) {
103                (OffsetEncoding::Utf8, _, _) => 0u8,
104                (_, true, true) => 1,
105                (_, true, false) => 2,
106                (_, false, true) => 3,
107                (_, false, false) => 4,
108            };
109            candidates.push((
110                rank,
111                CStringPatch {
112                    offsets: Cow::Owned(offsets),
113                    nul_pos: end,
114                    prefix,
115                },
116            ));
117        }
118    }
119
120    // Keep the best candidates whose c-strings do not overlap a better one. The kept c-strings
121    // are disjoint, so only the nearest kept c-string on either side can overlap a candidate.
122    candidates.sort_by_key(|(rank, patch)| (*rank, patch.offsets[0]));
123    let mut kept: BTreeMap<usize, CStringPatch<'a>> = BTreeMap::new();
124    for (_, candidate) in candidates {
125        let start = candidate.offsets[0];
126        let overlaps_earlier = kept
127            .range(..=start)
128            .next_back()
129            .is_some_and(|(_, earlier)| start < earlier.nul_pos);
130        let overlaps_later = kept
131            .range(start..)
132            .next()
133            .is_some_and(|(&later_start, _)| later_start < candidate.nul_pos);
134        if !overlaps_earlier && !overlaps_later {
135            kept.insert(start, candidate);
136        }
137    }
138    kept.into_values().collect()
139}

141/// Finds the end of the c-string containing a placeholder occurrence: the offset of the first zero
142/// code unit at or after `from`, or `None` when the c-string is unterminated at end-of-file.
143///
144/// `from` is the end of an occurrence, so the scan strides in code units from there. For a genuine
145/// occurrence that is the alignment of the string it sits in; for a spurious cross-encoding match
146/// it is not, which is exactly why a missing terminator has to be distinguishable from a found
147/// one.
148fn cstring_end(source_bytes: &[u8], from: usize, unit: usize) -> Option<usize> {
149    let mut pos = from;
150    while pos + unit <= source_bytes.len() {
151        if source_bytes[pos..pos + unit].iter().all(|&byte| byte == 0) {
152            return Some(pos);
153        }
154        pos += unit;
155    }
156    None
157}

159/// Writes `source_bytes` to `destination`, replacing the placeholder inside every patched
160/// c-string. The bytes that follow a replacement shift towards the start of the string and the gap
161/// before its NUL terminator is filled with zeros, so the file length is preserved.
162fn write_patched_cstrings(
163    mut destination: impl Write,
164    source_bytes: &[u8],
165    patches: &[CStringPatch<'_>],
166) -> Result<(), std::io::Error> {
167    let mut last_pos = 0;
168    for patch in patches {
169        let prefix = patch.prefix;
170        for &offset in patch.offsets.iter() {
171            write_replacement_range(&mut destination, source_bytes, last_pos, offset)?;
172            destination.write_all(&prefix.target)?;
173            last_pos = offset + prefix.placeholder.len();
174        }
175
176        // Write the remaining bytes of the c-string, which for an unterminated final c-string runs
177        // to the end of the file, and fill the gap the replacements left with zeros.
178        write_replacement_range(&mut destination, source_bytes, last_pos, patch.nul_pos)?;
179        let Some(shrinks_by) = prefix.shrinks_by() else {
180            return Err(prefix.growing_prefix_error());
181        };
182        let padding = patch.offsets.len() * shrinks_by;
183        if padding > 0 {
184            destination.write_all(&vec![0; padding])?;
185        }
186
187        last_pos = patch.nul_pos;
188    }
189
190    // Write any remaining bytes after the last c-string.
191    if last_pos < source_bytes.len() {
192        destination.write_all(&source_bytes[last_pos..])?;
193    }
194
195    Ok(())
196}

198/// Given the contents & offsets of a file, copies it to the `destination` and in the process
199/// replace any binary c-style string that contains the text `prefix_placeholder` with a binary
200/// compatible c-string where the `prefix_placeholder` text is replaced with the `target_prefix`
201/// text.
202///
203/// The length of the input will match the output.
204///
205/// Every group of `groups` is applied, each under its own encoding. A group's ranges are grouped
206/// by c-string: the inner lists hold the prefix start positions followed by the position of the
207/// first byte of the NUL terminator, or the file size when the final c-string is unterminated at
208/// end-of-file (the padding then runs to EOF, still preserving the length). For example,
209/// `[[5, 19], [22, 30, 39]]` means one c-string with the prefix at offset 5 (NUL at 19), and
210/// another with prefixes at 22 and 30 (NUL at 39).
211///
212/// The metadata is validated before anything is written, so a mismatch surfaces as
213/// [`OffsetReplaceError::InconsistentMetadata`] with an untouched destination the caller can hand
214/// to search-based replacement.
215pub fn copy_and_replace_cstring_placeholder_offsets(
216    source_bytes: &[u8],
217    destination: impl Write,
218    prefix_placeholder: &str,
219    target_prefix: &str,
220    groups: &[OffsetGroup],
221) -> Result<(), OffsetReplaceError> {
222    let prefixes = EncodedPrefix::all(prefix_placeholder, target_prefix);
223
224    let patches = cstring_patches_from_groups(groups, &prefixes)?;
225    validate_cstring_patches(source_bytes, &patches)?;
226    // Only the encodings whose recorded occurrences are present in the file constrain the target
227    // prefix. Checking this after the validation keeps bogus metadata a fallback rather than an
228    // install failure.
229    reject_growing_prefix(patches.iter().map(|patch| patch.prefix))?;
230
231    // --- The metadata is consistent; write the patched file. ---
232    write_patched_cstrings(destination, source_bytes, &patches)?;
233    Ok(())
234}

236/// Collects the c-strings to patch from the recorded offset groups, ordered by their position in
237/// the file so that the splice runs in file order regardless of which group a range came from.
238fn cstring_patches_from_groups<'a>(
239    groups: &'a [OffsetGroup],
240    prefixes: &'a [EncodedPrefix],
241) -> Result<Vec<CStringPatch<'a>>, InconsistentOffsetsError> {
242    let mut patches = Vec::new();
243    for group in groups {
244        let encoding = group.encoding();
245        let prefix = encoded_prefix_for(prefixes, encoding)?;
246        let OffsetRanges::Binary(cstrings) = group.ranges() else {
247            return Err(InvalidOffsetsError::RangesShapeMismatch(encoding).into());
248        };
249        for cstring in cstrings {
250            // Each c-string lists its prefix offsets followed by the NUL terminator position.
251            let Some((&nul_pos, offsets)) = cstring.split_last() else {
252                return Err(InvalidOffsetsError::ShortCStringRanges(encoding).into());
253            };
254            if offsets.is_empty() {
255                return Err(InvalidOffsetsError::ShortCStringRanges(encoding).into());
256            }
257            patches.push(CStringPatch {
258                offsets: Cow::Borrowed(offsets),
259                nul_pos,
260                prefix,
261            });
262        }
263    }
264
265    // The binary form must list at least one c-string.
266    if patches.is_empty() {
267        return Err(InvalidOffsetsError::EmptyList.into());
268    }
269
270    patches.sort_by_key(|patch| patch.offsets[0]);
271    Ok(patches)
272}

274/// Validates the c-strings to patch against the file contents.
275///
276/// Nothing may be written before this passes so that, on inconsistent metadata, the caller can
277/// fall back to search-based replacement using the still-empty destination. Within each c-string
278/// the prefix offsets must be in range (before the terminator), sorted in strictly increasing
279/// non-overlapping order (also across c-strings and encodings), and the placeholder bytes must be
280/// present at each offset. The recorded terminator must be a zero code unit of the group's
281/// encoding (or end-of-file) at a code-unit distance from every occurrence it terminates:
282/// otherwise the padding would be written into the middle of a live string and truncate it.
283fn validate_cstring_patches(
284    source_bytes: &[u8],
285    patches: &[CStringPatch<'_>],
286) -> Result<(), InconsistentOffsetsError> {
287    let mut prev_end = 0usize;
288    for patch in patches {
289        let terminator = patch.nul_pos;
290        let placeholder = patch.prefix.placeholder.as_slice();
291        let unit = patch.prefix.code_unit_size();
292        let Some(terminator_bytes) = source_bytes.get(terminator..) else {
293            return Err(InconsistentOffsetsError::TerminatorOutOfRange {
294                terminator,
295                file_size: source_bytes.len(),
296            });
297        };
298        if !terminator_bytes.is_empty()
299            && !terminator_bytes
300                .get(..unit)
301                .is_some_and(|unit| unit.iter().all(|&byte| byte == 0))
302        {
303            return Err(InconsistentOffsetsError::TerminatorNotZero { terminator });
304        }
305        for &offset in patch.offsets.iter() {
306            if offset < prev_end {
307                return Err(InconsistentOffsetsError::UnsortedOffset { offset });
308            }
309            let end = offset
310                .checked_add(placeholder.len())
311                .filter(|&end| end <= terminator)
312                .ok_or(InconsistentOffsetsError::OffsetPastTerminator { offset, terminator })?;
313            if (terminator - offset) % unit != 0 {
314                return Err(InconsistentOffsetsError::MisalignedOffset { offset, terminator });
315            }
316            if &source_bytes[offset..end] != placeholder {
317                return Err(InconsistentOffsetsError::PlaceholderNotFound { offset });
318            }
319            prev_end = end;
320        }
321        prev_end = terminator;
322    }
323    Ok(())
