1//! Replaces the prefix placeholder recorded in `paths.json` with the target prefix.
2//!
3//! Two paths do the same job. The search-based one scans the file contents for the placeholder;
4//! the offsets-based one splices the positions a producer recorded in `paths.json`, as proposed by
5//! the [draft CEP]. Both build the same patch lists ([`TextPatch`], [`CStringPatch`]) and hand them
6//! to the same splicers, which is what keeps them byte-for-byte identical.
7//!
8//! Text files live in [`text`], binary files in [`binary`], and this module holds what they share:
9//! the per-encoding prefixes, the patch types and the two entry points that dispatch on
10//! `file_mode`.
11//!
12//! [draft CEP]: https://github.com/conda/ceps/pull/179
13

32/// Given the contents of a file copy it to the `destination` and in the process replace the
33/// `prefix_placeholder` text with the `target_prefix` text.
34///
35/// This switches to more specialized functions that handle the replacement of either
36/// textual and binary placeholders, the [`FileMode`] enum switches between the two functions.
37/// See both [`copy_and_replace_cstring_placeholder`] and [`copy_and_replace_textual_placeholder`]
38pub fn copy_and_replace_placeholders(
39    source_bytes: &[u8],
40    mut destination: impl Write,
41    prefix_placeholder: &str,
42    target_prefix: &str,
43    target_platform: &Subdir,
44    file_mode: FileMode,
45) -> Result<(), std::io::Error> {
46    match file_mode {
47        FileMode::Text => {
48            copy_and_replace_textual_placeholder(
49                source_bytes,
50                destination,
51                prefix_placeholder,
52                target_prefix,
53                target_platform,
54            )?;
55        }
56        FileMode::Binary => {
57            // conda does not replace the prefix in the binary files on windows
58            // DLLs are loaded quite differently anyways (there is no rpath, for example).
59            if target_platform.is_windows() {
60                destination.write_all(source_bytes)?;
61            } else {
62                copy_and_replace_cstring_placeholder(
63                    source_bytes,
64                    destination,
65                    prefix_placeholder,
66                    target_prefix,
67                )?;
68            }
69        }
70    }
71    Ok(())
72}

74/// Error returned by the offset-based prefix replacement functions
75/// ([`copy_and_replace_placeholders_with_offsets`] and the specialized
76/// text/binary variants it dispatches to).
77///
78/// The offsets recorded in `paths.json` come from the package producer and
79/// are not trusted. When they are inconsistent with the file contents the
80/// install must not fail: the caller falls back to the search-based
81/// replacement path. IO errors while writing the patched file are surfaced
82/// separately.
83///
84/// The offset functions write nothing to the destination before returning
85/// [`OffsetReplaceError::InconsistentMetadata`], so the caller can reuse the
86/// same (still empty) destination for the fallback.
87#[derive(Debug, thiserror::Error)]
88pub enum OffsetReplaceError {
89    /// The recorded offsets are inconsistent with the file contents. Callers
90    /// should fall back to search-based replacement rather than failing the
91    /// install.
92    #[error("inconsistent prefix replacement metadata: {0}")]
93    InconsistentMetadata(#[from] InconsistentOffsetsError),
94
95    /// An IO error occurred while writing the patched file.
96    #[error(transparent)]
97    Io(#[from] std::io::Error),
98}
99
100/// How recorded offsets disagree with the file they were recorded for.
101#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
102pub enum InconsistentOffsetsError {
103    /// The offsets violate the draft CEP regardless of the file contents.
104    #[error(transparent)]
105    Invalid(#[from] InvalidOffsetsError),
106
107    /// The offsets were recorded for a different file mode than the one the file is installed
108    /// with.
109    #[error("the offsets were recorded for file mode {recorded:?}, not {installed:?}")]
110    FileModeMismatch {
111        /// The file mode the offsets were recorded for.
112        recorded: FileMode,
113        /// The file mode the file is installed with.
114        installed: FileMode,
115    },
116
117    /// The placeholder is empty, so no recorded offset can point at it.
118    #[error("the placeholder is empty")]
119    EmptyPlaceholder,
120
121    /// A `shebang_length` is recorded but the file does not start with `#!`.
122    #[error("shebang_length is recorded but the file does not start with #!")]
123    UnexpectedShebangLength,
124
125    /// The file starts with `#!` but no `shebang_length` is recorded.
126    #[error("the file starts with #! but no shebang_length is recorded")]
127    MissingShebangLength,
128
129    /// The recorded `shebang_length` is not the length of the file's first line.
130    #[error("shebang_length {shebang_length} is not the length of the first line")]
131    ShebangLengthMismatch {
132        /// The recorded `shebang_length`.
133        shebang_length: usize,
134    },
135
136    /// An offset lies inside the shebang region, which the draft CEP excludes from `offsets`.
137    #[error("offset {offset} lies inside the shebang region of {shebang_length} bytes")]
138    OffsetInShebangRegion {
139        /// The recorded offset.
140        offset: usize,
141        /// The length of the shebang region.
142        shebang_length: usize,
143    },
144
145    /// An offset starts before the end of the previous occurrence or c-string, across all
146    /// encodings.
147    #[error("offset {offset} overlaps or precedes the previous occurrence")]
148    UnsortedOffset {
149        /// The recorded offset.
150        offset: usize,
151    },
152
153    /// The placeholder at an offset extends past the end of the file.
154    #[error(
155        "the placeholder at offset {offset} extends past the end of the file ({file_size} bytes)"
156    )]
157    OffsetOutOfRange {
158        /// The recorded offset.
159        offset: usize,
160        /// The size of the file.
161        file_size: usize,
162    },
163
164    /// The placeholder at an offset extends past the terminator of its c-string.
165    #[error("the placeholder at offset {offset} extends past its terminator at {terminator}")]
166    OffsetPastTerminator {
167        /// The recorded offset.
168        offset: usize,
169        /// The recorded terminator position.
170        terminator: usize,
171    },
172
173    /// An offset is not a whole number of code units before the terminator of its c-string.
174    #[error(
175        "offset {offset} is not a whole number of code units before its terminator at {terminator}"
176    )]
177    MisalignedOffset {
178        /// The recorded offset.
179        offset: usize,
180        /// The recorded terminator position.
181        terminator: usize,
182    },
183
184    /// The bytes at an offset are not the encoded placeholder.
185    #[error("the placeholder is not present at offset {offset}")]
186    PlaceholderNotFound {
187        /// The recorded offset.
188        offset: usize,
189    },
190
191    /// A terminator position lies past the end of the file.
192    #[error("terminator {terminator} lies past the end of the file ({file_size} bytes)")]
193    TerminatorOutOfRange {
194        /// The recorded terminator position.
195        terminator: usize,
196        /// The size of the file.
197        file_size: usize,
198    },
199
200    /// The bytes at a terminator position are not a zero code unit.
201    #[error("the bytes at terminator {terminator} are not a zero code unit")]
202    TerminatorNotZero {
203        /// The recorded terminator position.
204        terminator: usize,
205    },
206}

208/// The placeholder and target prefix encoded with one of the encodings defined by the draft CEP.
209///
210/// Prefix replacement covers every one of these encodings, so paths that a binary stores as wide
211/// strings are patched too.
212struct EncodedPrefix {
213    encoding: OffsetEncoding,
214    placeholder: Vec<u8>,
215    target: Vec<u8>,
216}
217
218impl EncodedPrefix {
219    /// Encodes both prefixes with every encoding defined by the draft CEP, skipping encodings under
220    /// which the placeholder is empty because there is nothing to search for.
221    fn all(placeholder: &str, target: &str) -> Vec<EncodedPrefix> {
222        OffsetEncoding::DEFINED
223            .into_iter()
224            .filter_map(|encoding| {
225                let placeholder = encoding.encode(placeholder);
226                (!placeholder.is_empty()).then(|| EncodedPrefix {
227                    encoding,
228                    placeholder,
229                    target: encoding.encode(target),
230                })
231            })
232            .collect()
233    }
234
235    /// The number of bytes one replacement frees up, or `None` when the target prefix is longer
236    /// than the placeholder and therefore cannot be spliced into a fixed-size c-string.
237    fn shrinks_by(&self) -> Option<usize> {
238        self.placeholder.len().checked_sub(self.target.len())
239    }
240
241    /// The size of one code unit, which is also the size of a c-string's NUL terminator.
242    fn code_unit_size(&self) -> usize {
243        self.encoding.code_unit_size()
244    }
245
246    /// The error for a target prefix that does not fit in the space of the placeholder.
247    fn growing_prefix_error(&self) -> std::io::Error {
248        std::io::Error::new(
249            std::io::ErrorKind::InvalidData,
250            format!(
251                "target prefix cannot be longer than the placeholder prefix (encoding '{}')",
252                self.encoding
253            ),
254        )
255    }
256}

258/// Rejects a target prefix that is longer than the placeholder under one of the encodings that is
259/// actually replaced. Binary replacement preserves the file length, so the replacement must fit in
260/// the space the placeholder occupies.
261///
262/// The lengths differ per encoding once the placeholder leaves ASCII (UTF-8 counts bytes, the wide
263/// encodings count code units), so an encoding the file does not use must not fail the install.
264fn reject_growing_prefix<'a>(
265    prefixes: impl IntoIterator<Item = &'a EncodedPrefix>,
266) -> Result<(), std::io::Error> {
267    match prefixes
268        .into_iter()
269        .find(|prefix| prefix.shrinks_by().is_none())
270    {
271        Some(prefix) => Err(prefix.growing_prefix_error()),
272        None => Ok(()),
273    }
274}

276/// Looks up the encoded prefixes for the encoding of a recorded offset group.
277fn encoded_prefix_for(
278    prefixes: &[EncodedPrefix],
279    encoding: OffsetEncoding,
280) -> Result<&EncodedPrefix, InconsistentOffsetsError> {
281    prefixes
282        .iter()
283        .find(|prefix| prefix.encoding == encoding)
284        .ok_or(InconsistentOffsetsError::EmptyPlaceholder)
285}

304/// Given the contents of a file copy it to the `destination` and in the process replace the
305/// `prefix_placeholder` text with the `target_prefix` text, using the offsets recorded in
306/// `paths.json` instead of searching the file contents.
307///
308/// Per the [draft CEP], an installer applies exactly the groups whose encodings its own
309/// search-based replacement covers, so both paths produce the same bytes. rattler's search-based
310/// replacement covers every encoding the draft CEP defines, so every recorded group is spliced.
311/// Offsets with no groups at all mean there is nothing to splice: the file is copied through
312/// unchanged apart from the shebang handling of text files.
313///
314/// Returns [`OffsetReplaceError::InconsistentMetadata`] (having written nothing) when the offsets
315/// were recorded for a different `file_mode` or do not match the file, so the caller can fall
316/// back to search-based replacement.
317///
318/// Nothing is searched: the only bytes inspected are the ones the metadata points at, namely the
319/// encoded placeholder at each recorded offset, the zero code unit at each recorded c-string
320/// terminator, and, when the file starts with `#!`, the `shebang_length` bytes of the first line
321/// that the installer's shebang rules transform. Everything else is copied without being looked
322/// at, which is what lets a consumer compute the patched size from the metadata alone.
323///
324/// [draft CEP]: https://github.com/conda/ceps/pull/179
325pub fn copy_and_replace_placeholders_with_offsets(
326    source_bytes: &[u8],
327    mut destination: impl Write,
328    prefix_placeholder: &str,
329    target_prefix: &str,
330    target_platform: &Subdir,
331    file_mode: FileMode,
332    offsets: &PrefixOffsets,
333) -> Result<(), OffsetReplaceError> {
334    if offsets.file_mode() != file_mode {
335        return Err(InconsistentOffsetsError::FileModeMismatch {
336            recorded: offsets.file_mode(),
337            installed: file_mode,
338        }
339        .into());
340    }
341
342    match file_mode {
343        FileMode::Text => copy_and_replace_textual_placeholder_offsets(
344            source_bytes,
345            destination,
346            prefix_placeholder,
347            target_prefix,
348            target_platform,
349            offsets.groups(),
350            offsets.shebang_length(),
351        )?,
352        // conda does not replace the prefix in the binary files on windows
353        // DLLs are loaded quite differently anyways (there is no rpath, for example).
354        FileMode::Binary if target_platform.is_windows() => {
355            destination.write_all(source_bytes)?;
356        }
357        FileMode::Binary => copy_and_replace_cstring_placeholder_offsets(
358            source_bytes,
359            destination,
360            prefix_placeholder,
361            target_prefix,
362            offsets.groups(),
363        )?,
364    }
365    Ok(())
366}

368/// Writes `source[start..end]` to `destination`, returning an [`std::io::ErrorKind::InvalidData`]
369/// error instead of panicking when the range is invalid (out of bounds or out of order).
370///
371/// The offsets driving the prefix replacement come from a package's `paths.json`, which is not
372/// trusted input. A malformed or malicious entry (an offset past the end of the file, or offsets
373/// that are not sorted/overlapping) must surface as a recoverable error rather than crash the
374/// process (which for e.g. a FUSE/NFS mount would take down the whole mount).
375fn write_replacement_range<W: Write>(
376    destination: &mut W,
377    source: &[u8],
378    start: usize,
379    end: usize,
380) -> Result<(), std::io::Error> {
381    let slice = source.get(start..end).ok_or_else(|| {
382        std::io::Error::new(
383            std::io::ErrorKind::InvalidData,
384            format!(
385                "invalid prefix replacement offsets: range {start}..{end} is out of bounds or out \
386                 of order for content of length {}",
387                source.len()
388            ),
389        )
390    })?;
391    destination.write_all(slice)
392}
