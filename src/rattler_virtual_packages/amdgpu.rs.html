1//! Provides functionality to detect AMD GPUs present on the current system.
2//!
3//! This module backs two virtual packages, as specified by the `__amdgpu` CEP
4//! (<https://github.com/conda/ceps/pull/189>):
5//!
6//! ## AMD GPU presence (`__amdgpu`)
7//!
8//! A presence-only package (version and build string are always `0`) that is exposed when at least
9//! one AMD GPU is available through the host driver.
10//!
11//! ## AMD GPU architecture (`__amdgpu_arch`)
12//!
13//! The AMDGPU ISA version (`{major}.{minor}.{stepping}`) of the detected GPU with the most compute
14//! units; ties are broken by picking the highest ISA version. Unlike `__cuda_arch` this is *not* a
15//! minimum: AMDGPU binaries carry code objects for specific `gfx` targets and there is no
16//! forward-compatibility model like PTX, so the most capable device is the most useful one to
17//! describe.
18//!
19//! ## Detection
20//!
21//! * Linux: the KFD topology in `/sys/class/kfd/kfd/topology/nodes` provides the ISA version and
22//!   compute-unit count per device; `/sys/class/drm` is consulted for presence when KFD is not
23//!   available (e.g. a display-only driver setup). When KFD exposes no devices the HIP runtime is
24//!   tried as a fallback, which covers `ROCm` on WSL where the KFD sysfs tree does not exist.
25//! * Windows: the HIP runtime (`amdhip64_<major>.dll` / `amdhip64.dll`) installed by the AMD
26//!   driver in the system directory provides the ISA version and compute-unit count. Presence is
27//!   additionally derived from the AMD display adapters Windows knows about.
28

33/// The AMDGPU ISA version of a device, e.g. `9.0.10` for `gfx90a` or `11.5.1` for `gfx1151`.
34///
35/// The derived ordering is lexicographic over `(major, minor, stepping)`, which is the ordering the
36/// CEP prescribes for breaking compute-unit ties. It does *not* imply binary compatibility between
37/// architectures.
38#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
39pub struct AmdGpuArchInfo {
40    /// Major version, e.g. `11` for `gfx1151`.
41    pub major: u32,
42    /// Minor version, e.g. `5` for `gfx1151`.
43    pub minor: u32,
44    /// Stepping, e.g. `1` for `gfx1151` or `10` for `gfx90a`.
45    pub stepping: u32,
46}
47
48impl AmdGpuArchInfo {
49    /// Decodes the `gfx_target_version` property exposed by the Linux KFD topology.
50    ///
51    /// The value is encoded as `major * 10000 + minor * 100 + stepping`, e.g. `90010` for `gfx90a`.
52    /// A value of `0` marks a node that is not a GPU (KFD lists CPU nodes as well) and yields `None`.
53    pub fn from_gfx_target_version(value: u32) -> Option<Self> {
54        if value == 0 {
55            return None;
56        }
57        Some(Self {
58            major: (value / 10000) % 100,
59            minor: (value / 100) % 100,
60            stepping: value % 100,
61        })
62    }
63
64    /// Decodes an AMDGPU target id as reported by HIP's `gcnArchName`, e.g. `gfx90a` or
65    /// `gfx1100:xnack-`.
66    ///
67    /// Any target feature suffix after the first `:` is ignored. After the `gfx` prefix, the final
68    /// two characters are the hexadecimal minor and stepping and everything before them is the
69    /// decimal major version.
70    pub fn from_target_id(target_id: &str) -> Option<Self> {
71        let name = target_id.split(':').next().unwrap_or_default();
72        let digits = name.strip_prefix("gfx")?;
73        if digits.len() < 3 {
74            return None;
75        }
76        let (major, rest) = digits.split_at(digits.len() - 2);
77        let mut rest = rest.chars();
78        let minor = rest.next()?.to_digit(16)?;
79        let stepping = rest.next()?.to_digit(16)?;
80        if !major.bytes().all(|b| b.is_ascii_digit()) {
81            return None;
82        }
83        Some(Self {
84            major: major.parse().ok()?,
85            minor,
86            stepping,
87        })
88    }
89
90    /// Converts the ISA version into the [`Version`] used for the `__amdgpu_arch` virtual package.
91    pub fn to_version(self) -> Version {
92        Version::from_str(&self.to_string())
93            .expect("three dot separated decimal integers are a valid version")
94    }
95}
96
97impl fmt::Display for AmdGpuArchInfo {
98    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
99        write!(f, "{}.{}.{}", self.major, self.minor, self.stepping)
100    }
101}

103/// An error returned when parsing an AMDGPU ISA version from a string.
104#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
105#[error(
106    "invalid AMDGPU ISA version '{0}': expected 'major.minor.stepping' where all three are decimal integers (e.g. '11.0.0')"
107)]
108pub struct ParseAmdGpuArchError(String);
109
110impl FromStr for AmdGpuArchInfo {
111    type Err = ParseAmdGpuArchError;
112
113    /// Parses the `{major}.{minor}.{stepping}` form used by the `__amdgpu_arch` virtual package
114    /// and its `CONDA_OVERRIDE_AMDGPU_ARCH` override.
115    fn from_str(s: &str) -> Result<Self, Self::Err> {
116        let invalid = || ParseAmdGpuArchError(s.to_string());
117        let mut parts = s.split('.');
118        let mut component = || {
119            let part = parts.next().ok_or_else(invalid)?;
120            if part.is_empty() || !part.bytes().all(|b| b.is_ascii_digit()) {
121                return Err(invalid());
122            }
123            part.parse::<u32>().map_err(|_overflow| invalid())
124        };
125        let major = component()?;
126        let minor = component()?;
127        let stepping = component()?;
128        if parts.next().is_some() {
129            return Err(invalid());
130        }
131        Ok(Self {
132            major,
133            minor,
134            stepping,
135        })
136    }
137}

139/// A single detected AMD GPU.
140#[derive(Debug, Clone, PartialEq, Eq)]
141pub struct AmdGpuDevice {
142    /// The ISA version of the device.
143    pub arch: AmdGpuArchInfo,
144    /// The number of compute units, if it could be determined.
145    pub compute_units: Option<u32>,
146    /// The device's marketing name (`hipDeviceProp_t::name`), e.g. `"AMD Radeon RX 7800 XT"`.
147    ///
148    /// Only set for devices enumerated through the HIP runtime; the Linux KFD topology sysfs
149    /// this module also reads from has no equivalent, only the raw PCI device id.
150    pub name: Option<String>,
151}
152
153impl fmt::Display for AmdGpuDevice {
154    /// Renders as e.g. `"AMD Radeon RX 7800 XT (arch 11.0.1, 30 CUs)"`, or, when the marketing
155    /// name or compute-unit count is unavailable (only possible on the Linux KFD path), `"AMD
156    /// GPU (arch 11.0.1, CU count unknown)"`.
157    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
158        write!(f, "{}", self.name.as_deref().unwrap_or("AMD GPU"))?;
159        write!(f, " (arch {}", self.arch)?;
160        match self.compute_units {
161            Some(1) => write!(f, ", 1 CU)"),
162            Some(compute_units) => write!(f, ", {compute_units} CUs)"),
163            None => write!(f, ", CU count unknown)"),
164        }
165    }
166}

168/// AMD GPU information detected from the system.
169#[derive(Debug, Clone, Copy, PartialEq, Eq)]
170pub struct AmdGpuInfo {
171    /// Whether at least one AMD GPU is available through the host driver. This corresponds to the
172    /// `__amdgpu` virtual package.
173    pub present: bool,
174
175    /// The ISA version selected for the `__amdgpu_arch` virtual package: the device with the most
176    /// compute units, ties broken by the highest ISA version. `None` when no device could be
177    /// enumerated, which also happens for some presence-only detections (e.g. a Linux system whose
178    /// driver exposes the GPU through DRM but not through KFD).
179    ///
180    /// Always `None` when `present` is `false`.
181    pub arch_info: Option<AmdGpuArchInfo>,
182}
183
184/// Selects the device that represents the system in `__amdgpu_arch`: the one with the most compute
185/// units, ties (and unknown counts, which sort below any known count) broken by the highest ISA
186/// version.
187pub fn select_device(devices: &[AmdGpuDevice]) -> Option<&AmdGpuDevice> {
188    devices
189        .iter()
190        .max_by_key(|device| (device.compute_units, device.arch))
191}
192
193/// Returns the AMD GPU information of the current platform.
194///
195/// Detection runs at most once per process; the result is reused afterwards.
196pub fn amdgpu_info() -> &'static AmdGpuInfo {
197    static DETECTED: LazyLock<AmdGpuInfo> = LazyLock::new(|| {
198        let info = detect_amdgpu_info();
199        tracing::trace!(
200            present = info.present,
201            arch = %info.arch_info.map_or_else(|| "<none>".to_string(), |arch| arch.to_string()),
202            "detected AMD GPU info from host"
203        );
204        info
205    });
206    &DETECTED
207}
208
209/// Builds the final [`AmdGpuInfo`] from the enumerated devices and a separate presence signal.
210fn info_from_devices(devices: &[AmdGpuDevice], present_without_devices: bool) -> AmdGpuInfo {
211    let selected = select_device(devices);
212    if let Some(device) = selected {
213        tracing::debug!(
214            %device,
215            device_count = devices.len(),
216            "selected AMD GPU for __amdgpu_arch"
217        );
218    }
219    AmdGpuInfo {
220        present: selected.is_some() || present_without_devices,
221        arch_info: selected.map(|device| device.arch),
222    }
223}

225/// Detects AMD GPU information from the current system.
226#[cfg(target_os = "linux")]
227pub fn detect_amdgpu_info() -> AmdGpuInfo {
228    use std::path::Path;
229
230    let mut devices = linux::kfd_devices(Path::new(linux::KFD_TOPOLOGY_NODES));
231    if devices.is_empty() {
232        devices = linux_hip_fallback();
233    }
234    let drm_present = devices.is_empty() && linux::drm_has_amdgpu(Path::new(linux::DRM_CLASS));
235    info_from_devices(&devices, drm_present)
236}

271#[cfg(target_os = "linux")]
272mod linux {
273    use std::path::Path;
274
275    use super::{AmdGpuArchInfo, AmdGpuDevice};
276
277    /// The KFD topology; one directory per node with a `properties` file.
278    pub(super) const KFD_TOPOLOGY_NODES: &str = "/sys/class/kfd/kfd/topology/nodes";
279
280    /// The DRM class; one `cardN` directory per display adapter.
281    pub(super) const DRM_CLASS: &str = "/sys/class/drm";
282
283    /// Enumerates the GPUs in the KFD topology rooted at `nodes`.
284    ///
285    /// Every node has a `properties` file with `<key> <value>` lines. GPU nodes have a non-zero
286    /// `gfx_target_version`; the compute-unit count is `simd_count / simd_per_cu`.
287    pub(super) fn kfd_devices(nodes: &Path) -> Vec<AmdGpuDevice> {
288        let Ok(entries) = std::fs::read_dir(nodes) else {
289            tracing::trace!(path = %nodes.display(), "KFD topology is not available");
290            return Vec::new();
291        };
292        let mut devices: Vec<_> = entries
293            .filter_map(Result::ok)
294            .filter_map(|entry| {
295                let path = entry.path().join("properties");
296                let properties = std::fs::read_to_string(&path).ok()?;
297                let device = parse_kfd_properties(&properties)?;
298                tracing::trace!(path = %path.display(), %device, "found AMD GPU in KFD topology");
299                Some(device)
300            })
301            .collect();
302        // `read_dir` order is arbitrary; keep the result deterministic.
303        devices.sort_by_key(|device| (device.arch, device.compute_units));
304        devices
305    }
306
307    /// Parses the `properties` file of a KFD topology node. Returns `None` for non-GPU nodes.
308    pub(super) fn parse_kfd_properties(properties: &str) -> Option<AmdGpuDevice> {
309        let mut gfx_target_version = None;
310        let mut simd_count = None;
311        let mut simd_per_cu = None;
312        for line in properties.lines() {
313            let mut fields = line.split_ascii_whitespace();
314            let (Some(key), Some(value)) = (fields.next(), fields.next()) else {
315                continue;
316            };
317            let slot = match key {
318                "gfx_target_version" => &mut gfx_target_version,
319                "simd_count" => &mut simd_count,
320                "simd_per_cu" => &mut simd_per_cu,
321                _ => continue,
322            };
323            *slot = value.parse::<u32>().ok();
324        }
325        let arch = AmdGpuArchInfo::from_gfx_target_version(gfx_target_version?)?;
326        let compute_units = match (simd_count, simd_per_cu) {
327            (Some(simd_count), Some(simd_per_cu)) if simd_per_cu > 0 => {
328                Some(simd_count / simd_per_cu)
329            }
330            _ => None,
331        };
332        Some(AmdGpuDevice {
333            arch,
334            compute_units,
335            name: None,
336        })
337    }
338
339    /// Returns true if any DRM card in `drm` is an AMD device (PCI vendor `0x1002`) bound to the
340    /// `amdgpu` kernel driver.
341    pub(super) fn drm_has_amdgpu(drm: &Path) -> bool {
342        let Ok(entries) = std::fs::read_dir(drm) else {
343            return false;
344        };
345        entries.filter_map(Result::ok).any(|entry| {
346            let name = entry.file_name();
347            let Some(suffix) = name.to_str().and_then(|name| name.strip_prefix("card")) else {
348                return false;
349            };
350            if suffix.is_empty() || !suffix.bytes().all(|b| b.is_ascii_digit()) {
351                return false;
352            }
353            let device = entry.path().join("device");
354            let is_amd = std::fs::read_to_string(device.join("vendor"))
355                .is_ok_and(|vendor| vendor.trim().eq_ignore_ascii_case("0x1002"));
356            let is_amdgpu = std::fs::read_link(device.join("driver"))
357                .is_ok_and(|driver| driver.file_name().is_some_and(|name| name == "amdgpu"));
358            if is_amd && is_amdgpu {
359                tracing::trace!(path = %entry.path().display(), "found amdgpu DRM device");
360            }
361            is_amd && is_amdgpu
362        })
363    }

370    /// Returns true if Windows knows about a present display adapter with AMD's PCI vendor id
371    /// (`1002`). Only Plug and Play metadata is read, so this does not initialize the driver.
372    pub(super) fn has_amd_display_adapter() -> bool {
373        use windows_sys::Win32::Devices::DeviceAndDriverInstallation::{
374            CM_GETIDLIST_FILTER_CLASS, CM_GETIDLIST_FILTER_PRESENT,
375        };
376
377        // GUID_DEVCLASS_DISPLAY
378        const DISPLAY_CLASS: &str = "{4d36e968-e325-11ce-bfc1-08002be10318}";
379
380        win::device_instance_ids(
381            DISPLAY_CLASS,
382            CM_GETIDLIST_FILTER_CLASS | CM_GETIDLIST_FILTER_PRESENT,
383        )
384        .is_some_and(|mut ids| ids.any(|id| is_amd_pci_device_id(&id)))
385    }

395/// Device enumeration through the HIP runtime.
396///
397/// This is the only source of information on Windows, and the fallback on Linux when KFD is not
398/// available. Not compiled on musl: `libloading` itself would compile fine, but the
399/// statically-linked binaries that target produces have no dynamic loader to `dlopen` a driver
400/// library with, so it could never actually be reached.
401#[cfg(all(
402    any(target_os = "linux", target_os = "windows"),
403    not(target_env = "musl")
404))]
405mod hip {
406    use std::ffi::{c_int, c_void};
407
408    use libloading::{Library, Symbol};
409
410    use super::{AmdGpuArchInfo, AmdGpuDevice};
411
412    const HIP_SUCCESS: c_int = 0;
413
414    /// The byte offsets of the `hipDeviceProp_t` members this module reads, per struct revision.
415    ///
416    /// The HIP runtime versions its ABI by exporting one `hipGetDeviceProperties*` symbol per
417    /// struct revision. The struct is read from a zeroed, over-sized buffer at these offsets, so no
418    /// full definition of the struct is needed.
419    struct PropertiesLayout {
420        symbol: &'static [u8],
421        name: usize,
422        gcn_arch_name: usize,
423        multi_processor_count: usize,
424    }
425
426    /// Struct revisions in order of preference. Runtimes before HIP 6.0 only export the unversioned
427    /// symbol, which uses the `R0000` layout.
428    ///
429    /// `name` is `hipDeviceProp_t`'s first member in every revision, so its offset is 0
430    /// regardless of layout.
431    const LAYOUTS: [PropertiesLayout; 3] = [
432        PropertiesLayout {
433            symbol: b"hipGetDevicePropertiesR0600\0",
434            name: 0,
435            gcn_arch_name: 1160,
436            multi_processor_count: 388,
437        },
438        PropertiesLayout {
439            symbol: b"hipGetDevicePropertiesR0000\0",
440            name: 0,
441            gcn_arch_name: 396,
442            multi_processor_count: 336,
443        },
444        PropertiesLayout {
445            symbol: b"hipGetDeviceProperties\0",
446            name: 0,
447            gcn_arch_name: 396,
448            multi_processor_count: 336,
449        },
450    ];
451
452    /// `name` and `gcnArchName` are both `char[256]`.
453    const NAME_LEN: usize = 256;
454    const GCN_ARCH_NAME_LEN: usize = 256;
455
456    /// Comfortably larger than any `hipDeviceProp_t` revision (the `R0600` layout is 1480 bytes).
457    const PROPERTIES_BUFFER_LEN: usize = 4096;
458
459    type HipGetDeviceCount = unsafe extern "C" fn(*mut c_int) -> c_int;
460    type HipGetDeviceProperties = unsafe extern "C" fn(*mut c_void, c_int) -> c_int;
461
462    /// Enumerates the devices visible to the HIP runtime, or nothing if the runtime is not
463    /// installed or reports no devices.
464    pub(super) fn devices() -> Vec<AmdGpuDevice> {
465        let Some(library) = load_library() else {
466            tracing::debug!("could not load the HIP runtime from any known path");
467            return Vec::new();
468        };
469
470        let get_count: Symbol<'_, HipGetDeviceCount> =
471            match unsafe { library.get(b"hipGetDeviceCount\0") } {
472                Ok(symbol) => symbol,
473                Err(err) => {
474                    tracing::debug!(error = %err, "HIP runtime lacks hipGetDeviceCount");
475                    return Vec::new();
476                }
477            };
478        let mut count: c_int = 0;
479        let result = unsafe { get_count(&mut count) };
480        if result != HIP_SUCCESS || count <= 0 {
481            tracing::trace!(result, count, "HIP runtime reports no devices");
482            return Vec::new();
483        }
484
485        let Some((get_properties, layout)) = LAYOUTS.iter().find_map(|layout| {
486            let symbol: Symbol<'_, HipGetDeviceProperties> =
487                unsafe { library.get(layout.symbol) }.ok()?;
488            Some((symbol, layout))
489        }) else {
490            tracing::debug!("HIP runtime lacks every known hipGetDeviceProperties revision");
491            return Vec::new();
492        };
493
494        (0..count)
495            .filter_map(|device| {
496                // `u64` storage keeps the buffer suitably aligned for every member.
497                let mut buffer = [0u64; PROPERTIES_BUFFER_LEN / 8];
498                let result =
499                    unsafe { get_properties(buffer.as_mut_ptr().cast::<c_void>(), device) };
500                if result != HIP_SUCCESS {
501                    tracing::debug!(device, result, "hipGetDeviceProperties failed");
502                    return None;
503                }
504                let bytes: &[u8] = unsafe {
505                    std::slice::from_raw_parts(buffer.as_ptr().cast::<u8>(), PROPERTIES_BUFFER_LEN)
506                };
507
508                let target_id =
509                    &bytes[layout.gcn_arch_name..layout.gcn_arch_name + GCN_ARCH_NAME_LEN];
510                let target_id = &target_id[..target_id
511                    .iter()
512                    .position(|&b| b == 0)
513                    .unwrap_or(target_id.len())];
514                let target_id = std::str::from_utf8(target_id).ok()?;
515                let Some(arch) = AmdGpuArchInfo::from_target_id(target_id) else {
516                    tracing::debug!(device, target_id, "HIP reported an unparsable gcnArchName");
517                    return None;
518                };
519
520                let count = &bytes[layout.multi_processor_count..layout.multi_processor_count + 4];
521                let count = c_int::from_ne_bytes(count.try_into().expect("slice is 4 bytes"));
522                let compute_units = u32::try_from(count).ok().filter(|&count| count > 0);
523
524                // The marketing name is purely cosmetic: a non-UTF-8 or empty value just leaves
525                // it unset rather than dropping the device.
526                let name = &bytes[layout.name..layout.name + NAME_LEN];
527                let name = &name[..name.iter().position(|&b| b == 0).unwrap_or(name.len())];
528                let name = std::str::from_utf8(name)
529                    .ok()
530                    .filter(|name| !name.is_empty())
531                    .map(str::to_owned);
532
533                let device = AmdGpuDevice {
534                    arch,
535                    compute_units,
536                    name,
537                };
538                tracing::debug!(target_id, %device, "found AMD GPU through the HIP runtime");
539                Some(device)
540            })
541            .collect()
542    }

544    fn load_library() -> Option<Library> {
545        library_paths().into_iter().find_map(|path| {
546            match unsafe { Library::new(&path) } {
547                Ok(library) => {
548                    tracing::trace!(library_path = %path.display(), "loaded HIP runtime");
549                    Some(library)
550                }
551                Err(err) => {
552                    tracing::trace!(library_path = %path.display(), error = %err, "failed to load HIP runtime");
553                    None
554                }
555            }
556        })
557    }

559    /// The AMD driver installs the HIP runtime into the Windows system directory as
560    /// `amdhip64_<major>.dll` (newest first) or, for older drivers, `amdhip64.dll`.
561    #[cfg(target_os = "windows")]
562    fn library_paths() -> Vec<std::path::PathBuf> {
563        let system_root = std::env::var_os("SystemRoot").unwrap_or_else(|| r"C:\Windows".into());
564        let system32 = std::path::Path::new(&system_root).join("System32");
565
566        let mut versioned: Vec<(u32, std::path::PathBuf)> = std::fs::read_dir(&system32)
567            .into_iter()
568            .flatten()
569            .filter_map(Result::ok)
570            .filter_map(|entry| {
571                let name = entry.file_name();
572                let name = name.to_str()?.to_ascii_lowercase();
573                let version = name
574                    .strip_prefix("amdhip64_")?
575                    .strip_suffix(".dll")?
576                    .parse::<u32>()
577                    .ok()?;
578                Some((version, entry.path()))
579            })
580            .collect();
581        versioned.sort_by_key(|(version, _)| std::cmp::Reverse(*version));
582
583        versioned
584            .into_iter()
585            .map(|(_, path)| path)
586            .chain(std::iter::once(system32.join("amdhip64.dll")))
587            .collect()
588    }

828    #[test]
829    fn kfd_topology_and_drm_are_read_from_sysfs_layout() {
830        let root = tempfile::tempdir().unwrap();
831
832        // KFD topology: one CPU node and two GPU nodes.
833        let nodes = root.path().join("nodes");
834        for (name, properties) in [
835            ("0", "gfx_target_version 0\nsimd_count 0\nsimd_per_cu 0\n"),
836            (
837                "1",
838                "gfx_target_version 110003\nsimd_count 24\nsimd_per_cu 2\n",
839            ),
840            (
841                "2",
842                "gfx_target_version 110002\nsimd_count 64\nsimd_per_cu 2\n",
843            ),
844        ] {
845            let node = nodes.join(name);
846            std::fs::create_dir_all(&node).unwrap();
847            std::fs::write(node.join("properties"), properties).unwrap();
848        }
849        let devices = linux::kfd_devices(&nodes);
850        assert_eq!(devices.len(), 2);
851        assert_eq!(
852            select_device(&devices).map(|device| device.arch),
853            Some(arch(11, 0, 2))
854        );
855        assert!(linux::kfd_devices(&root.path().join("missing")).is_empty());
856
857        // DRM: an Intel card bound to i915 and an AMD card bound to amdgpu.
858        let drm = root.path().join("drm");
859        let drivers = root.path().join("drivers");
860        for (card, vendor, driver) in [
861            ("card0", "0x8086\n", "i915"),
862            ("card1", "0x1002\n", "amdgpu"),
863        ] {
864            let device = drm.join(card).join("device");
865            std::fs::create_dir_all(&device).unwrap();
866            std::fs::write(device.join("vendor"), vendor).unwrap();
867            let driver_dir = drivers.join(driver);
868            std::fs::create_dir_all(&driver_dir).unwrap();
869            std::os::unix::fs::symlink(&driver_dir, device.join("driver")).unwrap();
870        }
871        assert!(linux::drm_has_amdgpu(&drm));
872
873        // An AMD card without the amdgpu driver (e.g. bound to radeon or vfio) does not count.
874        std::fs::remove_file(drm.join("card1/device/driver")).unwrap();
875        std::os::unix::fs::symlink(drivers.join("i915"), drm.join("card1/device/driver")).unwrap();
876        assert!(!linux::drm_has_amdgpu(&drm));
877        assert!(!linux::drm_has_amdgpu(&root.path().join("missing")));
878    }
