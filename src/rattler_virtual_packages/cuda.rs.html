1//! Provides functionality to detect CUDA information present on the current system.
2//!
3//! This module detects two types of CUDA information:
4//!
5//! ## CUDA Driver Version (`__cuda`)
6//!
7//! The CUDA driver version represents the maximum CUDA version supported by the installed
8//! NVIDIA drivers. This is detected via:
9//!
10//! * NVIDIA Management Library (NVML): Standard method
11//! * CUDA driver library (libcuda) and the nvidia-smi command: Fallbacks for systems without
12//!   NVML, and for musl systems where dynamic library loading is not supported
13//!
14//! ## CUDA Compute Capability (`__cuda_arch`)
15//!
16//! The CUDA compute capability (also known as SM version or architecture version) represents
17//! the **minimum** compute capability of all CUDA devices detected on the system.

92/// Converts a CUDA driver version integer (as reported by NVML/libcuda) into a [`Version`].
93///
94/// The integer is encoded as `major * 1000 + minor * 10` (e.g. `12040` for CUDA 12.4). Because the
95/// FFI out-parameters are zero-initialized, an implausible value (such as `0` from an out-param the
96/// driver never wrote, a negative value, or a nonsensically large one) can slip through even on a
97/// `SUCCESS` return. Only values whose CUDA major version lies in `1..=99` are accepted; anything
98/// else is rejected so that garbage never propagates (or gets cached).
99fn parse_cuda_driver_version(version: c_int) -> Option<Version> {
100    // CUDA major version 1..=99, i.e. the encoded integer must be within [1000, 99990].
101    if !(1000..=99_990).contains(&version) {
102        tracing::trace!(version, "rejecting implausible CUDA driver version integer");
103        return None;
104    }
105    Version::from_str(&format!("{}.{}", version / 1000, (version % 1000) / 10)).ok()
106}

108/// Validates that a string is in the format "major.minor" where both parts are digits.
109///
110/// Returns `true` if the format is valid for CUDA compute capability.
111pub(crate) fn is_valid_cuda_version_format(s: &str) -> bool {
112    let mut parts = s.split('.');
113    match (parts.next(), parts.next(), parts.next()) {
114        (Some(major), Some(minor), None) => {
115            !major.is_empty()
116                && major.chars().all(|c| c.is_ascii_digit())
117                && !minor.is_empty()
118                && minor.chars().all(|c| c.is_ascii_digit())
119        }
120        _ => false,
121    }
122}

124/// Information about CUDA compute capability.
125///
126/// The compute capability (also called SM version) defines the set of features and
127/// instructions supported by a CUDA device. Higher compute capabilities generally
128/// support more features and newer instruction sets.
129///
130/// According to the CEP specification, this represents the minimum compute capability
131/// across all detected CUDA devices, formatted as `{major}.{minor}`.

135    pub major: u32,
136    /// Minor version of the compute capability (e.g., 6 for compute capability 8.6)
137    pub minor: u32,

140/// Combined CUDA information detected from the system.
141///
142/// This struct contains both the CUDA driver version and compute capability information.
143/// Each field is optional because detection can fail independently:
144///
145/// * `version` may be present even without GPUs (driver installed but no devices)
146/// * `arch_info` requires at least one GPU device to be present
147/// * Both may be `None` if CUDA is not available or detection fails
148#[derive(Debug, Clone)]
149pub struct CudaInfo {
150    /// The maximum CUDA version supported by the installed driver.
151    ///
152    /// This corresponds to the `__cuda` virtual package.
153    pub version: Option<Version>,

155    /// Information about the minimum compute capability across all detected devices.
156    ///
157    /// This corresponds to the `__cuda_arch` virtual package. Returns `None` if
158    /// no CUDA devices are detected or if device enumeration fails.
159    pub arch_info: Option<CudaArchInfo>,
160}

173/// Returns comprehensive CUDA information from the current platform.
174///
175/// This function returns both the CUDA driver version and compute capability information
176/// in a single cached result. The detection is performed only once per process and the
177/// result is cached for subsequent calls.
178///
179/// This is more efficient than calling [`cuda_version`] and [`cuda_arch`] separately
180/// because the CUDA library is loaded only once.
181///
182/// Detection runs at most once per process; the in-memory result is reused afterwards. The on-disk
183/// cache, however, is synced lazily: the first call that is given a `cache_dir` reads from and/or
184/// writes to it. A call that passes `None` (e.g. `EnvOverride::detect_from_host`) does not disable
185/// the disk cache for the rest of the process — a later call passing `Some(cache_dir)` still
186/// persists the already-detected result. Pass `None` from every call to fully disable the disk
187/// cache.
188pub fn cuda_info(cache_dir: Option<&Path>) -> &'static CudaInfo {
189    static DETECTED_CUDA_INFO: OnceCell<DetectedCudaInfo> = OnceCell::new();
190    // Whether the in-memory result has been synced with the on-disk cache (read from it or written
191    // to it). This lets a later call with a `cache_dir` persist a result first detected without one.
192    static PERSISTED: AtomicBool = AtomicBool::new(false);
193    cuda_info_impl(
194        &cache::CacheEnv::current(),
195        &DETECTED_CUDA_INFO,
196        &PERSISTED,
197        cache_dir,
198    )
199}

201/// Core of [`cuda_info`], generic over the state so it can be unit-tested with local state instead
202/// of the process-global statics.
203fn cuda_info_impl<'a>(
204    env: &cache::CacheEnv,
205    state: &'a OnceCell<DetectedCudaInfo>,
206    persisted: &AtomicBool,
207    cache_dir: Option<&Path>,
208) -> &'a CudaInfo {
209    if let Some(detected) = state.get() {
210        tracing::trace!(info = ?detected.info, "using process-cached CUDA info");
211        maybe_persist(env, detected, persisted, cache_dir);
212        return &detected.info;
213    }
214
215    let detected = state.get_or_init(|| {
216        // Initializing the driver to detect the GPU can be slow, so the result is cached on disk
217        // and reused until the cache is invalidated (reboot, driver change, GPU change, TTL).
218        if let Some(cache_dir) = cache_dir {
219            tracing::trace!(cache_dir = %cache_dir.display(), "checking CUDA info cache");
220            if let Some(cached) = cache::read_with_env(env, cache_dir) {
221                tracing::debug!(
222                    version = %display_cuda_version(cached.info.version.as_ref()),
223                    arch = %display_cuda_arch(cached.info.arch_info.as_ref()),
224                    version_source = cached.sources.version_str(),
225                    arch_source = cached.sources.arch_str(),
226                    "using disk-cached CUDA info"
227                );
228                // We are now in sync with disk; no need to write it back.
229                persisted.store(true, Ordering::Relaxed);
230                return cached;
231            }
232        } else {
233            tracing::trace!("CUDA info disk cache disabled");
234        }
235
236        tracing::trace!("detecting CUDA info from host");
237        let detected = detect_cuda_info();
238        tracing::debug!(
239            version = %display_cuda_version(detected.info.version.as_ref()),
240            arch = %display_cuda_arch(detected.info.arch_info.as_ref()),
241            version_source = detected.sources.version_str(),
242            arch_source = detected.sources.arch_str(),
243            "detected CUDA info from host"
244        );
245        detected
246    });
247
248    // Persist a freshly detected result if this (or a later) call supplied a cache directory.
249    maybe_persist(env, detected, persisted, cache_dir);
250    &detected.info
251}

253/// Writes the detected result to disk once, if a cache directory is available and it has not been
254/// synced with disk yet. A benign race may write twice, which is safe because the write replaces
255/// the file atomically.
256fn maybe_persist(
257    env: &cache::CacheEnv,
258    detected: &DetectedCudaInfo,
259    persisted: &AtomicBool,
260    cache_dir: Option<&Path>,
261) {
262    let Some(cache_dir) = cache_dir else {
263        return;
264    };
265    if persisted.load(Ordering::Relaxed) {
266        return;
267    }
268    cache::write_with_env(env, cache_dir, &detected.info, detected.sources);
269    // The flag means "we have synced with disk"; set it regardless of write's best-effort outcome.
270    persisted.store(true, Ordering::Relaxed);
271}

297/// Detects comprehensive CUDA information from the current system.
298///
299/// This function performs unified detection of both CUDA driver version and compute
300/// capability by loading NVML once and querying all necessary information.
301///
302/// The detection process:
303/// 1. Attempts to load NVML (`libnvidia-ml`/`nvml.dll`)
304/// 2. Queries the driver version (for `__cuda` virtual package); this does not require init
305/// 3. Initializes NVML and enumerates all CUDA devices to query their compute capabilities
306/// 4. Returns the minimum compute capability across all devices (for `__cuda_arch` virtual package)
307///
308/// On musl systems, both are detected via the `nvidia-smi` command since dynamic library
309/// loading is not supported.
310fn detect_cuda_info() -> DetectedCudaInfo {
311    let mut detected = if cfg!(target_env = "musl") {
312        tracing::trace!("detecting CUDA info via nvidia-smi because musl cannot load NVML");
313        // Dynamically loading a library is not supported on musl so we have to fall-back to using
314        // the nvidia-smi command.
315        let version = detect_cuda_version_via_nvidia_smi();
316        // Only query the compute capability when the driver version was found. On a GPU-less musl
317        // system the version query already failed, so running the arch query too would just spawn
318        // another doomed process and could produce an inconsistent `{version: None, arch: Some}`.
319        let arch_info = version
320            .as_ref()
321            .and_then(|_| detect_cuda_arch_via_nvidia_smi());
322        DetectedCudaInfo {
323            sources: CudaInfoSources {
324                version: version.is_some().then_some(CudaDetectionMethod::NvidiaSmi),
325                arch: arch_info
326                    .is_some()
327                    .then_some(CudaDetectionMethod::NvidiaSmi),
328            },
329            info: CudaInfo { version, arch_info },
330        }
331    } else {
332        tracing::trace!("detecting CUDA info via NVML");
333        // Prefer NVML because it is not affected by `CUDA_VISIBLE_DEVICES`, but fall back to the
334        // older probes so systems that expose libcuda (or nvidia-smi) without NVML still report
335        // `__cuda`.
336        let mut detected = detect_cuda_info_via_nvml();
337
338        if detected.info.version.is_none() {
339            tracing::debug!(
340                "NVML did not detect a CUDA driver version; trying libcuda/nvidia-smi fallbacks"
341            );
342            if let Some((version, source)) = detect_cuda_version_fallbacks() {
343                detected.info.version = Some(version);
344                detected.sources.version = Some(source);
345            }
346        }
347
348        if detected.info.version.is_some() && detected.info.arch_info.is_none() {
349            tracing::debug!(
350                "NVML did not detect CUDA compute capability; trying nvidia-smi fallback"
351            );
352            detected.info.arch_info = detect_cuda_arch_via_nvidia_smi();
353            if detected.info.arch_info.is_some() {
354                detected.sources.arch = Some(CudaDetectionMethod::NvidiaSmi);
355            }
356        }
357
358        // Last resort for platforms that ship libcuda but neither NVML nor nvidia-smi (e.g.
359        // Jetson/Tegra). libcuda comes last because its device enumeration is affected by
360        // `CUDA_VISIBLE_DEVICES`.
361        if detected.info.version.is_some() && detected.info.arch_info.is_none() {
362            tracing::debug!(
363                "nvidia-smi did not detect CUDA compute capability; trying libcuda fallback"
364            );
365            detected.info.arch_info = detect_cuda_arch_via_libcuda();
366            if detected.info.arch_info.is_some() {
367                detected.sources.arch = Some(CudaDetectionMethod::Libcuda);
368            }
369        }
370
371        detected
372    };
373
374    // Normalization for all paths: `__cuda_arch` is meaningless without `__cuda`. If no driver
375    // version was detected, drop any compute capability so callers can never observe
376    // arch-without-version.
377    if detected.info.version.is_none()
378        && (detected.info.arch_info.is_some() || detected.sources.arch.is_some())
379    {
380        tracing::debug!(
381            "dropping CUDA compute capability because no CUDA driver version was detected"
382        );
383        detected.info.arch_info = None;
384        detected.sources.arch = None;
385    }
386
387    detected
388}

390/// Detects CUDA version and architecture information via the NVIDIA Management Library.
391///
392/// The library is loaded once and used to query both the driver version and device compute
393/// capabilities. NVML is preferred over libcuda because it is not affected by
394/// `CUDA_VISIBLE_DEVICES`.
395///
396/// Returns a `CudaInfo` struct where:
397/// * `version` is `None` if the driver version cannot be determined
398/// * `arch_info` is `None` if no devices are present or device queries fail
399fn detect_cuda_info_via_nvml() -> DetectedCudaInfo {
400    let mut library = None;
401    for path in nvml_library_paths() {
402        match unsafe { Library::new(*path) } {
403            Ok(loaded) => {
404                tracing::trace!(library_path = *path, "loaded NVML library");
405                library = Some(loaded);
406                break;
407            }
408            Err(err) => {
409                tracing::trace!(library_path = *path, error = %err, "failed to load NVML library");
410            }
411        }
412    }
413
414    let Some(library) = library else {
415        tracing::debug!("could not load NVML library from any known path");
416        return DetectedCudaInfo {
417            info: CudaInfo {
418                version: None,
419                arch_info: None,
420            },
421            sources: CudaInfoSources::default(),
422        };
423    };
424
425    // Attempt the cheap no-init query. Some drivers answer `nvmlSystemGetCudaDriverVersion` before
426    // `nvmlInit`, but it is not officially supported, so this result is only used as a fall back for
427    // when `nvmlInit` itself fails below.
428    let no_init_version = match cuda_version_from_nvml_library(&library) {
429        Ok(version) => {
430            tracing::trace!(%version, "detected CUDA driver version via NVML without init");
431            Some(version)
432        }
433        Err(err) => {
434            tracing::trace!(
435                ?err,
436                "CUDA driver version query via NVML without init failed"
437            );
438            None
439        }
440    };
441
442    // Compute capability requires enumerating devices, which needs NVML to be initialized. Since
443    // NVML is being initialized anyway, query the driver version while initialized too and prefer
444    // that officially supported result over the no-init query.
445    let (initialized_version, arch_info) =
446        detect_cuda_initialized_info_via_nvml(&library, true, true);
447
448    // Prefer the version obtained from initialized NVML whenever it is available; only fall back to
449    // the no-init value when `nvmlInit` (and thus the initialized query) did not produce one.
450    let (version, version_source) = if let Some(version) = initialized_version {
451        (Some(version), Some(CudaDetectionMethod::NvmlInitialized))
452    } else if let Some(version) = no_init_version {
453        (Some(version), Some(CudaDetectionMethod::NvmlNoInit))
454    } else {
455        (None, None)
456    };
457
458    let arch_source = arch_info
459        .as_ref()
460        .map(|_| CudaDetectionMethod::NvmlInitialized);
461
462    DetectedCudaInfo {
463        info: CudaInfo { version, arch_info },
464        sources: CudaInfoSources {
465            version: version_source,
466            arch: arch_source,
467        },
468    }
469}

471/// Queries the CUDA driver version from an already-loaded NVML library.
472///
473/// Some drivers allow `nvmlSystemGetCudaDriverVersion` before `nvmlInit`, but others return
474/// `NVML_ERROR_UNINITIALIZED`; callers may retry after initializing NVML for that error.
475fn cuda_version_from_nvml_library(library: &Library) -> Result<Version, NvmlCudaVersionError> {
476    // Find the `nvmlSystemGetCudaDriverVersion_v2` function. If that function cannot be found, fall
477    // back to the `nvmlSystemGetCudaDriverVersion` function instead.
478    let nvml_system_get_cuda_driver_version: Symbol<'_, unsafe extern "C" fn(*mut c_int) -> c_int> =
479        unsafe {
480            library
481                .get(b"nvmlSystemGetCudaDriverVersion_v2\0")
482                .or_else(|_| library.get(b"nvmlSystemGetCudaDriverVersion\0"))
483        }
484        .map_err(|_err| NvmlCudaVersionError::MissingSymbol)?;
485
486    // Zero-initialize the out-parameter so that a driver returning `NVML_SUCCESS` without actually
487    // writing it yields a deterministic `0`, which `parse_cuda_driver_version` rejects, rather than
488    // undefined behavior from reading uninitialized memory.
489    let mut cuda_driver_version: c_int = 0;
490    let result = unsafe { nvml_system_get_cuda_driver_version(&mut cuda_driver_version) };
491    if result != NVML_SUCCESS {
492        return Err(NvmlCudaVersionError::Nvml(result));
493    }
494
495    parse_cuda_driver_version(cuda_driver_version).ok_or(NvmlCudaVersionError::InvalidVersion)
496}

498/// Queries information that requires initialized NVML.
499///
500/// Returns `(version, arch_info)`. Each field is only queried when the corresponding `query_*`
501/// argument is true. The initialized version query is a compatibility fallback for drivers that
502/// reject `nvmlSystemGetCudaDriverVersion` before `nvmlInit`.
503fn detect_cuda_initialized_info_via_nvml(
504    library: &Library,
505    query_version: bool,
506    query_arch: bool,
507) -> (Option<Version>, Option<CudaArchInfo>) {
508    // NVML device handle (`nvmlDevice_t`) is an opaque pointer.
509    type NvmlDevice = *mut c_void;
510
511    let Some(nvml_init): Option<Symbol<'_, unsafe extern "C" fn() -> c_int>> = (unsafe {
512        library
513            .get(b"nvmlInit_v2\0")
514            .or_else(|_| library.get(b"nvmlInit\0"))
515            .ok()
516    }) else {
517        tracing::debug!("missing nvmlInit symbol");
518        return (None, None);
519    };
520
521    let Some(nvml_shutdown): Option<Symbol<'_, unsafe extern "C" fn() -> c_int>> =
522        (unsafe { library.get(b"nvmlShutdown\0").ok() })
523    else {
524        tracing::debug!("missing nvmlShutdown symbol");
525        return (None, None);
526    };
527
528    tracing::trace!(query_version, query_arch, "initializing NVML");
529    let init_result = unsafe { nvml_init() };
530    if init_result != NVML_SUCCESS {
531        tracing::debug!(return_code = init_result, "nvmlInit failed");
532        return (None, None);
533    }
534
535    let version = if query_version {
536        match cuda_version_from_nvml_library(library) {
537            Ok(version) => {
538                tracing::debug!(%version, "detected CUDA driver version via initialized NVML");
539                Some(version)
540            }
541            Err(err) => {
542                tracing::debug!(
543                    ?err,
544                    "CUDA driver version query via initialized NVML failed"
545                );
546                None
547            }
548        }
549    } else {
550        None
551    };
552
553    // Enumerate devices to find the minimum compute capability. Wrapped in a closure so we always
554    // reach the `nvmlShutdown` call below regardless of the outcome.
555    let arch_info = query_arch
556        .then(|| {
557            tracing::trace!("querying CUDA compute capability via NVML");
558            let nvml_device_get_count: Symbol<'_, unsafe extern "C" fn(*mut c_uint) -> c_int> =
559                match unsafe {
560                    library
561                        .get(b"nvmlDeviceGetCount_v2\0")
562                        .or_else(|_| library.get(b"nvmlDeviceGetCount\0"))
563                } {
564                    Ok(symbol) => symbol,
565                    Err(err) => {
566                        tracing::trace!(error = %err, "missing NVML device count symbol");
567                        return None;
568                    }
569                };
570
571            let nvml_device_get_handle_by_index: Symbol<
572                '_,
573                unsafe extern "C" fn(c_uint, *mut NvmlDevice) -> c_int,
574            > = match unsafe {
575                library
576                    .get(b"nvmlDeviceGetHandleByIndex_v2\0")
577                    .or_else(|_| library.get(b"nvmlDeviceGetHandleByIndex\0"))
578            } {
579                Ok(symbol) => symbol,
580                Err(err) => {
581                    tracing::trace!(error = %err, "missing NVML device handle symbol");
582                    return None;
583                }
584            };
585
586            let nvml_device_get_cuda_compute_capability: Symbol<
587                '_,
588                unsafe extern "C" fn(NvmlDevice, *mut c_int, *mut c_int) -> c_int,
589            > = match unsafe { library.get(b"nvmlDeviceGetCudaComputeCapability\0") } {
590                Ok(symbol) => symbol,
591                Err(err) => {
592                    tracing::trace!(
593                        error = %err,
594                        "missing NVML CUDA compute capability symbol"
595                    );
596                    return None;
597                }
598            };
599
600            let mut device_count: c_uint = 0;
601            let device_count_result = unsafe { nvml_device_get_count(&mut device_count) };
602            if device_count_result != NVML_SUCCESS {
603                tracing::trace!(
604                    return_code = device_count_result,
605                    "nvmlDeviceGetCount failed"
606                );
607                return None;
608            }
609            tracing::trace!(device_count, "enumerating CUDA devices via NVML");
610
611            let mut min_arch: Option<CudaArchInfo> = None;
612            for device_idx in 0..device_count {
613                let mut device: NvmlDevice = ptr::null_mut();
614                let handle_result =
615                    unsafe { nvml_device_get_handle_by_index(device_idx, &mut device) };
616                if handle_result != NVML_SUCCESS {
617                    tracing::trace!(
618                        device_idx,
619                        return_code = handle_result,
620                        "failed to get NVML device handle"
621                    );
622                    continue;
623                }
624
625                let mut cc_major: c_int = 0;
626                let mut cc_minor: c_int = 0;
627                let compute_capability_result = unsafe {
628                    nvml_device_get_cuda_compute_capability(device, &mut cc_major, &mut cc_minor)
629                };
630                if compute_capability_result != NVML_SUCCESS {
631                    tracing::trace!(
632                        device_idx,
633                        return_code = compute_capability_result,
634                        "failed to get CUDA compute capability via NVML"
635                    );
636                    continue;
637                }
638                let cc_major = cc_major as u32;
639                let cc_minor = cc_minor as u32;
640                tracing::trace!(
641                    device_idx,
642                    major = cc_major,
643                    minor = cc_minor,
644                    "detected CUDA compute capability via NVML"
645                );
646
647                let is_new_minimum = min_arch.as_ref().is_none_or(|min| {
648                    cc_major < min.major || (cc_major == min.major && cc_minor < min.minor)
649                });
650                if is_new_minimum {
651                    min_arch = Some(CudaArchInfo {
652                        major: cc_major,
653                        minor: cc_minor,
654                    });
655                }
656            }
657            if let Some(arch) = min_arch.as_ref() {
658                tracing::debug!(
659                    major = arch.major,
660                    minor = arch.minor,
661                    "selected minimum CUDA compute capability"
662                );
663            } else {
664                tracing::debug!("no CUDA compute capability detected via NVML");
665            }
666            min_arch
667        })
668        .flatten();
669
670    // Whatever happens, after initializing NVML we have to call `nvmlShutdown`.
671    let shutdown_result = unsafe { nvml_shutdown() };
672    if shutdown_result != NVML_SUCCESS {
673        tracing::debug!(return_code = shutdown_result, "nvmlShutdown failed");
674    }
675
676    (version, arch_info)
677}

679/// Attempts to detect the version of CUDA present in the current operating system by employing the
680/// best technique available for the current environment.
681pub fn detect_cuda_version() -> Option<Version> {
682    if cfg!(target_env = "musl") {
683        // Dynamically loading a library is not supported on musl so we have to fall-back to using
684        // the nvidia-smi command.
685        detect_cuda_version_via_nvidia_smi()
686    } else {
687        detect_cuda_version_via_nvml().or_else(|| {
688            tracing::debug!(
689                "NVML did not detect a CUDA driver version; trying libcuda/nvidia-smi fallbacks"
690            );
691            detect_cuda_version_fallbacks().map(|(version, _source)| version)
692        })
693    }
694}

696fn detect_cuda_version_fallbacks() -> Option<(Version, CudaDetectionMethod)> {
697    if cfg!(target_env = "musl") {

709}

711/// Attempts to detect the version of CUDA present in the current operating system by loading the
712/// NVIDIA Management Library and querying the CUDA driver version. The method is preferred over
713/// [`detect_cuda_version_via_libcuda`] because that method might fail base on environment
714/// variables.
715///
716/// Although the required methods in the runtime are not implemented on much older machines it is
717/// considered old enough to be usable for our use case. Since Conda doesn't provide old versions of
718/// the CUDA SDK anyway this is considered a non-issue.
719///
720/// Some drivers can answer `nvmlSystemGetCudaDriverVersion` without `nvmlInit`, avoiding the
721/// expensive driver handshake / GPU attach that makes `nvmlInit` slow on Windows. If that no-init
722/// query fails, this falls back to querying while NVML is initialized.
723pub fn detect_cuda_version_via_nvml() -> Option<Version> {
724    // Try to open the library
725    let mut library = None;
726    for path in nvml_library_paths() {
727        match unsafe { libloading::Library::new(*path) } {
728            Ok(loaded) => {
729                tracing::trace!(library_path = *path, "loaded NVML library");
730                library = Some(loaded);
731                break;
732            }
733            Err(err) => {
734                tracing::trace!(library_path = *path, error = %err, "failed to load NVML library");
735            }
736        }
737    }
738    let library = library?;
739
740    match cuda_version_from_nvml_library(&library) {
741        Ok(version) => {
742            tracing::trace!(%version, "detected CUDA driver version via NVML without init");
743            Some(version)
744        }
745        Err(err) if err.should_retry_after_init() => {
746            tracing::debug!(?err, "retrying CUDA driver version query after nvmlInit");
747            detect_cuda_initialized_info_via_nvml(&library, true, false).0
748        }
749        Err(err) => {
750            tracing::debug!(?err, "CUDA driver version query via NVML failed");
751            None
752        }
753    }
754}

790/// Attempts to detect the version of CUDA present in the current operating system by loading the
791/// cuda runtime library and querying the CUDA driver version.
792///
793/// The behavior of functions from `libcuda` depend on the environment variable
794/// `CUDA_VISIBLE_DEVICES`. If users have this variable set in their environment this function will
795/// likely not return the correct value.
796///
797/// Therefore you should use the function [`detect_cuda_version_via_nvml`] instead which does not
798/// have this limitation.
799pub fn detect_cuda_version_via_libcuda() -> Option<Version> {
800    // Try to open the library
801    let mut cuda_library = None;
802    for path in cuda_library_paths() {
803        match unsafe { libloading::Library::new(*path) } {
804            Ok(loaded) => {
805                tracing::trace!(library_path = *path, "loaded CUDA driver library");
806                cuda_library = Some(loaded);
807                break;
808            }
809            Err(err) => {
810                tracing::trace!(
811                    library_path = *path,
812                    error = %err,
813                    "failed to load CUDA driver library"
814                );
815            }
816        }
817    }
818    let cuda_library = cuda_library?;
819
820    // Get entry points from the library. `CUresult` is a 32-bit enum, so these are declared to
821    // return `c_int` (matching the NVML declarations); on some ABIs the upper bits of a wider
822    // return register are unspecified.
823    let cu_init: Symbol<'_, unsafe extern "C" fn(c_uint) -> c_int> =
824        match unsafe { cuda_library.get(b"cuInit\0") } {
825            Ok(symbol) => symbol,
826            Err(err) => {
827                tracing::debug!(error = %err, "missing cuInit symbol");
828                return None;
829            }
830        };
831    let cu_driver_get_version: Symbol<'_, unsafe extern "C" fn(*mut c_int) -> c_int> =
832        match unsafe { cuda_library.get(b"cuDriverGetVersion\0") } {
833            Ok(symbol) => symbol,
834            Err(err) => {
835                tracing::debug!(error = %err, "missing cuDriverGetVersion symbol");
836                return None;
837            }
838        };
839
840    // Initialize the CUDA library
841    let init_result = unsafe { cu_init(0) };
842    if init_result != 0 {
843        tracing::debug!(return_code = init_result, "cuInit failed");
844        return None;
845    }
846
847    // Get the version from the library. The out-parameter is zero-initialized so that a driver
848    // returning success without writing it yields a deterministic `0`, which is rejected below.
849    let mut version_int: c_int = 0;
850    let version_result = unsafe { cu_driver_get_version(&mut version_int) };
851    if version_result != 0 {
852        tracing::debug!(return_code = version_result, "cuDriverGetVersion failed");
853        return None;
854    }
855
856    // Convert the version integer to a version string
857    let version = parse_cuda_driver_version(version_int);
858    if let Some(version) = &version {
859        tracing::trace!(%version, "detected CUDA driver version via libcuda");
860    } else {
861        tracing::trace!("failed to parse CUDA driver version reported by libcuda");
862    }
863    version
864}

866/// Attempts to detect the CUDA compute capability by loading the CUDA driver library and
867/// enumerating all devices, returning the **minimum** compute capability across all devices.
868///
869/// This is the fallback for platforms that ship libcuda but neither NVML nor nvidia-smi (e.g.
870/// Jetson/Tegra). Device enumeration through libcuda is affected by `CUDA_VISIBLE_DEVICES`, so
871/// the NVML and nvidia-smi probes are preferred.
872fn detect_cuda_arch_via_libcuda() -> Option<CudaArchInfo> {
873    // CUDA device attribute constants for querying compute capability
874    const CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MAJOR: c_int = 75;
875    const CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MINOR: c_int = 76;
876
877    // Try to open the library
878    let mut cuda_library = None;
879    for path in cuda_library_paths() {
880        match unsafe { libloading::Library::new(*path) } {
881            Ok(loaded) => {
882                tracing::trace!(library_path = *path, "loaded CUDA driver library");
883                cuda_library = Some(loaded);
884                break;
885            }
886            Err(err) => {
887                tracing::trace!(
888                    library_path = *path,
889                    error = %err,
890                    "failed to load CUDA driver library"
891                );
892            }
893        }
894    }
895    let cuda_library = cuda_library?;
896
897    // Get entry points from the library. `CUresult` is a 32-bit enum, so these are declared to
898    // return `c_int`.
899    let cu_init: Symbol<'_, unsafe extern "C" fn(c_uint) -> c_int> =
900        match unsafe { cuda_library.get(b"cuInit\0") } {
901            Ok(symbol) => symbol,
902            Err(err) => {
903                tracing::debug!(error = %err, "missing cuInit symbol");
904                return None;
905            }
906        };
907    let cu_device_get_count: Symbol<'_, unsafe extern "C" fn(*mut c_int) -> c_int> =
908        match unsafe { cuda_library.get(b"cuDeviceGetCount\0") } {
909            Ok(symbol) => symbol,
910            Err(err) => {
911                tracing::debug!(error = %err, "missing cuDeviceGetCount symbol");
912                return None;
913            }
914        };
915    let cu_device_get: Symbol<'_, unsafe extern "C" fn(*mut c_int, c_int) -> c_int> =
916        match unsafe { cuda_library.get(b"cuDeviceGet\0") } {
917            Ok(symbol) => symbol,
918            Err(err) => {
919                tracing::debug!(error = %err, "missing cuDeviceGet symbol");
920                return None;
921            }
922        };
923    let cu_device_get_attribute: Symbol<
924        '_,
925        unsafe extern "C" fn(*mut c_int, c_int, c_int) -> c_int,
926    > = match unsafe { cuda_library.get(b"cuDeviceGetAttribute\0") } {
927        Ok(symbol) => symbol,
928        Err(err) => {
929            tracing::debug!(error = %err, "missing cuDeviceGetAttribute symbol");
930            return None;
931        }
932    };
933
934    // Initialize the CUDA library
935    let init_result = unsafe { cu_init(0) };
936    if init_result != 0 {
937        tracing::debug!(return_code = init_result, "cuInit failed");
938        return None;
939    }
940
941    // Get the number of CUDA devices
942    let mut device_count: c_int = 0;
943    let device_count_result = unsafe { cu_device_get_count(&mut device_count) };
944    if device_count_result != 0 {
945        tracing::trace!(return_code = device_count_result, "cuDeviceGetCount failed");
946        return None;
947    }
948    tracing::trace!(device_count, "enumerating CUDA devices via libcuda");
949
950    // Iterate through all devices to find the minimum compute capability
951    let mut min_arch: Option<CudaArchInfo> = None;
952    for device_idx in 0..device_count {
953        let mut device: c_int = 0;
954        let device_result = unsafe { cu_device_get(&mut device, device_idx) };
955        if device_result != 0 {
956            tracing::trace!(
957                device_idx,
958                return_code = device_result,
959                "failed to get CUDA device handle"
960            );
961            continue;
962        }
963
964        let mut cc_major: c_int = 0;
965        let mut cc_minor: c_int = 0;
966        let major_result = unsafe {
967            cu_device_get_attribute(
968                &mut cc_major,
969                CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MAJOR,
970                device,
971            )
972        };
973        let minor_result = unsafe {
974            cu_device_get_attribute(
975                &mut cc_minor,
976                CU_DEVICE_ATTRIBUTE_COMPUTE_CAPABILITY_MINOR,
977                device,
978            )
979        };
980        if major_result != 0 || minor_result != 0 {
981            tracing::trace!(
982                device_idx,
983                major_return_code = major_result,
984                minor_return_code = minor_result,
985                "failed to get CUDA compute capability via libcuda"
986            );
987            continue;
988        }
989        let cc_major = cc_major as u32;
990        let cc_minor = cc_minor as u32;
991        tracing::trace!(
992            device_idx,
993            major = cc_major,
994            minor = cc_minor,
995            "detected CUDA compute capability via libcuda"
996        );
997
998        let is_new_minimum = min_arch.as_ref().is_none_or(|min| {
999            cc_major < min.major || (cc_major == min.major && cc_minor < min.minor)
1000        });
1001        if is_new_minimum {
1002            min_arch = Some(CudaArchInfo {
1003                major: cc_major,
1004                minor: cc_minor,
1005            });
1006        }
1007    }
1008
1009    if let Some(arch) = min_arch.as_ref() {
1010        tracing::debug!(
1011            major = arch.major,
1012            minor = arch.minor,
1013            "selected minimum CUDA compute capability from libcuda"
1014        );
1015    } else {
1016        tracing::debug!("no CUDA compute capability detected via libcuda");
1017    }
1018
1019    min_arch
1020}

1056/// Attempts to detect the version of CUDA present in the current operating system by executing the
1057/// "nvidia-smi" command and extracting the CUDA driver version from it.
1058///
1059/// The behavior of "nvidia-smi" depends on the environment variable `CUDA_VISIBLE_DEVICES`. If
1060/// users have this variable set in their environment this function will likely not return the
1061/// correct value. To ensure a consistent response this environment variable is unset when invoking
1062/// the command.
1063///
1064/// The upside of using this detection function over any of the others is that this method does not
1065/// dynamically load a library which might not be supported on all systems. The downside is that
1066/// executing a subprocess is generally slower and more prone to errors.
1067fn detect_cuda_version_via_nvidia_smi() -> Option<Version> {
1068    tracing::trace!("detecting CUDA driver version via nvidia-smi");
1069    // Invoke the "nvidia-smi" command to query the driver version that is usually installed when
1070    // Cuda drivers are installed.
1071    let nvidia_smi_output = match Command::new("nvidia-smi")
1072        // Display GPU or unit info
1073        .arg("--query")
1074        // Show unit, rather than GPU, attributes
1075        .arg("-u")
1076        // Produce XML output.
1077        .arg("-x")
1078        // The behavior of functions from `libcuda` depend on the environment variable
1079        // `CUDA_VISIBLE_DEVICES`. If users have this variable set in their environment this
1080        // function will likely not return the correct value. Therefor, we remove this variable
1081        // to ensure a consistent result.
1082        // TODO: Is this really the proper way to do it? Should we maybe clear the entire
1083        // environment.
1084        .env_remove("CUDA_VISIBLE_DEVICES")
1085        .output()
1086    {
1087        Ok(output) => output,
1088        Err(err) => {
1089            tracing::debug!(error = %err, "failed to run nvidia-smi for CUDA driver version");
1090            return None;
1091        }
1092    };
1093
1094    // nvidia-smi can exit non-zero in degraded-but-parseable states (e.g. one GPU lost while others
1095    // are healthy) where the XML still contains a usable `<cuda_version>`. Log the failure but still
1096    // attempt to parse stdout instead of bailing out on the exit status.
1097    if !nvidia_smi_output.status.success() {
1098        tracing::debug!(
1099            status = %nvidia_smi_output.status,
1100            stderr = %String::from_utf8_lossy(&nvidia_smi_output.stderr),
1101            "nvidia-smi CUDA driver version query exited non-zero; attempting to parse output anyway"
1102        );
1103    }
1104
1105    // Convert the output to Utf8. The conversion is lossy so it might contain some illegal
1106    // characters. If that is the case we simply assume the version in the file also wont make sense
1107    // during parsing.
1108    let output = String::from_utf8_lossy(&nvidia_smi_output.stdout);
1109    parse_nvidia_smi_cuda_version(&output)
1110}

1112/// Extracts the CUDA driver version from the XML output produced by `nvidia-smi --query -u -x`.
1113///
1114/// Returns `None` if the `<cuda_version>` element is missing or cannot be parsed as a [`Version`].
1115fn parse_nvidia_smi_cuda_version(output: &str) -> Option<Version> {
1116    static CUDA_VERSION_RE: once_cell::sync::Lazy<regex::Regex> =
1117        once_cell::sync::Lazy::new(|| {
1118            regex::Regex::new("<cuda_version>(.*)<\\/cuda_version>").unwrap()
1119        });
1120
1121    // Extract the version from the XML
1122    let Some(version_match) = CUDA_VERSION_RE.captures(output) else {
1123        tracing::trace!("nvidia-smi output did not contain a CUDA driver version");
1124        return None;
1125    };
1126    let Some(version_match) = version_match.get(1) else {
1127        tracing::trace!("nvidia-smi CUDA driver version match was empty");
1128        return None;
1129    };
1130    let version_str = version_match.as_str();
1131
1132    // Parse and return
1133    match Version::from_str(version_str) {
1134        Ok(version) => {
1135            tracing::trace!(%version, "detected CUDA driver version via nvidia-smi");
1136            Some(version)
1137        }
1138        Err(err) => {
1139            tracing::trace!(version = version_str, error = %err, "failed to parse nvidia-smi CUDA driver version");
1140            None
1141        }
1142    }
1143}

1145/// Attempts to detect the CUDA compute capability by executing the "nvidia-smi" command and
1146/// querying the `compute_cap` field of every GPU, returning the **minimum** across all devices.
1147///
1148/// Like [`detect_cuda_version_via_nvidia_smi`] this does not dynamically load a library and thus
1149/// also works on musl systems. The `compute_cap` query field requires a reasonably modern driver
1150/// (roughly R510+); on older drivers the command fails and `None` is returned.

1162        Err(err) => {

1164            return None;
1165        }
1166    };

1168    // On drivers that do not support the `compute_cap` field the command exits with an error, but it
1169    // can also exit non-zero while still reporting some healthy GPUs on stdout. Log the failure but
1170    // still attempt to parse whatever was produced instead of bailing out on the exit status.
1171    if !nvidia_smi_output.status.success() {
1172        tracing::debug!(
1173            status = %nvidia_smi_output.status,
1174            stderr = %String::from_utf8_lossy(&nvidia_smi_output.stderr),

1183/// Parses the CSV output of `nvidia-smi --query-gpu=compute_cap --format=csv,noheader` and returns
1184/// the **minimum** compute capability across all parseable GPU lines.
1185///
1186/// Each line is expected to be a `major.minor` value. Lines that are not in that format (such as
1187/// `[N/A]` reported for a GPU whose capability is unknown, or other junk) are skipped while still
1188/// using the valid ones. Returns `None` if no line could be parsed.
1189fn parse_nvidia_smi_compute_capabilities(output: &str) -> Option<CudaArchInfo> {
1190    // Find the minimum compute capability across all devices
1191    let mut min_arch: Option<CudaArchInfo> = None;
1192    for (device_idx, line) in output.lines().enumerate() {
1193        let line = line.trim();
1194        let Some((major, minor)) = line.split_once('.') else {
1195            tracing::trace!(
1196                device_idx,
1197                line,
1198                "ignoring invalid nvidia-smi compute capability line"
1199            );
1200            continue;
1201        };
1202        let (Ok(major), Ok(minor)) = (major.parse::<u32>(), minor.parse::<u32>()) else {
1203            tracing::trace!(
1204                device_idx,
1205                line,
1206                "ignoring unparsable nvidia-smi compute capability line"
1207            );
1208            continue;
1209        };
1210        tracing::trace!(
1211            device_idx,
1212            major,
1213            minor,
1214            "detected CUDA compute capability via nvidia-smi"
1215        );
1216
1217        let is_new_minimum = min_arch
1218            .as_ref()
1219            .is_none_or(|min| major < min.major || (major == min.major && minor < min.minor));
1220        if is_new_minimum {
1221            min_arch = Some(CudaArchInfo { major, minor });
1222        }
1223    }
1224
1225    if let Some(arch) = min_arch.as_ref() {
1226        tracing::debug!(
1227            major = arch.major,
1228            minor = arch.minor,
1229            "selected minimum CUDA compute capability from nvidia-smi"
1230        );
1231    } else {
1232        tracing::debug!("no CUDA compute capability detected via nvidia-smi");
1233    }
1234
1235    min_arch
1236}

1396        let smi_arch = time("arch via nvidia-smi", detect_cuda_arch_via_nvidia_smi);

1402            .copied()

1407            });

1441        println!("nvidia-smi:       {smi_version:?} / {smi_arch:?}");
