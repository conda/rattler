1//! On-disk cache for detected CUDA information, valid for the current boot session.
2//!
3//! Detecting CUDA can be slow (initializing NVML attaches to every GPU), so the result is cached
4//! on disk between processes. The cache is keyed on everything that can invalidate a previous
5//! detection without the code noticing:
6//!
7//! * the current **boot session**, because a reboot re-enumerates hardware and drivers;
8//! * a **driver fingerprint**, because drivers can be updated (and NVML reloaded) without a reboot;
9//! * a **device fingerprint** of the host-visible GPUs, because two containers sharing a cache
10//!   volume see different GPU subsets and topology can change within a session (eGPU hot-plug,
11//!   PCI hot-plug into VMs, suspend/resume);
12//! * a **TTL** as a staleness backstop for anything the fingerprints cannot catch (and to let
13//!   transient arch-detection failures self-heal quickly).
14//!
15//! Reads and writes are best-effort: any failure simply results in a fresh detection. Delete the
16//! file or use the `CONDA_OVERRIDE_CUDA*` variables to bypass it.
17
18use super::{CudaArchInfo, CudaDetectionMethod, CudaInfo, CudaInfoSources, DetectedCudaInfo};
19use rattler_conda_types::Version;
20use serde::{Deserialize, Serialize};
21use std::{
22    io::Write,
23    path::{Path, PathBuf},
24    str::FromStr,
25};

27const CACHE_FILE_NAME: &str = "cuda-info-v1.json";

29/// Full detections (with compute capability) are trusted for a day; the fingerprints catch most
30/// changes sooner, so this is only a staleness backstop.
31const FULL_TTL_SECS: u64 = 24 * 60 * 60;
32/// Entries without compute capability represent a transient arch-detection failure, so they expire
33/// quickly and self-heal instead of lingering for the whole boot session.
34const ARCH_MISSING_TTL_SECS: u64 = 10 * 60;
35/// Reject entries whose write time is further than this into the future, which means the clock
36/// stepped backwards and the recorded `written_at` can no longer be trusted for the TTL.
37const MAX_CLOCK_SKEW_SECS: u64 = 5 * 60;

39/// Identifies a single boot session of the machine.
40///
41/// All variants are compiled on every platform so the comparison logic can be unit-tested
42/// anywhere; `current` only ever produces the variants that exist on the host platform.
43#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
44#[serde(rename_all = "snake_case")]
45pub(super) enum BootId {
46    /// The kernel's per-boot UUID from `/proc/sys/kernel/random/boot_id` (Linux).
47    Uuid(String),
48    /// The prefetcher boot counter from the registry, incremented once per boot (Windows).
49    BootCount(u32),
50    /// A boot time in unix seconds derived from the uptime (Windows fallback). The derivation
51    /// drifts a little between processes, which `matches` absorbs with a tolerance.
52    BootTime(u64),
53}
54
55impl BootId {
56    /// Returns the identifier of the current boot session, or `None` if it cannot be determined
57    /// (in which case no caching takes place).
58    pub(super) fn current() -> Option<Self> {
59        #[cfg(target_os = "linux")]
60        {
61            // The kernel generates a fresh UUID on every boot.
62            let id = std::fs::read_to_string("/proc/sys/kernel/random/boot_id").ok()?;
63            Some(Self::Uuid(id.trim().to_owned()))
64        }
65        #[cfg(target_os = "windows")]
66        {
67            // Prefer the prefetcher boot counter: it increments exactly once per boot and involves
68            // no clock arithmetic, so it cannot be confused by reboots or clock steps.
69            if let Some(count) = windows_boot_count() {
70                return Some(Self::BootCount(count));
71            }
72            // Fall back to deriving the boot time from the uptime.
73            let uptime_secs =
74                unsafe { windows_sys::Win32::System::SystemInformation::GetTickCount64() } / 1000;
75            let now_secs = std::time::SystemTime::now()
76                .duration_since(std::time::UNIX_EPOCH)
77                .ok()?
78                .as_secs();
79            Some(Self::BootTime(now_secs.checked_sub(uptime_secs)?))
80        }
81        #[cfg(not(any(target_os = "linux", target_os = "windows")))]
82        {
83            None
84        }
85    }
86
87    /// Returns true if both identifiers refer to the same boot session.
88    pub(super) fn matches(&self, other: &Self) -> bool {
89        match (self, other) {
90            // Two derived boot times can drift a few seconds between processes; treat them as the
91            // same session when they are within tolerance.
92            (Self::BootTime(a), Self::BootTime(b)) => boot_times_within_tolerance(*a, *b),
93            // Everything else (boot UUIDs, boot counters, or mixed kinds) must match exactly; two
94            // different kinds never refer to the same session.
95            _ => self == other,
96        }
97    }
98}

100/// Returns true if two `boottime:` second values are close enough to be the same boot session.
101///
102/// Extracted as a plain function (not `cfg(windows)`-gated) so the tolerance logic is compiled and
103/// unit-tested on every platform. A real reboot shifts the derived boot time by at least the
104/// previous uptime, which is far larger than this tolerance.
105pub(super) fn boot_times_within_tolerance(a: u64, b: u64) -> bool {
106    /// The derived boot time drifts a little between processes.
107    const BOOT_TIME_TOLERANCE_SECS: u64 = 120;
108    a.abs_diff(b) <= BOOT_TIME_TOLERANCE_SECS
109}

111/// Reads the prefetcher boot counter from the registry, incremented once per boot.
112#[cfg(target_os = "windows")]
113fn windows_boot_count() -> Option<u32> {
114    use windows_sys::Win32::System::Registry::{
115        HKEY_LOCAL_MACHINE, RRF_RT_REG_DWORD, RegGetValueW,
116    };
117
118    fn wide(s: &str) -> Vec<u16> {
119        s.encode_utf16().chain(std::iter::once(0)).collect()
120    }
121
122    let subkey = wide(
123        "SYSTEM\\CurrentControlSet\\Control\\Session Manager\\Memory Management\\PrefetchParameters",
124    );
125    let value = wide("BootId");
126    let mut data: u32 = 0;
127    let mut data_size: u32 = std::mem::size_of::<u32>() as u32;
128    let status = unsafe {
129        RegGetValueW(
130            HKEY_LOCAL_MACHINE,
131            subkey.as_ptr(),
132            value.as_ptr(),
133            RRF_RT_REG_DWORD,
134            std::ptr::null_mut(),
135            std::ptr::addr_of_mut!(data).cast::<std::ffi::c_void>(),
136            &mut data_size,
137        )
138    };
139    // ERROR_SUCCESS
140    if status == 0 { Some(data) } else { None }
141}

143/// Identifies the installed NVIDIA driver.
144///
145/// Drivers can be updated without a reboot, so the boot session alone is not enough to key the
146/// cache on. All variants are compiled on every platform so they can be unit-tested anywhere.
147#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
148#[serde(rename_all = "snake_case")]
149pub(super) enum DriverFingerprint {
150    /// The version of the loaded `nvidia` kernel module (Linux), which changes when the driver is
151    /// updated and the module is reloaded.
152    Module { version: String },
153    /// The identity of the NVML library file on disk (Windows, WSL2), which driver updates
154    /// replace.
155    File {
156        path: PathBuf,
157        mtime_secs: u64,
158        len: u64,
159    },
160}

162/// Identifies the set of host-visible GPUs.
163///
164/// This distinguishes containers that share a cache volume but see different GPU subsets, and it
165/// changes when GPUs are hot-plugged or unplugged within a boot session.
166#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
167pub(super) struct DeviceFingerprint {
168    /// Sorted identifiers of the GPUs the host exposes: PCI bus ids from
169    /// `/proc/driver/nvidia/gpus` on Linux, Plug and Play device ids on Windows.
170    pub(super) gpus: Vec<String>,
171    /// Sorted `/dev/nvidiaN` device nodes that are present. Always empty on Windows, which has no
172    /// device nodes.
173    pub(super) device_nodes: Vec<String>,
174}

176#[derive(Serialize, Deserialize)]
177struct CacheFile {
178    /// The boot session during which detection ran.
179    boot_id: BootId,
180    /// Fingerprint of the installed driver when detection ran. Required: entries without a driver
181    /// fingerprint cannot be trusted, so they are neither written nor read.
182    driver_fingerprint: DriverFingerprint,
183    /// Fingerprint of the host-visible GPUs when detection ran, or `None` if it could not be
184    /// determined on this platform.
185    device_fingerprint: Option<DeviceFingerprint>,
186    /// Unix seconds at which the entry was written, used for the TTL.
187    written_at: u64,
188    version: String,
189    arch: Option<(u32, u32)>,
190    #[serde(default)]
191    version_source: Option<CudaDetectionMethod>,
192    #[serde(default)]
193    arch_source: Option<CudaDetectionMethod>,
194}

196/// The environment against which a cache entry is validated: everything that can invalidate a
197/// previous detection, plus the current time for the TTL.
198///
199/// [`read_with_env`] and [`write_with_env`] operate against an explicit `CacheEnv` so they can be
200/// unit-tested deterministically on any machine; [`CacheEnv::current`] gathers the real values for
201/// the call sites in `cuda.rs`.
202pub(super) struct CacheEnv {
203    pub(super) boot_id: Option<BootId>,
204    pub(super) driver_fingerprint: Option<DriverFingerprint>,
205    pub(super) device_fingerprint: Option<DeviceFingerprint>,
206    pub(super) now: u64,
207}

209impl CacheEnv {
210    /// Gathers the real cache environment from the current host.
211    pub(super) fn current() -> Self {
212        Self {
213            boot_id: BootId::current(),
214            driver_fingerprint: driver_fingerprint(),
215            device_fingerprint: device_fingerprint(),
216            now: now_unix_secs(),
217        }
218    }
219}

221/// Returns the current time in unix seconds, or `0` if the clock is before the epoch.
222fn now_unix_secs() -> u64 {
223    std::time::SystemTime::now()
224        .duration_since(std::time::UNIX_EPOCH)
225        .map_or(0, |d| d.as_secs())
226}

228/// Absolute libnvidia-ml paths probed by the detector, used as a driver-fingerprint fallback where
229/// `/sys/module/nvidia` does not exist (notably WSL2).
230///
231/// Keep in sync with the absolute entries of `nvml_library_paths()` in `cuda.rs`.
232#[cfg(target_os = "linux")]
233const LIBNVIDIA_ML_ABSOLUTE_PATHS: &[&str] = &[
234    "/usr/lib64/nvidia/libnvidia-ml.so.1", // RHEL/Centos/Fedora
235    "/usr/lib64/nvidia/libnvidia-ml.so",
236    "/usr/lib/x86_64-linux-gnu/libnvidia-ml.so.1", // Ubuntu
237    "/usr/lib/x86_64-linux-gnu/libnvidia-ml.so",
238    "/usr/lib/wsl/lib/libnvidia-ml.so.1", // WSL
239    "/usr/lib/wsl/lib/libnvidia-ml.so",
240];
241
242/// Fingerprints a file by its path, modification time and length, or `None` if it does not exist.
243#[cfg(any(target_os = "linux", target_os = "windows"))]
244fn file_fingerprint(path: &Path) -> Option<DriverFingerprint> {
245    let metadata = std::fs::metadata(path).ok()?;
246    let mtime = metadata
247        .modified()
248        .ok()?
249        .duration_since(std::time::UNIX_EPOCH)
250        .ok()?;
251    Some(DriverFingerprint::File {
252        path: path.to_path_buf(),
253        mtime_secs: mtime.as_secs(),
254        len: metadata.len(),
255    })
256}

258/// Returns a fingerprint of the installed NVIDIA driver, or `None` if it cannot be determined.
259fn driver_fingerprint() -> Option<DriverFingerprint> {
260    #[cfg(target_os = "linux")]
261    {
262        // The version of the loaded kernel module, which changes when the driver is updated and
263        // the module is reloaded.
264        if let Ok(version) = std::fs::read_to_string("/sys/module/nvidia/version") {
265            return Some(DriverFingerprint::Module {
266                version: version.trim().to_owned(),
267            });
268        }
269        // WSL2 (and similar setups) has no `/sys/module/nvidia`, so fall back to fingerprinting the
270        // libnvidia-ml file the detector would load.
271        for path in LIBNVIDIA_ML_ABSOLUTE_PATHS {
272            if let Some(fingerprint) = file_fingerprint(Path::new(path)) {
273                return Some(fingerprint);
274            }
275        }
276        None
277    }
278    #[cfg(target_os = "windows")]
279    {
280        // Driver updates on Windows usually complete without a reboot but replace nvml.dll. The DLL
281        // does not always load from System32, so also consider the NVSMI install location.
282        let mut candidates = Vec::new();
283        if let Some(windir) = std::env::var_os("WINDIR") {
284            candidates.push(Path::new(&windir).join("System32").join("nvml.dll"));
285        }
286        if let Some(program_files) = std::env::var_os("ProgramFiles") {
287            candidates.push(
288                Path::new(&program_files)
289                    .join("NVIDIA Corporation")
290                    .join("NVSMI")
291                    .join("nvml.dll"),
292            );
293        }
294        for path in candidates {
295            if let Some(fingerprint) = file_fingerprint(&path) {
296                return Some(fingerprint);
297            }
298        }
299        None
300    }
301    #[cfg(not(any(target_os = "linux", target_os = "windows")))]
302    {
303        None
304    }
305}

307/// Returns true if a `/dev` entry name is an `nvidiaN` device node (all-digits suffix), which
308/// excludes control nodes like `nvidiactl` and `nvidia-uvm`.
309#[cfg(target_os = "linux")]
310fn is_nvidia_device_node(name: &str) -> bool {
311    name.strip_prefix("nvidia")
312        .is_some_and(|rest| !rest.is_empty() && rest.bytes().all(|b| b.is_ascii_digit()))
313}

315/// Returns a fingerprint of the host-visible GPUs, or `None` if it cannot be determined.
316fn device_fingerprint() -> Option<DeviceFingerprint> {
317    #[cfg(target_os = "linux")]
318    {
319        // PCI bus ids of the GPUs the driver exposes to us.
320        let gpus_dir = std::fs::read_dir("/proc/driver/nvidia/gpus").ok();
321        let has_gpus_dir = gpus_dir.is_some();
322        let mut gpus: Vec<String> = gpus_dir
323            .map(|entries| {
324                entries
325                    .filter_map(Result::ok)
326                    .map(|entry| entry.file_name().to_string_lossy().into_owned())
327                    .collect()
328            })
329            .unwrap_or_default();
330        gpus.sort();
331
332        // The `/dev/nvidiaN` device nodes that are actually present.
333        let mut devices: Vec<String> = std::fs::read_dir("/dev")
334            .map(|entries| {
335                entries
336                    .filter_map(Result::ok)
337                    .map(|entry| entry.file_name().to_string_lossy().into_owned())
338                    .filter(|name| is_nvidia_device_node(name))
339                    .collect()
340            })
341            .unwrap_or_default();
342        devices.sort();
343
344        if !has_gpus_dir && devices.is_empty() {
345            return None;
346        }
347        Some(DeviceFingerprint {
348            gpus,
349            device_nodes: devices,
350        })
351    }
352    #[cfg(target_os = "windows")]
353    {
354        // Enumerate the NVIDIA PCI devices Windows knows about. This only reads Plug and Play metadata, so
355        // it does not initialize the driver or wake a powered-down GPU.
356        let mut gpus = windows_nvidia_device_ids()?;
357        gpus.sort();
358        Some(DeviceFingerprint {
359            gpus,
360            // Device nodes are a Linux concept.
361            device_nodes: Vec::new(),
362        })
363    }
364    #[cfg(not(any(target_os = "linux", target_os = "windows")))]
365    {
366        // No cheap per-container GPU enumeration is available; the TTL is the backstop here.
367        None
368    }
369}

371/// Returns the Plug and Play device ids of all NVIDIA PCI devices, or `None` if they cannot be
372/// enumerated.
373///
374/// Uses the configuration manager rather than the driver so that hot-plugged GPUs (for instance an
375/// external GPU over Thunderbolt) are noticed without paying for driver initialization.
376#[cfg(target_os = "windows")]
377fn windows_nvidia_device_ids() -> Option<Vec<String>> {
378    use windows_sys::Win32::Devices::DeviceAndDriverInstallation::{
379        CM_GETIDLIST_FILTER_ENUMERATOR, CM_GETIDLIST_FILTER_PRESENT, CM_Get_Device_ID_List_SizeW,
380        CM_Get_Device_ID_ListW, CR_SUCCESS,
381    };
382
383    // Only enumerate devices that are currently present on the PCI bus.
384    let filter: Vec<u16> = "PCI".encode_utf16().chain(std::iter::once(0)).collect();
385    let flags = CM_GETIDLIST_FILTER_ENUMERATOR | CM_GETIDLIST_FILTER_PRESENT;
386
387    let mut len: u32 = 0;
388    if unsafe { CM_Get_Device_ID_List_SizeW(&mut len, filter.as_ptr(), flags) } != CR_SUCCESS {
389        return None;
390    }
391
392    let mut buffer = vec![0u16; len as usize];
393    if unsafe { CM_Get_Device_ID_ListW(filter.as_ptr(), buffer.as_mut_ptr(), len, flags) }
394        != CR_SUCCESS
395    {
396        return None;
397    }
398
399    // The buffer is a sequence of null terminated strings, terminated by an empty string.
400    Some(
401        buffer
402            .split(|&c| c == 0)
403            .filter(|segment| !segment.is_empty())
404            .map(String::from_utf16_lossy)
405            .filter(|id| is_nvidia_pci_device_id(id))
406            .collect(),
407    )
408}

410/// Returns true if `id` is the Plug and Play device id of an NVIDIA PCI device.
411///
412/// NVIDIA's PCI vendor id is `10DE`. Windows reports device ids uppercased, but match
413/// case-insensitively to be safe.
414#[cfg(any(target_os = "windows", test))]
415pub(super) fn is_nvidia_pci_device_id(id: &str) -> bool {
416    id.to_ascii_uppercase().contains("VEN_10DE")
417}

419/// Reads a cached detection result, validating it against the given host environment.
420pub(super) fn read_with_env(env: &CacheEnv, cache_dir: &Path) -> Option<DetectedCudaInfo> {
421    let path = cache_dir.join(CACHE_FILE_NAME);
422    let Ok(content) = std::fs::read_to_string(&path) else {
423        tracing::trace!("no CUDA info cache found at {}", path.display());
424        return None;
425    };
426    let Ok(cached) = serde_json::from_str::<CacheFile>(&content) else {
427        tracing::debug!("ignoring invalid CUDA info cache at {}", path.display());
428        return None;
429    };
430    let Some(current_boot_id) = env.boot_id.as_ref() else {
431        tracing::debug!(
432            "ignoring CUDA info cache because the current boot id could not be determined"
433        );
434        return None;
435    };
436    if !cached.boot_id.matches(current_boot_id) {
437        tracing::info!(
438            cache_path = %path.display(),
439            cached_boot_id = ?cached.boot_id,
440            current_boot_id = ?current_boot_id,
441            "invalidating CUDA info cache from a previous boot session"
442        );
443        return None;
444    }
445    let Some(current_driver_fingerprint) = env.driver_fingerprint.as_ref() else {
446        tracing::debug!(
447            "ignoring CUDA info cache because the current driver fingerprint could not be determined"
448        );
449        return None;
450    };
451    if &cached.driver_fingerprint != current_driver_fingerprint {
452        tracing::info!(
453            cache_path = %path.display(),
454            cached_driver_fingerprint = ?cached.driver_fingerprint,
455            current_driver_fingerprint = ?current_driver_fingerprint,
456            "invalidating CUDA info cache because the driver changed"
457        );
458        return None;
459    }
460    if cached.device_fingerprint != env.device_fingerprint {
461        tracing::info!(
462            cache_path = %path.display(),
463            cached_device_fingerprint = ?cached.device_fingerprint,
464            current_device_fingerprint = ?env.device_fingerprint,
465            "invalidating CUDA info cache because the visible GPUs changed"
466        );
467        return None;
468    }
469    // Reject entries written in the future: the clock stepped backwards and the TTL below can no
470    // longer be trusted.
471    if cached.written_at.saturating_sub(env.now) > MAX_CLOCK_SKEW_SECS {
472        tracing::debug!(
473            cache_path = %path.display(),
474            written_at = cached.written_at,
475            now = env.now,
476            "ignoring CUDA info cache written in the future"
477        );
478        return None;
479    }
480    // Full detections are trusted for a day; transient arch failures self-heal within minutes.
481    let ttl = if cached.arch.is_some() {
482        FULL_TTL_SECS
483    } else {
484        ARCH_MISSING_TTL_SECS
485    };
486    if env.now.saturating_sub(cached.written_at) > ttl {
487        tracing::info!(
488            cache_path = %path.display(),
489            written_at = cached.written_at,
490            now = env.now,
491            ttl,
492            "invalidating expired CUDA info cache"
493        );
494        return None;
495    }
496    let version = match Version::from_str(&cached.version) {
497        Ok(version) => version,
498        Err(err) => {
499            tracing::debug!(
500                version = cached.version,
501                error = %err,
502                "ignoring CUDA info cache with invalid version"
503            );
504            return None;
505        }
506    };
507    tracing::trace!("using CUDA info cached at {}", path.display());
508    Some(DetectedCudaInfo {
509        info: CudaInfo {
510            version: Some(version),
511            arch_info: cached
512                .arch
513                .map(|(major, minor)| CudaArchInfo { major, minor }),
514        },
515        sources: CudaInfoSources {
516            version: cached.version_source,
517            arch: cached.arch_source,
518        },
519    })
520}

522/// Writes a detection result to the cache, keyed on the given host environment.
523pub(super) fn write_with_env(
524    env: &CacheEnv,
525    cache_dir: &Path,
526    info: &CudaInfo,
527    sources: CudaInfoSources,
528) {
529    // Only cache when a driver was found: detection without a driver is fast anyway, and not
530    // caching the negative result means a freshly installed driver is picked up immediately.
531    let Some(version) = &info.version else {
532        tracing::trace!("not caching CUDA info because no CUDA driver version was detected");
533        return;
534    };
535    let Some(boot_id) = env.boot_id.clone() else {
536        tracing::debug!(
537            "not caching CUDA info because the current boot id could not be determined"
538        );
539        return;
540    };
541    // The driver fingerprint is required: without it the cache is keyed on the boot session alone
542    // (a host driver update would serve stale data on e.g. WSL2), so we simply do not cache.
543    let Some(driver_fingerprint) = env.driver_fingerprint.clone() else {
544        tracing::debug!(
545            "not caching CUDA info because the current driver fingerprint could not be determined"
546        );
547        return;
548    };
549    if let Err(err) = std::fs::create_dir_all(cache_dir) {
550        tracing::debug!(
551            cache_dir = %cache_dir.display(),
552            error = %err,
553            "failed to create CUDA info cache directory"
554        );
555        return;
556    }
557    let cached = CacheFile {
558        boot_id,
559        driver_fingerprint,
560        device_fingerprint: env.device_fingerprint.clone(),
561        written_at: env.now,
562        version: version.to_string(),
563        arch: info.arch_info.as_ref().map(|arch| (arch.major, arch.minor)),
564        version_source: sources.version,
565        arch_source: sources.arch,
566    };
567    let Ok(content) = serde_json::to_string(&cached) else {
568        tracing::debug!("failed to serialize CUDA info cache entry");
569        return;
570    };
571    // Write to a temporary file in the cache directory and persist it into place so concurrent
572    // readers never see a partial cache file.
573    let path = cache_dir.join(CACHE_FILE_NAME);
574    let mut tmp = match tempfile::NamedTempFile::new_in(cache_dir) {
575        Ok(tmp) => tmp,
576        Err(err) => {
577            tracing::debug!(
578                cache_dir = %cache_dir.display(),
579                error = %err,
580                "failed to create temporary CUDA info cache file"
581            );
582            return;
583        }
584    };
585    if let Err(err) = tmp.write_all(content.as_bytes()) {
586        tracing::debug!(
587            cache_path = %path.display(),
588            error = %err,
589            "failed to write temporary CUDA info cache file"
590        );
591        return;
592    }
593    match tmp.persist(&path) {
594        Ok(_) => tracing::trace!("cached CUDA info at {}", path.display()),
595        Err(err) => {
596            tracing::debug!(
597                cache_path = %path.display(),
