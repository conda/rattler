1//! On-disk cache for detected CUDA information, valid for the current boot session.
2//!
3//! Detecting CUDA can be slow (initializing NVML attaches to every GPU), so the result is cached
4//! on disk between processes. The cache is keyed on everything that can invalidate a previous
5//! detection without the code noticing:
6//!
7//! * the current **boot session**, because a reboot re-enumerates hardware and drivers;
8//! * a **driver fingerprint**, because drivers can be updated (and NVML reloaded) without a reboot;
9//! * a **device fingerprint** of the host-visible GPUs, because two containers sharing a cache
10//!   volume see different GPU subsets and topology can change within a session (eGPU hot-plug,
11//!   PCI hot-plug into VMs, suspend/resume);
12//! * a **TTL** as a staleness backstop for anything the fingerprints cannot catch (and to let
13//!   transient arch-detection failures self-heal quickly).
14//!
15//! Reads and writes are best-effort: any failure simply results in a fresh detection. Delete the
16//! file or use the `CONDA_OVERRIDE_CUDA*` variables to bypass it.
17

30const CACHE_FILE_NAME: &str = "cuda-info-v1.json";
31
32/// Full detections (with compute capability) are trusted for a day; the fingerprints catch most
33/// changes sooner, so this is only a staleness backstop.
34const FULL_TTL_SECS: u64 = 24 * 60 * 60;
35/// Entries without compute capability represent a transient arch-detection failure, so they expire
36/// quickly and self-heal instead of lingering for the whole boot session.
37const ARCH_MISSING_TTL_SECS: u64 = 10 * 60;
38/// Reject entries whose write time is further than this into the future, which means the clock
39/// stepped backwards and the recorded `written_at` can no longer be trusted for the TTL.
40const MAX_CLOCK_SKEW_SECS: u64 = 5 * 60;

42/// Identifies the installed NVIDIA driver.
43///
44/// Drivers can be updated without a reboot, so the boot session alone is not enough to key the
45/// cache on. All variants are compiled on every platform so they can be unit-tested anywhere.
46#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
47#[serde(rename_all = "snake_case")]
48pub(super) enum DriverFingerprint {
49    /// The version of the loaded `nvidia` kernel module (Linux), which changes when the driver is
50    /// updated and the module is reloaded.
51    Module { version: String },
52    /// The identity of the NVML library file on disk (Windows, WSL2), which driver updates
53    /// replace.
54    File {
55        path: PathBuf,
56        mtime_secs: u64,
57        len: u64,
58    },
59}
60
61/// Identifies the set of host-visible GPUs.
62///
63/// This distinguishes containers that share a cache volume but see different GPU subsets, and it
64/// changes when GPUs are hot-plugged or unplugged within a boot session.
65#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
66pub(super) struct DeviceFingerprint {
67    /// Sorted identifiers of the GPUs the host exposes: PCI bus ids from
68    /// `/proc/driver/nvidia/gpus` on Linux, Plug and Play device ids on Windows.
69    pub(super) gpus: Vec<String>,
70    /// Sorted `/dev/nvidiaN` device nodes that are present. Always empty on Windows, which has no
71    /// device nodes.
72    pub(super) device_nodes: Vec<String>,
73}

75#[derive(Serialize, Deserialize)]
76struct CacheFile {
77    /// The boot session during which detection ran.
78    boot_id: BootId,
79    /// Fingerprint of the installed driver when detection ran. Required: entries without a driver
80    /// fingerprint cannot be trusted, so they are neither written nor read.
81    driver_fingerprint: DriverFingerprint,
82    /// Fingerprint of the host-visible GPUs when detection ran, or `None` if it could not be
83    /// determined on this platform.
84    device_fingerprint: Option<DeviceFingerprint>,
85    /// Unix seconds at which the entry was written, used for the TTL.
86    written_at: u64,
87    version: String,
88    arch: Option<(u32, u32)>,
89    #[serde(default)]
90    version_source: Option<CudaDetectionMethod>,
91    #[serde(default)]
92    arch_source: Option<CudaDetectionMethod>,
93}

95/// The environment against which a cache entry is validated: everything that can invalidate a
96/// previous detection, plus the current time for the TTL.
97///
98/// [`read_with_env`] and [`write_with_env`] operate against an explicit `CacheEnv` so they can be
99/// unit-tested deterministically on any machine; [`CacheEnv::current`] gathers the real values for
100/// the call sites in `cuda.rs`.
101pub(super) struct CacheEnv {
102    pub(super) boot_id: Option<BootId>,
103    pub(super) driver_fingerprint: Option<DriverFingerprint>,
104    pub(super) device_fingerprint: Option<DeviceFingerprint>,
105    pub(super) now: u64,
106}
107
108impl CacheEnv {
109    /// Gathers the real cache environment from the current host.
110    pub(super) fn current() -> Self {
111        Self {
112            boot_id: BootId::current(),
113            driver_fingerprint: driver_fingerprint(),
114            device_fingerprint: device_fingerprint(),
115            now: now_unix_secs(),
116        }
117    }
118}

120/// Returns the current time in unix seconds, or `0` if the clock is before the epoch.
121fn now_unix_secs() -> u64 {
122    std::time::SystemTime::now()
123        .duration_since(std::time::UNIX_EPOCH)
124        .map_or(0, |d| d.as_secs())
125}

127/// Absolute libnvidia-ml paths probed by the detector, used as a driver-fingerprint fallback where
128/// `/sys/module/nvidia` does not exist (notably WSL2).
129///
130/// Keep in sync with the absolute entries of `nvml_library_paths()` in `cuda.rs`.
131#[cfg(target_os = "linux")]
132const LIBNVIDIA_ML_ABSOLUTE_PATHS: &[&str] = &[
133    "/usr/lib64/nvidia/libnvidia-ml.so.1", // RHEL/Centos/Fedora
134    "/usr/lib64/nvidia/libnvidia-ml.so",
135    "/usr/lib/x86_64-linux-gnu/libnvidia-ml.so.1", // Ubuntu
136    "/usr/lib/x86_64-linux-gnu/libnvidia-ml.so",
137    "/usr/lib/wsl/lib/libnvidia-ml.so.1", // WSL
138    "/usr/lib/wsl/lib/libnvidia-ml.so",
139];
140
141/// Fingerprints a file by its path, modification time and length, or `None` if it does not exist.
142#[cfg(any(target_os = "linux", target_os = "windows"))]
143fn file_fingerprint(path: &Path) -> Option<DriverFingerprint> {
144    let metadata = std::fs::metadata(path).ok()?;
145    let mtime = metadata
146        .modified()
147        .ok()?
148        .duration_since(std::time::UNIX_EPOCH)
149        .ok()?;
150    Some(DriverFingerprint::File {
151        path: path.to_path_buf(),
152        mtime_secs: mtime.as_secs(),
153        len: metadata.len(),
154    })
155}

157/// Returns a fingerprint of the installed NVIDIA driver, or `None` if it cannot be determined.
158fn driver_fingerprint() -> Option<DriverFingerprint> {
159    #[cfg(target_os = "linux")]
160    {
161        // The version of the loaded kernel module, which changes when the driver is updated and
162        // the module is reloaded.
163        if let Ok(version) = std::fs::read_to_string("/sys/module/nvidia/version") {
164            return Some(DriverFingerprint::Module {
165                version: version.trim().to_owned(),
166            });
167        }
168        // WSL2 (and similar setups) has no `/sys/module/nvidia`, so fall back to fingerprinting the
169        // libnvidia-ml file the detector would load.
170        for path in LIBNVIDIA_ML_ABSOLUTE_PATHS {
171            if let Some(fingerprint) = file_fingerprint(Path::new(path)) {
172                return Some(fingerprint);
173            }
174        }
175        None
176    }
177    #[cfg(target_os = "windows")]
178    {
179        // Driver updates on Windows usually complete without a reboot but replace nvml.dll. The DLL
180        // does not always load from System32, so also consider the NVSMI install location.
181        let mut candidates = Vec::new();
182        if let Some(windir) = std::env::var_os("WINDIR") {
183            candidates.push(Path::new(&windir).join("System32").join("nvml.dll"));
184        }
185        if let Some(program_files) = std::env::var_os("ProgramFiles") {
186            candidates.push(
187                Path::new(&program_files)
188                    .join("NVIDIA Corporation")
189                    .join("NVSMI")
190                    .join("nvml.dll"),
191            );
192        }
193        for path in candidates {
194            if let Some(fingerprint) = file_fingerprint(&path) {
195                return Some(fingerprint);
196            }
197        }
198        None
199    }
200    #[cfg(not(any(target_os = "linux", target_os = "windows")))]
201    {
202        None
203    }
204}

206/// Returns true if a `/dev` entry name is an `nvidiaN` device node (all-digits suffix), which
207/// excludes control nodes like `nvidiactl` and `nvidia-uvm`.
208#[cfg(target_os = "linux")]
209fn is_nvidia_device_node(name: &str) -> bool {
210    name.strip_prefix("nvidia")
211        .is_some_and(|rest| !rest.is_empty() && rest.bytes().all(|b| b.is_ascii_digit()))
212}

214/// Returns a fingerprint of the host-visible GPUs, or `None` if it cannot be determined.
215fn device_fingerprint() -> Option<DeviceFingerprint> {
216    #[cfg(target_os = "linux")]
217    {
218        // PCI bus ids of the GPUs the driver exposes to us.
219        let gpus_dir = std::fs::read_dir("/proc/driver/nvidia/gpus").ok();
220        let has_gpus_dir = gpus_dir.is_some();
221        let mut gpus: Vec<String> = gpus_dir
222            .map(|entries| {
223                entries
224                    .filter_map(Result::ok)
225                    .map(|entry| entry.file_name().to_string_lossy().into_owned())
226                    .collect()
227            })
228            .unwrap_or_default();
229        gpus.sort();
230
231        // The `/dev/nvidiaN` device nodes that are actually present.
232        let mut devices: Vec<String> = std::fs::read_dir("/dev")
233            .map(|entries| {
234                entries
235                    .filter_map(Result::ok)
236                    .map(|entry| entry.file_name().to_string_lossy().into_owned())
237                    .filter(|name| is_nvidia_device_node(name))
238                    .collect()
239            })
240            .unwrap_or_default();
241        devices.sort();
242
243        if !has_gpus_dir && devices.is_empty() {
244            return None;
245        }
246        Some(DeviceFingerprint {
247            gpus,
248            device_nodes: devices,
249        })
250    }
251    #[cfg(target_os = "windows")]
252    {
253        // Enumerate the NVIDIA PCI devices Windows knows about. This only reads Plug and Play metadata, so
254        // it does not initialize the driver or wake a powered-down GPU.
255        let mut gpus = windows_nvidia_device_ids()?;
256        gpus.sort();
257        Some(DeviceFingerprint {
258            gpus,
259            // Device nodes are a Linux concept.
260            device_nodes: Vec::new(),
261        })
262    }
263    #[cfg(not(any(target_os = "linux", target_os = "windows")))]
264    {
265        // No cheap per-container GPU enumeration is available; the TTL is the backstop here.
266        None
267    }
268}

270/// Returns the Plug and Play device ids of all NVIDIA PCI devices, or `None` if they cannot be
271/// enumerated.
272///
273/// Uses the configuration manager rather than the driver so that hot-plugged GPUs (for instance an
274/// external GPU over Thunderbolt) are noticed without paying for driver initialization.
275#[cfg(target_os = "windows")]
276fn windows_nvidia_device_ids() -> Option<Vec<String>> {
277    use windows_sys::Win32::Devices::DeviceAndDriverInstallation::{
278        CM_GETIDLIST_FILTER_ENUMERATOR, CM_GETIDLIST_FILTER_PRESENT,
279    };
280
281    // Only enumerate devices that are currently present on the PCI bus.
282    let ids = win::device_instance_ids(
283        "PCI",
284        CM_GETIDLIST_FILTER_ENUMERATOR | CM_GETIDLIST_FILTER_PRESENT,
285    )?;
286    Some(ids.filter(|id| is_nvidia_pci_device_id(id)).collect())
287}

289/// Returns true if `id` is the Plug and Play device id of an NVIDIA PCI device.
290///
291/// NVIDIA's PCI vendor id is `10DE`. Windows reports device ids uppercased, but match
292/// case-insensitively to be safe.
293#[cfg(any(target_os = "windows", test))]
294pub(super) fn is_nvidia_pci_device_id(id: &str) -> bool {
295    id.to_ascii_uppercase().contains("VEN_10DE")
296}

298/// Reads a cached detection result, validating it against the given host environment.
299pub(super) fn read_with_env(env: &CacheEnv, cache_dir: &Path) -> Option<DetectedCudaInfo> {
300    let path = cache_dir.join(CACHE_FILE_NAME);
301    let Ok(content) = std::fs::read_to_string(&path) else {
302        tracing::trace!("no CUDA info cache found at {}", path.display());
303        return None;
304    };
305    let Ok(cached) = serde_json::from_str::<CacheFile>(&content) else {
306        tracing::debug!("ignoring invalid CUDA info cache at {}", path.display());
307        return None;
308    };
309    let Some(current_boot_id) = env.boot_id.as_ref() else {
310        tracing::debug!(
311            "ignoring CUDA info cache because the current boot id could not be determined"
312        );
313        return None;
314    };
315    if !cached.boot_id.matches(current_boot_id) {
316        tracing::info!(
317            cache_path = %path.display(),
318            cached_boot_id = ?cached.boot_id,
319            current_boot_id = ?current_boot_id,
320            "invalidating CUDA info cache from a previous boot session"
321        );
322        return None;
323    }
324    let Some(current_driver_fingerprint) = env.driver_fingerprint.as_ref() else {
325        tracing::debug!(
326            "ignoring CUDA info cache because the current driver fingerprint could not be determined"
327        );
328        return None;
329    };
330    if &cached.driver_fingerprint != current_driver_fingerprint {
331        tracing::info!(
332            cache_path = %path.display(),
333            cached_driver_fingerprint = ?cached.driver_fingerprint,
334            current_driver_fingerprint = ?current_driver_fingerprint,
335            "invalidating CUDA info cache because the driver changed"
336        );
337        return None;
338    }
339    if cached.device_fingerprint != env.device_fingerprint {
340        tracing::info!(
341            cache_path = %path.display(),
342            cached_device_fingerprint = ?cached.device_fingerprint,
343            current_device_fingerprint = ?env.device_fingerprint,
344            "invalidating CUDA info cache because the visible GPUs changed"
345        );
346        return None;
347    }
348    // Reject entries written in the future: the clock stepped backwards and the TTL below can no
349    // longer be trusted.
350    if cached.written_at.saturating_sub(env.now) > MAX_CLOCK_SKEW_SECS {
351        tracing::debug!(
352            cache_path = %path.display(),
353            written_at = cached.written_at,
354            now = env.now,
355            "ignoring CUDA info cache written in the future"
356        );
357        return None;
358    }
359    // Full detections are trusted for a day; transient arch failures self-heal within minutes.
360    let ttl = if cached.arch.is_some() {
361        FULL_TTL_SECS
362    } else {
363        ARCH_MISSING_TTL_SECS
364    };
365    if env.now.saturating_sub(cached.written_at) > ttl {
366        tracing::info!(
367            cache_path = %path.display(),
368            written_at = cached.written_at,
369            now = env.now,
370            ttl,
371            "invalidating expired CUDA info cache"
372        );
373        return None;
374    }
375    let version = match Version::from_str(&cached.version) {
376        Ok(version) => version,
377        Err(err) => {
378            tracing::debug!(
379                version = cached.version,
380                error = %err,
381                "ignoring CUDA info cache with invalid version"
382            );
383            return None;
384        }
385    };
386    tracing::trace!("using CUDA info cached at {}", path.display());
387    Some(DetectedCudaInfo {
388        info: CudaInfo {
389            version: Some(version),
390            arch_info: cached
391                .arch
392                .map(|(major, minor)| CudaArchInfo { major, minor }),
393        },
394        sources: CudaInfoSources {
395            version: cached.version_source,
396            arch: cached.arch_source,
397        },
398    })
399}

401/// Writes a detection result to the cache, keyed on the given host environment.
402pub(super) fn write_with_env(
403    env: &CacheEnv,
404    cache_dir: &Path,
405    info: &CudaInfo,
406    sources: CudaInfoSources,
407) {
408    // Only cache when a driver was found: detection without a driver is fast anyway, and not
409    // caching the negative result means a freshly installed driver is picked up immediately.
410    let Some(version) = &info.version else {
411        tracing::trace!("not caching CUDA info because no CUDA driver version was detected");
412        return;
413    };
414    let Some(boot_id) = env.boot_id.clone() else {
415        tracing::debug!(
416            "not caching CUDA info because the current boot id could not be determined"
417        );
418        return;
419    };
420    // The driver fingerprint is required: without it the cache is keyed on the boot session alone
421    // (a host driver update would serve stale data on e.g. WSL2), so we simply do not cache.
422    let Some(driver_fingerprint) = env.driver_fingerprint.clone() else {
423        tracing::debug!(
424            "not caching CUDA info because the current driver fingerprint could not be determined"
425        );
426        return;
427    };
428    if let Err(err) = std::fs::create_dir_all(cache_dir) {
429        tracing::debug!(
430            cache_dir = %cache_dir.display(),
431            error = %err,
432            "failed to create CUDA info cache directory"
433        );
434        return;
435    }
436    let cached = CacheFile {
437        boot_id,
438        driver_fingerprint,
439        device_fingerprint: env.device_fingerprint.clone(),
440        written_at: env.now,
441        version: version.to_string(),
442        arch: info.arch_info.as_ref().map(|arch| (arch.major, arch.minor)),
443        version_source: sources.version,
444        arch_source: sources.arch,
445    };
446    let Ok(content) = serde_json::to_string(&cached) else {
447        tracing::debug!("failed to serialize CUDA info cache entry");
448        return;
449    };
450    // Write to a temporary file in the cache directory and persist it into place so concurrent
451    // readers never see a partial cache file.
452    let path = cache_dir.join(CACHE_FILE_NAME);
453    let mut tmp = match tempfile::NamedTempFile::new_in(cache_dir) {
454        Ok(tmp) => tmp,
455        Err(err) => {
456            tracing::debug!(
457                cache_dir = %cache_dir.display(),
458                error = %err,
459                "failed to create temporary CUDA info cache file"
460            );
461            return;
462        }
463    };
464    if let Err(err) = tmp.write_all(content.as_bytes()) {
465        tracing::debug!(
466            cache_path = %path.display(),
467            error = %err,
468            "failed to write temporary CUDA info cache file"
469        );
470        return;
471    }
472    match tmp.persist(&path) {
473        Ok(_) => tracing::trace!("cached CUDA info at {}", path.display()),
474        Err(err) => {
475            tracing::debug!(
476                cache_path = %path.display(),
477                error = %err.error,
478                "failed to persist CUDA info cache"
479            );
480        }
481    }
