1//! Caching detector results between solves.
2//!
3//! An entry is keyed on the registration identity, the registered names and
4//! the environment digest, and it always expires: after the lifetime the
5//! report asked for, clamped to the protocol's maximum, at the next reboot for
6//! `"REBOOT"` results, and early when a watched path or variable changes.
7//! Entries are JSON files written atomically, so concurrent readers never see
8//! a partial file.
9

29/// What identifies a cached result.
30#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
31pub struct CacheKey {
32    /// The registration's origin.
33    pub origin: ChannelUrl,
34    /// The detector.
35    pub detector: PackageName,
36    /// The registered virtual package names, sorted.
37    pub names: Vec<PackageName>,
38    /// The environment digest, as lowercase hexadecimal.
39    pub digest: String,
40}
41
42impl CacheKey {
43    /// Builds the key for a detector environment.
44    pub fn new(
45        origin: ChannelUrl,
46        detector: PackageName,
47        names: impl IntoIterator<Item = PackageName>,
48        digest: &Sha256Hash,
49    ) -> Self {
50        let mut names: Vec<PackageName> = names.into_iter().collect();
51        names.sort();
52        Self {
53            origin,
54            detector,
55            names,
56            digest: hex::encode(digest),
57        }
58    }
59
60    /// The normalized detector name followed by the full hash of the key.
61    pub fn file_name(&self) -> String {
62        let mut material = String::new();
63        material.push_str(self.origin.as_str());
64        material.push('\0');
65        material.push_str(self.detector.as_normalized());
66        material.push('\0');
67        for name in &self.names {
68            material.push_str(name.as_normalized());
69            material.push(',');
70        }
71        material.push('\0');
72        material.push_str(&self.digest);
73        format!(
74            "{}-{}.json",
75            self.detector.as_normalized(),
76            hex::encode(compute_bytes_digest::<Sha256>(material))
77        )
78    }
79}

81/// The observed state of a watched path.
82#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
83pub enum PathState {
84    /// The path does not exist or cannot be accessed.
85    Absent,
86    /// The path exists with this modification time.
87    Present {
88        /// Seconds since the Unix epoch.
89        modified_secs: u64,
90        /// Nanoseconds within that second.
91        modified_nanos: u32,
92    },
93}
94
95impl PathState {
96    async fn observe(path: &Path) -> Self {
97        let Ok(metadata) = tokio::fs::metadata(path).await else {
98            return Self::Absent;
99        };
100        let Ok(modified) = metadata.modified() else {
101            return Self::Absent;
102        };
103        let since_epoch = modified.duration_since(UNIX_EPOCH).unwrap_or_default();
104        Self::Present {
105            modified_secs: since_epoch.as_secs(),
106            modified_nanos: since_epoch.subsec_nanos(),
107        }
108    }
109}

111/// A watched path and its state when the entry was written.
112#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
113pub struct WatchedPath {
114    /// The absolute path.
115    pub path: PathBuf,
116    /// Its state at write time.
117    pub state: PathState,
118}
119
120/// A watched environment variable and a fingerprint of its value when the
121/// entry was written. Only the fingerprint is stored: a detector may watch
122/// variables that hold credentials, and the cache file must not leak them.
123#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
124pub struct WatchedVariable {
125    /// The variable's name.
126    pub name: String,
127    /// The SHA-256 of its value at write time, as lowercase hexadecimal, or
128    /// `None` if unset.
129    pub value_digest: Option<String>,
130}
131
132impl WatchedVariable {
133    fn observe(name: &str, env: &EnvironmentSnapshot) -> Self {
134        let value = env.get(name);
135        Self {
136            name: name.to_string(),
137            value_digest: value
138                .map(|value| hex::encode(compute_bytes_digest::<Sha256>(value.as_encoded_bytes()))),
139        }
140    }
141}

143/// A cached result as stored on disk.
144#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
145pub struct CachedResult {
146    /// The key the entry was written for.
147    pub key: CacheKey,
148    /// When the entry was written, in seconds since the Unix epoch.
149    pub written_at: u64,
150    /// When the entry expires, in seconds since the Unix epoch.
151    pub expires_at: u64,
152    /// The boot session the entry is valid in, for `"REBOOT"` results.
153    #[serde(default, skip_serializing_if = "Option::is_none")]
154    pub boot_id: Option<BootId>,
155    /// The watched paths and their states at write time.
156    #[serde(default, skip_serializing_if = "Vec::is_empty")]
157    pub watch_paths: Vec<WatchedPath>,
158    /// The watched variables and their values at write time.
159    #[serde(default, skip_serializing_if = "Vec::is_empty")]
160    pub watch_env: Vec<WatchedVariable>,
161    /// The detector's results.
162    pub virtual_packages: IndexMap<PackageName, Option<DetectedVersion>>,
163    /// Nonempty standard error from the successful invocation.
164    #[serde(default, skip_serializing_if = "Option::is_none")]
165    pub stderr: Option<String>,
166}

168#[derive(Serialize)]
169struct BorrowedCachedResult<'a> {
170    key: &'a CacheKey,
171    written_at: u64,
172    expires_at: u64,
173    #[serde(skip_serializing_if = "Option::is_none")]
174    boot_id: Option<BootId>,
175    #[serde(skip_serializing_if = "Vec::is_empty")]
176    watch_paths: Vec<WatchedPath>,
177    #[serde(skip_serializing_if = "Vec::is_empty")]
178    watch_env: Vec<WatchedVariable>,
179    virtual_packages: &'a IndexMap<PackageName, Option<DetectedVersion>>,
180    #[serde(skip_serializing_if = "Option::is_none")]
181    stderr: Option<&'a str>,
182}

184/// The time and boot session against which entries are validated. Passed
185/// explicitly so the expiry rules can be tested deterministically.
186#[derive(Clone, Debug)]
187pub struct CacheClock {
188    /// The current time in seconds since the Unix epoch.
189    pub now: u64,
190    /// The current boot session, if it can be observed on this host.
191    pub boot_id: Option<BootId>,
192}
193
194impl CacheClock {
195    /// The real clock and boot session.
196    pub fn current() -> Self {
197        Self {
198            now: SystemTime::now()
199                .duration_since(UNIX_EPOCH)
200                .map_or(0, |elapsed| elapsed.as_secs()),
201            boot_id: BootId::current(),
202        }
203    }
204}

206/// Why an entry could not be written.
207#[derive(Debug, Error)]
208pub enum CacheError {
209    /// The entry could not be serialized.
210    #[error("failed to serialize the cached result")]
211    Serialize(#[from] serde_json::Error),
212
213    /// The entry could not be written.
214    #[error("failed to write the cached result to {}", path.display())]
215    Io {
216        /// The entry's path.
217        path: PathBuf,
218        /// The underlying error.
219        #[source]
220        source: std::io::Error,
221    },
222}

224/// The on-disk cache of detector results.
225#[derive(Clone, Debug)]
226pub struct ResultCache {
227    root: PathBuf,
228}
229
230impl ResultCache {
231    /// A cache whose entries live directly in `root`.
232    pub fn new(root: impl Into<PathBuf>) -> Self {
233        Self { root: root.into() }
234    }
235
236    /// The directory the entries live in.
237    pub fn root(&self) -> &Path {
238        &self.root
239    }
240
241    fn path_for(&self, key: &CacheKey) -> PathBuf {
242        self.root.join(key.file_name())
243    }
244
245    /// Returns the valid entry for `key`, if there is one.
246    ///
247    /// An unreadable or malformed entry counts as absent.
248    pub async fn read(
249        &self,
250        key: &CacheKey,
251        clock: &CacheClock,
252        environment: &EnvironmentSnapshot,
253    ) -> Option<CachedResult> {
254        let path = self.path_for(key);
255        let bytes = fs_err::tokio::read(&path).await.ok()?;
256        let entry: CachedResult = serde_json::from_slice(&bytes).ok()?;
257        if &entry.key != key {
258            tracing::debug!(path = %path.display(), "ignoring cached result for a different key");
259            return None;
260        }
261        match entry.is_valid(clock, environment).await {
262            Ok(()) => Some(entry),
263            Err(reason) => {
264                tracing::debug!(path = %path.display(), reason, "cached result expired");
265                None
266            }
267        }
268    }
269
270    /// Stores `report`'s results and diagnostics for `key`, honoring its cache hints.
271    ///
272    /// Writes nothing when the report asked not to be reused.
273    pub async fn write(
274        &self,
275        key: &CacheKey,
276        report: &DetectorReport,
277        stderr: Option<&str>,
278        clock: &CacheClock,
279        environment: &EnvironmentSnapshot,
280    ) -> Result<(), CacheError> {
281        let Some(lifetime) = lifetime(report.cache.ttl, clock.boot_id.is_some()) else {
282            return Ok(());
283        };
284        let boot_id = match report.cache.ttl {
285            Some(CacheLifetime::Reboot) => clock.boot_id.clone(),
286            _ => None,
287        };
288        let mut watch_paths = Vec::with_capacity(report.cache.watch_paths.len());
289        for path in &report.cache.watch_paths {
290            watch_paths.push(WatchedPath {
291                path: path.clone(),
292                state: PathState::observe(path).await,
293            });
294        }
295        let entry = BorrowedCachedResult {
296            key,
297            written_at: clock.now,
298            expires_at: clock.now.saturating_add(lifetime.as_secs()),
299            boot_id,
300            watch_paths,
301            watch_env: report
302                .cache
303                .watch_env
304                .iter()
305                .map(|name| WatchedVariable::observe(name, environment))
306                .collect(),
307            virtual_packages: &report.virtual_packages,
308            stderr: stderr.filter(|stderr| !stderr.is_empty()),
309        };
310
311        let path = self.path_for(key);
312        let io_error = |source| CacheError::Io {
313            path: path.clone(),
314            source,
315        };
316        fs_err::tokio::create_dir_all(&self.root)
317            .await
318            .map_err(io_error)?;
319        let bytes = serde_json::to_vec_pretty(&entry)?;
320        // Written next to the entry and renamed into place, so a reader never
321        // sees a partial file.
322        let (file, temporary) = tempfile::NamedTempFile::new_in(&self.root)
323            .map_err(io_error)?
324            .into_parts();
325        let mut file = tokio::fs::File::from_std(file);
326        file.write_all(&bytes).await.map_err(io_error)?;
327        file.flush().await.map_err(io_error)?;
328        drop(file);
329        fs_err::tokio::rename(&temporary, &path)
330            .await
331            .map_err(io_error)?;
332        Ok(())
333    }
334
335    /// Removes every entry.
336    pub async fn clear(&self) -> std::io::Result<()> {
337        match fs_err::tokio::remove_dir_all(&self.root).await {
338            Ok(()) => Ok(()),
339            Err(err) if err.kind() == std::io::ErrorKind::NotFound => Ok(()),
340            Err(err) => Err(err),
341        }
342    }
343}

345impl CachedResult {
346    /// Checks the entry against `clock`, returning why it is no longer valid.
347    pub async fn is_valid(
348        &self,
349        clock: &CacheClock,
350        environment: &EnvironmentSnapshot,
351    ) -> Result<(), &'static str> {
352        if clock.now >= self.expires_at {
353            return Err("lifetime elapsed");
354        }
355        if let Some(boot_id) = &self.boot_id {
356            match &clock.boot_id {
357                Some(current) if boot_id.matches(current) => {}
358                _ => return Err("boot session changed"),
359            }
360        }
361        for watched in &self.watch_paths {
362            if PathState::observe(&watched.path).await != watched.state {
363                return Err("watched path changed");
364            }
365        }
366        for watched in &self.watch_env {
367            if WatchedVariable::observe(&watched.name, environment) != *watched {
368                return Err("watched variable changed");
369            }
370        }
371        Ok(())
372    }
373}

375/// The lifetime an entry gets for the requested `ttl`, or `None` when the
376/// result must not be reused.
377fn lifetime(ttl: Option<CacheLifetime>, boot_observable: bool) -> Option<Duration> {
378    match ttl {
379        None => Some(DEFAULT_CACHE_LIFETIME),
380        Some(CacheLifetime::Seconds(0)) => None,
381        Some(CacheLifetime::Seconds(seconds)) => {
382            Some(Duration::from_secs(seconds).min(MAX_CACHE_LIFETIME))
383        }
384        Some(CacheLifetime::Reboot) if boot_observable => Some(MAX_CACHE_LIFETIME),
385        Some(CacheLifetime::Reboot) => Some(REBOOT_FALLBACK_LIFETIME),
386    }
387}

531    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
532    async fn concurrent_publications_leave_one_complete_report() {
533        const WRITERS: usize = 64;
534        let dir = tempfile::tempdir().unwrap();
535        let cache = ResultCache::new(dir.path());
536        let key = key();
537        let barrier = Arc::new(tokio::sync::Barrier::new(WRITERS));
538        let mut writers = tokio::task::JoinSet::new();
539        for index in 0..WRITERS {
540            let cache = cache.clone();
541            let key = key.clone();
542            let barrier = barrier.clone();
543            writers.spawn(async move {
544                let mut report = report(CacheHints::default());
545                report
546                    .virtual_packages
547                    .values_mut()
548                    .next()
549                    .unwrap()
550                    .as_mut()
551                    .unwrap()
552                    .build_string = index.to_string().repeat(index + 1);
553                barrier.wait().await;
554                let stderr = format!("publication {index}");
555                cache
556                    .write(&key, &report, Some(&stderr), &clock(1), &env())
557                    .await?;
558                Ok::<_, CacheError>((report.virtual_packages, stderr))
559            });
560        }
561        let mut publications = Vec::new();
562        while let Some(result) = writers.join_next().await {
563            publications.push(result.unwrap());
564        }
565        let publications = publications
566            .into_iter()
567            .map(|result| result.expect("a concurrent publication failed"))
568            .collect::<Vec<_>>();
569        let cached = cache
570            .read(&key, &clock(2), &env())
571            .await
572            .expect("concurrent publications left an unreadable cache entry");
573        assert!(
574            publications.contains(&(cached.virtual_packages, cached.stderr.unwrap())),
575            "the cached entry combines data from different publications"
576        );
577    }

672        assert!(cache.read(&key(), &clock(2), &env()).await.is_none());
673
674        // Present at write time: disappearing expires the entry.
675        cache
676            .write(&key(), &report(hints), None, &clock(3), &env())
677            .await
678            .unwrap();
679        assert!(cache.read(&key(), &clock(4), &env()).await.is_some());
680        std::fs::remove_file(&watched).unwrap();
681        assert!(cache.read(&key(), &clock(4), &env()).await.is_none());
682    }
