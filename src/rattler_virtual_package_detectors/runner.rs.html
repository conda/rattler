1//! Running a detector executable within the protocol's limits.
2//!
3//! The executable is looked up only in the prefix's `PATH` directories, in CEP
4//! 32 order, and started with no arguments, no standard input and the
5//! activated environment. The clock starts at spawn. When the timeout is
6//! reached, or when standard output and standard error together reach the
7//! output limit, the process and its descendants are terminated and the run
8//! fails. Standard error captured until then is kept for diagnostics.
9

11    path::{Path, PathBuf},
12    process::ExitStatus,
13    time::Duration,
14};

18    activation::prefix_path_entries,

20    process::{ProcessError, ProcessLimits, run_bounded},
21};
22use thiserror::Error;

26/// The bounds of one run.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub struct RunLimits {
29    /// How long the process may run after it was spawned.
30    pub timeout: Duration,
31    /// How many bytes of standard output and standard error, combined, the
32    /// process may write.
33    pub output_limit: usize,
34}
35
36impl Default for RunLimits {
37    fn default() -> Self {
38        Self {
39            timeout: DEFAULT_TIMEOUT,
40            output_limit: OUTPUT_LIMIT,
41        }
42    }
43}

45/// A successful run: the process exited with status `0` within the limits.
46#[derive(Clone, Debug)]
47pub struct DetectorRun {
48    /// The executable that ran.
49    pub executable: PathBuf,
50    /// Everything the process wrote to standard output.
51    pub stdout: Vec<u8>,
52    /// Everything the process wrote to standard error, lossily decoded.
53    pub stderr: String,
54    /// How long the process ran.
55    pub duration: Duration,
56}
57
58/// Why a run failed.
59#[derive(Debug, Error)]
60pub enum RunError {
61    /// No executable with the detector's name exists in the prefix's `PATH`
62    /// directories.
63    #[error("no executable named '{name}' in {}", searched.iter().map(|p| p.display().to_string()).collect::<Vec<_>>().join(", "))]
64    ExecutableNotFound {
65        /// The executable name that was looked for.
66        name: String,
67        /// The directories that were searched, in order.
68        searched: Vec<PathBuf>,
69    },
70
71    /// The process could not be started or its output could not be read.
72    #[error("failed to run {executable}")]
73    Io {
74        /// The executable.
75        executable: PathBuf,
76        /// The underlying error.
77        #[source]
78        source: std::io::Error,
79        /// Standard error captured before the failure.
80        stderr: String,
81    },
82
83    /// The process did not exit within the timeout and was terminated.
84    #[error("the detector did not finish within {timeout:?} and was terminated")]
85    TimedOut {
86        /// The timeout that was reached.
87        timeout: Duration,
88        /// Standard error captured before termination.
89        stderr: String,
90    },
91
92    /// The process reached the combined output limit and was terminated.
93    #[error("the detector reached the {limit}-byte output limit and was terminated")]
94    OutputLimitExceeded {
95        /// The limit that was exceeded.
96        limit: usize,
97        /// Standard error captured before termination.
98        stderr: String,
99    },
100
101    /// The process exited with a nonzero status.
102    #[error("the detector exited with {status}")]
103    Exited {
104        /// The exit status.
105        status: ExitStatus,
106        /// Everything the process wrote to standard error.
107        stderr: String,
108    },
109}
110
111impl RunError {
112    /// The standard error captured before the run failed, where a process ran
113    /// at all.
114    pub fn stderr(&self) -> Option<&str> {
115        match self {
116            Self::Io { stderr, .. }
117            | Self::TimedOut { stderr, .. }
118            | Self::OutputLimitExceeded { stderr, .. }
119            | Self::Exited { stderr, .. } => Some(stderr),
120            Self::ExecutableNotFound { .. } => None,
121        }
122    }
123}

125/// Finds the detector executable named after `detector` in the `PATH`
126/// directories of `prefix` for `platform`.
127///
128/// On Windows the `.exe`, `.cmd` and `.bat` extensions are tried in that order
129/// within each directory; elsewhere the bare name must be an executable file.
130pub fn find_executable(
131    prefix: &Path,
132    detector: &PackageName,
133    platform: Subdir,
134) -> Result<PathBuf, RunError> {
135    let directories = prefix_path_entries(prefix, &platform);
136    let name = detector.as_normalized();
137    let candidates: Vec<String> = if platform.is_windows() {
138        ["exe", "cmd", "bat"]
139            .iter()
140            .map(|extension| format!("{name}.{extension}"))
141            .collect()
142    } else {
143        vec![name.to_string()]
144    };
145    for directory in &directories {
146        for candidate in &candidates {
147            let path = directory.join(candidate);
148            if is_executable_file(&path, platform) {
149                return Ok(path);
150            }
151        }
152    }
153    Err(RunError::ExecutableNotFound {
154        name: name.to_string(),
155        searched: directories,
156    })
157}
158
159/// Whether `path` is a regular file the platform would execute.
160fn is_executable_file(path: &Path, platform: Subdir) -> bool {
161    let Ok(metadata) = std::fs::metadata(path) else {
162        return false;
163    };
164    if !metadata.is_file() {
165        return false;
166    }
167    if platform.is_windows() {
168        return true;
169    }
170    #[cfg(unix)]
171    {
172        use std::os::unix::fs::PermissionsExt;
173        metadata.permissions().mode() & 0o111 != 0
174    }
175    #[cfg(not(unix))]
176    {
177        true
178    }
179}

181/// Runs the detector `detector` installed in `prefix` with the activated
182/// environment `env`.
183pub async fn run_detector(
184    prefix: &Path,
185    detector: &PackageName,
186    platform: Subdir,
187    env: &EnvironmentSnapshot,
188    limits: RunLimits,
189) -> Result<DetectorRun, RunError> {
190    let executable = find_executable(prefix, detector, platform)?;
191    let mut command = tokio::process::Command::new(&executable);
192    command.env_clear().envs(env.iter());
193
194    let lossy = |bytes: &[u8]| String::from_utf8_lossy(bytes).into_owned();
195    let output = run_bounded(
196        &mut command,
197        ProcessLimits {
198            timeout: limits.timeout,
199            output_limit: limits.output_limit,
200        },
201    )
202    .await
203    .map_err(|error| match error {
204        ProcessError::Spawn(source) => RunError::Io {
205            executable: executable.clone(),
206            source,
207            stderr: String::new(),
208        },
209        ProcessError::Io { source, stderr } => RunError::Io {
210            executable: executable.clone(),
211            source,
212            stderr: lossy(&stderr),
213        },
214        ProcessError::TimedOut { timeout, stderr } => RunError::TimedOut {
215            timeout,
216            stderr: lossy(&stderr),
217        },
218        ProcessError::OutputLimitExceeded { limit, stderr } => RunError::OutputLimitExceeded {
219            limit,
220            stderr: lossy(&stderr),
221        },
222    })?;
223
224    let stderr = lossy(&output.stderr);
225    if !output.status.success() {
226        return Err(RunError::Exited {
227            status: output.status,
228            stderr,
229        });
230    }
231    Ok(DetectorRun {
232        executable,
233        stdout: output.stdout,
234        stderr,
235        duration: output.duration,
236    })
237}

243    #[cfg(unix)]
244    fn write_script(dir: &Path, name: &str, body: &str) -> PathBuf {
245        use std::os::unix::fs::PermissionsExt;
246        let bin = dir.join("bin");
247        std::fs::create_dir_all(&bin).unwrap();
248        let path = bin.join(name);
249        std::fs::write(&path, format!("#!/bin/sh\n{body}\n")).unwrap();
250        std::fs::set_permissions(&path, std::fs::Permissions::from_mode(0o755)).unwrap();
251        path
252    }
253
254    #[cfg(windows)]
255    fn write_script(dir: &Path, name: &str, body: &str) -> PathBuf {
256        let scripts = dir.join("Scripts");
257        std::fs::create_dir_all(&scripts).unwrap();
258        let path = scripts.join(format!("{name}.bat"));
259        std::fs::write(&path, format!("@echo off\r\n{body}\r\n")).unwrap();
260        path
261    }
262
263    fn name(name: &str) -> PackageName {
264        PackageName::try_from(name).unwrap()
265    }

379    #[test]
380    fn windows_extension_order() {
381        let dir = tempfile::tempdir().unwrap();
382        let scripts = dir.path().join("Scripts");
383        std::fs::create_dir_all(&scripts).unwrap();
384        std::fs::write(scripts.join("x-detect.bat"), "").unwrap();
385        std::fs::write(scripts.join("x-detect.cmd"), "").unwrap();
386        // A bare file never matches on Windows.
387        std::fs::write(scripts.join("x-detect"), "").unwrap();
388        let found = find_executable(dir.path(), &name("x-detect"), Subdir::Win64).unwrap();
389        assert_eq!(found, scripts.join("x-detect.cmd"));
390        std::fs::write(scripts.join("x-detect.exe"), "").unwrap();
391        let found = find_executable(dir.path(), &name("x-detect"), Subdir::Win64).unwrap();
392        assert_eq!(found, scripts.join("x-detect.exe"));
393        // `Library/bin` comes before `Scripts` in CEP 32 order.
394        let library_bin = dir.path().join("Library/bin");
395        std::fs::create_dir_all(&library_bin).unwrap();
396        std::fs::write(library_bin.join("x-detect.bat"), "").unwrap();
397        let found = find_executable(dir.path(), &name("x-detect"), Subdir::Win64).unwrap();
398        assert_eq!(found, library_bin.join("x-detect.bat"));
399    }
