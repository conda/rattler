1//! The environment variables of an activated detector environment.
2//!
3//! Activation evaluates the prefix's activation scripts, including those of
4//! the detector's dependencies, the same way a normal activation would, and
5//! prepends the prefix's `PATH` directories in CEP 32 order to the inherited
6//! `PATH`. It runs in the platform's default shell under the same time and
7//! output bounds as the detector itself, and its process tree is terminated
8//! when a bound is reached.
9

16    },
17    environment::EnvironmentSnapshot,
18    process::{ProcessError, ProcessLimits},
19    shell::ShellEnum,
20};
21use thiserror::Error;

23use crate::limits::OUTPUT_LIMIT;

25/// Why activation failed.
26#[derive(Debug, Error)]
27pub enum ActivationError {
28    /// The activation script could not be assembled.
29    #[error("failed to build the activation script")]
30    Script(#[source] rattler_shell::activation::ActivationError),
31
32    /// The activation script could not be written.
33    #[error("failed to write the activation script")]
34    Io(#[from] std::io::Error),
35
36    /// The shell could not be started or read.
37    #[error("failed to run the activation script")]
38    Process(#[source] ProcessError),
39
40    /// The shell did not finish within the timeout.
41    #[error("activation did not finish within {timeout:?}")]
42    TimedOut {
43        /// The timeout that was reached.
44        timeout: Duration,
45        /// What the shell wrote to standard error before termination.
46        stderr: String,
47    },
48
49    /// The shell reached the combined output limit.
50    #[error("activation reached the {limit}-byte output limit and was terminated")]
51    OutputLimitExceeded {
52        /// The limit that was exceeded.
53        limit: usize,
54        /// What the shell wrote to standard error before termination.
55        stderr: String,
56    },
57
58    /// The shell exited unsuccessfully.
59    #[error("activation exited with {status}")]
60    Failed {
61        /// The shell's exit status.
62        status: std::process::ExitStatus,
63        /// What the shell wrote to standard error.
64        stderr: String,
65    },
66
67    /// The shell's output did not contain the environment dumps.
68    #[error("activation produced no environment listing")]
69    NoEnvironment,
70}
71
72impl ActivationError {
73    /// What the shell wrote to standard error, where it ran at all.
74    pub fn stderr(&self) -> Option<String> {
75        match self {
76            Self::TimedOut { stderr, .. }
77            | Self::OutputLimitExceeded { stderr, .. }
78            | Self::Failed { stderr, .. } => Some(stderr.clone()),
79            Self::Process(error) => error
80                .stderr()
81                .map(|bytes| String::from_utf8_lossy(bytes).into_owned()),
82            Self::Script(_) | Self::Io(_) | Self::NoEnvironment => None,
83        }
84    }
85}

87/// Computes the environment of `prefix` after activation on `platform`,
88/// starting from an explicit native-string environment snapshot.
89///
90/// The result is the complete environment to start the detector with.
91pub async fn activated_environment(
92    prefix: &Path,
93    platform: Subdir,
94    timeout: Duration,
95    inherited: &EnvironmentSnapshot,
96) -> Result<EnvironmentSnapshot, ActivationError> {
97    let shell = ShellEnum::default();
98    let activator = {
99        let prefix = prefix.to_path_buf();
100        let shell = shell.clone();
101        tokio::task::spawn_blocking(move || Activator::from_path(&prefix, shell, platform))
102            .await
103            .map_err(|error| ActivationError::Io(std::io::Error::other(error)))?
104            .map_err(ActivationError::Script)?
105    };
106    activator
107        .run_activation_bounded(
108            ActivationVariables {
109                conda_prefix: None,
110                path: None,
111                path_modification_behavior: PathModificationBehavior::Prepend,
112                current_env: HashMap::default(),
113            },
114            inherited,
115            ProcessLimits {
116                timeout,
117                output_limit: OUTPUT_LIMIT,
118            },
119        )
120        .await
121        .map_err(|error| {
122            let lossy = |bytes: &[u8]| String::from_utf8_lossy(bytes).into_owned();
123            match error {
124                BoundedActivationError::Script(error) => ActivationError::Script(error),
125                BoundedActivationError::Process(ProcessError::TimedOut { timeout, stderr }) => {
126                    ActivationError::TimedOut {
127                        timeout,
128                        stderr: lossy(&stderr),
129                    }
130                }
131                BoundedActivationError::Process(ProcessError::OutputLimitExceeded {
132                    limit,
133                    stderr,
134                }) => ActivationError::OutputLimitExceeded {
135                    limit,
136                    stderr: lossy(&stderr),
137                },
138                BoundedActivationError::Process(error) => ActivationError::Process(error),
139                BoundedActivationError::Failed { status, stderr } => ActivationError::Failed {
140                    status,
141                    stderr: lossy(&stderr),
142                },
143                BoundedActivationError::NoEnvironment => ActivationError::NoEnvironment,
144            }
145        })
146}

148#[cfg(all(test, unix))]
149mod tests {
150    use super::*;
151    use std::{ffi::OsString, os::unix::ffi::OsStringExt};

153    #[tokio::test]
154    async fn activation_prepends_the_prefix_and_applies_scripts() {
155        let dir = tempfile::tempdir().unwrap();
156        let prefix = dir.path();
157        std::fs::create_dir_all(prefix.join("bin")).unwrap();
158        std::fs::create_dir_all(prefix.join("etc/conda/activate.d")).unwrap();
159        std::fs::write(
160            prefix.join("etc/conda/activate.d/test.sh"),
161            "export DETECTOR_TEST_ACTIVATED=1\nunset DETECTOR_TEST_REMOVED\n",
162        )
163        .unwrap();
164
165        let mut inherited = EnvironmentSnapshot::from_system();
166        inherited.insert("DETECTOR_TEST_REMOVED", "present");
167        let env = activated_environment(
168            prefix,
169            Subdir::current().unwrap(),
170            Duration::from_secs(30),
171            &inherited,
172        )
173        .await
174        .unwrap();
175        assert_eq!(
176            env.get("DETECTOR_TEST_ACTIVATED")
177                .and_then(|value| value.to_str()),
178            Some("1")
179        );
180        assert!(env.get("DETECTOR_TEST_REMOVED").is_none());
181
182        let path = env.get("PATH").unwrap();
183        let entries: Vec<_> = std::env::split_paths(path).collect();
184        assert_eq!(entries[0], prefix.join("bin"));
185        let inherited: Vec<_> = std::env::split_paths(inherited.get("PATH").unwrap()).collect();
186        assert_eq!(&entries[1..], inherited.as_slice());
187    }

189    #[tokio::test]
190    async fn activation_preserves_native_values_and_removes_unset_variables() {
191        let dir = tempfile::tempdir().unwrap();
192        std::fs::create_dir_all(dir.path().join("etc/conda/activate.d")).unwrap();
193        std::fs::write(
194            dir.path().join("etc/conda/activate.d/native.sh"),
195            "export COPIED=\"$NATIVE\"\nexport CHANGED=\"${NATIVE}suffix\"\nunset REMOVED\nexport EMPTY=\nexport COLLISION=____RATTLER_ENV_START____after\n",
196        ).unwrap();
197        let native = OsString::from_vec(b"native-\xff".to_vec());
198        let inherited: EnvironmentSnapshot = [
199            ("PATH", OsString::from("/usr/bin:/bin")),
200            ("NATIVE", native.clone()),
201            ("REMOVED", native.clone()),
202            (
203                "COLLISION",
204                OsString::from("____RATTLER_ENV_START____before"),
205            ),
206        ]
207        .into_iter()
208        .collect();
209        let result = activated_environment(
210            dir.path(),
211            Subdir::current().unwrap(),
212            Duration::from_secs(30),
213            &inherited,
214        )
215        .await
216        .unwrap();
217        assert_eq!(result.get("NATIVE"), Some(native.as_os_str()));
218        assert_eq!(result.get("COPIED"), Some(native.as_os_str()));
219        let mut changed = native.clone();
220        changed.push("suffix");
221        assert_eq!(result.get("CHANGED"), Some(changed.as_os_str()));
222        assert!(result.get("REMOVED").is_none());
223        assert_eq!(result.get("EMPTY"), Some(std::ffi::OsStr::new("")));
224        assert_eq!(
225            result.get("COLLISION"),
226            Some(std::ffi::OsStr::new("____RATTLER_ENV_START____after"))
227        );
228        assert_eq!(inherited.get("REMOVED"), Some(native.as_os_str()));
229    }
