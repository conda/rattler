1//! Resolving a detector and installing it into an environment of its own.
2//!
3//! Clients can implement [`DetectorEnvironmentProvider`] to use their own
4//! environment flow. The standalone [`RattlerEnvironmentProvider`] resolves the
5//! detector by name, qualified with its registering channel, against the
6//! registration's resolution channels for the host platform. The
7//! only virtual packages available to that solve are the client's own. The
8//! resolved records are fingerprinted with the
9//! [`environment_digest`], and the environment lives in a directory named
10//! after that digest, so an environment is reused exactly when the resolved
11//! records are unchanged and reinstalled otherwise.
12//!
13//! Installation never runs link scripts. A prefix guard makes concurrent
14//! installations of the same digest wait for each other and marks an
15//! environment ready only once installation completed, so a detector never
16//! runs from an incomplete installation.
17

38/// What resolving and installing a detector needs from the client.
39#[derive(Clone)]
40pub struct EnvironmentOptions<'a> {
41    /// The gateway to load repodata with.
42    pub gateway: &'a Gateway,
43    /// The package cache to install from.
44    pub package_cache: &'a PackageCache,
45    /// The client to download packages with.
46    pub download_client: LazyClient,
47    /// The directory that holds every detector environment.
48    pub root: &'a Path,
49    /// The platform of the machine running the client.
50    pub host_platform: Subdir,
51    /// The client's own virtual packages, with overrides applied. Detector
52    /// results never take part in resolving a detector.
53    pub virtual_packages: Vec<GenericVirtualPackage>,
54}

56/// Lets a client resolve and create isolated detector environments using its
57/// own environment machinery.
58///
59/// Resolution must use only the client's builtin virtual packages, qualify the
60/// detector with its registering channel, and use the registration's already
61/// expanded resolution channels without expanding them again. Detector results
62/// must never participate in this solve. Installation must disable link scripts,
63/// guard concurrent creation and repair interrupted installations. The engine
64/// calls `install` only after consent and only when no valid result is cached.
65#[async_trait]
66pub trait DetectorEnvironmentProvider: Send + Sync {
67    /// Resolves the detector against current repodata without installing it.
68    async fn resolve(
69        &self,
70        registration: &AcceptedDetectorRegistration,
71    ) -> Result<ResolvedDetector, EnvironmentError>;
72
73    /// Returns a complete environment for the resolved detector.
74    async fn install(
75        &self,
76        resolved: ResolvedDetector,
77    ) -> Result<DetectorEnvironment, EnvironmentError>;
78}

80/// The standalone Rattler solver and installer implementation.
81pub struct RattlerEnvironmentProvider<'a> {
82    options: EnvironmentOptions<'a>,
83}

85impl<'a> RattlerEnvironmentProvider<'a> {
86    /// Creates a provider whose environment root and solve configuration are
87    /// supplied by the client.
88    pub fn new(options: EnvironmentOptions<'a>) -> Self {
89        Self { options }
90    }
91}

93#[async_trait]
94impl DetectorEnvironmentProvider for RattlerEnvironmentProvider<'_> {
95    async fn resolve(
96        &self,
97        registration: &AcceptedDetectorRegistration,
98    ) -> Result<ResolvedDetector, EnvironmentError> {
99        resolve_detector(registration, &self.options).await
100    }
101
102    async fn install(
103        &self,
104        resolved: ResolvedDetector,
105    ) -> Result<DetectorEnvironment, EnvironmentError> {
106        ensure_environment(resolved, &self.options).await
107    }
108}

120/// An installed, ready detector environment.
121#[derive(Clone, Debug)]
122pub struct DetectorEnvironment {
123    /// The prefix the detector is installed in.
124    pub prefix: PathBuf,
125    /// Whether this call installed the environment rather than reusing it.
126    pub installed: bool,

129/// Why a detector environment could not be prepared.
130#[derive(Debug, Error)]
131pub enum EnvironmentError {
132    /// The repodata of the resolution channels could not be loaded.
133    #[error("failed to load repodata for the detector")]
134    Repodata(#[from] GatewayError),
135
136    /// The detector could not be resolved.
137    #[error("failed to resolve the detector")]
138    Solve(#[from] SolveError),
139
140    /// The solve task panicked or was cancelled.
141    #[error("the solve was interrupted")]
142    SolveInterrupted(#[source] tokio::task::JoinError),
143
144    /// The detector could not be installed.
145    #[error("failed to install the detector")]
146    Install(#[from] rattler::install::InstallerError),
147
148    /// The prefix guard could not be acquired or updated.
149    #[error("failed to guard the detector environment at {}", prefix.display())]
150    Guard {
151        /// The prefix.
152        prefix: PathBuf,
153        /// The underlying error.
154        #[source]
155        source: std::io::Error,
156    },
157
158    /// A client-provided environment implementation failed.
159    #[error("failed to prepare the detector environment: {0}")]
160    Provider(#[source] Box<dyn std::error::Error + Send + Sync>),
161}

163/// The prefix a detector environment with `digest` lives in under `root`.
164pub fn prefix_for(root: &Path, digest: &Sha256Hash) -> PathBuf {
165    root.join(hex::encode(digest))
166}

168/// The `MatchSpec` a detector is resolved with: its name qualified with the
169/// registering channel, without version or build constraints.
170pub fn detector_spec(registration: &AcceptedDetectorRegistration) -> MatchSpec {
171    MatchSpec {
172        name: registration.registration.detector.clone().into(),
173        channel: Some(Arc::new(registration.channel.clone())),
174        ..MatchSpec::default()
175    }
176}

178/// Resolves `registration`'s detector against current repodata and computes
179/// the environment digest.
180pub async fn resolve_detector(
181    registration: &AcceptedDetectorRegistration,
182    options: &EnvironmentOptions<'_>,
183) -> Result<ResolvedDetector, EnvironmentError> {
184    let spec = detector_spec(registration);
185    let repodata = options
186        .gateway
187        .query(
188            registration.resolution_channels.iter().cloned(),
189            [options.host_platform, Subdir::NoArch],
190            [spec.clone()],
191        )
192        .recursive(true)
193        // The resolution channels are already the fully resolved set.
194        .channel_relations(ChannelRelationsMode::Disabled)
195        .await?;
196
197    let virtual_packages = options.virtual_packages.clone();
198    let records = tokio::task::spawn_blocking(move || {
199        let task = SolverTask {
200            specs: vec![spec],
201            virtual_packages,
202            ..SolverTask::from_iter(&repodata.repodata)
203        };
204        rattler_solve::resolvo::Solver.solve(task)
205    })
206    .await
207    .map_err(EnvironmentError::SolveInterrupted)??
208    .records;
209
210    let digest = environment_digest(&records);
211    Ok(ResolvedDetector { records, digest })
212}

214/// Makes sure the environment for `resolved` exists and is complete,
215/// installing it when it is missing or was never completed.
216pub async fn ensure_environment(
217    resolved: ResolvedDetector,
218    options: &EnvironmentOptions<'_>,
219) -> Result<DetectorEnvironment, EnvironmentError> {
220    let prefix = prefix_for(options.root, &resolved.digest);
221    let guard_error = |source| EnvironmentError::Guard {
222        prefix: prefix.clone(),
223        source,
224    };
225    let guard = AsyncPrefixGuard::new(&prefix).await.map_err(guard_error)?;
226    let mut write_guard = guard.write().await.map_err(guard_error)?;
227    if write_guard.is_ready() {
228        tracing::debug!(prefix = %prefix.display(), "reusing detector environment");
229        return Ok(DetectorEnvironment {
230            prefix,
231            installed: false,
232        });
233    }
234
235    tracing::debug!(prefix = %prefix.display(), "installing detector environment");
236    write_guard.begin().await.map_err(guard_error)?;
237    Installer::new()
238        .with_package_cache(options.package_cache.clone())
239        .with_download_client(options.download_client.clone())
240        .with_target_platform(options.host_platform)
241        .with_execute_link_scripts(false)
242        .install(&prefix, resolved.records)
243        .await?;
244    write_guard.finish().await.map_err(guard_error)?;
245
246    Ok(DetectorEnvironment {
247        prefix,
248        installed: true,
249    })
