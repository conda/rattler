1//! Dependency translation between solver specs and CEP-37 dependency mappings.
2//!
3//! CEP-37 stores dependencies as `name: constraint`, where the full spec is the
4//! concatenation `name + " " + constraint`. Conda records keep a list of
5//! `MatchSpec` strings and Python records keep PEP 508 requirements, so the
6//! helpers here answer one question per dependency: does this spec survive that
7//! representation unchanged? Anything that would change which packages a
8//! dependency matches is an error, never a silent simplification.
9
10use std::collections::BTreeMap;
11
12use pep508_rs::{MarkerTree, Requirement, VersionOrUrl};
13use rattler_conda_lock::NodePath;
14use rattler_conda_types::{
15    MatchSpec, NamelessMatchSpec, PackageNameMatcher, ParseStrictness, VersionSpec,
16};
17
18use super::error::{CondaLockError, CondaLockErrorKind};
19
20/// One entry of a CEP-37 `dependencies` mapping.
21#[derive(Clone, Debug, PartialEq, Eq)]
22pub(super) struct Dependency {
23    /// The mapping key: the normalized dependency name.
24    pub name: String,
25    /// The mapping value: the constraint without the name, empty when the
26    /// dependency is unconstrained.
27    pub constraint: String,
28}
29
30/// The `dependencies` mapping of one package under construction.
31///
32/// Owns the rejection of repeated names: the source is a list, the destination
33/// is a mapping, so inserting twice would drop a dependency.
34#[derive(Debug, Default)]
35pub(super) struct Dependencies {
36    values: BTreeMap<String, String>,
37    origins: BTreeMap<String, NodePath>,
38}
39
40impl Dependencies {
41    /// Add one dependency, or fail if the package already has one by that name.
42    pub(super) fn insert(
43        &mut self,
44        dependency: Dependency,
45        path: NodePath,
46    ) -> Result<(), CondaLockError> {
47        let Dependency { name, constraint } = dependency;
48        if let Some(origin) = self.origins.get(&name) {
49            return Err(CondaLockError::new(
50                path,
51                CondaLockErrorKind::RepeatedDependencyName { name },
52            )
53            .with_related_path(origin.clone(), "first dependency with this name"));
54        }
55        self.origins.insert(name.clone(), path);
56        self.values.insert(name, constraint);
57        Ok(())
58    }
59
60    /// The finished mapping, in key order.
61    pub(super) fn into_values(self) -> BTreeMap<String, String> {
62        self.values
63    }
64}

66/// Converts one conda `MatchSpec` string into a CEP-37 dependency entry.
67///
68/// Rejects specs whose meaning the CEP-37 spelling cannot carry: a name matcher
69/// instead of an exact name, any field besides version and build (channel,
70/// subdir, checksums, URL, license, ...), and specs that do not parse back to
71/// themselves after being split into name and constraint.
72pub(super) fn conda_spec(value: &str, path: &NodePath) -> Result<Dependency, CondaLockError> {
73    let spec = MatchSpec::from_str(value, ParseStrictness::Lenient).map_err(|error| {
74        CondaLockError::new(path.clone(), CondaLockErrorKind::InvalidMatchSpec(error))
75    })?;
76    let PackageNameMatcher::Exact(name) = &spec.name else {
77        return Err(CondaLockError::new(
78            path.clone(),
79            CondaLockErrorKind::InexactDependencyName,
80        ));
81    };
82    let supported = MatchSpec {
83        name: spec.name.clone(),
84        version: spec.version.clone(),
85        build: spec.build.clone(),
86        ..MatchSpec::default()
87    };
88    if spec != supported {
89        return Err(CondaLockError::new(
90            path.clone(),
91            CondaLockErrorKind::UnsupportedSpecFields,
92        ));
93    }
94    // CEP-37 concatenates name and constraint, so an unconstrained dependency
95    // has an empty value rather than an explicit `*`.
96    let unconstrained = spec.build.is_none()
97        && spec
98            .version
99            .as_ref()
100            .is_none_or(|version| *version == VersionSpec::Any);
101    if unconstrained {
102        return Ok(Dependency {
103            name: name.as_normalized().to_owned(),
104            constraint: String::new(),
105        });
106    }
107    let constraint = NamelessMatchSpec::from(spec.clone()).to_string();
108    let reconstructed = MatchSpec::from_str(
109        &format!("{} {constraint}", name.as_source()),
110        ParseStrictness::Lenient,
111    )
112    .map_err(|error| {
113        CondaLockError::new(path.clone(), CondaLockErrorKind::InvalidMatchSpec(error))
114    })?;
115    if reconstructed != spec {
116        return Err(CondaLockError::new(
117            path.clone(),
118            CondaLockErrorKind::LossyDependency,
119        ));
120    }
121    Ok(Dependency {
122        name: name.as_normalized().to_owned(),
123        constraint,
124    })
125}

127/// Converts one PEP 508 requirement into a CEP-37 dependency entry.
128///
129/// Rejects requirements whose installation semantics a bare constraint cannot
130/// carry: extras, environment markers, and direct URLs.
131pub(super) fn python_spec(
132    requirement: &Requirement,
133    path: &NodePath,
134) -> Result<Dependency, CondaLockError> {
135    if !requirement.extras.is_empty() || requirement.marker != MarkerTree::TRUE {
136        return Err(CondaLockError::new(
137            path.clone(),
138            CondaLockErrorKind::PythonExtrasOrMarkers,
139        ));
140    }
141    let constraint = match &requirement.version_or_url {
142        None => String::new(),
143        Some(VersionOrUrl::VersionSpecifier(spec)) => spec.to_string(),
144        Some(VersionOrUrl::Url(_)) => {
145            return Err(CondaLockError::new(
146                path.clone(),
147                CondaLockErrorKind::DirectPythonUrl,
148            ));
149        }
150    };
151    Ok(Dependency {
152        name: requirement.name.to_string(),
153        constraint,
154    })
155}
