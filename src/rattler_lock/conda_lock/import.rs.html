22/// Exact category selections for each destination pixi environment.
23#[derive(Clone, Debug, PartialEq, Eq)]
24pub struct ImportOptions {
25    /// Environment names mapped to category sets. No implicit `main` category is added.
26    pub environments: BTreeMap<String, BTreeSet<String>>,
27}
28
29impl Default for ImportOptions {
30    fn default() -> Self {
31        Self {
32            environments: BTreeMap::from([("default".into(), BTreeSet::from(["main".into()]))]),
33        }
34    }
35}

37impl LockFile {
38    /// Imports selected categories of a CEP-37 lock file, without solving,
39    /// downloading, or expanding variables.
40    ///
41    /// Empty declared platforms remain present in every destination environment.
42    /// Errors refer to CEP-37 model paths, including both sides of conflicting
43    /// selections; pass a [`Document`] to [`Self::from_conda_lock_document`] to
44    /// have those paths resolved to source spans as well.
45    ///
46    /// ```
47    /// # fn example(cep: &rattler_conda_lock::LockFile) -> Result<(), rattler_lock::conda_lock::CondaLockError> {
48    /// use rattler_lock::{LockFile, conda_lock::ImportOptions};
49    ///
50    /// let pixi = LockFile::from_conda_lock(cep, &ImportOptions::default())?;
51    /// assert!(pixi.default_environment().is_some());
52    /// # Ok(())
53    /// # }
54    /// ```
55    pub fn from_conda_lock(
56        lock_file: &CepLockFile,
57        options: &ImportOptions,
58    ) -> Result<Self, CondaLockError> {
59        import(lock_file, options)
60    }
61
62    /// Imports a parsed document, attaching its source spans to any error.
63    ///
64    /// ```
65    /// # fn example(yaml: &str) -> Result<(), rattler_lock::conda_lock::CondaLockError> {
66    /// use rattler_lock::{LockFile, conda_lock::ImportOptions};
67    ///
68    /// let document = rattler_conda_lock::Document::parse(yaml)?;
69    /// let pixi = LockFile::from_conda_lock_document(&document, &ImportOptions::default())?;
70    /// # Ok(())
71    /// # }
72    /// ```
73    pub fn from_conda_lock_document(
74        document: &Document,
75        options: &ImportOptions,
76    ) -> Result<Self, CondaLockError> {
77        import(document.lock_file(), options).map_err(|mut error| {
78            document.locate(error.labels_mut());
79            error
80        })
81    }
82}

84fn import(lock_file: &CepLockFile, options: &ImportOptions) -> Result<LockFile, CondaLockError> {
85    lock_file.validate()?;
86    let options_path = NodePath::root().field("options").field("environments");
87    if options.environments.is_empty() {
88        return Err(CondaLockError::new(
89            options_path,
90            CondaLockErrorKind::NoEnvironments,
91        ));
92    }
93    let mut categories: BTreeSet<&str> = lock_file
94        .package
95        .iter()
96        .map(|package| package.category.as_str())
97        .collect();
98    // An empty lock still has a meaningful empty main environment.
99    categories.insert("main");
100    for (name, selected) in &options.environments {
101        let path = options_path.field(name);
102        if name.trim().is_empty() || selected.is_empty() {
103            return Err(CondaLockError::new(
104                path,
105                CondaLockErrorKind::EmptySelection,
106            ));
107        }
108        for category in selected {
109            if !categories.contains(category.as_str()) {
110                return Err(CondaLockError::new(
111                    path.clone(),
112                    CondaLockErrorKind::UnknownCategory {
113                        category: category.clone(),
114                    },
115                ));
116            }
117        }
118    }
119    let platforms_path = NodePath::root().field("metadata").field("platforms");
120    let platforms = lock_file
121        .metadata
122        .platforms
123        .iter()
124        .enumerate()
125        .map(|(index, name)| {
126            let subdir = Subdir::from_str(name).map_err(|error| {
127                CondaLockError::new(
128                    platforms_path.index(index),
129                    CondaLockErrorKind::UnsupportedPlatform(error),
130                )
131            })?;
132            Ok(PlatformData {
133                name: (&subdir).into(),
134                subdir,
135                virtual_packages: Vec::new(),
136            })
137        })
138        .collect::<Result<Vec<_>, CondaLockError>>()?;
139    let mut builder = LockFile::builder()
140        .with_platforms(platforms)
141        .map_err(|error| {
142            CondaLockError::new(
143                platforms_path.clone(),
144                CondaLockErrorKind::Builder(Box::new(error)),
145            )
146        })?;
147    let packages_path = NodePath::root().field("package");
148    // The builder merges packages by artifact identity. Reject divergent records
149    // before registration so one environment cannot change another's semantics.
150    let mut artifacts: BTreeMap<ArtifactKey<'_>, (LockedPackage, usize)> = BTreeMap::new();
151    for (environment, categories) in &options.environments {
152        // Declared platforms stay in the environment even when the selected
153        // categories contain no package for them.
154        for (index, platform) in lock_file.metadata.platforms.iter().enumerate() {
155            builder
156                .add_environment_platform(environment, platform)
157                .map_err(|error| {
158                    CondaLockError::new(
159                        platforms_path.index(index),
160                        CondaLockErrorKind::Builder(Box::new(error)),
161                    )
162                })?;
163        }
164        builder.set_channels(
165            environment,
166            lock_file
167                .metadata
168                .channels
169                .iter()
170                .map(|channel| crate::Channel {
171                    url: channel.url.clone(),
172                    used_env_vars: channel.used_env_vars.clone(),
173                }),
174        );
175        let mut selected = BTreeMap::<EnvironmentSlot<'_>, (LockedPackage, usize)>::new();
176        for (index, package) in lock_file.package.iter().enumerate() {
177            if !categories.contains(&package.category) {
178                continue;
179            }
180            let path = packages_path.index(index);
181            let converted = convert(package, &path)?;
182            // One destination environment holds one package per normalized
183            // name, whatever categories selected it.
184            let slot = EnvironmentSlot::new(package, &converted);
185            if let Some((original, original_index)) = selected.get(&slot) {
186                if original != &converted {
187                    return Err(conflict(
188                        &packages_path,
189                        index,
190                        *original_index,
191                        CondaLockErrorKind::IncompatibleSelection,
192                    ));
193                }
194                continue;
195            }
196            let artifact = ArtifactKey {
197                manager: package.manager,
198                url: package.url.as_str(),
199            };
200            if let Some((original, original_index)) = artifacts.get(&artifact) {
201                if original != &converted {
202                    return Err(conflict(
203                        &packages_path,
204                        index,
205                        *original_index,
206                        CondaLockErrorKind::ConflictingArtifact,
207                    ));
208                }
209            } else {
210                artifacts.insert(artifact, (converted.clone(), index));
211            }
212            selected.insert(slot, (converted.clone(), index));
213            builder
214                .add_package(environment, &package.platform, converted)
215                .map_err(|error| {
216                    CondaLockError::new(path.clone(), CondaLockErrorKind::Builder(Box::new(error)))
217                })?;
218        }
219    }
220    Ok(builder.finish())
221}

223/// The slot a package occupies in one destination environment. A pixi
224/// environment holds one package per manager, platform and normalized name,
225/// however many CEP-37 categories selected it.
226#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
227struct EnvironmentSlot<'a> {
228    manager: Manager,
229    platform: &'a str,
230    name: String,
231}
232
233impl<'a> EnvironmentSlot<'a> {
234    fn new(package: &'a Package, converted: &LockedPackage) -> Self {
235        Self {
236            manager: package.manager,
237            platform: &package.platform,
238            name: match converted {
239                LockedPackage::Conda(data) => data.name().as_normalized().to_owned(),
240                LockedPackage::Pypi(data) => data.name().to_string(),
241            },
242        }
243    }
244}

246/// The artifact a package resolves to. The pixi lock file stores one record per
247/// artifact, so two CEP-37 packages sharing a URL must agree on its metadata.
248#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
249struct ArtifactKey<'a> {
250    manager: Manager,
251    url: &'a str,
252}

254fn conflict(
255    packages: &NodePath,
256    index: usize,
257    original: usize,
258    kind: CondaLockErrorKind,
259) -> CondaLockError {
260    CondaLockError::new(packages.index(index), kind)
261        .with_related_path(packages.index(original), "original package selection")
262}

264fn convert(package: &Package, path: &NodePath) -> Result<LockedPackage, CondaLockError> {
265    if package.source.is_some() {
266        return Err(CondaLockError::new(
267            path.field("source"),
268            CondaLockErrorKind::SourcePackage,
269        ));
270    }
271    let url_path = path.field("url");
272    let url = super::artifact_url(&package.url, &url_path)?;
273    let hash_path = path.field("hash");
274    let md5 = package
275        .hash
276        .md5
277        .as_deref()
278        .map(|value| {
279            parse_digest_from_hex::<Md5>(value).ok_or_else(|| {
280                CondaLockError::new(
281                    hash_path.field("md5"),
282                    CondaLockErrorKind::InvalidDigest {
283                        algorithm: ChecksumAlgorithm::Md5,
284                    },
285                )
286            })
287        })
288        .transpose()?;
289    let sha256 = package
290        .hash
291        .sha256
292        .as_deref()
293        .map(|value| {
294            parse_digest_from_hex::<Sha256>(value).ok_or_else(|| {
295                CondaLockError::new(
296                    hash_path.field("sha256"),
297                    CondaLockErrorKind::InvalidDigest {
298                        algorithm: ChecksumAlgorithm::Sha256,
299                    },
300                )
301            })
302        })
303        .transpose()?;
304    let dependencies_path = path.field("dependencies");
305    match package.manager {
306        Manager::Conda => {
307            if url.as_str() != package.url {
308                return Err(CondaLockError::new(
309                    url_path,
310                    CondaLockErrorKind::UnnormalizedUrl,
311                ));
312            }
313            let location = UrlOrPath::Url(url);
314            let derived = LocationDerivedFields::new(&location);
315            let file_name = derived.identifier.ok_or_else(|| {
316                CondaLockError::new(url_path.clone(), CondaLockErrorKind::MissingFileName)
317            })?;
318            let build = package.build.clone().or(derived.build).ok_or_else(|| {
319                CondaLockError::new(path.field("build"), CondaLockErrorKind::MissingBuildString)
320            })?;
321            let name = PackageName::from_str(&package.name).map_err(|error| {
322                CondaLockError::new(
323                    path.field("name"),
324                    CondaLockErrorKind::InvalidCondaName(error),
325                )
326            })?;
327            let version = VersionWithSource::from_str(&package.version).map_err(|error| {
328                CondaLockError::new(
329                    path.field("version"),
330                    CondaLockErrorKind::InvalidCondaVersion(error),
331                )
332            })?;
333            let mut record = PackageRecord::new(name, version, build);
334            // Matches the historical importer: a build string without a trailing
335            // number carries no build number, and CEP-37 never stores one.
336            record.build_number = derive_build_number_from_build(&record.build).unwrap_or(0);
337            record.subdir = derived.subdir.unwrap_or_else(|| package.platform.clone());
338            if record.subdir != "noarch" && record.subdir != package.platform {
339                return Err(
340                    CondaLockError::new(url_path, CondaLockErrorKind::SubdirMismatch)
341                        .with_related_path(path.field("platform"), "selected platform"),
342                );
343            }
344            (record.arch, record.platform) = derive_arch_and_platform(&record.subdir);
345            record.noarch = derive_noarch_type(&record.subdir, &record.build);
346            record.md5 = md5;
347            record.sha256 = sha256;
348            let mut normalized = BTreeMap::new();
349            for (name, value) in &package.dependencies {
350                let dependency_path = dependencies_path.field(name);
351                let text = format!("{name} {value}");
352                let dependency = conda_spec(&text, &dependency_path)?;
353                if let Some(original) = normalized.insert(dependency.name, dependency_path.clone())
354                {
355                    return Err(CondaLockError::new(
356                        dependency_path,
357                        CondaLockErrorKind::AmbiguousDependencyName,
358                    )
359                    .with_related_path(original, "original dependency"));
360                }
361                record.depends.push(text);
362            }
363            Ok(LockedPackage::Conda(
364                CondaBinaryData {
365                    package_record: record,
366                    location,
367                    file_name,
368                    channel: derived.channel,
369                }
370                .into(),
371            ))
372        }
373        Manager::Pip => {
374            if package.build.is_some() {
375                return Err(CondaLockError::new(
376                    path.field("build"),
377                    CondaLockErrorKind::PythonBuildString,
378                ));
379            }
380            let mut requires_dist = Vec::with_capacity(package.dependencies.len());
381            let mut normalized = BTreeMap::new();
382            for (name, value) in &package.dependencies {
383                let dependency_path = dependencies_path.field(name);
384                let constraint = python_constraint(value, &dependency_path)?;
385                let requirement =
386                    Requirement::from_str(&format!("{name}{constraint}")).map_err(|error| {
387                        CondaLockError::new(
388                            dependency_path.clone(),
389                            CondaLockErrorKind::InvalidRequirement(Box::new(error)),
390                        )
391                    })?;
392                let dependency = python_spec(&requirement, &dependency_path)?;
393                if let Some(original) = normalized.insert(dependency.name, dependency_path.clone())
394                {
395                    return Err(CondaLockError::new(
396                        dependency_path,
397                        CondaLockErrorKind::AmbiguousDependencyName,
398                    )
399                    .with_related_path(original, "original dependency"));
400                }
401                requires_dist.push(requirement);
402            }
403            let data = PypiDistributionData {
404                name: package.name.parse().map_err(|error| {
405                    CondaLockError::new(
406                        path.field("name"),
407                        CondaLockErrorKind::InvalidPythonName(error),
408                    )
409                })?,
410                version: package.version.parse().map_err(|error| {
411                    CondaLockError::new(
412                        path.field("version"),
413                        CondaLockErrorKind::InvalidPythonVersion(error),
414                    )
415                })?,
416                location: Verbatim::new_with_given(UrlOrPath::Url(url), package.url.clone()),
417                index_url: None,
418                hash: PackageHashes::from_hashes(md5, sha256),
419                requires_dist,
420                requires_python: None,
421            };
422            Ok(LockedPackage::Pypi(data.into()))
423        }
424    }
425}

427/// Translates the constraint spellings conda-lock writes into a PEP 508 suffix.
428///
429/// `*` is unconstrained and a bare version literal is an exact pin. Alternatives
430/// separated by `||` have no PEP 508 equivalent and are rejected.
431fn python_constraint(value: &str, path: &NodePath) -> Result<String, CondaLockError> {
432    let value = value.trim();
433    if value == "*" || value.is_empty() {
434        return Ok(String::new());
435    }
436    if value.contains("||") {
437        return Err(CondaLockError::new(
438            path.clone(),
439            CondaLockErrorKind::AlternativeConstraints,
440        ));
441    }
442    if pep440_rs::Version::from_str(value).is_ok() {
443        return Ok(format!("=={value}"));
444    }
445    Ok(value.to_owned())
