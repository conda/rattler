1/// Why a channel URL, host or container name failed to parse.
2#[derive(Debug, thiserror::Error)]
3pub enum AzureUrlError {
4    #[error("no host in Azure blob URL")]
5    NoHost,
6
7    #[error(
8        "Azure blob URL must not contain userinfo (`user:pass@host`), userinfo is invalid in blob URLs"
9    )]
10    UserInfoNotAllowed,
11
12    #[error("`{authority}` is not a valid Azure host: {reason}; expected `host` or `host:port`")]
13    InvalidHostAuthority { authority: String, reason: String },
14
15    #[error(
16        "`{0}` is not a valid Azure blob container name: container names are 3-63 characters of \
17         lowercase letters, digits and hyphens, must start and end with a letter or digit, and \
18         must not contain consecutive hyphens"
19    )]
20    InvalidContainerName(String),
21
22    #[error("`{value}` is not a valid URL")]
23    InvalidUrl {
24        value: String,
25        #[source]
26        source: url::ParseError,
27    },
28
29    #[error(
30        "Azure blob channel URL segment `{0}` is a relative path segment; a channel URL must name \
31         the container it addresses directly, so write the path without `.` or `..`"
32    )]
33    DotSegmentInPath(String),
34
35    #[error(
36        "Azure blob channel URL segment `{0}` percent-decodes to a path separator; write `/` \
37         unencoded between segments"
38    )]
39    EncodedSeparatorInPath(String),
40
41    #[error(
42        "Azure blob channel URL path `{path}` has an empty segment; a doubled `/` names nothing, \
43         so write the path with single separators"
44    )]
45    EmptyPathSegment { path: String },
46
47    #[error(
48        "Azure blob channel URL segment `{segment}` contains `{escape}`, which is not a valid \
49         percent-escape; write a literal `%` as `%25`"
50    )]
51    MalformedPercentEscape { segment: String, escape: String },
52
53    /// Blob names are UTF-8. Decoding lossily would substitute U+FFFD and silently
54    /// address a different blob than the URL names.
55    #[error(
56        "Azure blob channel URL segment `{segment}` percent-decodes to bytes that are not UTF-8, \
57         so it cannot name a blob"
58    )]
59    NonUtf8Path {
60        segment: String,
61        #[source]
62        source: std::str::Utf8Error,
63    },
64
65    #[error("Azure blob channel URL must use the `az://` scheme, got `{0}`")]
66    InvalidScheme(String),
67
68    #[error(
69        "{0:?} contains a control character or a leading or trailing space, which URL parsing \
70         would silently drop; remove it"
71    )]
72    StrippedCharacter(String),
73}
