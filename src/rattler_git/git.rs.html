65impl GitLfs {
66    fn probe() -> Result<Self, GitBinaryError> {
67        let git = GIT.as_ref().map_err(Clone::clone)?.clone();
68        let ok = Command::new(&git)
69            .args(["lfs", "version"])
70            .env(GIT_TERMINAL_PROMPT, "0")
71            .output()
72            .is_ok_and(|o| o.status.success());
73        if ok {
74            Ok(Self { git })
75        } else {
76            Err(GitBinaryError::GitNotFound)
77        }
78    }
79
80    /// Fresh `git lfs` command with `GIT_TERMINAL_PROMPT=0` already set.
81    pub fn cmd(&self) -> Command {
82        let mut c = Command::new(&self.git);
83        c.arg("lfs").env(GIT_TERMINAL_PROMPT, "0");
84        c
85    }
86}

88/// Runs a prepared `git` command and returns its output, failing when git
89/// exits with a non-zero status. Failed commands can write misleading output;
90/// `git rev-parse`, for example, echoes unresolved refnames to stdout.
91fn git_output(cmd: &mut Command) -> Result<std::process::Output, GitError> {
92    let output = cmd.output()?;
93    if !output.status.success() {
94        let args = cmd
95            .get_args()
96            .map(|arg| arg.to_string_lossy())
97            .collect::<Vec<_>>()
98            .join(" ");
99        return Err(GitError::Command(
100            args,
101            String::from_utf8_lossy(&output.stderr).trim().to_string(),
102        ));
103    }
104    Ok(output)
105}

122/// A reference to commit or commit-ish.
123#[derive(
124    Debug,
125    Clone,
126    PartialEq,
127    Eq,
128    Hash,
129    PartialOrd,
130    Ord,
131    serde::Serialize,
132    serde::Deserialize,
133    Default,
134)]
135#[serde(rename_all = "kebab-case")]
136pub enum GitReference {
137    /// A specific branch.
138    Branch(String),
139    /// A specific tag.
140    Tag(String),
141    /// A specific (short) commit.
142    ShortCommit(String),
143    /// From a reference that's ambiguously a branch or tag.
144    BranchOrTag(String),
145    /// From a reference that's ambiguously a short commit, a branch, or a tag.
146    BranchOrTagOrCommit(String),
147    /// From a named reference, like `refs/pull/493/head`.
148    NamedRef(String),
149    /// From a specific revision, using a full 40-character commit hash.
150    FullCommit(String),
151    /// The default branch of the repository, the reference named `HEAD`.
152    #[default]
153    DefaultBranch,
154}
155
156impl GitReference {
157    /// Creates a [`GitReference`] from an arbitrary revision string, which could represent a
158    /// branch, tag, commit, or named ref.
159    pub fn from_rev(rev: String) -> Self {
160        if rev.starts_with("refs/") {
161            Self::NamedRef(rev)
162        } else if GitReference::looks_like_commit_hash(&rev) {
163            if rev.len() == 40 {
164                Self::FullCommit(rev)
165            } else {
166                Self::BranchOrTagOrCommit(rev)
167            }
168        } else {
169            Self::BranchOrTag(rev)
170        }
171    }
172
173    /// Converts the [`GitReference`] to a `str`.
174    pub fn as_str(&self) -> Option<&str> {
175        match self {
176            Self::Tag(rev)
177            | Self::Branch(rev)
178            | Self::ShortCommit(rev)
179            | Self::BranchOrTag(rev)
180            | Self::BranchOrTagOrCommit(rev)
181            | Self::FullCommit(rev)
182            | Self::NamedRef(rev) => Some(rev),
183            Self::DefaultBranch => None,
184        }
185    }
186
187    /// Converts the [`GitReference`] to a `str` that can be used as a revision.
188    pub(crate) fn as_rev(&self) -> &str {
189        match self {
190            Self::Tag(rev)
191            | Self::Branch(rev)
192            | Self::ShortCommit(rev)
193            | Self::BranchOrTag(rev)
194            | Self::BranchOrTagOrCommit(rev)
195            | Self::FullCommit(rev)
196            | Self::NamedRef(rev) => rev,
197            Self::DefaultBranch => "HEAD",
198        }
199    }
200
201    /// Returns the precise [`GitSha`] of this reference, if it's a full commit.
202    pub(crate) fn as_sha(&self) -> Option<GitSha> {
203        if let Self::FullCommit(rev) = self {
204            Some(GitSha::from_str(rev).expect("Full commit should be exactly 40 characters"))
205        } else {
206            None
207        }
208    }
209    /// Resolves self to an object ID with objects the `repo` currently has.
210    pub(crate) fn resolve(&self, repo: &GitRepository) -> Result<GitOid, GitError> {
211        match self {
212            // Resolve the commit pointed to by the tag.
213            //
214            // `^0` recursively peels away from the revision to the underlying commit object.
215            // This also verifies that the tag indeed refers to a commit.
216            Self::Tag(s) => repo.rev_parse(&format!("refs/remotes/origin/tags/{s}^0")),
217
218            // Resolve the commit pointed to by the branch.
219            Self::Branch(s) => repo.rev_parse(&format!("origin/{s}^0")),
220
221            // Attempt to resolve the branch, then the tag.
222            Self::BranchOrTag(s) => repo
223                .rev_parse(&format!("origin/{s}^0"))
224                .or_else(|_| repo.rev_parse(&format!("refs/remotes/origin/tags/{s}^0"))),
225
226            // Attempt to resolve the commit, the tag then the branch.
227            Self::BranchOrTagOrCommit(s) => repo
228                .rev_parse(&format!("{s}^0"))
229                .or_else(|_| repo.rev_parse(&format!("refs/remotes/origin/tags/{s}^0")))
230                .or_else(|_| repo.rev_parse(&format!("origin/{s}^0"))),
231
232            // We'll be using the HEAD commit.
233            Self::DefaultBranch => repo.rev_parse("refs/remotes/origin/HEAD"),
234
235            // Resolve a direct commit reference.
236            Self::FullCommit(s) | Self::ShortCommit(s) | Self::NamedRef(s) => {
237                repo.rev_parse(&format!("{s}^0"))
238            }
239        }
240    }
241
242    /// Whether a `rev` looks like a commit hash (ASCII hex digits).
243    pub fn looks_like_commit_hash(rev: &str) -> bool {
244        rev.len() >= 7 && rev.chars().all(|ch| ch.is_ascii_hexdigit())
245    }
246
247    /// Whether a `rev` looks like a commit hash (ASCII hex digits).
248    pub fn looks_like_full_commit_hash(rev: &str) -> bool {
249        rev.len() == 40 && rev.chars().all(|ch| ch.is_ascii_hexdigit())
250    }
251
252    pub fn is_default(&self) -> bool {
253        matches!(self, Self::DefaultBranch)
254    }
255}

257impl Display for GitReference {
258    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
259        write!(f, "{}", self.as_str().unwrap_or("HEAD"))
260    }
261}

263/// A remote repository. It gets cloned into a local [`GitDatabase`].
264#[derive(PartialEq, Clone, Debug)]
265pub(crate) struct GitRemote {
266    /// URL to a remote repository.
267    url: Url,
268}
269
270impl GitRemote {
271    /// Creates an instance for a remote repository URL.
272    pub(crate) fn new(url: &Url) -> Self {
273        Self { url: url.clone() }
274    }
275
276    /// Fetches and checkouts to a reference or a revision from this remote
277    /// into a local path.
278    ///
279    /// This ensures that it gets the up-to-date commit when a named reference
280    /// is given (tag, branch, refs/*). Thus, network connection is involved.
281    ///
282    /// When `locked_rev` is provided, it takes precedence over `reference`.
283    ///
284    /// If we have a previous instance of [`GitDatabase`] then fetch into that
285    /// if we can. If that can successfully load our revision then we've
286    /// populated the database with the latest version of `reference`, so
287    /// return that database and the rev we resolve to.
288    pub(crate) fn checkout(
289        &self,
290        into: &Path,
291        db: Option<GitDatabase>,
292        reference: &GitReference,
293        locked_rev: Option<GitOid>,
294        client: &LazyClient,
295        options: &CheckoutOptions,
296    ) -> Result<(GitDatabase, GitOid), GitError> {
297        let locked_ref = locked_rev.map(|oid| GitReference::FullCommit(oid.to_string()));
298        let reference = locked_ref.as_ref().unwrap_or(reference);
299        if let Some(mut db) = db {
300            fetch(&mut db.repo, self.url.as_str(), reference, client)?;
301
302            let resolved_commit_hash = match locked_rev {
303                Some(rev) => db.contains(rev).then_some(rev),
304                None => reference.resolve(&db.repo).ok(),
305            };
306
307            if let Some(rev) = resolved_commit_hash {
308                let ready = (options.lfs == Some(true))
309                    .then(|| {
310                        maybe_fetch_lfs(&mut db.repo, self.url.as_str(), rev, &options.lfs_filter)
311                    })
312                    .flatten();
313                return Ok((db.with_lfs_ready(ready), rev));
314            }
315        }
316
317        // Otherwise start from scratch to handle corrupt git repositories.
318        // After our fetch (which is interpreted as a clone now) we do the same
319        // resolution to figure out what we cloned.
320        if into.exists() {
321            fs_err::remove_dir_all(into)?;
322        }
323
324        fs_err::create_dir_all(into)?;
325        let mut repo = GitRepository::init(into)?;
326        fetch(&mut repo, self.url.as_str(), reference, client)?;
327        let rev = match locked_rev {
328            Some(rev) => rev,
329            None => reference.resolve(&repo).map_err(|err| {
330                let mut repository = self.url.clone();
331                let _ = repository.set_password(None);
332                let _ = repository.set_username("");
333                GitError::ReferenceNotFound {
334                    reference: reference.as_rev().to_string(),
335                    repository: repository.to_string(),
336                    source: Box::new(err),
337                }
338            })?,
339        };
340
341        let ready = (options.lfs == Some(true))
342            .then(|| maybe_fetch_lfs(&mut repo, self.url.as_str(), rev, &options.lfs_filter))
343            .flatten();
344
345        Ok((
346            GitDatabase {
347                repo,
348                lfs_ready: None,
349            }
350            .with_lfs_ready(ready),
351            rev,
352        ))
353    }
354
355    /// Creates a [`GitDatabase`] of this remote at `db_path`.
356    #[allow(clippy::unused_self)]
357    pub(crate) fn db_at(&self, db_path: &Path) -> Result<GitDatabase, GitError> {
358        let repo = GitRepository::open(db_path)?;
359        Ok(GitDatabase {
360            repo,
361            lfs_ready: None,
362        })
363    }
364
365    pub fn url(&self) -> &Url {
366        &self.url
367    }

370/// Path filters passed to Git LFS. Values use the comma-separated gitignore
371/// pattern syntax accepted by `git lfs fetch --include/--exclude`.
372#[derive(Debug, Clone, Default, PartialEq, Eq, Hash)]
373pub struct LfsFilter {
374    /// Only materialize LFS objects whose paths match these patterns.
375    pub include: Option<String>,
376    /// Do not materialize LFS objects whose paths match these patterns.
377    pub exclude: Option<String>,
378}
379
380impl LfsFilter {
381    pub fn is_empty(&self) -> bool {
382        self.include.is_none() && self.exclude.is_none()
383    }
384
385    fn configure_git(&self, command: &mut Command) {
386        if let Some(include) = &self.include {
387            command.arg("-c").arg(format!("lfs.fetchinclude={include}"));
388        }
389        if let Some(exclude) = &self.exclude {
390            command.arg("-c").arg(format!("lfs.fetchexclude={exclude}"));
391        }
392    }
393
394    fn configure_lfs_fetch(&self, command: &mut Command) {
395        if let Some(include) = &self.include {
396            command.arg(format!("--include={include}"));
397        }
398        if let Some(exclude) = &self.exclude {
399            command.arg(format!("--exclude={exclude}"));
400        }
401    }
402}

404/// Options controlling checkout behavior (submodules, LFS, etc.).
405#[derive(Debug, Clone)]
406pub struct CheckoutOptions {
407    /// Whether to recursively initialize and update submodules.
408    pub update_submodules: bool,
409
410    /// Git LFS handling, tri-state:
411    /// * `Some(true)`: fetch LFS objects into the database (`git lfs fetch`)
412    ///   and run the smudge filter during checkout so pointer files are
413    ///   materialised into real content.
414    /// * `Some(false)`: force-skip the smudge filter (`GIT_LFS_SKIP_SMUDGE=1`)
415    ///   so checkouts always contain pointer files. Callers can handle LFS
416    ///   themselves afterwards.
417    /// * `None`: LFS is not requested and the smudge filter is skipped. This
418    ///   differs from `Some(false)` only in preserving the caller's preference.
419    pub lfs: Option<bool>,
420
421    /// Optional path filters limiting which LFS objects are fetched and
422    /// materialized. Filters have no effect unless `lfs == Some(true)`.
423    pub lfs_filter: LfsFilter,
424}
425
426impl Default for CheckoutOptions {
427    fn default() -> Self {
428        Self {
429            update_submodules: true,
430            // Skipping the smudge filter is the safe default: the checkout's
431            // origin points at the local database, which has no LFS objects
432            // unless `lfs == Some(true)` fetched them.
433            lfs: Some(false),
434            lfs_filter: LfsFilter::default(),
435        }
436    }
437}

439/// A local clone of a remote repository's database. Multiple [`GitCheckout`]s
440/// can be cloned from a single [`GitDatabase`].
441pub(crate) struct GitDatabase {
442    /// Underlying Git repository instance for this database.
443    repo: GitRepository,
444    /// `Some(true)` = fsck passed, `Some(false)` = fsck failed or git-lfs
445    /// missing, `None` = LFS never requested.
446    lfs_ready: Option<bool>,
447}
448
449impl GitDatabase {
450    pub(crate) fn lfs_ready(&self) -> Option<bool> {
451        self.lfs_ready
452    }
453
454    /// Builder: set [`Self::lfs_ready`] and return self. Mirrors uv's
455    /// `with_lfs_ready` so callers can chain after a fetch or cache hit.
456    #[must_use]
457    pub(crate) fn with_lfs_ready(mut self, value: Option<bool>) -> Self {
458        self.lfs_ready = value;
459        self
460    }
461
462    /// True if the LFS objects reachable from `revision` are already present
463    /// and valid in this database (i.e. `git lfs fsck --objects` passes).
464    /// Used to decide whether a cached DB satisfies an LFS-aware request.
465    pub(crate) fn contains_lfs_artifacts(&self, revision: GitOid, filter: &LfsFilter) -> bool {
466        // A filtered fetch is cheap when its objects are already cached and is
467        // the authoritative way to apply Git LFS's path matching semantics.
468        // Do not claim a DB-only cache hit for filtered requests.
469        filter.is_empty() && self.repo.lfs_fsck_objects(revision)
470    }
471
472    /// Checkouts to a revision at `destination` from this database.
473    pub(crate) fn copy_to(
474        &self,
475        rev: GitOid,
476        destination: &Path,
477        source_url: &Url,
478        options: &CheckoutOptions,
479    ) -> Result<GitCheckout, GitError> {
480        // If the existing checkout exists, and it is fresh, use it.
481        // A non-fresh checkout can happen if the checkout operation was
482        // interrupted. In that case, the checkout gets deleted and a new
483        // clone is created.
484        let checkout = match GitRepository::open(destination)
485            .ok()
486            .map(|repo| GitCheckout::new(rev, repo))
487            .filter(|checkout| checkout.is_fresh(GIT_LFS.is_err()))
488        {
489            Some(co) => co,
490            None => GitCheckout::clone_into(destination, self, rev, source_url, options)?,
491        };
492        Ok(checkout)
493    }
494
495    /// Get a short OID for a `revision`, usually 7 chars or more if ambiguous.
496    pub(crate) fn to_short_id(&self, revision: GitOid) -> Result<String, GitError> {
497        let output = git_output(
498            Command::new(GIT.as_ref().map_err(Clone::clone)?)
499                .arg("rev-parse")
500                .arg("--short")
501                .arg(revision.as_str())
502                .current_dir(&self.repo.path),
503        )?;
504
505        let mut result = String::from_utf8(output.stdout)?;
506
507        result.truncate(result.trim_end().len());
508        tracing::debug!("result of short id is  {:?}", result);
509        Ok(result)
510    }
511
512    /// Checks if `oid` resolves to a commit in this database.
513    pub(crate) fn contains(&self, oid: GitOid) -> bool {
514        self.repo.rev_parse(&format!("{oid}^0")).is_ok()
515    }
516}

518/// A local Git repository.
519pub(crate) struct GitRepository {
520    /// Path to the underlying Git repository on the local filesystem.
521    path: PathBuf,
522}
523
524impl GitRepository {
525    /// Opens an existing Git repository at `path`.
526    ///
527    /// Returns an error if the path is not a valid git repository (e.g., missing .git directory,
528    /// corrupted repository, etc.)
529    pub(crate) fn open(path: &Path) -> Result<GitRepository, GitError> {
530        // Make sure there is a Git repository at the specified path.
531        // Use --git-dir to verify this is a valid git repository.
532        let output = Command::new(GIT.as_ref().map_err(Clone::clone)?)
533            .args(["rev-parse", "--git-dir"])
534            .current_dir(path)
535            .output()?;
536
537        if !output.status.success() {
538            return Err(GitError::InvalidRepository(path.to_path_buf()));
539        }
540
541        Ok(GitRepository {
542            path: path.to_path_buf(),
543        })
544    }
545
546    /// Initializes a Git repository at `path`.
547    fn init(path: &Path) -> Result<GitRepository, GitError> {
548        // Initialize the repository.
549        git_output(
550            Command::new(GIT.as_ref().map_err(Clone::clone)?)
551                .arg("init")
552                .current_dir(path),
553        )?;
554
555        Ok(GitRepository {
556            path: path.to_path_buf(),
557        })
558    }
559
560    /// Parses the object ID of the given `refname`.
561    fn rev_parse(&self, refname: &str) -> Result<GitOid, GitError> {
562        let result = git_output(
563            Command::new(GIT.as_ref().map_err(Clone::clone)?)
564                .arg("rev-parse")
565                .arg(refname)
566                .current_dir(&self.path),
567        )?;
568
569        let mut result = String::from_utf8(result.stdout)?;
570
571        result.truncate(result.trim_end().len());
572        result.parse().map_err(GitError::OidParse)
573    }
574
575    /// `git lfs fsck --objects <revision>`. Returns `true` iff fsck passes;
576    /// any failure (missing git-lfs, non-zero exit) is warned and returns
577    /// `false`. Informational — see [`GitDatabase::lfs_ready`].
578    fn lfs_fsck_objects(&self, revision: GitOid) -> bool {
579        let Ok(lfs) = GIT_LFS.as_ref() else {
580            return false;
581        };
582        let output = lfs
583            .cmd()
584            .arg("fsck")
585            .arg("--objects")
586            .arg(revision.as_str())
587            .env_remove(GIT_DIR)
588            .current_dir(&self.path)
589            .output();
590        match output {
591            Ok(out) if out.status.success() => true,
592            Ok(out) => {
593                tracing::warn!(
594                    "`git lfs fsck` reported problems for {revision} in {}: {}",
595                    self.path.display(),
596                    String::from_utf8_lossy(&out.stderr).trim()
597                );
598                false
599            }
600            Err(err) => {
601                tracing::warn!(
602                    "failed to run `git lfs fsck` for {revision} in {}: {err}",
603                    self.path.display()
604                );
605                false
606            }
607        }
608    }
609}

611/// A local checkout of a particular revision from a [`GitRepository`].
612pub(crate) struct GitCheckout {
613    /// The git revision this checkout is for.
614    revision: GitOid,
615    /// Underlying Git repository instance for this checkout.
616    repo: GitRepository,
617}
618
619impl GitCheckout {
620    /// Creates an instance of [`GitCheckout`]. This doesn't imply the checkout
621    /// is done. Use [`GitCheckout::is_fresh`] to check.
622    ///
623    /// * The `repo` will be the checked out Git repository.
624    fn new(revision: GitOid, repo: GitRepository) -> Self {
625        Self { revision, repo }
626    }
627
628    /// Clone a repo for a `revision` into a local path from a `database`.
629    /// This is a filesystem-to-filesystem clone.
630    fn clone_into(
631        into: &Path,
632        database: &GitDatabase,
633        revision: GitOid,
634        source_url: &Url,
635        options: &CheckoutOptions,
636    ) -> Result<Self, GitError> {
637        tracing::debug!("cloning into {:?} from {:?}", database.repo.path, into);
638        let dirname = into.parent().expect("into path must have a parent");
639        fs_err::create_dir_all(dirname)?;
640        if into.exists() {
641            fs_err::remove_dir_all(into)?;
642        }
643
644        // Perform a local clone of the repository, which will attempt to use
645        // hardlinks to set up the repository. This should speed up the clone operation
646        // quite a bit if it works.
647        //
648        // The smudge filter is controlled by `options.lfs`: when LFS was not
649        // requested we skip it, because the database the clone originates
650        // from has no LFS objects. When LFS was requested, the objects were
651        // fetched into the database beforehand, so smudging can succeed.
652        let mut clone_cmd = Command::new(GIT.as_ref().map_err(Clone::clone)?);
653        if options.lfs == Some(true) {
654            options.lfs_filter.configure_git(&mut clone_cmd);
655        }
656        clone_cmd
657            .arg("clone")
658            .arg("--local")
659            // Make sure to pass the local file path and not a file://... url. If given a url,
660            // Git treats the repository as a remote origin and gets confused because we don't
661            // have a HEAD checked out.
662            .arg(dunce::simplified(&database.repo.path).display().to_string())
663            .arg(dunce::simplified(into).display().to_string());
664        clone_cmd.env(GIT_LFS_SKIP_SMUDGE, lfs_skip_smudge_env(options.lfs));
665        let output = git_output(&mut clone_cmd)?;
666
667        tracing::debug!("output after cloning {:?}", output);
668
669        let repo = GitRepository::open(into)?;
670        let checkout = GitCheckout::new(revision, repo);
671        checkout.reset(source_url, options)?;
672        Ok(checkout)
673    }
674
675    /// Checks if the `HEAD` points to the expected revision and its ready
676    /// marker is usable. An LFS-degraded checkout is only reusable while the
677    /// current process still has no usable `git-lfs`.
678    fn is_fresh(&self, accept_lfs_degraded: bool) -> bool {
679        match self.repo.rev_parse("HEAD") {
680            Ok(id) if id == self.revision => {
681                // See comments in reset() for why we check this
682                match fs_err::read_to_string(self.repo.path.join(CHECKOUT_READY_LOCK)) {
683                    Ok(contents) => contents != CHECKOUT_LFS_DEGRADED || accept_lfs_degraded,
684                    Err(_) => false,
685                }
686            }
687            _ => false,
688        }
689    }
690
691    /// This performs `git reset --hard` to the revision of this checkout, with
692    /// additional interrupt protection by a dummy file [`CHECKOUT_READY_LOCK`].
693    ///
694    /// If we're interrupted while performing a `git reset` (e.g., we die
695    /// because of a signal) Cargo needs to be sure to try to check out this
696    /// repo again on the next go-round.
697    ///
698    /// To enable this we have a dummy file in our checkout, [`.ok`],
699    /// which if present means that the repo has been successfully reset and is
700    /// ready to go. Hence if we start to do a reset, we make sure this file
701    /// *doesn't* exist, and then once we're done we create the file.
702    ///
703    /// [`.ok`]: CHECKOUT_READY_LOCK
704    fn reset(&self, source_url: &Url, options: &CheckoutOptions) -> Result<(), GitError> {
705        let ok_file = self.repo.path.join(CHECKOUT_READY_LOCK);
706        let _ = fs_err::remove_file(&ok_file);
707
708        tracing::debug!("reset {} to {}", self.repo.path.display(), self.revision);
709
710        let skip_smudge = lfs_skip_smudge_env(options.lfs);
711
712        // Perform the hard reset. Configure the filter explicitly when LFS is
713        // enabled so materialization does not depend on a prior global
714        // `git lfs install`.
715        let mut reset_cmd = Command::new(GIT.as_ref().map_err(Clone::clone)?);
716        if options.lfs == Some(true) && GIT_LFS.is_ok() {
717            options.lfs_filter.configure_git(&mut reset_cmd);
718            reset_cmd
719                .arg("-c")
720                .arg("filter.lfs.smudge=git-lfs smudge -- %f")
721                .arg("-c")
722                .arg("filter.lfs.process=git-lfs filter-process")
723                .arg("-c")
724                .arg("filter.lfs.required=true");
725        }
726        reset_cmd
727            .arg("reset")
728            .arg("--hard")
729            .arg(self.revision.as_str())
730            .current_dir(&self.repo.path)
731            .env(GIT_LFS_SKIP_SMUDGE, skip_smudge);
732        git_output(&mut reset_cmd)?;
733
734        if options.update_submodules {
735            // The checkout's origin points to the local bare cache database
736            // (set by `git clone --local`). Submodules with relative URLs
737            // would resolve against that local path and fail. Resolve them
738            // against the real source URL first.
739            resolve_submodule_urls(&self.repo.path, source_url)?;
740
741            // Update submodules (`git submodule update --recursive`).
742            // Submodules may contain LFS files too, so apply the same smudge
743            // policy. Allow file:// protocol so local clones and file-based
744            // submodule URLs work on modern Git (>= 2.38.1).
745            let mut submodule_cmd = Command::new(GIT.as_ref().map_err(Clone::clone)?);
746            if options.lfs == Some(true) {
747                options.lfs_filter.configure_git(&mut submodule_cmd);
748            }
749            submodule_cmd
750                .args(["-c", "protocol.file.allow=always"])
751                .arg("submodule")
752                .arg("update")
753                .arg("--recursive")
754                .arg("--init")
755                .current_dir(&self.repo.path);
756            submodule_cmd.env(GIT_LFS_SKIP_SMUDGE, skip_smudge);
757            git_output(&mut submodule_cmd)?;
758        }
759
760        if options.lfs == Some(true) && GIT_LFS.is_err() {
761            fs_err::write(ok_file, CHECKOUT_LFS_DEGRADED)?;
762        } else {
763            fs_err::File::create(ok_file)?;
764        }
765        Ok(())
766    }
767}

769/// Attempts to fetch the given git `reference` for a Git repository.
770///
771/// This is the main entry for git clone/fetch. It does the following:
772///
773/// * Turns [`GitReference`] into refspecs accordingly.
774/// * Dispatches `git fetch` using the git CLI.
775///
776/// The `remote_url` argument is the git remote URL where we want to fetch from.
777pub(crate) fn fetch(
778    repo: &mut GitRepository,
779    remote_url: &str,
780    reference: &GitReference,
781    client: &LazyClient,
782) -> Result<(), GitError> {
783    let oid_to_fetch = match github_fast_path(repo, remote_url, reference, client) {
784        Ok(FastPathRev::UpToDate) => return Ok(()),
785        Ok(FastPathRev::NeedsFetch(rev)) => Some(rev),
786        Ok(FastPathRev::Indeterminate) => None,
787        Err(e) => {
788            tracing::debug!("failed to check github fast path {:?}", e);
789            None
790        }
791    };
792
793    // Translate the reference desired here into an actual list of refspecs
794    // which need to get fetched. Additionally record if we're fetching tags.
795    let mut refspecs = Vec::new();
796    let mut tags = false;
797    let mut refspec_strategy = RefspecStrategy::All;
798    // The `+` symbol on the refspec means to allow a forced (fast-forward)
799    // update which is needed if there is ever a force push that requires a
800    // fast-forward.
801    match reference {
802        // For branches and tags we can fetch simply one reference and copy it
803        // locally, no need to fetch other branches/tags.
804        GitReference::Branch(branch) => {
805            refspecs.push(format!("+refs/heads/{branch}:refs/remotes/origin/{branch}"));
806        }
807
808        GitReference::Tag(tag) => {
809            refspecs.push(format!("+refs/tags/{tag}:refs/remotes/origin/tags/{tag}"));
810        }
811
812        GitReference::BranchOrTag(branch_or_tag) => {
813            refspecs.push(format!(
814                "+refs/heads/{branch_or_tag}:refs/remotes/origin/{branch_or_tag}"
815            ));
816            refspecs.push(format!(
817                "+refs/tags/{branch_or_tag}:refs/remotes/origin/tags/{branch_or_tag}"
818            ));
819            refspec_strategy = RefspecStrategy::First;
820        }
821
822        // For ambiguous references, we can fetch the exact commit (if known); otherwise,
823        // we fetch all branches and tags.
824        GitReference::ShortCommit(branch_or_tag_or_commit)
825        | GitReference::BranchOrTagOrCommit(branch_or_tag_or_commit) => {
826            // The `oid_to_fetch` is the exact commit we want to fetch. But it could be the exact
827            // commit of a branch or tag. We should only fetch it directly if it's the exact commit
828            // of a short commit hash.
829            if let Some(oid_to_fetch) =
830                oid_to_fetch.filter(|oid| is_short_hash_of(branch_or_tag_or_commit, *oid))
831            {
832                refspecs.push(format!("+{oid_to_fetch}:refs/commit/{oid_to_fetch}"));
833            } else {
834                // We don't know what the rev will point to. To handle this
835                // situation we fetch all branches and tags, and then we pray
836                // it's somewhere in there.
837                refspecs.push(String::from("+refs/heads/*:refs/remotes/origin/*"));
838                refspecs.push(String::from("+HEAD:refs/remotes/origin/HEAD"));
839                tags = true;
840            }
841        }
842
843        GitReference::DefaultBranch => {
844            refspecs.push(String::from("+HEAD:refs/remotes/origin/HEAD"));
845        }
846
847        GitReference::NamedRef(rev) => {
848            refspecs.push(format!("+{rev}:{rev}"));
849        }
850
851        GitReference::FullCommit(rev) => {
852            if let Some(oid_to_fetch) = oid_to_fetch {
853                refspecs.push(format!("+{oid_to_fetch}:refs/commit/{oid_to_fetch}"));
854            } else {
855                // There is a specific commit to fetch and we will do so in shallow-mode only
856                // to not disturb the previous logic.
857                // Note that with typical settings for shallowing, we will just fetch a single `rev`
858                // as single commit.
859                // The reason we write to `refs/remotes/origin/HEAD` is that it's of special significance
860                // when during `GitReference::resolve()`, but otherwise it shouldn't matter.
861                refspecs.push(format!("+{rev}:refs/remotes/origin/HEAD"));
862            }
863        }
864    }
865
866    tracing::debug!(
867        "Performing a Git fetch for: {remote_url} with repo path {}",
868        repo.path.display()
869    );
870    let result = match refspec_strategy {
871        RefspecStrategy::All => fetch_with_cli(repo, remote_url, refspecs.as_slice(), tags),
872        RefspecStrategy::First => {
873            // Try each refspec
874            let mut errors = refspecs
875                .iter()
876                .map_while(|refspec| {
877                    let fetch_result =
878                        fetch_with_cli(repo, remote_url, std::slice::from_ref(refspec), tags);
879
880                    // Stop after the first success and log failures
881                    match fetch_result {
882                        Err(ref err) => {
883                            tracing::debug!("failed to fetch refspec `{refspec}`: {err}");
884                            Some(fetch_result)
885                        }
886                        Ok(()) => None,
887                    }
888                })
889                .collect::<Vec<_>>();
890
891            if errors.len() == refspecs.len() {
892                if let Some(result) = errors.pop() {
893                    // Use the last error for the message
894                    result
895                } else {
896                    // Can only occur if there were no refspecs to fetch
897                    Ok(())
898                }
899            } else {
900                Ok(())
901            }
902        }
903    };
904    tracing::debug!("fetched with cli {:?}", result);
905    result
906}

934/// `git lfs fetch <url> <revision>` then `git lfs fsck --objects <revision>`.
935/// Scoping to the resolved rev (not just HEAD) means LFS objects on feature
936/// branches and tags are fetched too. Returns the fsck result; the bool of
937/// `Ok` is `true` if fsck passes. `GIT_LFS_SKIP_SMUDGE` is removed from the
938/// env so an inherited value can't suppress smudge mid-fetch.
939fn fetch_lfs(
940    lfs: &GitLfs,
941    repo: &mut GitRepository,
942    url: &str,
943    revision: GitOid,
944    filter: &LfsFilter,
945) -> Result<bool, GitError> {
946    let remote = lfs_remote_url(url);
947    tracing::debug!("fetching LFS objects for {remote} at {revision}");
948
949    let mut command = lfs.cmd();
950    command.arg("fetch");
951    filter.configure_lfs_fetch(&mut command);
952    let output = command
953        .arg(&*remote)
954        .arg(revision.as_str())
955        .env_remove(GIT_DIR)
956        .env_remove(GIT_LFS_SKIP_SMUDGE)
957        .current_dir(&repo.path)
958        .output()?;
959    if !output.status.success() {
960        let stderr = String::from_utf8(output.stderr)?;
961        return Err(GitError::LfsFetch(remote.into_owned(), stderr));
962    }
963    tracing::debug!("git lfs fetch output: {:?}", output);
964
965    // `git lfs fsck` supports exclusions but not includes. A successful
966    // filtered fetch is therefore the best available validation for a subset;
967    // unfiltered requests retain the full object-integrity check.
968    Ok(!filter.is_empty() || repo.lfs_fsck_objects(revision))
969}

971/// The remote to pass to `git lfs fetch`. git-lfs' standalone file transfer
972/// agent only accepts a literal `file://` URL (or the name of a configured
973/// remote) — plain local paths fail with "missing protocol" and Windows
974/// drive-letter paths (`D:/repo`, parsed with a single-letter URL scheme)
975/// fail with "no valid file:// URLs found". Convert such local paths to
976/// canonical `file://` URLs; everything else passes through unchanged.
977fn lfs_remote_url(url: &str) -> std::borrow::Cow<'_, str> {
978    if let Ok(parsed) = Url::parse(url)
979        && parsed.scheme().len() == 1
980        && parsed.scheme().chars().all(|c| c.is_ascii_alphabetic())
981        && let Ok(file_url) = Url::from_file_path(Path::new(url))
982    {
983        return std::borrow::Cow::Owned(file_url.to_string());
984    }
985    std::borrow::Cow::Borrowed(url)
986}

988/// Attempts to use `git` CLI installed on the system to fetch a repository.
989fn fetch_with_cli(
990    repo: &mut GitRepository,
991    url: &str,
992    refspecs: &[String],
993    tags: bool,
994) -> Result<(), GitError> {
995    let mut cmd = Command::new(GIT.as_ref().map_err(Clone::clone)?);
996    cmd.arg("fetch");
997    if tags {
998        cmd.arg("--tags");
999    }
1000    cmd.arg("--force") // handle force pushes
1001        .arg("--update-head-ok") // see discussion in #2078
1002        .arg(url)
1003        .args(refspecs)
1004        //     // If cargo is run by git (for example, the `exec` command in `git
1005        //     // rebase`), the GIT_DIR is set by git and will point to the wrong
1006        //     // location (this takes precedence over the cwd). Make sure this is
1007        //     // unset so git will look at cwd for the repo.
1008        .env_remove(GIT_DIR)
1009        // Disable interactive credential prompts so an unreachable or
1010        // non-existent remote fails fast instead of hanging waiting for
1011        // input on the controlling TTY.
1012        .env(GIT_TERMINAL_PROMPT, "0")
1013        .current_dir(&repo.path);
1014
1015    // // We capture the output to avoid streaming it to the user's console during clones.
1016    // // The required `on...line` callbacks currently do nothing.
1017    // // The output appears to be included in error messages by default.
1018    let output = cmd.output()?;
1019    if !output.status.success() {
1020        let stderr = String::from_utf8(output.stderr)?;
1021        return Err(GitError::Fetch(url.to_string(), stderr));
1022    }
1023    tracing::debug!("git fetch output: {:?}", output);
1024    Ok(())
1025}

1027/// The result of GitHub fast path check. See [`github_fast_path`] for more.
1028enum FastPathRev {
1029    /// The local rev (determined by `reference.resolve(repo)`) is already up to
1030    /// date with what this rev resolves to on GitHub's server.
1031    UpToDate,
1032    /// The following SHA must be fetched in order for the local rev to become
1033    /// up-to-date.
1034    NeedsFetch(GitOid),
1035    /// Don't know whether local rev is up-to-date. We'll fetch _all_ branches
1036    /// and tags from the server and see what happens.
1037    Indeterminate,
1038}

1040/// Attempts GitHub's special fast path for testing if we've already got an
1041/// up-to-date copy of the repository.
1042///
1043/// Updating the index is done pretty regularly so we want it to be as fast as
1044/// possible. For registries hosted on GitHub (like the crates.io index) there's
1045/// a fast path available to use[^1] to tell us that there's no updates to be
1046/// made.
1047///
1048/// Note that this function should never cause an actual failure because it's
1049/// just a fast path. As a result, a caller should ignore `Err` returned from
1050/// this function and move forward on the normal path.
1051///
1052/// [^1]: <https://developer.github.com/v3/repos/commits/#get-the-sha-1-of-a-commit-reference>
1053fn github_fast_path(
1054    repo: &mut GitRepository,
1055    url: &str,
1056    reference: &GitReference,
1057    client: &LazyClient,
1058) -> Result<FastPathRev, GitError> {
1059    let url = Url::parse(url)?;
1060    if !is_github(&url) {
1061        return Ok(FastPathRev::Indeterminate);
1062    }
1063
1064    let local_object = reference.resolve(repo).ok();
1065    let github_branch_name = match reference {
1066        GitReference::Branch(branch) => branch,
1067        GitReference::Tag(tag) => tag,
1068        GitReference::BranchOrTag(branch_or_tag) => branch_or_tag,
1069        GitReference::DefaultBranch => "HEAD",
1070        GitReference::NamedRef(rev) => rev,
1071        GitReference::FullCommit(rev)
1072        | GitReference::ShortCommit(rev)
1073        | GitReference::BranchOrTagOrCommit(rev) => {
1074            // `revparse_single` (used by `resolve`) is the only way to turn
1075            // short hash -> long hash, but it also parses other things,
1076            // like branch and tag names, which might coincidentally be
1077            // valid hex.
1078            //
1079            // We only return early if `rev` is a prefix of the object found
1080            // by `revparse_single`. Don't bother talking to GitHub in that
1081            // case, since commit hashes are permanent. If a commit with the
1082            // requested hash is already present in the local clone, its
1083            // contents must be the same as what is on the server for that
1084            // hash.
1085            //
1086            // If `rev` is not found locally by `revparse_single`, we'll
1087            // need GitHub to resolve it and get a hash. If `rev` is found
1088            // but is not a short hash of the found object, it's probably a
1089            // branch and we also need to get a hash from GitHub, in case
1090            // the branch has moved.
1091            if let Some(ref local_object) = local_object
1092                && is_short_hash_of(rev, *local_object)
1093            {
1094                return Ok(FastPathRev::UpToDate);
1095            }
1096            rev
1097        }
1098    };
1099
1100    // This expects GitHub urls in the form `github.com/user/repo` and nothing
1101    // else
1102    let mut pieces = url.path_segments().ok_or_else(|| {
1103        GitError::GitUrlFormat(
1104            url.as_str().to_string(),
1105            "no path segments on url".to_string(),
1106        )
1107    })?;
1108    let username = pieces.next().ok_or_else(|| {
1109        GitError::GitUrlFormat(
1110            url.as_str().to_string(),
1111            "couldn't find username or organisation name".to_string(),
1112        )
1113    })?;
1114    let repository = pieces.next().ok_or_else(|| {
1115        GitError::GitUrlFormat(
1116            url.as_str().to_string(),
1117            "couldn't find repository name".to_string(),
1118        )
1119    })?;
1120    if pieces.next().is_some() {
1121        return Err(GitError::GitUrlFormat(
1122            url.as_str().to_string(),
1123            "too many segments in the url".to_string(),
1124        ));
1125    }
1126
1127    // Trim off the `.git` from the repository, if present, since that's
1128    // optional for GitHub and won't work when we try to use the API as well.
1129    let repository = repository.strip_suffix(".git").unwrap_or(repository);
1130
1131    let url = format!(
1132        "https://api.github.com/repos/{username}/{repository}/commits/{github_branch_name}"
1133    );
1134
1135    let runtime = tokio::runtime::Builder::new_current_thread()
1136        .enable_all()
1137        .build()?;
1138
1139    runtime.block_on(async move {
1140        tracing::debug!("Attempting GitHub fast path for: {url}");
1141        let mut request = client
1142            .client()
1143            .get(&url)
1144            .header("Accept", "application/vnd.github.3.sha");
1145        if let Some(local_object) = local_object {
1146            request = request.header("If-None-Match", local_object.to_string());
1147        }
1148        let mut request = request.build()?;
1149        if !request.headers().contains_key("User-Agent") {
1150            request
1151                .headers_mut()
1152                .insert("User-Agent", "rattler".parse().unwrap());
1153        }
1154
1155        let response = client.client().execute(request).await?;
1156        let response_code = response.status();
1157        if response_code == StatusCode::NOT_MODIFIED {
1158            Ok(FastPathRev::UpToDate)
1159        } else if response_code == StatusCode::OK {
1160            let oid_to_fetch = response.text().await?.parse()?;
1161            Ok(FastPathRev::NeedsFetch(oid_to_fetch))
1162        } else {
1163            // The fast path is only an optimization; any non-success status
1164            // (404 for a missing repo, 422 when the rev cannot be resolved,
1165            // 403 when rate-limited, 5xx, etc.) just falls back to a normal
1166            // git fetch.
1167            tracing::debug!("GitHub fast path returned {response_code}, falling back to git fetch");
1168            Ok(FastPathRev::Indeterminate)
1169        }
1170    })
1171}

1178/// Whether `rev` is a shorter hash of `oid`.
1179fn is_short_hash_of(rev: &str, oid: GitOid) -> bool {
1180    let long_hash = oid.to_string();
1181    match long_hash.get(..rev.len()) {
1182        Some(truncated_long_hash) => truncated_long_hash.eq_ignore_ascii_case(rev),
1183        None => false,
1184    }

1187/// Resolve a relative submodule URL against a base URL.
1188///
1189/// This mirrors Cargo's `absolute_submodule_url`: if the base URL is
1190/// parseable (http, https, file, ssh, etc.) we use `Url::join` which
1191/// handles `../` normalization. The base URL gets a trailing `/`
1192/// appended to its path so that `join` resolves relative to the
1193/// directory rather than replacing the last path segment.
1194pub fn resolve_relative_url(base: &Url, relative: &str) -> Result<String, GitError> {
1195    let mut base = base.clone();
1196
1197    // Ensure the base path ends with `/` so `join` treats it as a directory.
1198    if !base.path().ends_with('/') {
1199        base.set_path(&format!("{}/", base.path()));
1200    }
1201
1202    let resolved = base.join(relative)?;
1203    Ok(resolved.to_string())
1204}
1205
1206/// Resolve relative submodule URLs against the source URL.
1207///
1208/// Reads `.gitmodules`, finds entries with relative URLs (`./` or `../`),
1209/// resolves them against `source_url`, and writes the absolute URL into
1210/// the repo-level git config so that `git submodule update` uses it.
1211fn resolve_submodule_urls(repo_path: &Path, source_url: &Url) -> Result<(), GitError> {
1212    let gitmodules_path = repo_path.join(".gitmodules");
1213    if !gitmodules_path.exists() {
1214        return Ok(());
1215    }
1216
1217    // List all submodule URLs from .gitmodules
1218    let output = Command::new(GIT.as_ref().map_err(Clone::clone)?)
1219        .current_dir(repo_path)
1220        .args([
1221            "config",
1222            "--file",
1223            ".gitmodules",
1224            "--get-regexp",
1225            r"submodule\..*\.url",
1226        ])
1227        .output()?;
1228
1229    if !output.status.success() {
1230        // No submodule entries — nothing to resolve
1231        return Ok(());
1232    }
1233
1234    let stdout = String::from_utf8(output.stdout)?;
1235    for line in stdout.lines() {
1236        // Each line is: submodule.<name>.url <url>
1237        let Some((key, submodule_url)) = line.split_once(' ') else {
1238            continue;
1239        };
1240
1241        if !submodule_url.starts_with("./") && !submodule_url.starts_with("../") {
1242            continue;
1243        }
1244
1245        let resolved = resolve_relative_url(source_url, submodule_url)?;
1246
1247        // Write the resolved URL into the repo config (not .gitmodules).
1248        // `git submodule update --init` reads from the repo config,
1249        // falling back to .gitmodules only for `submodule init`.
1250        let output = Command::new(GIT.as_ref().map_err(Clone::clone)?)
1251            .current_dir(repo_path)
1252            .args(["config", key, &resolved])
1253            .output()?;
1254
1255        if !output.status.success() {
1256            let stderr = String::from_utf8(output.stderr)?;
1257            return Err(GitError::SubmoduleUrl(key.to_string(), stderr));
1258        }
1259    }
1260
1261    Ok(())
1262}
