24/// Parses a tri-state LFS preference from the environment variable named
25/// `var_name`. Accepts `1`/`0`, `true`/`false`, `yes`/`no`, `on`/`off`
26/// (case-insensitive). Unset/empty → `None` (LFS is not requested).
27///
28/// Callers pick the variable name (e.g. pixi uses `PIXI_GIT_LFS`) and pass
29/// the result to [`CheckoutOptions::lfs`].
30pub fn lfs_enabled_from_env(var_name: &str) -> Option<bool> {
31    let raw = std::env::var(var_name).ok()?;
32    let value = raw.trim();
33    if value.is_empty() {
34        return None;
35    }
36    if value == "0"
37        || value.eq_ignore_ascii_case("false")
38        || value.eq_ignore_ascii_case("no")
39        || value.eq_ignore_ascii_case("off")
40    {
41        return Some(false);
42    }
43    if value == "1"
44        || value.eq_ignore_ascii_case("true")
45        || value.eq_ignore_ascii_case("yes")
46        || value.eq_ignore_ascii_case("on")
47    {
48        return Some(true);
49    }
50    tracing::warn!("unrecognised value for {var_name}: {raw:?}; treating as enabled");
51    Some(true)
52}

54/// A remote Git source that can be checked out locally.
55pub struct GitSource {
56    /// The Git reference from the manifest file.
57    git: GitUrl,
58    /// The HTTP client to use for fetching.
59    client: LazyClient,
60    /// The path to the Git source database.
61    cache: PathBuf,
62    /// The reporter to use for this source.
63    reporter: Option<Arc<dyn Reporter>>,
64    /// Options controlling checkout behavior (submodules, LFS, etc.).
65    checkout_options: CheckoutOptions,
66}

68impl GitSource {
69    /// Initialize a new Git source.
70    pub fn new(git: GitUrl, client: impl Into<LazyClient>, cache: impl Into<PathBuf>) -> Self {
71        Self {
72            git,
73            client: client.into(),
74            cache: cache.into(),
75            reporter: None,
76            checkout_options: CheckoutOptions::default(),

98    /// Override the LFS preference. See [`CheckoutOptions::lfs`] for the
99    /// tri-state semantics.
100    #[must_use]
101    pub fn with_lfs(mut self, lfs: Option<bool>) -> Self {
102        self.checkout_options.lfs = lfs;
103        self
104    }
105
106    /// Limit which LFS paths are fetched and materialized. The filter only has
107    /// an effect when LFS is enabled.
108    #[must_use]
109    pub fn with_lfs_filter(mut self, filter: LfsFilter) -> Self {
110        self.checkout_options.lfs_filter = filter;
111        self
112    }

114    /// Fetch the underlying Git repository at the given revision.
115    #[instrument(skip(self), fields(repository = %self.git.repository, rev = self.git.precise.map(tracing::field::display)))]
116    pub fn fetch(self) -> Result<Fetch, GitError> {
117        // Compute the canonical URL for the repository.
118        let canonical = RepositoryUrl::new(&self.git.repository);
119
120        // The path to the repo, within the Git database.
121        let ident = cache_digest(&canonical);
122        let db_path = self.cache.join("db").join(&ident);
123
124        // Authenticate the URL, if necessary.
125        let remote = if let Some(credentials) = GIT_STORE.get(&canonical) {
126            Cow::Owned(credentials.apply(self.git.repository.clone()))
127        } else {
128            Cow::Borrowed(&self.git.repository)
129        };
130
131        let remote = GitRemote::new(&remote);
132
133        // Try to open the existing database, logging a warning if it's corrupted
134        let existing_db = match remote.db_at(&db_path) {
135            Ok(db) => Some(db),
136            Err(GitError::InvalidRepository(path)) => {
137                tracing::warn!(
138                    "Detected corrupted git cache at {} (not a valid git repository), removing and re-cloning",
139                    path.display()
140                );
141                None
142            }
143            Err(_) => None,
144        };
145
146        let lfs_requested = self.checkout_options.lfs == Some(true);
147        let (db, actual_rev, task) = match (self.git.precise, existing_db) {
148            // Cache hit: the DB has the locked revision and, if LFS was
149            // requested, its LFS objects validate. Skip the regular fetch.
150            (Some(rev), Some(db))
151                if db.contains(rev.into())
152                    && (!lfs_requested
153                        || db.contains_lfs_artifacts(
154                            rev.into(),
155                            &self.checkout_options.lfs_filter,
156                        )) =>
157            {
158                tracing::debug!(
159                    "Using existing Git source `{}` pointed at `{}`",
160                    self.git.repository,
161                    rev
162                );
163                let db = db.with_lfs_ready(lfs_requested.then_some(true));
164                (db, rev, None)
165            }
166
167            // ... otherwise we use this state to update the git database. Note
168            // that we still check for being offline here, for example in the
169            // situation that we have a locked revision but the database
170            // doesn't have it.
171            (locked_rev, db) => {
172                tracing::debug!("Updating Git source `{}`", self.git.repository);
173
174                // Report the checkout operation to the reporter.
175                let task = self.reporter.as_ref().map(|reporter| {
176                    reporter.on_checkout_start(remote.url(), self.git.reference.as_rev())
177                });
178
179                let (db, actual_rev) = remote.checkout(
180                    &db_path,
181                    db,
182                    &self.git.reference,
183                    locked_rev.map(GitOid::from),
184                    &self.client,
185                    &self.checkout_options,
186                )?;
187
188                (db, GitSha::from(actual_rev), task)
189            }
190        };
191
192        // Don’t use the full hash, in order to contribute less to reaching the
193        // path length limit on Windows.
194        let short_id = db.to_short_id(actual_rev.into())?;
195
196        // Check out `actual_rev` from the database to a scoped location on the
197        // filesystem. This will use hard links and such to ideally make the
198        // checkout operation here pretty fast. LFS-enabled checkouts contain
199        // different files than plain checkouts and must not share their path.
200        let checkout_name = if lfs_requested && !self.checkout_options.lfs_filter.is_empty() {
201            let mut hasher = DefaultHasher::new();
202            self.checkout_options.lfs_filter.hash(&mut hasher);
203            format!("{short_id}-lfs-{:x}", hasher.finish())
204        } else if lfs_requested {
205            format!("{short_id}-lfs")
206        } else {
207            short_id.clone()
208        };
209        let checkout_path = self
210            .cache
211            .join("checkouts")
212            .join(&ident)
213            .join(checkout_name);
214
215        tracing::debug!(
216            "Copying git revision `{}` to path `{}`",
217            actual_rev,
218            checkout_path.display()
219        );
220        db.copy_to(
221            actual_rev.into(),
222            &checkout_path,
223            &self.git.repository,
224            &self.checkout_options,
225        )?;
226
227        // Report the checkout operation to the reporter.
228        if let (Some(task), Some(reporter)) = (task, self.reporter.as_ref()) {
229            reporter.on_checkout_complete(remote.url(), short_id.as_str(), task);
230        }
231
232        tracing::trace!("Finished fetching Git source `{}`", self.git.repository);
233
234        Ok(Fetch {
235            repository: RepositoryReference {
236                url: canonical,
237                reference: self.git.reference.clone(),
238            },
239            commit: actual_rev,
240            path: checkout_path,
241            lfs_ready: db.lfs_ready() == Some(true),
242        })
243    }
244}

246#[derive(Debug, Clone)]
247pub struct Fetch {
248    /// The [`RepositoryReference`] reference that was fetched.
249    repository: RepositoryReference,
250
251    /// The precise git checkout
252    commit: GitSha,
253
254    /// The path to the checked-out repository.
255    path: PathBuf,
256
257    /// True iff LFS was requested for this fetch and `git lfs fsck` passed.
258    /// `false` means either LFS wasn't requested or it wasn't ready.
259    lfs_ready: bool,
260}
261
262impl Fetch {
263    pub fn repository(&self) -> &RepositoryReference {
264        &self.repository
265    }
266
267    pub fn commit(&self) -> GitSha {
268        self.commit
269    }
270
271    pub fn path(&self) -> &Path {
272        &self.path
273    }
274
275    pub fn into_path(self) -> PathBuf {
276        self.path
277    }
278
279    /// True iff LFS was requested for this fetch and the LFS objects for
280    /// the checked-out revision validated (`git lfs fsck --objects`).
281    pub fn lfs_ready(&self) -> bool {
282        self.lfs_ready
283    }
284}

286pub fn cache_digest(url: &RepositoryUrl) -> String {
287    let mut hasher = DefaultHasher::new();
288    url.hash(&mut hasher);
289    let hash = hasher.finish();
290    format!("{hash:x}")
291}

299    /// Serialised env-var swap to keep parallel tests from racing.
300    fn with_env<R>(value: Option<&str>, body: impl FnOnce() -> R) -> R {
301        use std::sync::Mutex;
302        static LOCK: Mutex<()> = Mutex::new(());
303        let _g = LOCK
304            .lock()
305            .unwrap_or_else(std::sync::PoisonError::into_inner);
306        let previous = std::env::var(TEST_LFS_ENV).ok();
307        // SAFETY: tests are serialised by LOCK above.
308        match value {
309            Some(v) => unsafe { std::env::set_var(TEST_LFS_ENV, v) },
310            None => unsafe { std::env::remove_var(TEST_LFS_ENV) },
311        }
312        let out = body();
313        match previous {
314            Some(v) => unsafe { std::env::set_var(TEST_LFS_ENV, v) },
315            None => unsafe { std::env::remove_var(TEST_LFS_ENV) },
316        }
317        out
318    }

322        with_env(None, || {
323            assert_eq!(lfs_enabled_from_env(TEST_LFS_ENV), None);

325    }
