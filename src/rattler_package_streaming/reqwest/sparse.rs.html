1//! Sparse remote access to files inside `.conda` archives.
2//!
3//! Thin wrappers around [`crate::archive::PackageArchive`] that require range
4//! request support and never fall back to a full download. For fallback
5//! behavior see [`super::fetch`]; to read multiple files from one package,
6//! use [`crate::archive::PackageArchive`] directly.
7
8use std::path::Path;

11use rattler_redaction::{DEFAULT_REDACTION_STR, redact_known_secrets_from_url};
12use reqwest_middleware::ClientWithMiddleware;
13use tracing::instrument;
14use url::Url;
15
16use crate::ExtractError;
17use crate::archive::PackageArchive;

19/// Fetch the raw bytes of a single file from a remote `.conda` package using
20/// HTTP range requests. Returns `Ok(None)` if the file is not in the archive.
21///
22/// Only the bytes needed to reach the target file are downloaded. Errors if
23/// the URL is not a `.conda` archive or the server does not support ranges.
24#[instrument(skip_all, fields(url = %redact_known_secrets_from_url(&url, DEFAULT_REDACTION_STR).as_ref().unwrap_or(&url), path = %target_path.display()))]
25pub async fn fetch_file_from_remote_sparse(
26    client: ClientWithMiddleware,
27    url: Url,
28    target_path: &Path,
29) -> Result<Option<Vec<u8>>, ExtractError> {
30    if CondaArchiveType::try_from(Path::new(url.path())) != Some(CondaArchiveType::Conda) {
31        return Err(ExtractError::UnsupportedArchiveType);
32    }
33    let archive = PackageArchive::open_sparse(client, url).await?;
34    archive.read_file(target_path).await
35}

37/// Fetch and parse a typed [`PackageFile`] from a remote `.conda` package
38/// using HTTP range requests.
39///
40/// # Example
41///
42/// ```rust,no_run
43/// # #[tokio::main]
44/// # async fn main() {
45/// use rattler_conda_types::package::IndexJson;
46/// use rattler_package_streaming::reqwest::sparse::fetch_package_file_sparse;
47/// use reqwest::Client;
48/// use reqwest_middleware::ClientWithMiddleware;
49/// use url::Url;
50///
51/// let client = ClientWithMiddleware::from(Client::new());
52/// let url = Url::parse("https://conda.anaconda.org/conda-forge/noarch/tzdata-2024b-hc8b5060_0.conda").unwrap();
53///
54/// let index_json: IndexJson = fetch_package_file_sparse(client, url).await.unwrap();
55/// println!("Package: {}", index_json.name.as_normalized());
56/// # }
57/// ```
58pub async fn fetch_package_file_sparse<P: PackageFile>(
59    client: ClientWithMiddleware,
60    url: Url,
61) -> Result<P, ExtractError> {
62    if CondaArchiveType::try_from(Path::new(url.path())) != Some(CondaArchiveType::Conda) {
63        return Err(ExtractError::UnsupportedArchiveType);
64    }
65    let archive = PackageArchive::open_sparse(client, url).await?;
66    archive.read_package_file().await
67}

81    async fn test_fetch_package_file_sparse() {
82        use rattler_conda_types::package::{AboutJson, IndexJson};
83
84        let url = test_server::serve_file(test_file()).await;
85        let client = reqwest_middleware::ClientWithMiddleware::from(reqwest::Client::new());
86
87        let index_json: IndexJson = fetch_package_file_sparse(client.clone(), url.clone())
88            .await
89            .unwrap();
90        insta::assert_yaml_snapshot!(index_json);
91
92        let about_json: AboutJson = fetch_package_file_sparse(client, url).await.unwrap();
93        insta::assert_yaml_snapshot!(about_json);
94    }
