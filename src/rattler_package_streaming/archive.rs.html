1//! Read individual files from conda packages, local or remote, with as few
2//! HTTP requests as possible.
3//!
4//! A [`PackageArchive`] is opened once and queried many times. Remote
5//! `.conda` archives on range-capable servers are opened with a single
6//! request for the archive tail (ZIP central directory and usually the whole
7//! info section); reads then cost at most one streaming ranged request per
8//! touched section, aborted once the last requested file has been read.
9//! `.tar.bz2` archives and servers without range support transparently fall
10//! back to downloading the archive once into a temporary spool file.
11//!
12//! Reads are not retried internally: a network error mid-read surfaces as an
13//! [`ExtractError`] (check [`ExtractError::should_retry`]) and the call can
14//! simply be repeated. Symbolic links inside the archive are surfaced but
15//! never followed; reading one is an error.
16//!
17//! # Example
18//!
19//! ```rust,no_run
20//! # #[tokio::main]
21//! # async fn main() {
22//! use rattler_conda_types::package::PathsJson;
23//! use rattler_package_streaming::archive::PackageArchive;
24//! use reqwest::Client;
25//! use reqwest_middleware::ClientWithMiddleware;
26//! use url::Url;
27//!
28//! let client = ClientWithMiddleware::from(Client::new());
29//! let url = Url::parse("https://conda.anaconda.org/conda-forge/linux-64/python-3.12.7-hc5c86c4_0_cpython.conda").unwrap();
30//!
31//! // One HTTP range request.
32//! let archive = PackageArchive::from_url(client, url).await.unwrap();
33//!
34//! // Usually free: the info section often sits inside the cached tail.
35//! let paths: PathsJson = archive.read_package_file().await.unwrap();
36//!
37//! // One streaming pass over the payload, aborted after the last hit.
38//! let files = archive
39//!     .read_files(paths.paths.iter().map(|entry| entry.relative_path.clone()))
40//!     .await
41//!     .unwrap();
42//! # drop(files);
43//! # }
44//! ```

70/// Bytes fetched from the end of a remote archive on open: enough for the
71/// ZIP central directory, with the surplus acting as a cache that often
72/// contains the entire info section.
73const TAIL_SIZE: u64 = 64 * 1024;
74
75/// Buffer size used for the decompression pipelines.
76const STREAM_BUF_SIZE: usize = 128 * 1024;

84/// The two sections of a conda package.
85#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
86pub enum Section {
87    /// Package metadata: everything under `info/`. Stored in the
88    /// `info-*.tar.zst` member of a `.conda` archive.
89    Info,
90    /// The package payload. Stored in the `pkg-*.tar.zst` member of a
91    /// `.conda` archive.
92    Content,
93}
94
95impl Section {
96    /// Returns the section a path inside the package belongs to.
97    pub(crate) fn containing(path: &Path) -> Section {
98        let first = path
99            .components()
100            .find(|c| !matches!(c, std::path::Component::CurDir));
101        match first {
102            Some(std::path::Component::Normal(first)) if first == "info" => Section::Info,
103            _ => Section::Content,
104        }
105    }
106
107    /// The file name prefix of the ZIP member holding this section.
108    pub(crate) fn zip_prefix(self) -> &'static str {
109        match self {
110            Section::Info => "info-",
111            Section::Content => "pkg-",
112        }
113    }
114}

116/// How a remote archive should be opened.

158/// How a [`PackageArchive`] accesses the underlying archive.
159///
160/// This is diagnostic information. Use [`RemoteArchiveOptions`] to control

162#[derive(Debug, Clone, Copy, PartialEq, Eq)]
163#[non_exhaustive]
164pub enum ArchiveAccess {
165    /// Remote archive read sparsely with HTTP range requests.
166    Sparse,
167    /// Local file on disk.

171    Spooled,

174/// Byte span of a stored ZIP member inside a `.conda` archive.
175///
176/// `end` is the offset of the next member's local header (or the central

179#[derive(Debug, Clone)]

182    /// Offset of the member's local file header.
183    header_offset: u64,
184    /// Size of the stored (uncompressed) member data.

186    /// Exclusive upper bound of the member's bytes in the archive.
187    end: u64,

318    /// Opens a remote package archive using the supplied access policy.
319    pub async fn from_url_with_options(
320        client: ClientWithMiddleware,
321        url: Url,
322        options: RemoteArchiveOptions,
323    ) -> Result<Self, ExtractError> {
324        let archive_type = CondaArchiveType::try_from(Path::new(url.path()))
325            .ok_or(ExtractError::UnsupportedArchiveType)?;
326
327        if archive_type == CondaArchiveType::Conda && options.sparse_policy != SparsePolicy::Disable
328        {
329            if let Some(archive) = Self::try_open_sparse(client.clone(), url.clone()).await? {
330                return Ok(archive);
331            }
332            if options.sparse_policy == SparsePolicy::Require {
333                return Err(ExtractError::SparseAccessUnsupported);
334            }
335        } else if options.sparse_policy == SparsePolicy::Require {
336            return Err(ExtractError::SparseAccessUnsupported);
337        }
338
339        Self::open_spooled(client, url, archive_type, options.max_spool_size).await
340    }

362    /// Returns how this handle accesses the archive.
363    pub fn access(&self) -> ArchiveAccess {
364        let temp = match &*self.backend {
365            Backend::Conda {
366                source: CondaSource::Sparse { .. },
367                ..
368            } => return ArchiveAccess::Sparse,
369            Backend::Conda {
370                source: CondaSource::Local { temp, .. },
371                ..
372            }
373            | Backend::TarBz2 { temp, .. } => temp,
374        };
375        if temp.is_some() {
376            ArchiveAccess::Spooled
377        } else {
378            ArchiveAccess::Local
379        }

382    /// Reads a single file from the package, or `None` if the path does not
383    /// exist.
384    ///
385    /// Contents are not cached: every call streams the containing section
386    /// again up to the requested file. Prefer [`PackageArchive::read_files`]
387    /// with one batch over repeated calls. Requesting a path that is a
388    /// symbolic or hard link is an error; links are not followed.
389    ///
390    /// ```rust,no_run
391    /// # #[tokio::main]
392    /// # async fn main() {
393    /// # let archive = rattler_package_streaming::archive::PackageArchive::from_path("pkg.conda").await.unwrap();
394    /// match archive.read_file("info/recipe/meta.yaml").await.unwrap() {
395    ///     Some(bytes) => println!("recipe: {}", String::from_utf8_lossy(&bytes)),
396    ///     None => println!("package has no recipe"),
397    /// }
398    /// # }
399    /// ```
400    pub async fn read_file(&self, path: impl AsRef<Path>) -> Result<Option<Vec<u8>>, ExtractError> {
401        let path = normalize(path.as_ref())?.into_owned();
402        let mut result = self.read_files([path.clone()]).await?;
403        Ok(result.remove(&path).flatten())
404    }

406    /// Reads multiple files in one pass per touched section (sections are
407    /// fetched concurrently), aborting each stream after its last requested
408    /// file. Maps every requested path to its contents, or `None` when
409    /// absent.
410    ///
411    /// Calls are independent and may run concurrently, but contents are not
412    /// cached: a repeated call streams its sections again, so batch all
413    /// needed paths into a single call where possible. Requesting a path
414    /// that is a symbolic or hard link is an error; links are not followed.
415    ///
416    /// ```rust,no_run
417    /// # #[tokio::main]
418    /// # async fn main() {
419    /// # let archive = rattler_package_streaming::archive::PackageArchive::from_path("pkg.conda").await.unwrap();
420    /// // One pass over the payload, one over info, fetched concurrently.
421    /// let files = archive
422    ///     .read_files(["info/index.json", "lib/libfoo.so", "bin/foo"])
423    ///     .await
424    ///     .unwrap();
425    /// for (path, contents) in &files {
426    ///     match contents {
427    ///         Some(bytes) => println!("{}: {} bytes", path.display(), bytes.len()),
428    ///         None => println!("{}: not in archive", path.display()),
429    ///     }
430    /// }
431    /// # }
432    /// ```
433    pub async fn read_files(
434        &self,
435        paths: impl IntoIterator<Item = impl Into<PathBuf>>,
436    ) -> Result<HashMap<PathBuf, Option<Vec<u8>>>, ExtractError> {
437        let paths: Vec<PathBuf> = paths
438            .into_iter()
439            .map(|path| {
440                let path: PathBuf = path.into();
441                normalize(&path).map(std::borrow::Cow::into_owned)
442            })
443            .collect::<Result<_, _>>()?;
444        if paths.is_empty() {
445            return Ok(HashMap::new());
446        }
447
448        // A .tar.bz2 archive is one flat tar: serve everything in a single
449        // unfiltered pass. Grouping per section here would decompress the
450        // whole bz2 stream once per section.
451        if let Backend::TarBz2 { path, .. } = &*self.backend {
452            let mut stream = Self::tar_bz2_stream(path, None).await?;
453            return scan_stream(&mut stream, paths).await;
454        }
455
456        let mut groups: HashMap<Section, Vec<PathBuf>> = HashMap::new();
457        for path in paths {
458            groups
459                .entry(Section::containing(&path))
460                .or_default()
461                .push(path);
462        }
463
464        let passes = groups.into_iter().map(|(section, group)| async move {
465            match self.stream(section).await {
466                Ok(mut stream) => scan_stream(&mut stream, group).await,
467                // A section that is absent from the archive simply does not
468                // contain any of the requested paths.
469                Err(ExtractError::MissingComponent) => {
470                    Ok(group.into_iter().map(|path| (path, None)).collect())
471                }
472                Err(err) => Err(err),
473            }
474        });
475        let results = futures::future::try_join_all(passes).await?;
476
477        Ok(results.into_iter().flatten().collect())
478    }

508    /// Lists the paths of all files (including symbolic links) in one
509    /// section.
510    ///
511    /// For [`Section::Info`] this is usually served from the cached archive
512    /// tail without extra requests. For [`Section::Content`] it streams the
513    /// entire section; prefer reading `info/paths.json` when only paths are
514    /// needed.
515    ///
516    /// ```rust,no_run
517    /// # #[tokio::main]
518    /// # async fn main() {
519    /// # let archive = rattler_package_streaming::archive::PackageArchive::from_path("pkg.conda").await.unwrap();
520    /// use rattler_package_streaming::archive::Section;
521    ///
522    /// // Usually free: the info section tends to sit in the cached tail.
523    /// for path in archive.list_files(Section::Info).await.unwrap() {
524    ///     println!("{}", path.display());
525    /// }
526    /// # }
527    /// ```
528    pub async fn list_files(&self, section: Section) -> Result<Vec<PathBuf>, ExtractError> {
529        let mut stream = self.stream(section).await?;
530        let mut paths = Vec::new();
531        while let Some(entry) = stream.next_entry().await? {
532            if matches!(
533                entry.kind(),
534                ArchiveEntryKind::File | ArchiveEntryKind::Symlink | ArchiveEntryKind::Hardlink
535            ) {
536                paths.push(entry.path().to_owned());
537            }
538        }
539        Ok(paths)
540    }

542    /// Streams the tar entries of one section. Unread entries are skipped
543    /// cheaply; dropping the stream aborts any underlying HTTP transfer.
544    ///
545    /// Every call opens a new independent forward-only stream (for remote
546    /// archives: a new request).
547    ///
548    /// ```rust,no_run
549    /// # #[tokio::main]
550    /// # async fn main() {
551    /// # let archive = rattler_package_streaming::archive::PackageArchive::from_path("pkg.conda").await.unwrap();
552    /// use rattler_package_streaming::archive::Section;
553    ///
554    /// let mut stream = archive.stream(Section::Content).await.unwrap();
555    /// while let Some(mut entry) = stream.next_entry().await.unwrap() {
556    ///     let path = entry.path().to_owned();
557    ///     if path.extension().is_some_and(|ext| ext == "so") {
558    ///         let bytes = entry.read().await.unwrap();
559    ///         println!("{}: {} bytes", path.display(), bytes.len());
560    ///     } // entries that are not read are skipped cheaply
561    /// }
562    /// # }
563    /// ```
564    pub async fn stream(&self, section: Section) -> Result<SectionStream, ExtractError> {
565        match &*self.backend {
566            Backend::Conda { source, members } => {
567                let span = find_section_member(members, section)?;
568                let raw = Self::conda_member_reader(source, span).await?;
569                let decoder =
570                    ZstdDecoder::new(tokio::io::BufReader::with_capacity(STREAM_BUF_SIZE, raw));
571                Ok(SectionStream::new(Box::new(decoder), None))
572            }
573            // `read_files` bypasses the filter with `tar_bz2_stream(None)`
574            // to serve both sections from a single pass.
575            Backend::TarBz2 { path, .. } => Self::tar_bz2_stream(path, Some(section)).await,
576        }
577    }

583    /// Opens a remote `.conda` archive sparsely, or `None` when the server

585    /// fall back to a full download.
586    pub(crate) async fn try_open_sparse(

591            Ok(archive) => Ok(Some(archive)),

594                Ok(None)
595            }
596            Err(err) => Err(err),
597        }

600    /// Opens a remote `.conda` archive sparsely, without full-download fallback.
601    pub(crate) async fn open_sparse(
602        client: ClientWithMiddleware,
603        url: Url,
604    ) -> Result<Self, ExtractError> {
605        // One suffix range request: fetches the last TAIL_SIZE bytes and
606        // reveals the total archive size.
607        let (reader, headers) = AsyncHttpRangeReader::new(
608            client.clone(),
609            url.clone(),
610            CheckSupportMethod::NegativeRangeRequest(TAIL_SIZE),
611            HeaderMap::default(),
612        )
613        .await?;
614        // A weak ETag must not be sent in `If-Range` (RFC 9110 §13.1.5);
615        // fall back to `Last-Modified` in that case.
616        let validator = headers
617            .get(ETAG)
618            .filter(|v| !v.as_bytes().starts_with(b"W/"))
619            .or_else(|| headers.get(LAST_MODIFIED))
620            .cloned();
621        let size = reader.len();
622        debug!("opened remote archive ({size} bytes) with a {TAIL_SIZE} byte tail request");
623
624        // Parse the central directory. The needed bytes are already cached
625        // from the tail request; if the central directory is unusually large
626        // the range reader transparently fetches the difference.
627        let buf_reader = futures::io::BufReader::new(reader.compat());
628        let zip = ZipFileReader::new(buf_reader).await?;
629        let members = collect_members(zip.file(), size)?;
630
631        // Recover the range reader and keep a copy of the tail bytes so
632        // members that live inside the tail (usually the info section) can be
633        // served without further requests.
634        let mut reader = zip.into_inner().into_inner().into_inner();
635        let tail_offset = size.saturating_sub(TAIL_SIZE);
636        let mut tail = vec![0u8; (size - tail_offset) as usize];
637        reader.seek(SeekFrom::Start(tail_offset)).await?;
638        reader.read_exact(&mut tail).await?;
639
640        Ok(Self {
641            backend: Arc::new(Backend::Conda {
642                source: CondaSource::Sparse {
643                    client,
644                    url,
645                    validator,
646                    tail_offset,
647                    tail: tail.into(),
648                },
649                members,
650            }),
651        })
652    }

654    async fn open_spooled(
655        client: ClientWithMiddleware,
656        url: Url,
657        archive_type: CondaArchiveType,
658        max_spool_size: Option<u64>,
659    ) -> Result<Self, ExtractError> {
660        let response = client
661            .get(url.clone())
662            .send()
663            .await?
664            .error_for_status()
665            .map_err(|e| ExtractError::ReqwestError(e.into()))?;
666
667        if let (Some(limit), Some(content_length)) = (max_spool_size, response.content_length())
668            && content_length > limit
669        {
670            return Err(ExtractError::SpoolLimitExceeded { limit });
671        }
672
673        // Spool to disk rather than memory: packages can be arbitrarily
674        // large (multi-GB), so an in-memory copy is not an option.
675        let temp = tempfile::NamedTempFile::new()?;
676        let (file, temp_path) = temp.into_parts();
677        let mut file = tokio::fs::File::from_std(file);
678        let body = StreamReader::new(response.bytes_stream().map_err(std::io::Error::other));
679        let copied = if let Some(limit) = max_spool_size {
680            let mut body = body.take(limit.saturating_add(1));
681            tokio::io::copy(&mut body, &mut file).await?
682        } else {
683            let mut body = body;
684            tokio::io::copy(&mut body, &mut file).await?
685        };
686        if let Some(limit) = max_spool_size
687            && copied > limit
688        {
689            return Err(ExtractError::SpoolLimitExceeded { limit });
690        }
691        file.flush().await?;
692
693        Self::open_local(temp_path.to_path_buf(), archive_type, Some(temp_path)).await
694    }
695
696    async fn open_local(
697        path: PathBuf,
698        archive_type: CondaArchiveType,
699        temp: Option<tempfile::TempPath>,
700    ) -> Result<Self, ExtractError> {
701        let backend = match archive_type {
702            CondaArchiveType::Conda => {
703                let file = tokio::fs::File::open(&path).await?;
704                let size = file.metadata().await?.len();
705                let buf_reader =
706                    futures::io::BufReader::new(tokio::io::BufReader::new(file).compat());
707                let zip = ZipFileReader::new(buf_reader).await?;
708                let members = collect_members(zip.file(), size)?;
709                Backend::Conda {
710                    source: CondaSource::Local { path, temp },
711                    members,
712                }
713            }
714            CondaArchiveType::TarBz2 => Backend::TarBz2 { path, temp },
715        };
716        Ok(Self {
717            backend: Arc::new(backend),
718        })
719    }

725    /// Returns a reader over the stored bytes of a ZIP member.
726    async fn conda_member_reader(
727        source: &CondaSource,
728        span: &MemberSpan,
729    ) -> Result<DynReader, ExtractError> {
730        match source {
731            CondaSource::Sparse {
732                client,
733                url,
734                validator,
735                tail_offset,
736                tail,
737            } => {
738                // Serve from the cached tail when the whole member is inside it.
739                if span.header_offset >= *tail_offset {
740                    let rel = (span.header_offset - tail_offset) as usize;
741                    if let Some(range) = member_data_range(&tail[rel..], span.size) {
742                        debug!("serving member {} from the cached tail", span.name);
743                        let data = tail.slice(rel + range.start..rel + range.end);
744                        return Ok(Box::new(std::io::Cursor::new(data)));
745                    }
746                }
747
748                // One bounded streaming ranged GET for the member. Dropping
749                // the returned reader aborts the transfer.
750                debug!(
751                    "requesting range {}-{} for member {}",
752                    span.header_offset,
753                    span.end - 1,
754                    span.name
755                );
756                let mut request = client
757                    .get(url.clone())
758                    .header(
759                        RANGE,
760                        format!("bytes={}-{}", span.header_offset, span.end - 1),
761                    )
762                    // Forbid content-coding: byte math relies on the exact
763                    // stored representation.
764                    .header(http::header::ACCEPT_ENCODING, "identity");
765                if let Some(validator) = validator {
766                    request = request.header(IF_RANGE, validator);
767                }
768                let response = request
769                    .send()
770                    .await?
771                    .error_for_status()
772                    .map_err(|e| ExtractError::ReqwestError(e.into()))?;
773                if response.status() != ::reqwest::StatusCode::PARTIAL_CONTENT {
774                    // An honored `If-Range` mismatch (the archive changed since
775                    // it was opened) or a server that stopped honoring ranges.
776                    return Err(ExtractError::RemoteArchiveChanged);
777                }
778                let mut reader =
779                    StreamReader::new(response.bytes_stream().map_err(std::io::Error::other));
780                skip_local_header(&mut reader).await?;
781                Ok(Box::new(reader.take(span.size)))
782            }
783            CondaSource::Local { path, .. } => {
784                let mut file = tokio::fs::File::open(path).await?;
785                file.seek(SeekFrom::Start(span.header_offset)).await?;
786                let mut reader = tokio::io::BufReader::new(file);
787                skip_local_header(&mut reader).await?;
788                Ok(Box::new(reader.take(span.size)))
789            }
790        }
791    }

804/// A streaming view over the tar entries of one package section.
805pub struct SectionStream {
806    entries: tokio_tar::Entries<DynReader>,
807    /// For `.tar.bz2` archives (one flat tar), entries are filtered to the
808    /// requested section. `None` yields every entry.
809    filter: Option<Section>,
810}
811
812impl SectionStream {
813    fn new(reader: DynReader, filter: Option<Section>) -> Self {
814        let mut archive = tokio_tar::Archive::new(reader);
815        let entries = archive
816            .entries()
817            .expect("entries() cannot fail on a fresh archive");
818        Self { entries, filter }
819    }
820
821    /// Advances to the next tar entry of the section, or `None` at the end of
822    /// the section.
823    pub async fn next_entry(&mut self) -> Result<Option<SectionEntry>, ExtractError> {
824        use futures_util::StreamExt;
825        while let Some(entry) = self.entries.next().await {
826            let entry = entry?;
827            let path = {
828                let path = entry.path()?;
829                normalize(&path)?.into_owned()
830            };
831            if let Some(section) = self.filter
832                && Section::containing(&path) != section
833            {
834                continue;
835            }
836            return Ok(Some(SectionEntry { inner: entry, path }));
837        }
838        Ok(None)
839    }
840}

842/// Reads the requested paths out of a section stream, aborting as soon as the
843/// last one has been found.
844async fn scan_stream(
845    stream: &mut SectionStream,
846    paths: Vec<PathBuf>,
847) -> Result<HashMap<PathBuf, Option<Vec<u8>>>, ExtractError> {
848    let mut remaining: HashSet<PathBuf> = paths.into_iter().collect();
849    let mut out = HashMap::with_capacity(remaining.len());
850    let mut links: Vec<String> = Vec::new();
851    while !remaining.is_empty() {
852        let Some(mut entry) = stream.next_entry().await? else {
853            break;
854        };
855        let path = entry.path().to_owned();
856        if remaining.remove(&path) {
857            // Finish the scan so the error names every offending link
858            // instead of discarding the whole batch on the first one.
859            if let Some(link) = describe_link(&entry)? {
860                links.push(link);
861                continue;
862            }
863            let buf = entry.read().await?;
864            out.insert(path, Some(buf));
865        }
866    }
867    if !links.is_empty() {
868        return Err(ExtractError::LinksNotFollowed(links));
869    }
870    for path in remaining {
871        out.insert(path, None);
872    }
873    Ok(out)
874}

876/// Collects the member spans of a `.conda` ZIP archive from its parsed
877/// central directory. The exclusive end bound of each member is the offset of
878/// the next member (or the end of the archive), which over-approximates by at
879/// most the size of the central directory for the last member.
880fn collect_members(
881    zip: &async_zip::ZipFile,
882    archive_size: u64,
883) -> Result<Vec<MemberSpan>, ExtractError> {
884    let entries = zip.entries();
885    let mut members = Vec::with_capacity(entries.len());
886    for entry in entries {
887        let name = entry
888            .filename()
889            .as_str()
890            .map_err(|e| {
891                ExtractError::IoError(std::io::Error::new(std::io::ErrorKind::InvalidData, e))
892            })?
893            .to_owned();
894        if name.ends_with(".tar.zst") && entry.compression() != Compression::Stored {
895            return Err(ExtractError::UnsupportedCompressionMethod);
896        }
897        members.push(MemberSpan {
898            name,
899            header_offset: entry.header_offset(),
900            size: entry.compressed_size(),
901            end: archive_size,
902        });
903    }
904    // Bound each member by the next member's local header offset.
905    members.sort_unstable_by_key(|m| m.header_offset);
906    for i in 1..members.len() {
907        members[i - 1].end = members[i].header_offset;
908    }
909    Ok(members)
910}
911
912fn find_section_member(
913    members: &[MemberSpan],
914    section: Section,
915) -> Result<&MemberSpan, ExtractError> {
916    let prefix = section.zip_prefix();
917    members
918        .iter()
919        .find(|m| m.name.starts_with(prefix) && m.name.ends_with(".tar.zst"))
920        .ok_or(ExtractError::MissingComponent)
921}

923/// Describes a link entry for [`ExtractError::LinksNotFollowed`], or `None`
924/// for regular entries.
925fn describe_link(entry: &SectionEntry) -> Result<Option<String>, ExtractError> {
926    if !entry.kind().is_link() {
927        return Ok(None);
928    }
929    let target = entry
930        .link_target()?
931        .map(|target| target.display().to_string())
932        .unwrap_or_default();
933    Ok(Some(format!(
934        "'{}' (links to '{target}')",
935        entry.path().display()
936    )))
937}

939/// Reads the contents of a raw tar entry while capping the upfront allocation
940/// derived from its untrusted header.
941pub(crate) async fn read_raw_entry_contents<R: AsyncRead + Unpin>(
942    entry: &mut tokio_tar::Entry<R>,
943) -> Result<Vec<u8>, ExtractError> {
944    let kind = entry.header().entry_type();
945    if kind.is_symlink() || kind.is_hard_link() {
946        let path = normalize(&entry.path()?)?.into_owned();
947        let target = entry
948            .link_name()?
949            .map(std::borrow::Cow::into_owned)
950            .unwrap_or_default();
951        return Err(ExtractError::LinksNotFollowed(vec![format!(
952            "'{}' (links to '{}')",
953            path.display(),
954            target.display()
955        )]));
956    }
957
958    let size = entry.header().size()?;
959    let mut buf = Vec::with_capacity(size.min(MAX_PREALLOC) as usize);
960    entry.read_to_end(&mut buf).await?;
961    Ok(buf)
962}

970/// Validates a package-relative path and strips `.` components.
971///
972/// Package paths may not be empty, absolute, or contain parent components.
973pub(crate) fn normalize(path: &Path) -> Result<std::borrow::Cow<'_, Path>, ExtractError> {
974    let mut needs_normalization = false;
975    let mut has_component = false;
976    for component in path.components() {
977        match component {
978            std::path::Component::Normal(_) => has_component = true,
979            std::path::Component::CurDir => needs_normalization = true,
980            std::path::Component::ParentDir
981            | std::path::Component::RootDir
982            | std::path::Component::Prefix(_) => {
983                return Err(ExtractError::InvalidArchivePath(path.to_owned()));
984            }
985        }
986    }
987    if !has_component {
988        return Err(ExtractError::InvalidArchivePath(path.to_owned()));
989    }
990    if needs_normalization {
991        Ok(std::borrow::Cow::Owned(
992            path.components()
993                .filter(|component| !matches!(component, std::path::Component::CurDir))
994                .collect(),
995        ))
996    } else {
997        Ok(std::borrow::Cow::Borrowed(path))
998    }
999}

1001/// Parses a ZIP local file header at the start of `buf` and returns the
1002/// range of the member data if `buf` contains all of it.
1003fn member_data_range(buf: &[u8], size: u64) -> Option<std::ops::Range<usize>> {
1004    if buf.len() < 30 || buf[0..4] != LOCAL_HEADER_MAGIC {
1005        return None;
1006    }
1007    let name_len = u16::from_le_bytes([buf[26], buf[27]]) as usize;
1008    let extra_len = u16::from_le_bytes([buf[28], buf[29]]) as usize;
1009    let data_start = 30 + name_len + extra_len;
1010    let data_end = data_start.checked_add(size as usize)?;
1011    (data_end <= buf.len()).then_some(data_start..data_end)
1012}
1013
1014/// Reads and skips a ZIP local file header from a stream, leaving the reader
1015/// positioned at the start of the member data.
1016async fn skip_local_header<R: AsyncRead + Unpin>(reader: &mut R) -> Result<(), ExtractError> {
1017    let mut header = [0u8; 30];
1018    reader.read_exact(&mut header).await?;
1019    if header[0..4] != LOCAL_HEADER_MAGIC {
1020        return Err(ExtractError::IoError(std::io::Error::new(
1021            std::io::ErrorKind::InvalidData,
1022            "expected a ZIP local file header",
1023        )));
1024    }
1025    let name_len = u64::from(u16::from_le_bytes([header[26], header[27]]));
1026    let extra_len = u64::from(u16::from_le_bytes([header[28], header[29]]));
1027    let mut skip = reader.take(name_len + extra_len);
1028    tokio::io::copy(&mut skip, &mut tokio::io::sink()).await?;
1029    Ok(())
1030}

1032/// Returns true for errors that mean "sparse access is unavailable" and the
1033/// caller should fall back to a full download.
1034fn sparse_unsupported(err: &ExtractError) -> bool {
1035    match err {
1036        // Servers that ignore the `Range` header answer with a plain `200 OK`
1037        // that carries no `Content-Range` header.
1038        ExtractError::AsyncHttpRangeReaderError(
1039            AsyncHttpRangeReaderError::HttpRangeRequestUnsupported
1040            | AsyncHttpRangeReaderError::ContentRangeMissing,
1041        ) => true,
1042        // JFrog Artifactory returns 416 when querying more than the object length.
1043        ExtractError::AsyncHttpRangeReaderError(AsyncHttpRangeReaderError::HttpError(err)) => {
1044            err.status() == Some(::reqwest::StatusCode::RANGE_NOT_SATISFIABLE)
1045        }
1046        _ => false,
1047    }
1048}
