1//! Functions that enable extracting or streaming a Conda package from local
2//! files in an async context. Extraction runs on a blocking worker thread
3//! with plain file I/O, which is the fastest path for a file that is already
4//! on disk.
5
6use crate::{ExtractError, ExtractResult};
7use rattler_conda_types::package::CondaArchiveType;
8use std::path::Path;
9
10/// Runs a blocking extractor on a worker thread.
11async fn spawn_extract(
12    archive: &Path,
13    destination: &Path,
14    extract: fn(&Path, &Path) -> Result<ExtractResult, ExtractError>,
15) -> Result<ExtractResult, ExtractError> {
16    let archive = archive.to_owned();
17    let destination = destination.to_owned();
18    match tokio::task::spawn_blocking(move || extract(&archive, &destination)).await {
19        Ok(result) => result,
20        Err(err) => {
21            if let Ok(reason) = err.try_into_panic() {
22                std::panic::resume_unwind(reason);
23            }
24            Err(ExtractError::Cancelled)
25        }
26    }
27}

29/// Extracts the contents a `.tar.bz2` package archive at the specified path to a directory.
30///
31/// ```rust,no_run
32/// # #[tokio::main]
33/// # async fn main() {
34/// # use std::path::Path;
35/// use rattler_package_streaming::tokio::fs::extract_tar_bz2;
36/// let _ = extract_tar_bz2(
37///     Path::new("conda-forge/win-64/python-3.11.0-hcf16a7b_0_cpython.tar.bz2"),
38///     Path::new("/tmp"))
39///     .await
40///     .unwrap();
41/// # }
42/// ```
43pub async fn extract_tar_bz2(
44    archive: &Path,
45    destination: &Path,
46) -> Result<ExtractResult, ExtractError> {
47    spawn_extract(archive, destination, crate::fs::extract_tar_bz2).await
48}

71/// Extracts the contents a package archive at the specified path to a directory. The type of
72/// package is determined based on the file extension of the archive path.
73///
74/// ```rust,no_run
75/// # #[tokio::main]
76/// # async fn main() {
77/// # use std::path::Path;
78/// use rattler_package_streaming::tokio::fs::extract;
79/// let _ = extract(
80///     Path::new("conda-forge/win-64/python-3.11.0-hcf16a7b_0_cpython.conda"),
81///     Path::new("/tmp"))
82///     .await
83///     .unwrap();
84/// # }
85/// ```
86pub async fn extract(archive: &Path, destination: &Path) -> Result<ExtractResult, ExtractError> {
87    match CondaArchiveType::try_from(archive).ok_or(ExtractError::UnsupportedArchiveType)? {
88        CondaArchiveType::TarBz2 => extract_tar_bz2(archive, destination).await,
89        CondaArchiveType::Conda => extract_conda(archive, destination).await,
90    }
