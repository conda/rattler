1//! RECORD generation for the cache layout, not final installed paths.
2
3use std::{io, path::Path};
4
5use base64::{Engine, engine::general_purpose::URL_SAFE_NO_PAD};
6use rattler_conda_types::package::PathsEntry;
7
8use super::SITE_PACKAGES;
9use crate::ExtractError;
10
11/// Builds RECORD relative to the cache's `site-packages`, not an installed prefix.
12pub(super) fn for_cache(paths: &[PathsEntry], dist_info: &Path) -> Result<Vec<u8>, ExtractError> {
13    let mut writer = csv::WriterBuilder::new()
14        .has_headers(false)
15        .from_writer(Vec::new());
16    for entry in paths {
17        let relative = match entry.relative_path.strip_prefix(SITE_PACKAGES) {
18            Ok(path) => posix_path(path),
19            Err(_) => format!("../{}", posix_path(&entry.relative_path)),
20        };
21        let hash = entry
22            .sha256
23            .map(|hash| format!("sha256={}", URL_SAFE_NO_PAD.encode(hash)))
24            .unwrap_or_default();
25        writer
26            .serialize((relative, hash, entry.size_in_bytes))
27            .map_err(io::Error::other)?;
28    }
29    writer
30        .write_record([
31            format!("{}/RECORD", posix_path(dist_info)),
32            String::new(),
33            String::new(),
34        ])
35        .map_err(io::Error::other)?;
36    writer
37        .into_inner()
38        .map_err(|err| io::Error::other(err).into())
39}
40
41fn posix_path(path: &Path) -> String {
42    path.components()
43        .map(|component| component.as_os_str().to_string_lossy())
44        .collect::<Vec<_>>()
45        .join("/")
46}
