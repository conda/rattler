10/// The records of each package, grouped by their [Version]
11/// on first use.
12#[derive(Default)]
13pub(super) struct VersionIndex<'a> {
14    group_of_solvable: Vec<u32>,
15    /// Ranges never overlap, so other packages' candidates are matched in full.
16    groups_of_package: HashMap<NameId, Range<u32>>,
17    versions: Vec<&'a Version>,
18    group_matches: Vec<bool>,
19}
20
21const NO_GROUP: u32 = u32::MAX;
22
23impl<'a> VersionIndex<'a> {
24    /// Returns the `candidates` that match `spec`, or the others if `inverse` is
25    /// set. The version spec is tested once per group of `name`; candidates in a
26    /// passing group, or in none (like virtual packages), are then matched in full.
27    pub(super) fn filter(
28        &mut self,
29        pool: &Pool<SolverMatchSpec<'a>, NameType>,
30        records: &HashMap<NameId, Candidates>,
31        name: NameId,
32        candidates: &[SolvableId],
33        spec: &NamelessMatchSpec,
34        inverse: bool,
35    ) -> Vec<SolvableId> {
36        let groups = match &spec.version {
37            Some(version_spec) if !matches!(version_spec, VersionSpec::Any) => {
38                let groups = self.package_groups(pool, records, name);
39                let versions = &self.versions[groups.start as usize..groups.end as usize];
40                self.group_matches.clear();
41                self.group_matches
42                    .extend(versions.iter().map(|version| version_spec.matches(version)));
43                groups
44            }
45            _ => 0..0,
46        };
47
48        candidates
49            .iter()
50            .copied()
51            .filter(|&candidate| {
52                let group = self.group_of_solvable.get(candidate.to_index()).copied();
53                let fails_version = group.is_some_and(|group| {
54                    groups.contains(&group) && !self.group_matches[(group - groups.start) as usize]
55                });
56                (!fails_version && matches_one_by_one(pool, candidate, spec)) != inverse
57            })
58            .collect()
59    }
60
61    fn package_groups(
62        &mut self,
63        pool: &Pool<SolverMatchSpec<'a>, NameType>,
64        records: &HashMap<NameId, Candidates>,
65        name: NameId,
66    ) -> Range<u32> {
67        if let Some(groups) = self.groups_of_package.get(&name) {
68            return groups.clone();
69        }
70
71        let first_group = self.versions.len() as u32;
72        let mut group_of_versions = HashMap::new();
73        for &solvable in records.get(&name).map_or(&[][..], |c| &c.candidates) {
74            let SolverPackageRecord::Record(record) = pool.resolve_solvable(solvable).record else {
75                continue;
76            };
77            let version = record.package_record.version.version();
78            let group = *group_of_versions.entry(version).or_insert_with(|| {
79                self.versions.push(version);
80                self.versions.len() as u32 - 1
81            });
82
83            let index = solvable.to_index();
84            if index >= self.group_of_solvable.len() {
85                self.group_of_solvable.resize(index + 1, NO_GROUP);
86            }
87            self.group_of_solvable[index] = group;
88        }
89
90        let groups = first_group..self.versions.len() as u32;
91        self.groups_of_package.insert(name, groups.clone());
92        groups
93    }
94}
95
96fn matches_one_by_one(
97    pool: &Pool<SolverMatchSpec<'_>, NameType>,
98    candidate: SolvableId,
99    spec: &NamelessMatchSpec,
100) -> bool {
101    match &pool.resolve_solvable(candidate).record {
102        SolverPackageRecord::Record(record) => spec.matches(*record),
103        SolverPackageRecord::VirtualPackage(GenericVirtualPackage {
104            version,
105            build_string,
106            ..
107        }) => {
108            spec.version.as_ref().is_none_or(|v| v.matches(version))
109                && spec
110                    .build
111                    .as_ref()
112                    .is_none_or(|b| b.matches(build_string.as_str()))
113        }
114        SolverPackageRecord::Extra { .. } => {
115            unreachable!("extras should never be compared to matchspecs")
116        }
117    }
118}
