1//! Classification of package files by archive format, shared by every
2//! repodata source.
3//!
4//! Files that share an
5//! [`ArchiveIdentifier`](rattler_conda_types::package::ArchiveIdentifier)
6//! (the `name-version-build` stem) are the same build in different archive
7//! formats. After removed files are dropped and `v3` entries have replaced
8//! legacy entries of the same file, every remaining file of a package falls
9//! into exactly one [`FormatBucket`]. A [`PackageFormatSelection`] is a fixed
10//! set of buckets, so every source that classifies its files this way returns
11//! the same records for the same selection.
12

25/// The class of a package file, determined by its archive format and by which
26/// other formats of the same build are available.
27#[derive(Debug, Clone, Copy, PartialEq, Eq)]
28pub(crate) enum FormatBucket {
29    /// A `.conda` file.
30    Conda,
31    /// A `.whl` file of a build that has no `.conda` file.
32    WhlAlone,
33    /// A `.whl` file of a build that also has a `.conda` file.
34    WhlBehindConda,
35    /// A `.tar.bz2` file of a build that has neither a `.conda` nor a `.whl`
36    /// file.
37    TarAlone,
38    /// A `.tar.bz2` file of a build that has a `.whl` but no `.conda` file.
39    TarBehindWhl,
40    /// A `.tar.bz2` file of a build that also has a `.conda` file.
41    TarBehindConda,
42}
43
44impl FormatBucket {
45    /// Every bucket, in the order used by [`FormatBucketMap`].
46    pub(crate) const ALL: [FormatBucket; 6] = [
47        FormatBucket::Conda,
48        FormatBucket::WhlAlone,
49        FormatBucket::WhlBehindConda,
50        FormatBucket::TarAlone,
51        FormatBucket::TarBehindWhl,
52        FormatBucket::TarBehindConda,
53    ];
54
55    /// Classifies a file with the given archive type. `has_twin` reports
56    /// whether the same build is also available in another archive type; it
57    /// is only consulted for the formats that matter for this file.
58    pub(crate) fn classify(
59        archive_type: DistArchiveType,
60        mut has_twin: impl FnMut(DistArchiveType) -> bool,
61    ) -> Self {
62        let conda = DistArchiveType::Conda(CondaArchiveType::Conda);
63        let whl = DistArchiveType::Wheel(WheelArchiveType::Whl);
64        match archive_type {
65            DistArchiveType::Conda(CondaArchiveType::Conda) => FormatBucket::Conda,
66            DistArchiveType::Wheel(WheelArchiveType::Whl) => {
67                if has_twin(conda) {
68                    FormatBucket::WhlBehindConda
69                } else {
70                    FormatBucket::WhlAlone
71                }
72            }
73            DistArchiveType::Conda(CondaArchiveType::TarBz2) => {
74                if has_twin(conda) {
75                    FormatBucket::TarBehindConda
76                } else if has_twin(whl) {
77                    FormatBucket::TarBehindWhl
78                } else {
79                    FormatBucket::TarAlone
80                }
81            }
82        }
83    }
84
85    /// The archive type of every file in this bucket.
86    pub(crate) fn archive_type(self) -> DistArchiveType {
87        match self {
88            FormatBucket::Conda => CondaArchiveType::Conda.into(),
89            FormatBucket::WhlAlone | FormatBucket::WhlBehindConda => WheelArchiveType::Whl.into(),
90            FormatBucket::TarAlone | FormatBucket::TarBehindWhl | FormatBucket::TarBehindConda => {
91                CondaArchiveType::TarBz2.into()
92            }
93        }
94    }
95
96    /// The position of this bucket in [`Self::ALL`].
97    const fn index(self) -> usize {
98        match self {
99            FormatBucket::Conda => 0,
100            FormatBucket::WhlAlone => 1,
101            FormatBucket::WhlBehindConda => 2,
102            FormatBucket::TarAlone => 3,
103            FormatBucket::TarBehindWhl => 4,
104            FormatBucket::TarBehindConda => 5,
105        }
106    }
107}

109impl PackageFormatSelection {
110    /// The buckets whose files this selection uses.
111    pub(crate) fn buckets(self) -> FormatBucketSet {
112        match self {
113            PackageFormatSelection::OnlyConda => {
114                FormatBucketSet::from_buckets(&[FormatBucket::Conda])
115            }
116            PackageFormatSelection::OnlyTarBz2 => FormatBucketSet::from_buckets(&[
117                FormatBucket::TarAlone,
118                FormatBucket::TarBehindWhl,
119                FormatBucket::TarBehindConda,
120            ]),
121            PackageFormatSelection::PreferConda => FormatBucketSet::from_buckets(&[
122                FormatBucket::Conda,
123                FormatBucket::TarAlone,
124                FormatBucket::TarBehindWhl,
125            ]),
126            PackageFormatSelection::PreferCondaWithWhl => FormatBucketSet::from_buckets(&[
127                FormatBucket::Conda,
128                FormatBucket::WhlAlone,
129                FormatBucket::TarAlone,
130            ]),
131            PackageFormatSelection::Both => FormatBucketSet::from_buckets(&[
132                FormatBucket::Conda,
133                FormatBucket::TarAlone,
134                FormatBucket::TarBehindWhl,
135                FormatBucket::TarBehindConda,
136            ]),
137            PackageFormatSelection::All => FormatBucketSet::ALL,
138        }
139    }
140}

142/// A set of [`FormatBucket`]s.
143#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
144pub(crate) struct FormatBucketSet(u8);
145
146impl FormatBucketSet {
147    /// The set that contains every bucket.
148    pub(crate) const ALL: FormatBucketSet = FormatBucketSet::from_buckets(&FormatBucket::ALL);
149
150    /// Constructs the set of the given buckets.
151    pub(crate) const fn from_buckets(buckets: &[FormatBucket]) -> Self {
152        let mut bits = 0;
153        let mut position = 0;
154        while position < buckets.len() {
155            bits |= 1 << buckets[position].index();
156            position += 1;
157        }
158        FormatBucketSet(bits)
159    }
160
161    /// Returns true if `bucket` is part of this set.
162    pub(crate) fn contains(self, bucket: FormatBucket) -> bool {
163        self.0 & (1 << bucket.index()) != 0
164    }
165
166    /// Returns true if this set contains no bucket.
167    #[cfg(feature = "gateway")]
168    pub(crate) fn is_empty(self) -> bool {
169        self.0 == 0
170    }
171
172    /// Returns true if this set contains a bucket with files of the given
173    /// archive type.
174    pub(crate) fn contains_archive_type(self, archive_type: DistArchiveType) -> bool {
175        self.iter()
176            .any(|bucket| bucket.archive_type() == archive_type)
177    }
178
179    /// Iterates over the buckets of this set in the order of
180    /// [`FormatBucket::ALL`].
181    pub(crate) fn iter(self) -> impl FusedIterator<Item = FormatBucket> {
182        FormatBucket::ALL
183            .into_iter()
184            .filter(move |bucket| self.contains(*bucket))
185    }
186}
187
188#[cfg(feature = "gateway")]
189impl FromIterator<FormatBucket> for FormatBucketSet {
190    fn from_iter<I: IntoIterator<Item = FormatBucket>>(buckets: I) -> Self {
191        FormatBucketSet(
192            buckets
193                .into_iter()
194                .fold(0, |bits, bucket| bits | (1 << bucket.index())),
195        )
196    }
197}

199/// A value for every [`FormatBucket`].
200#[cfg(feature = "gateway")]
201#[derive(Debug, Default)]
202pub(crate) struct FormatBucketMap<T>([T; 6]);
203
204#[cfg(feature = "gateway")]
205impl<T> IntoIterator for FormatBucketMap<T> {
206    type Item = (FormatBucket, T);
207    type IntoIter = Zip<array::IntoIter<FormatBucket, 6>, array::IntoIter<T, 6>>;
208
209    fn into_iter(self) -> Self::IntoIter {
210        FormatBucket::ALL.into_iter().zip(self.0)
211    }
212}
213
214#[cfg(feature = "gateway")]
215impl<T> Index<FormatBucket> for FormatBucketMap<T> {
216    type Output = T;
217
218    fn index(&self, bucket: FormatBucket) -> &T {
219        &self.0[bucket.index()]
220    }
221}
222
223#[cfg(feature = "gateway")]
224impl<T> IndexMut<FormatBucket> for FormatBucketMap<T> {
225    fn index_mut(&mut self, bucket: FormatBucket) -> &mut T {
226        &mut self.0[bucket.index()]
227    }
