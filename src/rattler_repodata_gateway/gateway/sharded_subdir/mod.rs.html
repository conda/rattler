19/// Returns `true` if the HTTP status indicates that the server does not expose
20/// sharded repodata. We treat 404 (Not Found) and 501 (Not Implemented) the
21/// same: the resource is unavailable and we should fall back to repodata.json.
22pub(super) fn is_missing_sharded_repodata_status(status: StatusCode) -> bool {
23    status == StatusCode::NOT_FOUND || status == StatusCode::NOT_IMPLEMENTED
24}

39/// Returns the URL with a trailing slash if it doesn't already have one.
40fn add_trailing_slash(url: &Url) -> Cow<'_, Url> {
41    let path = url.path();
42    if path.ends_with('/') {
43        Cow::Borrowed(url)
44    } else {
45        let mut url = url.clone();
46        url.set_path(&format!("{path}/"));
47        Cow::Owned(url)
48    }
49}

51async fn decode_zst_bytes_async<R: AsRef<[u8]> + Send + 'static>(
52    bytes: R,
53    url: Url,
54) -> Result<Vec<u8>, GatewayError> {
55    let decode = move || {
56        let bytes_ref = bytes.as_ref();
57
58        // Check for empty response which indicates a misconfigured server
59        if bytes_ref.is_empty() {
60            return Err(GatewayError::IoError(
61                format!(
62                    "failed to decode zstd shard from '{}': received empty response (0 bytes). \
63                    This usually indicates a misconfigured server.",
64                    url.redact()
65                ),
66                std::io::Error::new(std::io::ErrorKind::UnexpectedEof, "empty response"),
67            ));
68        }
69
70        match zstd::decode_all(bytes_ref) {
71            Ok(decoded) => Ok(decoded),
72            Err(err) => Err(GatewayError::IoError(
73                format!(
74                    "failed to decode zstd shard from '{}' ({} bytes received). \
75                    The server may have returned invalid or truncated data.",
76                    url.redact(),
77                    bytes_ref.len()
78                ),
79                err,
80            )),
81        }
82    };
83
84    #[cfg(target_arch = "wasm32")]
85    return decode();
86
87    #[cfg(not(target_arch = "wasm32"))]
88    simple_spawn_blocking::tokio::run_blocking_task(decode).await
89}

91async fn parse_records<R: AsRef<[u8]> + Send + 'static>(
92    bytes: R,
93    channel_base_url: ChannelUrl,
94    base_url: Url,
95) -> Result<PackageRecords, GatewayError> {
96    let parse =
97        move || {
98            let shard = rmp_serde::from_slice::<Shard>(bytes.as_ref())
99                .map_err(|e| std::io::Error::new(std::io::ErrorKind::InvalidData, e.to_string()))
100                .map_err(FetchRepoDataError::IoError)?;
101
102            // Chain v3 tar.bz2/conda packages into the main iteration
103            let v3_tar_bz2 = shard.v3.tar_bz2.into_iter().map(|(id, rec)| {
104                (
105                    DistArchiveIdentifier::new(id, CondaArchiveType::TarBz2),
106                    rec,
107                )
108            });
109            let v3_conda =
110                shard.v3.conda.into_iter().map(|(id, rec)| {
111                    (DistArchiveIdentifier::new(id, CondaArchiveType::Conda), rec)
112                });
113
114            let packages = itertools::chain(shard.packages, shard.conda_packages)
115                .chain(v3_tar_bz2)
116                .chain(v3_conda)
117                .filter(|(name, _record)| !shard.removed.contains(name));
118
119            let channel_str = channel_base_url.url().clone().redact().to_string();
120            let base_url_str = base_url.as_str();
121            let mut records: Vec<Arc<RepoDataRecord>> = packages
122                .map(|(file_name, package_record)| {
123                    let file_name_str = file_name.to_file_name();
124                    Arc::new(RepoDataRecord {
125                        url: Url::parse(&format!("{base_url_str}{file_name_str}"))
126                            .expect("filename is not a valid url"),
127                        channel: Some(channel_str.clone()),
128                        package_record,
129                        identifier: file_name,
130                    })
131                })
132                .collect();
133
134            // Handle v3 whl packages separately (different URL resolution)
135            for (
136                id,
137                WhlPackageRecord {
138                    url,
139                    package_record,
140                },
141            ) in shard.v3.whl
142            {
143                let dist_id = DistArchiveIdentifier::new(id, WheelArchiveType::Whl);
144                let url = match url {
145                    UrlOrPath::Path(path) => Url::parse(&format!("{base_url_str}{path}"))
146                        .expect("path is not a valid url"),
147                    UrlOrPath::Url(url) => url,
148                };
149                records.push(Arc::new(RepoDataRecord {
150                    url,
151                    channel: Some(channel_str.clone()),
152                    package_record,
153                    identifier: dist_id,
154                }));
155            }
156
157            let (unique_base_deps, unique_extra_deps) =
158                extract_unique_deps_split(records.iter().map(|r| &**r));
159            Ok(PackageRecords {
160                records,
161                unique_base_deps,
162                unique_extra_deps,
163            })
164        };
165
166    #[cfg(target_arch = "wasm32")]
167    return parse();
168
169    #[cfg(not(target_arch = "wasm32"))]
170    simple_spawn_blocking::tokio::run_blocking_task(parse).await
171}

199    /// A mock server that serves a sharded repodata index but returns
200    /// configurable responses for shard requests.
201    struct MockShardedServer {
202        local_addr: SocketAddr,
203        shard_requests: Arc<AtomicUsize>,
204        _shutdown_sender: oneshot::Sender<()>,
205    }
206
207    impl MockShardedServer {
208        async fn new(shard_response: MockShardResponse) -> Self {
209            // Create a minimal sharded index with one package
210            let mut shards = ahash::HashMap::default();
211            // Use a known hash for the "test-package" shard (SHA256 of empty string)
212            let shard_hash = parse_digest_from_hex::<Sha256>(
213                "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855",
214            )
215            .unwrap();
216            shards.insert("test-package".to_string(), shard_hash);
217
218            let sharded_index = ShardedRepodata {
219                info: ShardedSubdirInfo {
220                    subdir: "linux-64".to_string(),
221                    base_url: "./".to_string(),
222                    shards_base_url: "./shards/".to_string(),
223                    created_at: Some(jiff::Timestamp::now()),
224                    repodata_revisions: RepodataRevisions::default(),
225                    channel_relations: None,
226                },
227                shards,
228            };
229
230            // Encode the index as msgpack and compress with zstd
231            let index_bytes = rmp_serde::to_vec_named(&sharded_index).unwrap();
232            let compressed_index = zstd::encode_all(index_bytes.as_slice(), 3).unwrap();
233
234            let shard_requests = Arc::new(AtomicUsize::new(0));
235            let app = Router::new()
236                .route(
237                    "/linux-64/repodata_shards.msgpack.zst",
238                    get(move || async move {
239                        Response::builder()
240                            .status(StatusCode::OK)
241                            .header("Content-Type", "application/octet-stream")
242                            // Keep the cached copy fresh, so `UseCacheOnly`
243                            // accepts it in the cold-shard tests.
244                            .header("Cache-Control", "max-age=3600")
245                            .body(Body::from(compressed_index.clone()))
246                            .unwrap()
247                    }),
248                )
249                .route(
250                    "/linux-64/shards/{shard_file}",
251                    get({
252                        let shard_requests = Arc::clone(&shard_requests);
253                        move || async move {
254                            shard_requests.fetch_add(1, Ordering::SeqCst);
255                            match shard_response {
256                                MockShardResponse::Empty => Response::builder()
257                                    .status(StatusCode::OK)
258                                    .body(Body::empty())
259                                    .unwrap(),
260                                MockShardResponse::Truncated => {
261                                    // Return some bytes that look like zstd but are truncated
262                                    Response::builder()
263                                        .status(StatusCode::OK)
264                                        .body(Body::from(vec![0x28, 0xb5, 0x2f, 0xfd]))
265                                        .unwrap()
266                                }
267                            }
268                        }
269                    }),
270                );
271
272            let addr = SocketAddr::new([127, 0, 0, 1].into(), 0);
273            let listener = tokio::net::TcpListener::bind(&addr).await.unwrap();
274            let local_addr = listener.local_addr().unwrap();
275
276            let (tx, rx) = oneshot::channel();
277            let server = axum::serve(listener, app)
278                .with_graceful_shutdown(async {
279                    rx.await.ok();
280                })
281                .into_future();
282
283            tokio::spawn(server);
284
285            Self {
286                local_addr,
287                shard_requests,
288                _shutdown_sender: tx,
289            }
290        }
291
292        fn url(&self) -> Url {
293            Url::parse(&format!("http://localhost:{}", self.local_addr.port())).unwrap()
294        }
295
296        fn channel(&self) -> Channel {
297            Channel::from_url(self.url())
298        }
299
300        /// How many shard downloads the server has answered so far. The index
301        /// request is not counted.
302        fn shard_request_count(&self) -> usize {
303            self.shard_requests.load(Ordering::SeqCst)
304        }
305    }

512    /// A cache-only build with no index cached must report

544    /// Without the opt-in, a cold shard fails a cache-only query with a
545    /// distinct error: nothing is known about the package, which is not the
546    /// same as the package having no records. Neither mode may touch the
547    /// network to find out.
548    #[tokio::test]
549    async fn cold_shard_is_an_error_by_default() {
550        for action in CACHE_ONLY_ACTIONS {
551            let server = MockShardedServer::new(MockShardResponse::Empty).await;
552            let cache_dir = tempfile::tempdir().unwrap();
553
554            let subdir =
555                cache_only_subdir_with_cold_shard(cache_dir.path(), &server, action, false).await;
556
557            let err = subdir
558                .fetch_package_records(&"test-package".parse().unwrap(), None)
559                .await
560                .expect_err("a cold shard fails a cache-only query");
561
562            assert!(
563                matches!(err, GatewayError::ShardNotCached(name) if name == "test-package"),
564                "the error names the package whose shard is missing"
565            );
566            assert_eq!(
567                server.shard_request_count(),
568                0,
569                "{action:?} may not download a shard"
570            );
571        }
572    }

574    /// With `missing_shards_are_empty` the same query reports the package as

578    #[tokio::test]
579    async fn cold_shard_is_empty_when_opted_in() {

587            let records = subdir

589                .await
590                .expect("a cold shard is not an error when opted in");

592            assert!(records.records.is_empty());
593            assert_eq!(
594                server.shard_request_count(),
595                0,
596                "{action:?} may not download a shard"
597            );
598        }
