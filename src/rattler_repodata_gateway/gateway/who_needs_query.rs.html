1//! A streaming reverse-dependency query over a [`Gateway`](super::Gateway).
2//!
3//! Rather than materializing the repodata of the queried platforms (as a
4//! wildcard [`RepoDataQuery`](super::RepoDataQuery) would), this query
5//! scans the records of every package name one name at a time, keeps only
6//! the matching records, and never inserts the scanned records into the
7//! gateway's long-lived per-name cache. The memory held by the scan is
8//! therefore bounded by the in-flight package scans instead of the complete
9//! repodata of the queried platforms.
10//!
11//! What the scan does not bound is the result set, which for a common
12//! package is the larger cost of the two. The query is therefore exposed as
13//! a stream so a consumer can reduce each match and drop the record it came
14//! from; [`WhoNeedsQuery::execute`] is the collecting convenience over it.
15//!
16//! This module answers *where* the scanned records come from; what counts
17//! as a reverse dependency is decided by [`crate::who_needs`].
18

39/// How many package names one batch task fetches and scans sequentially.
40/// Batching amortizes the per-task overhead over many cheap per-name
41/// fetches.
42///
43/// A batch collects its matches before yielding them, so together with
44/// [`BATCH_CONCURRENCY`] this sets how many matches a stream buffers ahead
45/// of its consumer. Scanning conda-forge for the dependents of `python`
46/// (~490k matching records) holds ~190 MB at 100 names and 16 batches,
47/// against ~500 MB at 500 names; larger batches do not scan faster.
48const NAME_BATCH_SIZE: usize = 100;
49
50/// A reverse-dependency query created through
51/// [`Gateway::who_needs`](super::Gateway::who_needs).
52///
53/// When executed, every package of the queried sources and platforms is
54/// scanned against the target. Records of scanned packages are dropped as
55/// soon as the package is processed and are *not* inserted into the
56/// gateway's per-name record cache (previously cached entries are still
57/// reused), so a query over a large channel does not permanently grow the
58/// gateway's memory footprint. Only the matching records are retained,
59/// shared via `Arc` in the returned [`Dependent`]s.
60///
61/// Channel sources always use full repodata, even if sharding is enabled
62/// or a previous query has already loaded a sharded subdir.
63///
64/// The matches themselves can still be numerous enough to dominate memory —
65/// half a million records depend on `python` in conda-forge. Use
66/// [`stream`](Self::stream) to fold them as they arrive;
67/// [`execute`](Self::execute) keeps every one of them.
68///
69/// Platforms are scanned one after another in the order they were passed in
70/// (duplicate platforms are scanned once), and the name batches within a
71/// subdir are scanned concurrently. See [`WhoNeedsTarget`] for the matching
72/// semantics of its variants.
73///
74/// Unlike the other gateway queries, this one does not follow CEP-42
75/// `channel_relations`: only the subdirs of the sources passed in are
76/// scanned.
77#[derive(Clone)]
78pub struct WhoNeedsQuery {
79    gateway: Arc<GatewayInner>,
80    sources: Vec<Source>,
81    platforms: Vec<Platform>,
82    target: WhoNeedsTarget,
83    reporter: Option<Arc<dyn Reporter>>,
84}

87    /// Constructs a new instance. This should not be called directly, use
88    /// [`Gateway::who_needs`](super::Gateway::who_needs) instead.
89    pub(super) fn new(
90        gateway: Arc<GatewayInner>,
91        sources: Vec<Source>,
92        platforms: Vec<Platform>,
93        target: WhoNeedsTarget,
94    ) -> Self {
95        Self {
96            gateway,
97            sources,
98            platforms,
99            target,
100            reporter: None,
101        }
102    }
103
104    /// Sets the reporter to use for this query.
105    pub fn with_reporter(self, reporter: impl Reporter + 'static) -> Self {
106        Self {
107            reporter: Some(Arc::new(reporter)),
108            ..self
109        }
110    }
111
112    /// Execute the query and return all reverse dependencies of the target,
113    /// in the order described by [`Self::stream`].
114    ///
115    /// This collects the entire result set into memory. A channel-wide
116    /// query can match a lot of records — every record depending on
117    /// `python` in conda-forge is roughly half a million, about a gigabyte
118    /// once each is retained — so prefer [`Self::stream`] when the results
119    /// can be folded into something smaller as they arrive.
120    pub async fn execute(self) -> Result<Vec<Dependent>, GatewayError> {
121        self.stream().try_collect().await
122    }
123
124    /// Execute the query as a stream of reverse dependencies.
125    ///
126    /// Records are fetched, scanned and dropped as the stream is polled, so
127    /// a consumer that aggregates each [`Dependent`] and drops it keeps only
128    /// its own aggregate in memory rather than the whole result set. Nothing
129    /// is fetched until the stream is first polled, and dropping the stream
130    /// stops the scan.
131    ///
132    /// Items arrive in the same order as [`Self::execute`] returns them:
133    /// grouped by the platform order passed to
134    /// [`Gateway::who_needs`](super::Gateway::who_needs), then by source
135    /// order, with the order within a subdir unspecified.
136    ///
137    /// The stream is boxed so it can be polled without pinning it first.
138    ///
139    /// ```no_run
140    /// # use futures::TryStreamExt;
141    /// # use rattler_conda_types::{Channel, PackageName, Platform};
142    /// # use rattler_repodata_gateway::Gateway;
143    /// # async fn example(gateway: Gateway, channel: Channel, name: PackageName) -> anyhow::Result<()> {
144    /// let mut stream = gateway
145    ///     .who_needs(vec![channel], vec![Platform::Linux64], name)
146    ///     .stream();
147    ///
148    /// // Count the dependents while holding only one record at a time.
149    /// let mut count = 0;
150    /// while let Some(dependent) = stream.try_next().await? {
151    ///     count += 1;
152    ///     drop(dependent);
153    /// }
154    /// # Ok(())
155    /// # }
156    /// ```
157    pub fn stream(self) -> BoxStream<Result<Dependent, GatewayError>> {
158        // Deduplicate platforms while keeping the input order, so the
159        // result order stays deterministic and no subdir is scanned twice.
160        let mut seen_platforms = std::collections::HashSet::new();
161        let platforms: Vec<Platform> = self
162            .platforms
163            .iter()
164            .copied()
165            .filter(|p| seen_platforms.insert(*p))
166            .collect();
167
168        // Platforms are scanned in sequence rather than concurrently. That
169        // is what makes the output order match the input platform order
170        // without buffering a whole platform's results to re-sort them, and
171        // it bounds the number of in-flight scans; the concurrency that
172        // matters is between the name batches of a subdir.
173        box_stream(stream::iter(platforms).flat_map(move |platform| {
174            scan_platform(
175                self.gateway.clone(),
176                self.sources.clone(),
177                platform,
178                self.target.clone(),
179                self.reporter.clone(),
180            )
181        }))
182    }
183}

194/// Runs a scan on its own tokio task so scans run on the runtime's worker
195/// threads instead of interleaving on the task driving the query.
196#[cfg(not(target_arch = "wasm32"))]
197async fn spawn_scan<T: Send + 'static>(
198    scan: impl Future<Output = Result<T, GatewayError>> + Send + 'static,
199) -> Result<T, GatewayError> {
200    tokio::spawn(scan).await.expect("the scan task panicked")
201}
202
203/// On wasm there are no threads to parallelize over; run the scan in place.
204#[cfg(target_arch = "wasm32")]
205async fn spawn_scan<T>(
206    scan: impl Future<Output = Result<T, GatewayError>>,
207) -> Result<T, GatewayError> {
208    scan.await
209}

211/// A resolved subdir tagged with the index of the source it came from.
212type IndexedSubdir = (usize, Arc<Subdir>);
213
214/// Streams the dependents of `target` found on `platform`, across the
215/// subdirs of every source, in the caller's source order.
216fn scan_platform(
217    gateway: Arc<GatewayInner>,
218    sources: Vec<Source>,
219    platform: Platform,
220    target: WhoNeedsTarget,
221    reporter: Option<Arc<dyn Reporter>>,
222) -> BoxStream<Result<Dependent, GatewayError>> {
223    // The subdirs of a platform are resolved when the platform is first
224    // polled, not when the stream is built, so a consumer that stops early
225    // never pays to resolve the platforms it did not reach.
226    box_stream(
227        stream::once(resolve_subdirs(
228            gateway,
229            sources,
230            platform,
231            reporter.clone(),
232        ))
233        .map_ok(move |subdirs| {
234            let target = target.clone();
235            let reporter = reporter.clone();
236            stream::iter(subdirs)
237                .map(move |(_, subdir)| scan_subdir(subdir, target.clone(), reporter.clone()))
238                // Subdirs are scanned one after another so the result order
239                // follows the caller's source order.
240                .flatten()
241        })
242        .try_flatten(),
243    )
244}

246/// Resolves the subdirs of every source for `platform`, in the caller's
247/// source order.
248async fn resolve_subdirs(
249    gateway: Arc<GatewayInner>,
250    sources: Vec<Source>,
251    platform: Platform,
252    reporter: Option<Arc<dyn Reporter>>,
253) -> Result<Vec<IndexedSubdir>, GatewayError> {
254    // Kick off the subdir fetch of every channel source; custom and sparse
255    // sources resolve immediately. Each subdir is tagged with the index of
256    // the source it came from so the scan order follows the caller's
257    // source order regardless of which fetch finishes first.
258    let mut pending: FuturesUnordered<BoxFuture<Result<IndexedSubdir, GatewayError>>> =
259        FuturesUnordered::new();
260    let mut subdirs: Vec<IndexedSubdir> = Vec::new();
261
262    for (source_index, source) in sources.into_iter().enumerate() {
263        match source {
264            Source::Channel(channel) => {
265                let gateway = gateway.clone();
266                let reporter = reporter.clone();
267                pending.push(box_future(async move {
268                    let subdir = gateway
269                        .get_or_create_subdir(&channel, platform, reporter, false)
270                        .await?;
271                    Ok((source_index, subdir))
272                }));
273            }
274            Source::Custom(custom_source) => {
275                let client = CustomSourceClient::new(custom_source, platform);
276                subdirs.push((
277                    source_index,
278                    Arc::new(Subdir::Found(SubdirData::from_client(client))),
279                ));
280            }
281            Source::SparseRepoData(sparse_list) => {
282                let subdir = match sparse_list
283                    .iter()
284                    .find(|sparse| platform.as_str() == sparse.subdir())
285                {
286                    Some(sparse) => Arc::new(Subdir::Found(SubdirData::from_client(
287                        LocalSubdirClient::new(sparse.clone()),
288                    ))),
289                    None => Arc::new(Subdir::NotFound),
290                };
291                subdirs.push((source_index, subdir));
292            }
293        }
294    }
295
296    while let Some(result) = pending.next().await {
297        subdirs.push(result?);
298    }
299    subdirs.sort_by_key(|(source_index, _)| *source_index);
300    Ok(subdirs)
301}

303/// How many name batches of one subdir are scanned concurrently.
304///
305/// This is what bounds the memory of a stream: at most this many batches'
306/// worth of matches are buffered before the consumer sees them, so a
307/// consumer that folds results as they arrive never holds the whole result
308/// set. Concurrency is per subdir and subdirs are scanned in sequence, so
309/// the in-flight batch count of a whole query stays bounded by this.
310const BATCH_CONCURRENCY: usize = 16;
311
312/// Streams the dependents found in `subdir`, scanning it in batches of
313/// [`NAME_BATCH_SIZE`] names.
314///
315/// Each batch runs as its own task (see [`spawn_scan`]) that fetches one
316/// package at a time, keeps the matches, and drops the scanned records
317/// before fetching the next package. The scanned records are never inserted
318/// into the subdir's per-name record cache. Batches are polled in order and
319/// at most [`BATCH_CONCURRENCY`] run at once, so matches reach the consumer
320/// while the rest of the subdir is still being scanned.
321fn scan_subdir(
322    subdir: Arc<Subdir>,
323    target: WhoNeedsTarget,
324    reporter: Option<Arc<dyn Reporter>>,
325) -> BoxStream<Result<Dependent, GatewayError>> {
326    let names: Vec<PackageName> = match subdir.as_ref() {
327        Subdir::Found(subdir_data) => subdir_data
328            .package_names()
329            .into_iter()
330            .filter_map(|name| PackageName::try_from(name).ok())
331            .collect(),
332        Subdir::NotFound => Vec::new(),
333    };
334    let batches: Vec<Vec<PackageName>> = names
335        .chunks(NAME_BATCH_SIZE)
336        .map(<[PackageName]>::to_vec)
337        .collect();
338
339    box_stream(
340        stream::iter(batches)
341            .map(move |batch| {
342                let subdir = subdir.clone();
343                let target = target.clone();
344                let reporter = reporter.clone();
345                spawn_scan(async move {
346                    let Subdir::Found(subdir_data) = subdir.as_ref() else {
347                        return Ok(Vec::new());
348                    };
349                    let mut matches = Vec::new();
350                    for name in batch {
351                        let records = subdir_data
352                            .fetch_package_records_uncached(&name, reporter.as_deref())
353                            .await?;
354                        matches.extend(who_needs(&records, &target));
355                        // The scanned records are dropped here; only the
356                        // matches survive.
357                    }
358                    Ok(matches)
359                })
360            })
361            // `buffered` keeps the batch order while running up to
362            // `BATCH_CONCURRENCY` of them at once, which is what caps how
363            // many matches are in memory ahead of the consumer.
364            .buffered(BATCH_CONCURRENCY)
365            // Flatten each batch's `Vec<Dependent>` into individual items so
366            // a consumer can drop each dependent as it goes.
367            .map_ok(|matches| stream::iter(matches.into_iter().map(Ok)))
368            .try_flatten(),
369    )
370}

397    /// Renders the dependents as one sorted `kind | package | dependency`
398    /// line each, for snapshot comparison. Sorting makes the rendering
399    /// independent of the unspecified within-platform result order.
400    fn render(dependents: &[Dependent]) -> String {
401        let mut lines: Vec<String> = dependents
402            .iter()
403            .map(|dependent| {
404                let record = &dependent.record.package_record;
405                let kind = match &dependent.kind {
406                    DependencyKind::Depends => "depends".to_string(),
407                    DependencyKind::Constrains => "constrains".to_string(),
408                    DependencyKind::ExtraDepends(extra) => format!("extra_depends[{extra}]"),
409                    DependencyKind::RunExport(kind) => format!("run_export[{kind:?}]"),
410                };
411                format!(
412                    "{kind} | {}-{}-{} | {}",
413                    record.name.as_normalized(),
414                    record.version,
415                    record.build,
416                    dependent.dependency
417                )
418            })
419            .collect();
420        lines.sort();
421        lines.join("\n")
422    }

520    /// Fetches the record of `name` at `version` from `channel`, to use as
521    /// a concrete [`WhoNeedsTarget`].
522    async fn record(
523        channel: &Channel,
524        platform: Platform,
525        name: &str,
526        version: &str,
527    ) -> rattler_conda_types::PackageRecord {
528        Gateway::new()
529            .query(
530                vec![channel.clone()],
531                vec![platform],
532                vec![PackageName::from_str(name).unwrap()],
533            )
534            .recursive(false)
535            .execute()
536            .await
537            .unwrap()
538            .repodata
539            .iter()
540            .flat_map(|repodata| repodata.iter())
541            .find(|record| record.package_record.version.to_string() == version)
542            .unwrap_or_else(|| panic!("{name} {version} is missing from the test channel"))
543            .package_record
544            .clone()
545    }

641            panic!("expected the noarch subdir to exist");
642        };
643        assert_eq!(noarch_data.cached_package_count(), 0);
644
645        // The previously cached records are still usable afterwards.
646        let records = gateway
647            .query(vec![channel], vec![Platform::Linux64], vec![python])
648            .recursive(false)
649            .execute()
650            .await
651            .unwrap();
652        assert!(records.repodata.iter().any(|repodata| !repodata.is_empty()));
653    }

674    #[rstest::rstest]
675    #[tokio::test]
676    async fn test_who_needs_uses_full_repodata(
677        #[values(false, true)] prime_cache: bool,
678        #[values(false, true)] sharded_enabled: bool,
679    ) {
680        use rattler_conda_types::{RepodataRevisions, ShardedRepodata, ShardedSubdirInfo};
681
682        use crate::{
683            ChannelConfig, SourceConfig, utils::simple_channel_server::SimpleChannelServer,
684        };
685
686        let channel_dir = tempfile::tempdir().unwrap();
687        let subdir = channel_dir.path().join("linux-64");
688        std::fs::create_dir(&subdir).unwrap();
689        std::fs::write(
690            subdir.join("repodata.json"),
691            include_str!("../../../../test-data/channels/dummy/linux-64/repodata.json"),
692        )
693        .unwrap();
694
695        // Advertise a shard that cannot be fetched. The scan must use full
696        // repodata even when an earlier names query has cached this index.
697        let index = ShardedRepodata {
698            info: ShardedSubdirInfo {
699                subdir: "linux-64".into(),
700                base_url: "./".into(),
701                shards_base_url: "./shards/".into(),
702                created_at: None,
703                repodata_revisions: RepodataRevisions::default(),
704                channel_relations: None,
705            },
706            shards: [("missing-shard".into(), [0u8; 32].into())]
707                .into_iter()
708                .collect(),
709        };
710        let index_bytes = rmp_serde::to_vec_named(&index).unwrap();
711        std::fs::write(
712            subdir.join("repodata_shards.msgpack.zst"),
713            zstd::encode_all(index_bytes.as_slice(), 0).unwrap(),
714        )
715        .unwrap();
716
717        let server = SimpleChannelServer::new(channel_dir.path()).await;
718        let url = server.url();
719        let channel = Channel::from_url(url.clone());
720        let cache_dir = tempfile::tempdir().unwrap();
721        let gateway = Gateway::builder()
722            .with_client(reqwest::Client::builder().no_proxy().build().unwrap())
723            .with_cache_dir(cache_dir.path())
724            .with_channel_config(ChannelConfig {
725                default: SourceConfig {
726                    sharded_enabled: false,
727                    ..SourceConfig::default()
728                },
729                // Exercise the per-channel sharding configuration.
730                per_channel: [(
731                    url,
732                    SourceConfig {
733                        sharded_enabled,
734                        ..SourceConfig::default()
735                    },
736                )]
737                .into_iter()
738                .collect(),
739            })
740            .finish();
741
742        if prime_cache {
743            gateway
744                .names([channel.clone()], [Platform::Linux64])
745                .await
746                .unwrap();
747        }
748
749        let dependents = gateway
750            .who_needs(
751                [channel.clone()],
752                [Platform::Linux64],
753                PackageName::new_unchecked("bors"),
754            )
755            .await
756            .unwrap();
757        assert_eq!(dependents.len(), 3);
758
759        // Ordinary queries must still follow the sharding configuration,
760        // including when who_needs was the first query on this gateway.
761        let names = gateway.names([channel], [Platform::Linux64]).await.unwrap();
762        if sharded_enabled {
763            assert_eq!(
764                names.names,
765                vec![PackageName::new_unchecked("missing-shard")]
766            );
767        } else {
768            assert!(names.names.contains(&PackageName::new_unchecked("bors")));
769            assert!(
770                !names
771                    .names
772                    .contains(&PackageName::new_unchecked("missing-shard"))
773            );
774        }
775    }
