1//! A streaming reverse-dependency query over a [`Gateway`](super::Gateway).
2//!
3//! Rather than materializing the repodata of the queried platforms (as a
4//! wildcard [`RepoDataQuery`](super::RepoDataQuery) would), this query
5//! scans the records of every package name one name at a time, keeps only
6//! the matching records, and never inserts the scanned records into the
7//! gateway's long-lived per-name cache. The memory held by the scan is
8//! therefore bounded by the in-flight package scans instead of the complete
9//! repodata of the queried platforms.
10//!
11//! What the scan does not bound is the result set, which for a common
12//! package is the larger cost of the two. The query is therefore exposed as
13//! a stream so a consumer can reduce each match and drop the record it came
14//! from; [`WhoNeedsQuery::execute`] is the collecting convenience over it.
15//!
16//! This module answers *where* the scanned records come from; what counts
17//! as a reverse dependency is decided by [`crate::who_needs`].
18

39/// How many package names one batch task fetches and scans sequentially.
40/// Batching amortizes the per-task overhead over many cheap per-name
41/// fetches.
42///
43/// A batch collects its matches before yielding them, so together with
44/// [`BATCH_CONCURRENCY`] this sets how many matches a stream buffers ahead
45/// of its consumer. Scanning conda-forge for the dependents of `python`
46/// (~490k matching records) holds ~190 MB at 100 names and 16 batches,
47/// against ~500 MB at 500 names; larger batches do not scan faster.
48const NAME_BATCH_SIZE: usize = 100;
49
50/// A reverse-dependency query created through
51/// [`Gateway::who_needs`](super::Gateway::who_needs).
52///
53/// When executed, every package of the queried sources and platforms is
54/// scanned against the target. Records of scanned packages are dropped as
55/// soon as the package is processed and are *not* inserted into the
56/// gateway's per-name record cache (previously cached entries are still
57/// reused), so a query over a large channel does not permanently grow the
58/// gateway's memory footprint. Only the matching records are retained,
59/// shared via `Arc` in the returned [`Dependent`]s.
60///
61/// The matches themselves can still be numerous enough to dominate memory —
62/// half a million records depend on `python` in conda-forge. Use
63/// [`stream`](Self::stream) to fold them as they arrive;
64/// [`execute`](Self::execute) keeps every one of them.
65///
66/// Platforms are scanned one after another in the order they were passed in
67/// (duplicate platforms are scanned once), and the name batches within a
68/// subdir are scanned concurrently. See [`WhoNeedsTarget`] for the matching
69/// semantics of its variants.
70///
71/// Unlike the other gateway queries, this one does not follow CEP-42
72/// `channel_relations`: only the subdirs of the sources passed in are
73/// scanned.
74#[derive(Clone)]
75pub struct WhoNeedsQuery {
76    gateway: Arc<GatewayInner>,
77    sources: Vec<Source>,
78    platforms: Vec<Platform>,
79    target: WhoNeedsTarget,
80    reporter: Option<Arc<dyn Reporter>>,
81}

83impl WhoNeedsQuery {
84    /// Constructs a new instance. This should not be called directly, use
85    /// [`Gateway::who_needs`](super::Gateway::who_needs) instead.
86    pub(super) fn new(
87        gateway: Arc<GatewayInner>,
88        sources: Vec<Source>,
89        platforms: Vec<Platform>,
90        target: WhoNeedsTarget,
91    ) -> Self {
92        Self {
93            gateway,
94            sources,
95            platforms,
96            target,
97            reporter: None,
98        }
99    }
100
101    /// Sets the reporter to use for this query.
102    pub fn with_reporter(self, reporter: impl Reporter + 'static) -> Self {
103        Self {
104            reporter: Some(Arc::new(reporter)),
105            ..self
106        }
107    }
108
109    /// Execute the query and return all reverse dependencies of the target,
110    /// in the order described by [`Self::stream`].
111    ///
112    /// This collects the entire result set into memory. A channel-wide
113    /// query can match a lot of records — every record depending on
114    /// `python` in conda-forge is roughly half a million, about a gigabyte
115    /// once each is retained — so prefer [`Self::stream`] when the results
116    /// can be folded into something smaller as they arrive.
117    pub async fn execute(self) -> Result<Vec<Dependent>, GatewayError> {
118        self.stream().try_collect().await
119    }
120
121    /// Execute the query as a stream of reverse dependencies.
122    ///
123    /// Records are fetched, scanned and dropped as the stream is polled, so
124    /// a consumer that aggregates each [`Dependent`] and drops it keeps only
125    /// its own aggregate in memory rather than the whole result set. Nothing
126    /// is fetched until the stream is first polled, and dropping the stream
127    /// stops the scan.
128    ///
129    /// Items arrive in the same order as [`Self::execute`] returns them:
130    /// grouped by the platform order passed to
131    /// [`Gateway::who_needs`](super::Gateway::who_needs), then by source
132    /// order, with the order within a subdir unspecified.
133    ///
134    /// The stream is boxed so it can be polled without pinning it first.
135    ///
136    /// ```no_run
137    /// # use futures::TryStreamExt;
138    /// # use rattler_conda_types::{Channel, PackageName, Platform};
139    /// # use rattler_repodata_gateway::Gateway;
140    /// # async fn example(gateway: Gateway, channel: Channel, name: PackageName) -> anyhow::Result<()> {
141    /// let mut stream = gateway
142    ///     .who_needs(vec![channel], vec![Platform::Linux64], name)
143    ///     .stream();
144    ///
145    /// // Count the dependents while holding only one record at a time.
146    /// let mut count = 0;
147    /// while let Some(dependent) = stream.try_next().await? {
148    ///     count += 1;
149    ///     drop(dependent);
150    /// }
151    /// # Ok(())
152    /// # }
153    /// ```
154    pub fn stream(self) -> BoxStream<Result<Dependent, GatewayError>> {
155        // Deduplicate platforms while keeping the input order, so the
156        // result order stays deterministic and no subdir is scanned twice.
157        let mut seen_platforms = std::collections::HashSet::new();
158        let platforms: Vec<Platform> = self
159            .platforms
160            .iter()
161            .copied()
162            .filter(|p| seen_platforms.insert(*p))
163            .collect();
164
165        // Platforms are scanned in sequence rather than concurrently. That
166        // is what makes the output order match the input platform order
167        // without buffering a whole platform's results to re-sort them, and
168        // it bounds the number of in-flight scans; the concurrency that
169        // matters is between the name batches of a subdir.
170        box_stream(stream::iter(platforms).flat_map(move |platform| {
171            scan_platform(
172                self.gateway.clone(),
173                self.sources.clone(),
174                platform,
175                self.target.clone(),
176                self.reporter.clone(),
177            )
178        }))
179    }
180}

186    fn into_future(self) -> Self::IntoFuture {
187        box_future(self.execute())
188    }

191/// Runs a scan on its own tokio task so scans run on the runtime's worker
192/// threads instead of interleaving on the task driving the query.
193#[cfg(not(target_arch = "wasm32"))]
194async fn spawn_scan<T: Send + 'static>(
195    scan: impl Future<Output = Result<T, GatewayError>> + Send + 'static,
196) -> Result<T, GatewayError> {
197    tokio::spawn(scan).await.expect("the scan task panicked")
198}

200/// On wasm there are no threads to parallelize over; run the scan in place.
201#[cfg(target_arch = "wasm32")]
202async fn spawn_scan<T>(
203    scan: impl Future<Output = Result<T, GatewayError>>,
204) -> Result<T, GatewayError> {
205    scan.await
206}

208/// A resolved subdir tagged with the index of the source it came from.
209type IndexedSubdir = (usize, Arc<Subdir>);
210
211/// Streams the dependents of `target` found on `platform`, across the
212/// subdirs of every source, in the caller's source order.
213fn scan_platform(
214    gateway: Arc<GatewayInner>,
215    sources: Vec<Source>,
216    platform: Platform,
217    target: WhoNeedsTarget,
218    reporter: Option<Arc<dyn Reporter>>,
219) -> BoxStream<Result<Dependent, GatewayError>> {
220    // The subdirs of a platform are resolved when the platform is first
221    // polled, not when the stream is built, so a consumer that stops early
222    // never pays to resolve the platforms it did not reach.
223    box_stream(
224        stream::once(resolve_subdirs(
225            gateway,
226            sources,
227            platform,
228            reporter.clone(),
229        ))
230        .map_ok(move |subdirs| {
231            let target = target.clone();
232            let reporter = reporter.clone();
233            stream::iter(subdirs)
234                .map(move |(_, subdir)| scan_subdir(subdir, target.clone(), reporter.clone()))
235                // Subdirs are scanned one after another so the result order
236                // follows the caller's source order.
237                .flatten()
238        })
239        .try_flatten(),
240    )
241}

243/// Resolves the subdirs of every source for `platform`, in the caller's
244/// source order.
245async fn resolve_subdirs(
246    gateway: Arc<GatewayInner>,
247    sources: Vec<Source>,
248    platform: Platform,
249    reporter: Option<Arc<dyn Reporter>>,
250) -> Result<Vec<IndexedSubdir>, GatewayError> {
251    // Kick off the subdir fetch of every channel source; custom and sparse
252    // sources resolve immediately. Each subdir is tagged with the index of
253    // the source it came from so the scan order follows the caller's
254    // source order regardless of which fetch finishes first.
255    let mut pending: FuturesUnordered<BoxFuture<Result<IndexedSubdir, GatewayError>>> =
256        FuturesUnordered::new();
257    let mut subdirs: Vec<IndexedSubdir> = Vec::new();
258
259    for (source_index, source) in sources.into_iter().enumerate() {
260        match source {
261            Source::Channel(channel) => {
262                let gateway = gateway.clone();
263                let reporter = reporter.clone();
264                pending.push(box_future(async move {
265                    let subdir = gateway
266                        .get_or_create_subdir(&channel, platform, reporter)
267                        .await?;
268                    Ok((source_index, subdir))
269                }));
270            }
271            Source::Custom(custom_source) => {
272                let client = CustomSourceClient::new(custom_source, platform);
273                subdirs.push((
274                    source_index,
275                    Arc::new(Subdir::Found(SubdirData::from_client(client))),
276                ));
277            }
278            Source::SparseRepoData(sparse_list) => {
279                let subdir = match sparse_list
280                    .iter()
281                    .find(|sparse| platform.as_str() == sparse.subdir())
282                {
283                    Some(sparse) => Arc::new(Subdir::Found(SubdirData::from_client(
284                        LocalSubdirClient::new(sparse.clone()),
285                    ))),
286                    None => Arc::new(Subdir::NotFound),
287                };
288                subdirs.push((source_index, subdir));
289            }
290        }
291    }
292
293    while let Some(result) = pending.next().await {
294        subdirs.push(result?);
295    }
296    subdirs.sort_by_key(|(source_index, _)| *source_index);
297    Ok(subdirs)
298}

300/// How many name batches of one subdir are scanned concurrently.
301///
302/// This is what bounds the memory of a stream: at most this many batches'
303/// worth of matches are buffered before the consumer sees them, so a
304/// consumer that folds results as they arrive never holds the whole result
305/// set. Concurrency is per subdir and subdirs are scanned in sequence, so
306/// the in-flight batch count of a whole query stays bounded by this.
307const BATCH_CONCURRENCY: usize = 16;
308
309/// Streams the dependents found in `subdir`, scanning it in batches of
310/// [`NAME_BATCH_SIZE`] names.
311///
312/// Each batch runs as its own task (see [`spawn_scan`]) that fetches one
313/// package at a time, keeps the matches, and drops the scanned records
314/// before fetching the next package. The scanned records are never inserted
315/// into the subdir's per-name record cache. Batches are polled in order and
316/// at most [`BATCH_CONCURRENCY`] run at once, so matches reach the consumer
317/// while the rest of the subdir is still being scanned.
318fn scan_subdir(
319    subdir: Arc<Subdir>,
320    target: WhoNeedsTarget,
321    reporter: Option<Arc<dyn Reporter>>,
322) -> BoxStream<Result<Dependent, GatewayError>> {
323    let names: Vec<PackageName> = match subdir.as_ref() {
324        Subdir::Found(subdir_data) => subdir_data
325            .package_names()
326            .into_iter()
327            .filter_map(|name| PackageName::try_from(name).ok())
328            .collect(),
329        Subdir::NotFound => Vec::new(),
330    };
331    let batches: Vec<Vec<PackageName>> = names
332        .chunks(NAME_BATCH_SIZE)
333        .map(<[PackageName]>::to_vec)
334        .collect();
335
336    box_stream(
337        stream::iter(batches)
338            .map(move |batch| {
339                let subdir = subdir.clone();
340                let target = target.clone();
341                let reporter = reporter.clone();
342                spawn_scan(async move {
343                    let Subdir::Found(subdir_data) = subdir.as_ref() else {
344                        return Ok(Vec::new());
345                    };
346                    let mut matches = Vec::new();
347                    for name in batch {
348                        let records = subdir_data
349                            .fetch_package_records_uncached(&name, reporter.as_deref())
350                            .await?;
351                        matches.extend(who_needs(&records, &target));
352                        // The scanned records are dropped here; only the
353                        // matches survive.
354                    }
355                    Ok(matches)
356                })
357            })
358            // `buffered` keeps the batch order while running up to
359            // `BATCH_CONCURRENCY` of them at once, which is what caps how
360            // many matches are in memory ahead of the consumer.
361            .buffered(BATCH_CONCURRENCY)
362            // Flatten each batch's `Vec<Dependent>` into individual items so
363            // a consumer can drop each dependent as it goes.
364            .map_ok(|matches| stream::iter(matches.into_iter().map(Ok)))
365            .try_flatten(),
366    )
367}

394    /// Renders the dependents as one sorted `kind | package | dependency`
395    /// line each, for snapshot comparison. Sorting makes the rendering
396    /// independent of the unspecified within-platform result order.
397    fn render(dependents: &[Dependent]) -> String {
398        let mut lines: Vec<String> = dependents
399            .iter()
400            .map(|dependent| {
401                let record = &dependent.record.package_record;
402                let kind = match &dependent.kind {
403                    DependencyKind::Depends => "depends".to_string(),
404                    DependencyKind::Constrains => "constrains".to_string(),
405                    DependencyKind::ExtraDepends(extra) => format!("extra_depends[{extra}]"),
406                    DependencyKind::RunExport(kind) => format!("run_export[{kind:?}]"),
407                };
408                format!(
409                    "{kind} | {}-{}-{} | {}",
410                    record.name.as_normalized(),
411                    record.version,
412                    record.build,
413                    dependent.dependency
414                )
415            })
416            .collect();
417        lines.sort();
418        lines.join("\n")
419    }

517    /// Fetches the record of `name` at `version` from `channel`, to use as
518    /// a concrete [`WhoNeedsTarget`].
519    async fn record(
520        channel: &Channel,
521        platform: Platform,
522        name: &str,
523        version: &str,
524    ) -> rattler_conda_types::PackageRecord {
525        Gateway::new()
526            .query(
527                vec![channel.clone()],
528                vec![platform],
529                vec![PackageName::from_str(name).unwrap()],
530            )
531            .recursive(false)
532            .execute()
533            .await
534            .unwrap()
535            .repodata
536            .iter()
537            .flat_map(|repodata| repodata.iter())
538            .find(|record| record.package_record.version.to_string() == version)
539            .unwrap_or_else(|| panic!("{name} {version} is missing from the test channel"))
540            .package_record
541            .clone()
542    }
