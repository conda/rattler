9/// A container for [`RepoDataRecord`]s that are returned from the [`super::Gateway`].
10///
11/// Records are stored as `Arc<RepoDataRecord>` so that cloning is cheap
12/// (reference count bump rather than deep copy).
13///
14/// `RepoData` uses internal reference counting, therefor it is relatively
15/// cheap to clone.
16#[derive(Debug, Default, Clone)]
17pub struct RepoData {
18    pub(crate) records: Vec<Arc<RepoDataRecord>>,
19    pub(crate) removed: RemovedPackages,
20}
21
22impl RepoData {
23    /// Returns the packages that the source lists as removed for the package
24    /// names the query fetched.
25    ///
26    /// Removed packages are never part of [`Self::iter`]. The match specs of
27    /// the query do not filter this collection: every removed entry of a
28    /// fetched package name is included, so callers look up the URL they are
29    /// interested in with [`RemovedPackages::contains`] or
30    /// [`RemovedPackages::get`].
31    pub fn removed(&self) -> &RemovedPackages {
32        &self.removed
33    }
34
35    /// Returns an iterator over all the records in this instance.
36    pub fn iter(&self) -> RepoDataIterator<'_> {
37        RepoDataIterator {
38            inner: self.records.iter(),
39        }
40    }
41
42    /// Returns the total number of records stored in this instance.
43    pub fn len(&self) -> usize {
44        self.records.len()
45    }
46
47    /// Returns true if there are no records stored in this instance.
48    pub fn is_empty(&self) -> bool {
49        self.records.is_empty()
50    }
51
52    /// Returns an iterator over the Arc-wrapped records.
53    ///
54    /// This is useful when you want to clone records cheaply (Arc clone
55    /// instead of deep copy).
56    pub fn iter_arc(&self) -> impl ExactSizeIterator<Item = &Arc<RepoDataRecord>> {
57        self.records.iter()
58    }
59
60    /// Returns true if there is at least one [`RepoDataRecord`] with
61    /// [`rattler_conda_types::package::RunExportsJson`] missing.
62    pub fn is_run_exports_missing(&self) -> bool {
63        self.iter().any(|r| r.package_record.run_exports.is_none())
64    }
65
66    /// Returns number of missing run exports from the underlying
67    /// [`RepoDataRecord`]s.
68    pub fn run_exports_missing(&self) -> usize {
69        self.iter()
70            .filter(|r| r.package_record.run_exports.is_none())
71            .count()
72    }
73}
74
75impl<'r> IntoIterator for &'r RepoData {
76    type Item = &'r RepoDataRecord;
77    type IntoIter = RepoDataIterator<'r>;
78
79    fn into_iter(self) -> Self::IntoIter {
80        self.iter()
81    }
82}

84/// The [`RemovedPackage`]s of a [`RepoData`] instance, indexed by their URL so
85/// that a lock file entry or a previously fetched record can be checked in
86/// constant time.
87#[derive(Debug, Default, Clone)]
88pub struct RemovedPackages {
89    by_url: HashMap<Url, RemovedPackage>,
90}
91
92impl RemovedPackages {
93    /// Returns true if the package served from `url` is listed as removed.
94    pub fn contains(&self, url: &Url) -> bool {
95        self.by_url.contains_key(url)
96    }
97
98    /// Returns the removed package that was served from `url`, if any.
99    pub fn get(&self, url: &Url) -> Option<&RemovedPackage> {
100        self.by_url.get(url)
101    }
102
103    /// Returns an iterator over all removed packages in no particular order.
104    pub fn iter(&self) -> impl ExactSizeIterator<Item = &RemovedPackage> {
105        self.by_url.values()
106    }
107
108    /// Returns the number of removed packages.
109    pub fn len(&self) -> usize {
110        self.by_url.len()
111    }
112
113    /// Returns true if no packages are listed as removed.
114    pub fn is_empty(&self) -> bool {
115        self.by_url.is_empty()
116    }
117
118    pub(crate) fn extend(&mut self, removed: impl IntoIterator<Item = RemovedPackage>) {
119        self.by_url.extend(
120            removed
121                .into_iter()
122                .map(|package| (package.url.clone(), package)),
123        );
124    }
125}
126
127impl<'r> IntoIterator for &'r RemovedPackages {
128    type Item = &'r RemovedPackage;
129    type IntoIter = std::collections::hash_map::Values<'r, Url, RemovedPackage>;
130
131    fn into_iter(self) -> Self::IntoIter {
132        self.by_url.values()
133    }
134}

136/// An iterator over the records in a [`RepoData`] instance.
137pub struct RepoDataIterator<'r> {
138    inner: std::slice::Iter<'r, Arc<RepoDataRecord>>,
139}
140
141impl<'r> Iterator for RepoDataIterator<'r> {
142    type Item = &'r RepoDataRecord;
143
144    fn next(&mut self) -> Option<Self::Item> {
145        self.inner.next().map(std::convert::AsRef::as_ref)
146    }
147
148    fn size_hint(&self) -> (usize, Option<usize>) {
149        self.inner.size_hint()
150    }
151}
152
153impl std::iter::FusedIterator for RepoDataIterator<'_> {}
154
155impl ExactSizeIterator for RepoDataIterator<'_> {
156    fn len(&self) -> usize {
157        self.inner.len()
158    }
