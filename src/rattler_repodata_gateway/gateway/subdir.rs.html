11/// Records for a single package, with precomputed unique dependency strings
12/// split between unconditional (base) deps and per-extra deps.
13///
14/// The split lets the gateway walk only the extras that are actually active
15/// for a name instead of cascading through every extra's dependencies. For
16/// packages without any extras (the common case), `unique_extra_deps` is
17/// empty.
18#[derive(Clone, Debug, Default)]
19pub struct PackageRecords {
20    /// All repodata records for this package.
21    pub records: Vec<Arc<RepoDataRecord>>,
22
23    /// Unique base dependency strings across all records.
24    pub unique_base_deps: Arc<[String]>,
25
26    /// Unique dependency strings per extra, deduplicated across all records.
27    pub unique_extra_deps: ExtraDeps,
28}
29
30/// Per-extra deduplicated dependency strings. Empty for packages without any
31/// extras (the common case).
32pub type ExtraDeps = Arc<HashMap<String, Arc<[String]>>>;
33
34/// Extract the unique dependency strings from a set of records, split into
35/// base deps and per-extra deps. Each output list is deduplicated, and a dep
36/// that appears in any record's base list is removed from every extra list
37/// (a base requirement is unconditional, so the solver does not need it gated
38/// on an extra).
39pub(crate) fn extract_unique_deps_split<'a>(
40    records: impl IntoIterator<Item = &'a RepoDataRecord>,
41) -> (Arc<[String]>, ExtraDeps) {
42    let mut base_seen = ahash::HashSet::<String>::default();
43    let mut base = Vec::new();
44    let mut per_extra: HashMap<String, (ahash::HashSet<String>, Vec<String>)> = HashMap::default();
45
46    for record in records {
47        for dep in &record.package_record.depends {
48            if base_seen.insert(dep.clone()) {
49                base.push(dep.clone());
50            }
51        }
52        for (extra, extra_deps) in record.package_record.extra_depends.iter() {
53            let entry = per_extra.entry(extra.clone()).or_default();
54            for dep in extra_deps {
55                if entry.0.insert(dep.clone()) {
56                    entry.1.push(dep.clone());
57                }
58            }
59        }
60    }
61
62    // Final pass: a dep that ended up in base must not appear in any extra
63    // list, regardless of the order records were visited.
64    let per_extra: HashMap<String, Arc<[String]>> = per_extra
65        .into_iter()
66        .filter_map(|(extra, (_, deps))| {
67            let filtered: Vec<String> = deps
68                .into_iter()
69                .filter(|d| !base_seen.contains(d))
70                .collect();
71            if filtered.is_empty() {
72                None
73            } else {
74                Some((extra, Arc::from(filtered)))
75            }
76        })
77        .collect();
78
79    (Arc::from(base), Arc::new(per_extra))
80}

82pub enum Subdir {
83    /// The subdirectory is missing from the channel, it is considered empty.
84    NotFound,
85
86    /// A subdirectory and the data associated with it.
87    Found(SubdirData),
88}
89
90impl Subdir {
91    /// Returns the names of all packages in the subdirectory.
92    pub fn package_names(&self) -> Option<Vec<String>> {
93        match self {
94            Subdir::Found(subdir) => Some(subdir.package_names()),
95            Subdir::NotFound => None,
96        }
97    }
98
99    /// Returns repodata revisions advertised by this subdirectory.
100    pub fn repodata_revisions(&self) -> &RepodataRevisions {
101        match self {
102            Subdir::Found(subdir) => subdir.repodata_revisions(),
103            Subdir::NotFound => empty_repodata_revisions(),
104        }
105    }
106
107    /// [CEP-42] channel relations from this subdir's repodata, or
108    /// `None` if absent / subdir not found.
109    ///
110    /// [CEP-42]: https://github.com/conda/ceps/blob/main/cep-0042.md
111    pub fn channel_relations(&self) -> Option<&ChannelRelations> {
112        match self {
113            Subdir::Found(subdir) => subdir.channel_relations(),
114            Subdir::NotFound => None,
115        }
116    }
117}

119/// Fetches and caches repodata records by package name for a specific
120/// subdirectory of a channel.
121pub struct SubdirData {
122    /// The client to use to fetch repodata.
123    client: Arc<dyn SubdirClient>,
124
125    /// Previously fetched or currently pending records (with precomputed deps).
126    records: CoalescedMap<PackageName, PackageRecords>,
127}
128
129impl SubdirData {
130    pub fn from_client<C: SubdirClient + 'static>(client: C) -> Self {
131        Self {
132            client: Arc::new(client),
133            records: CoalescedMap::new(),
134        }
135    }
136
137    pub async fn get_or_fetch_package_records(
138        &self,
139        name: &PackageName,
140        reporter: Option<Arc<dyn Reporter>>,
141    ) -> Result<PackageRecords, GatewayError> {
142        let client = self.client.clone();
143        let name_clone = name.clone();
144
145        self.records
146            .get_or_try_init(name.clone(), || async move {
147                client
148                    .fetch_package_records(&name_clone, reporter.as_deref())
149                    .await
150            })
151            .await
152            .map_err(|e| match e {
153                CoalescedGetError::Init(gateway_err) => gateway_err,
154                CoalescedGetError::CoalescedRequestFailed => GatewayError::IoError(
155                    "a coalesced request failed".to_string(),
156                    std::io::ErrorKind::Other.into(),
157                ),
158            })
159    }
160
161    /// Fetches the records for `name` without inserting them into the
162    /// long-lived per-name cache. A previously cached entry is still reused.
163    /// Used by streaming scans (e.g. the gateway's `who_needs` query) that
164    /// visit every package of a subdir exactly once and would otherwise
165    /// permanently fill the cache with millions of records.
166    pub async fn fetch_package_records_uncached(
167        &self,
168        name: &PackageName,
169        reporter: Option<&dyn Reporter>,
170    ) -> Result<Vec<Arc<RepoDataRecord>>, GatewayError> {
171        if let Some(cached) = self.records.get(name) {
172            return Ok(cached.records);
173        }
174        Ok(self
175            .client
176            .fetch_package_records(name, reporter)
177            .await?
178            .records)
179    }
180
181    /// The number of package names currently held in the per-name record
182    /// cache.
183    #[cfg(test)]
184    pub(crate) fn cached_package_count(&self) -> usize {
185        self.records.len()
186    }
187
188    pub fn package_names(&self) -> Vec<String> {
189        self.client.package_names()
190    }
191
192    pub fn repodata_revisions(&self) -> &RepodataRevisions {
193        self.client.repodata_revisions()
194    }
195
196    /// [CEP-42] channel relations from this subdir's repodata, if any.
197    ///
198    /// [CEP-42]: https://github.com/conda/ceps/blob/main/cep-0042.md
199    pub fn channel_relations(&self) -> Option<&ChannelRelations> {
200        self.client.channel_relations()
201    }
202}

204/// A client that can be used to fetch repodata for a specific subdirectory.
205#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
206#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
207pub trait SubdirClient: Send + Sync {
208    /// Fetches all repodata records for the package with the given name in a
209    /// channel subdirectory.
210    async fn fetch_package_records(
211        &self,
212        name: &PackageName,
213        reporter: Option<&dyn Reporter>,
214    ) -> Result<PackageRecords, GatewayError>;
215
216    /// Returns the names of all packages in the subdirectory.
217    fn package_names(&self) -> Vec<String>;
218
219    /// Returns repodata revisions advertised by the subdirectory.
220    fn repodata_revisions(&self) -> &RepodataRevisions {
221        empty_repodata_revisions()
222    }
223
224    /// [CEP-42] channel relations from this subdir's repodata, if any.
225    /// Sources without CEP-42 metadata (e.g. custom) keep the default.
226    ///
227    /// [CEP-42]: https://github.com/conda/ceps/blob/main/cep-0042.md
228    fn channel_relations(&self) -> Option<&ChannelRelations> {
229        None
230    }
231}
