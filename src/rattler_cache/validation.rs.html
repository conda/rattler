1//! Functionality to validate the contents of a Conda package.
2//!
3//! Almost all Conda packages contain a file `info/paths.json` that describes
4//! all the files the package contains. The [`validate_package_directory`]
5//! function validates that a directory containing an extracted Conda package
6//! archive actually contains the files as described by the `paths.json` file.
7//!
8//! Very old Conda packages do not contain a `paths.json` file. These packages
9//! contain a (deprecated) `files` file as well as optionally a `has_prefix` and
10//! some other files. If the `paths.json` file is missing these deprecated files
11//! are used instead to reconstruct a [`PathsJson`] object. See
12//! [`PathsJson::from_deprecated_package_directory`] for more information.
13

25/// The mode in which the validation should be performed.
26#[derive(Debug, Copy, Clone, Eq, PartialEq, Default, Hash)]
27pub enum ValidationMode {
28    /// Only check if the package directory exists and contains a valid index.json.
29    /// Does not validate individual files or paths.json. This is the fastest validation
30    /// mode but provides minimal guarantees about package integrity.
31    #[default]
32    Skip,
33
34    /// Only check if the files exists. Do not check if the hashes match.
35    Fast,
36
37    /// Check if the files exists and the content matches the hashes.
38    Full,
39}

41/// An error that is returned by [`validate_package_directory`] if the contents
42/// of the directory seems to be corrupted.
43#[derive(Debug, thiserror::Error)]
44pub enum PackageValidationError {
45    /// Neither a `paths.json` file nor a deprecated `files` file was found.
46    #[error("neither a 'paths.json' or a deprecated 'files' file was found")]
47    MetadataMissing,
48
49    /// An error occurred while reading the `paths.json` file.
50    #[error("failed to read 'paths.json' file")]
51    ReadPathsJsonError(#[source] std::io::Error),
52
53    /// An error occurred while reading the deprecated `files` file.
54    #[error("failed to read validation data from deprecated files")]
55    ReadDeprecatedPathsJsonError(#[source] std::io::Error),
56
57    /// The path seems to be corrupted.
58    #[error("the path '{0}' seems to be corrupted")]
59    CorruptedEntry(PathBuf, #[source] PackageEntryValidationError),
60
61    /// An error occurred while reading the `index.json` file.
62    #[error("failed to read 'index.json'")]
63    ReadIndexJsonError(#[source] std::io::Error),
64
65    /// The `wheel.json` file could not be read.
66    #[error("failed to read 'wheel.json'")]
67    ReadWheelJsonError(#[source] std::io::Error),
68
69    /// The cached wheel layout was produced by a different converter version.
70    #[error(
71        "the package was converted from a wheel with conversion version {found}, expected {expected}"
72    )]
73    OutdatedWheelConversion {
74        /// The version recorded in `wheel.json`.
75        found: u64,
76        /// The version this build of rattler writes.
77        expected: u64,
78    },
79}

81/// An error that indicates that a specific file in a package archive directory
82/// seems to be corrupted.
83#[derive(Debug, thiserror::Error)]
84pub enum PackageEntryValidationError {
85    /// An error occurred while reading the metadata of the file.
86    #[error("failed to retrieve file metadata'")]
87    GetMetadataFailed(#[source] std::io::Error),
88
89    /// The file does not exist.
90    #[error("the file does not exist")]
91    NotFound,
92
93    /// The file is not a symbolic link.
94    #[error("expected a symbolic link")]
95    ExpectedSymlink,
96
97    /// The file is not a directory.
98    #[error("expected a directory")]
99    ExpectedDirectory,
100
101    /// The size of the file does not match the expected size.
102    #[error("incorrect size, expected {0} but file on disk is {1}")]
103    IncorrectSize(u64, u64),
104
105    /// An IO error occurred while reading the file.
106    #[error("an io error occurred")]
107    IoError(#[from] std::io::Error),
108
109    /// The SHA256 hash of the file does not match the expected hash.
110    #[error("sha256 hash mismatch, expected '{0}' but file on disk is '{1}'")]
111    HashMismatch(String, String),
112}

114/// Determine whether the files in the specified directory match what is
115/// expected according to the `info/paths.json` file in the same directory.
116///
117/// If the `info/paths.json` file could not be found this function tries to
118/// reconstruct the information from older deprecated methods. See
119/// [`PathsJson::from_deprecated_package_directory`].
120///
121/// If validation succeeds the parsed [`PathsJson`] object is returned which
122/// contains information about the files in the archive.
123pub fn validate_package_directory(
124    package_dir: &Path,
125    mode: ValidationMode,
126) -> Result<(IndexJson, PathsJson), PackageValidationError> {
127    // Validate that there is a valid IndexJson
128    let index_json = IndexJson::from_package_directory(package_dir)
129        .map_err(PackageValidationError::ReadIndexJsonError)?;
130
131    // Read the 'paths.json' file which describes all files that should be present.
132    // If the file could not be found try reconstructing the paths information
133    // from deprecated files in the package directory.
134    let paths = match PathsJson::from_package_directory(package_dir) {
135        Err(e) if e.kind() == ErrorKind::NotFound => {
136            match PathsJson::from_deprecated_package_directory(package_dir) {
137                Ok(paths) => paths,
138                Err(e) if e.kind() == ErrorKind::NotFound => {
139                    return Err(PackageValidationError::MetadataMissing);
140                }
141                Err(e) => return Err(PackageValidationError::ReadDeprecatedPathsJsonError(e)),
142            }
143        }
144        Err(e) => return Err(PackageValidationError::ReadPathsJsonError(e)),
145        Ok(paths) => paths,
146    };
147
148    check_wheel_conversion(package_dir)?;
149
150    // In Skip mode, only validate that index.json and paths.json exist and are readable.
151    // Skip all file validation checks.
152    if mode == ValidationMode::Skip {
153        return Ok((index_json, paths));
154    }
155
156    // Validate all the entries
157    validate_package_directory_from_paths(package_dir, &paths, mode)
158        .map_err(|(path, err)| PackageValidationError::CorruptedEntry(path, err))?;
159
160    Ok((index_json, paths))
161}

163/// Archive hashes cannot detect changes to the wheel conversion layout, so a
164/// package converted from a wheel by an older layout is invalid.
165fn check_wheel_conversion(package_dir: &Path) -> Result<(), PackageValidationError> {
166    match WheelJson::from_package_directory(package_dir) {
167        Ok(wheel) if wheel.conversion_version != CONVERSION_VERSION => {
168            Err(PackageValidationError::OutdatedWheelConversion {
169                found: wheel.conversion_version,
170                expected: CONVERSION_VERSION,
171            })
172        }
173        Ok(_) => Ok(()),
174        Err(e) if e.kind() == ErrorKind::NotFound => Ok(()),
175        Err(e) => Err(PackageValidationError::ReadWheelJsonError(e)),
176    }
177}

179/// Like [`validate_package_directory`], but in [`ValidationMode::Skip`] only
180/// checks that the package's file list exists instead of parsing it, and
181/// returns no [`PathsJson`].
182///
183/// Parsing `paths.json` is a large part of validating a cached package (it
184/// lists, and hex-decodes a hash for, every file), and in skip mode nothing
185/// uses the result unless the caller needs the file list, which it can read
186/// itself. The package cache uses this so that consumers that don't need the
187/// file list (or read it elsewhere) don't pay for parsing it.
188pub(crate) fn validate_package_directory_lazily(
189    package_dir: &Path,
190    mode: ValidationMode,
191) -> Result<(IndexJson, Option<PathsJson>), PackageValidationError> {
192    if mode != ValidationMode::Skip {
193        let (index_json, paths_json) = validate_package_directory(package_dir, mode)?;
194        return Ok((index_json, Some(paths_json)));
195    }
196
197    let index_json = IndexJson::from_package_directory(package_dir)
198        .map_err(PackageValidationError::ReadIndexJsonError)?;
199    let info = package_dir.join("info");
200    if !info.join("paths.json").is_file() && !info.join("files").is_file() {
201        return Err(PackageValidationError::MetadataMissing);
202    }
203    check_wheel_conversion(package_dir)?;
204    Ok((index_json, None))
205}

207/// Determine whether the files in the specified directory match wat is expected
208/// according to the passed in [`PathsJson`].
209pub fn validate_package_directory_from_paths(
210    package_dir: &Path,
211    paths: &PathsJson,
212    mode: ValidationMode,
213) -> Result<(), (PathBuf, PackageEntryValidationError)> {
214    // Check every entry in the PathsJson object
215    paths
216        .paths
217        .par_iter()
218        .with_min_len(1000)
219        .try_for_each(|entry| {
220            validate_package_entry(package_dir, entry, mode)
221                .map_err(|e| (entry.relative_path.clone(), e))
222        })
223}

225/// Determine whether the information in the [`PathsEntry`] matches the file in
226/// the package directory.
227fn validate_package_entry(
228    package_dir: &Path,
229    entry: &PathsEntry,
230    mode: ValidationMode,
231) -> Result<(), PackageEntryValidationError> {
232    let path = package_dir.join(&entry.relative_path);
233
234    // Validate based on the type of path
235    match entry.path_type {
236        PathType::HardLink => validate_package_hard_link_entry(path, entry, mode),
237        PathType::SoftLink => validate_package_soft_link_entry(path, entry, mode),
238        PathType::Directory => validate_package_directory_entry(path, entry, mode),
239    }
240}

242/// Determine whether the information in the [`PathsEntry`] matches the file at
243/// the specified path.
244fn validate_package_hard_link_entry(
245    path: PathBuf,
246    entry: &PathsEntry,
247    mode: ValidationMode,
248) -> Result<(), PackageEntryValidationError> {
249    debug_assert!(entry.path_type == PathType::HardLink);
250
251    if mode == ValidationMode::Fast {
252        if !path.is_file() {
253            return Err(PackageEntryValidationError::NotFound);
254        }
255        return Ok(());
256    }
257
258    // Short-circuit if we have no validation reference
259    if entry.sha256.is_none() && entry.size_in_bytes.is_none() {
260        if !path.is_file() {
261            return Err(PackageEntryValidationError::NotFound);
262        }
263        return Ok(());
264    }
265
266    // Open the file for reading
267    let file = match std::fs::File::open(&path) {
268        Ok(file) => file,
269        Err(e) if e.kind() == ErrorKind::NotFound => {
270            return Err(PackageEntryValidationError::NotFound);
271        }
272        Err(e) => return Err(PackageEntryValidationError::IoError(e)),
273    };
274
275    // Validate the size of the file
276    if let Some(size_in_bytes) = entry.size_in_bytes {
277        let actual_file_len = file
278            .metadata()
279            .map_err(PackageEntryValidationError::IoError)?
280            .len();
281        if size_in_bytes != actual_file_len {
282            return Err(PackageEntryValidationError::IncorrectSize(
283                size_in_bytes,
284                actual_file_len,
285            ));
286        }
287    }
288
289    // Check the SHA256 hash of the file
290    if let Some(expected_hash) = &entry.sha256 {
291        // Determine the hash of the file on disk
292        let mut file = BufReader::with_capacity(64 * 1024, file);
293        let mut writer = HashingWriter::<_, Sha256>::new(std::io::sink());
294        std::io::copy(&mut file, &mut writer)?;
295        let (_, hash) = writer.finalize();
296
297        // Compare the two hashes
298        if expected_hash != &hash {
299            return Err(PackageEntryValidationError::HashMismatch(
300                hex::encode(expected_hash),
301                hex::encode(hash),
302            ));
303        }
304    }
305
306    Ok(())
307}

309/// Determine whether the information in the [`PathsEntry`] matches the symbolic
310/// link at the specified path.
311fn validate_package_soft_link_entry(
312    path: PathBuf,
313    entry: &PathsEntry,
314    _mode: ValidationMode,
315) -> Result<(), PackageEntryValidationError> {
316    debug_assert!(entry.path_type == PathType::SoftLink);
317
318    if !path.is_symlink() {
319        return Err(PackageEntryValidationError::ExpectedSymlink);
320    }
321
322    // TODO: Validate symlink content. Dont validate the SHA256 hash of the file
323    // because since a symlink will most likely point to another file added as a
324    // hardlink by the package this is double work. Instead check that the
325    // symlink is correct e.g. `../a` points to the same file as `b/../../a` but
326    // they are different.
327
328    Ok(())
329}

331/// Determine whether the information in the [`PathsEntry`] matches the
332/// directory at the specified path.
333fn validate_package_directory_entry(
334    path: PathBuf,
335    entry: &PathsEntry,
336    _mode: ValidationMode,
337) -> Result<(), PackageEntryValidationError> {
338    debug_assert!(entry.path_type == PathType::Directory);
339
340    if path.is_dir() {
341        Ok(())
342    } else {
343        Err(PackageEntryValidationError::ExpectedDirectory)
344    }
345}

473    /// In skip mode the file list is only checked for existence, not parsed;
474    /// other modes still return it.
475    #[test]
476    fn test_lazy_validation() {
477        let temp_dir = tempfile::tempdir().unwrap();
478        let info = temp_dir.path().join("info");
479        std::fs::create_dir_all(&info).unwrap();
480        std::fs::write(
481            info.join("index.json"),
482            r#"{"name": "foo", "version": "1.0", "build": "0", "build_number": 0}"#,
483        )
484        .unwrap();
485
486        // No file list at all.
487        assert_matches!(
488            validate_package_directory_lazily(temp_dir.path(), ValidationMode::Skip),
489            Err(PackageValidationError::MetadataMissing)
490        );
491
492        std::fs::write(
493            info.join("paths.json"),
494            r#"{"paths": [], "paths_version": 1}"#,
495        )
496        .unwrap();
497        let (index, paths) =
498            validate_package_directory_lazily(temp_dir.path(), ValidationMode::Skip).unwrap();
499        assert_eq!(index.name.as_normalized(), "foo");
500        assert!(paths.is_none());
501        let (_, paths) =
502            validate_package_directory_lazily(temp_dir.path(), ValidationMode::Fast).unwrap();
503        assert!(paths.unwrap().paths.is_empty());
504    }
