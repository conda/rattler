42/// A [`PackageCache`] manages a cache of extracted Conda packages on disk.
43///
44/// The store does not provide an implementation to get the data into the store.
45/// Instead, this is left up to the user when the package is requested. If the
46/// package is found in the cache it is returned immediately. However, if the
47/// cache is stale a user defined function is called to populate the cache. This
48/// separates the concerns between caching and fetching of the content.
49#[derive(Clone)]
50pub struct PackageCache {
51    inner: Arc<PackageCacheInner>,
52    cache_origin: bool,
53}

66/// A snapshot of the packages present in a [`PackageCache`] at the moment it
67/// was created.
68///
69/// Building the snapshot reads each layer directory once, which is far cheaper
70/// than probing packages one by one. In exchange the snapshot is fixed: a
71/// package added to the cache afterwards is not reported as present. Create a
72/// new snapshot when a current view is needed.
73///
74/// Presence means the package directory exists, not that its contents are
75/// complete or valid. An interrupted extraction leaves a directory behind that
76/// is reported as present and is only rejected by validation on use.
77#[derive(Debug, Clone)]
78pub struct CacheIndex {
79    /// Directory names as produced by [`CacheKey::to_path_segment`], mapped to
80    /// the sha256 each layer records for that entry (absent when the entry
81    /// predates hash recording, or its metadata could not be read). Every
82    /// layer holding the name contributes, in layer order: lookup walks past
83    /// a layer whose hash does not match, so the index must too.
84    entries: HashMap<String, Vec<Option<Sha256Hash>>>,
85    cache_origin: bool,
86}
87
88impl CacheIndex {
89    /// Returns whether the package a [`RepoDataRecord`] describes is present
90    /// in the cache.
91    ///
92    /// Both halves of the query come from the record, so its hashes and its
93    /// origin cannot disagree - which is the failure mode of assembling the
94    /// two by hand with [`Self::contains_url`].
95    ///
96    /// A `file:` URL is looked up by its path, because that is how the
97    /// installer stores it: local packages go through
98    /// [`PackageCache::get_or_fetch_from_path`], everything else through
99    /// [`PackageCache::get_or_fetch_from_url_with_retry`].
100    pub fn contains_record(&self, record: &RepoDataRecord) -> bool {
101        if record.url.scheme() == "file"
102            && let Ok(path) = record.url.to_file_path()
103        {
104            return self.contains_path(&record.package_record, &path);
105        }
106        self.contains_url(&record.package_record, &record.url)
107    }
108
109    /// Returns whether a package fetched from `url` is present in the cache.
110    ///
111    /// The origin is applied here rather than by the caller so that the answer
112    /// always matches the way
113    /// [`PackageCache::get_or_fetch_from_url_with_retry`] stores packages. It
114    /// only affects caches built with [`PackageCache::with_cached_origin`].
115    pub fn contains_url(&self, pkg: impl Into<CacheKey>, url: &Url) -> bool {
116        let mut cache_key = pkg.into();
117        if self.cache_origin {
118            cache_key = cache_key.with_url(url.clone());
119        }
120        self.contains_key(&cache_key)
121    }
122
123    /// Returns whether a package fetched from `path` is present in the cache.
124    ///
125    /// The path counterpart of [`Self::contains_url`], matching the way
126    /// [`PackageCache::get_or_fetch_from_path`] stores packages.
127    pub fn contains_path(&self, pkg: impl Into<CacheKey>, path: &Path) -> bool {
128        let mut cache_key = pkg.into();
129        if self.cache_origin {
130            cache_key = cache_key.with_path(path);
131        }
132        self.contains_key(&cache_key)
133    }
134
135    fn contains_key(&self, cache_key: &CacheKey) -> bool {
136        let Ok(segment) = cache_key.to_path_segment() else {
137            return false;
138        };
139        let Some(cached_sha256s) = self.entries.get(&segment) else {
140            return false;
141        };
142
143        // The directory name does not include the hash, so a package rebuilt
144        // under the same name, version and build string shares an entry with
145        // the one on disk. `get_or_fetch` re-downloads on a hash mismatch, so
146        // reporting such an entry as present would promise an install that
147        // still needs the network. Apply the very same rule here, to every
148        // layer: lookup continues past a mismatching layer, so a match
149        // anywhere is a match.
150        cached_sha256s.iter().any(|cached_sha256| {
151            !cache_lock::sha256_mismatch(cache_key.sha256().as_ref(), cached_sha256.as_ref())
152        })
153    }
154
155    /// Returns the number of packages in the snapshot.
156    pub fn len(&self) -> usize {
157        self.entries.len()
158    }
159
160    /// Returns true if the snapshot contains no packages.
161    pub fn is_empty(&self) -> bool {
162        self.entries.is_empty()
163    }
164}

264/// Returns `true` if `path` lives on a filesystem that is *mounted* read-only.
265///
266/// Permission bits miss filesystems mounted read-only (CVMFS, squashfs,
267/// read-only bind mounts): a mode-0755 directory still reports as writable, so
268/// we additionally check the `ST_RDONLY` mount flag via `statvfs(3)`.
269#[cfg(unix)]
270fn is_mounted_readonly(path: &Path) -> bool {
271    rustix::fs::statvfs(path)
272        .is_ok_and(|s| s.f_flag.contains(rustix::fs::StatVfsMountFlags::RDONLY))
273}
274
275#[cfg(not(unix))]
276fn is_mounted_readonly(_path: &Path) -> bool {
277    false
278}

289    /// Validate the packages.
290    ///
291    /// An entry this cache object has not touched before is validated from
292    /// disk, just as [`Self::validate_or_fetch`] would: layers hold entries
293    /// written by other processes or earlier runs, and a lookup that only
294    /// believed its own in-memory bookkeeping would refetch all of those.
295    pub async fn try_validate(
296        &self,
297        cache_key: &CacheKey,
298    ) -> Result<CacheMetadata, PackageCacheLayerError> {
299        let cache_entry = self
300            .packages
301            .entry(cache_key.clone().into())
302            .or_default()
303            .clone();
304        let mut cache_entry = cache_entry.lock().await;
305        let cache_path = self.path.join(cache_key.to_path_segment()?);
306
307        match validate_package_common::<
308            fn(PathBuf) -> _,
309            Pin<Box<dyn Future<Output = Result<(), _>> + Send>>,
310            std::io::Error,
311        >(
312            cache_path,
313            cache_entry.last_revision,
314            cache_key.sha256.as_ref(),
315            None,
316            None,
317            self.validation_mode,
318        )
319        .await
320        {
321            Ok(cache_metadata) => {
322                cache_entry.last_revision = Some(cache_metadata.revision);
323                cache_entry.last_sha256 = cache_metadata.sha256;
324                Ok(cache_metadata)
325            }
326            Err(err) => Err(err),
327        }
328    }
329
330    /// Validate the package, and fetch it if invalid.
331    pub async fn validate_or_fetch<F, Fut, E>(
332        &self,
333        fetch: F,
334        cache_key: &CacheKey,
335        reporter: Option<Arc<dyn CacheReporter>>,
336    ) -> Result<CacheMetadata, PackageCacheLayerError>
337    where
338        F: (Fn(PathBuf) -> Fut) + Send + 'static,
339        Fut: Future<Output = Result<(), E>> + Send + 'static,
340        E: std::error::Error + Send + Sync + 'static,
341    {
342        let entry = self
343            .packages
344            .entry(cache_key.clone().into())
345            .or_default()
346            .clone();
347
348        let mut cache_entry = entry.lock().await;
349        let cache_path = self.path.join(cache_key.to_path_segment()?);
350
351        match validate_package_common(
352            cache_path,
353            cache_entry.last_revision,
354            cache_key.sha256.as_ref(),
355            Some(fetch),
356            reporter,
357            self.validation_mode,
358        )
359        .await
360        {
361            Ok(cache_metadata) => {
362                cache_entry.last_revision = Some(cache_metadata.revision);
363                cache_entry.last_sha256 = cache_metadata.sha256;
364                Ok(cache_metadata)
365            }
366            Err(e) => Err(e),
367        }
368    }

390    /// Acquires a global lock on the package cache.
391    ///
392    /// This lock can be used to coordinate multiple package operations,
393    /// reducing the overhead of acquiring individual locks for each package.
394    /// The lock is held until the returned `CacheGlobalLock` is dropped.
395    ///
396    /// This is particularly useful when installing many packages at once,
397    /// as it significantly reduces the number of file locking syscalls.
398    pub async fn acquire_global_lock(&self) -> Result<CacheGlobalLock, PackageCacheError> {
399        // Use the first writable layer's path for the global cache lock
400        let (_, writable_layers) = self.split_layers();
401        let cache_layer = writable_layers
402            .first()
403            .ok_or(PackageCacheError::NoWritableLayers)?;
404
405        let lock_file_path = cache_layer.path.join(".cache.lock");
406
407        // Ensure the directory exists
408        tokio_fs::create_dir_all(&cache_layer.path)
409            .await
410            .map_err(|e| {
411                PackageCacheError::LayerError(Box::new(PackageCacheLayerError::LockError(
412                    format!(
413                        "failed to create cache directory: '{}'",
414                        cache_layer.path.display()
415                    ),
416                    e,
417                )))
418            })?;
419
420        CacheGlobalLock::acquire(&lock_file_path)
421            .await
422            .map_err(|e| PackageCacheError::LayerError(Box::new(e)))
423    }

448    /// Returns a snapshot of the packages currently present in the cache.
449    ///
450    /// All layers are scanned and merged into a single view. Layers that do
451    /// not exist on disk yet contribute nothing. See [`CacheIndex`] for what
452    /// the snapshot does and does not guarantee.
453    ///
454    /// The scan touches every entry in every layer, so on a warm cache this is
455    /// thousands of filesystem calls. It runs on a blocking thread to keep
456    /// them off the async runtime.
457    pub async fn index(&self) -> std::io::Result<CacheIndex> {
458        let layer_paths: Vec<PathBuf> = self
459            .inner
460            .layers
461            .iter()
462            .map(|layer| layer.path.clone())
463            .collect();
464        let cache_origin = self.cache_origin;
465
466        let scan = tokio::task::spawn_blocking(move || {
467            let mut entries: HashMap<String, Vec<Option<Sha256Hash>>> = HashMap::new();
468            for layer_path in layer_paths {
469                let dir = match fs_err::read_dir(&layer_path) {
470                    Ok(dir) => dir,
471                    Err(err) if err.kind() == std::io::ErrorKind::NotFound => continue,
472                    Err(err) => return Err(err),
473                };
474
475                for entry in dir {
476                    let entry = entry?;
477                    if entry.file_type()?.is_dir()
478                        && let Some(name) = entry.file_name().to_str()
479                    {
480                        let sha256 = cache_lock::peek_sha256(&entry.path());
481                        // Every layer holding the name contributes: lookup
482                        // walks past a layer whose hash does not match.
483                        entries.entry(name.to_owned()).or_default().push(sha256);
484                    }
485                }
486            }
487            Ok(entries)
488        })
489        .await;
490
491        let entries = match scan {
492            Ok(entries) => entries?,
493            Err(err) => match err.try_into_panic() {
494                Ok(panic) => std::panic::resume_unwind(panic),
495                Err(err) => return Err(std::io::Error::other(err)),
496            },
497        };
498
499        Ok(CacheIndex {
500            entries,
501            cache_origin,
502        })
503    }

517    /// Returns the directory that contains the specified package.
518    ///
519    /// If the package was previously successfully fetched and stored in the
520    /// cache the directory containing the data is returned immediately. If
521    /// the package was not previously fetch the filesystem is checked to
522    /// see if a directory with valid package content exists. Otherwise, the
523    /// user provided `fetch` function is called to populate the cache.
524    ///
525    /// ## Layer Priority
526    ///
527    /// Layers are checked in the order they were provided to [`PackageCache::new_layered`].
528    /// If a valid package is found in any layer, it is returned immediately. If no valid
529    /// package is found in any layer, the package is fetched and written to the first
530    /// writable layer.
531    ///
532    /// If the package is already being fetched by another task/thread the
533    /// request is coalesced. No duplicate fetch is performed.
534    pub async fn get_or_fetch<F, Fut, E>(
535        &self,
536        pkg: impl Into<CacheKey>,
537        fetch: F,
538        reporter: Option<Arc<dyn CacheReporter>>,
539    ) -> Result<CacheMetadata, PackageCacheError>
540    where
541        F: (Fn(PathBuf) -> Fut) + Send + 'static,
542        Fut: Future<Output = Result<(), E>> + Send + 'static,
543        E: std::error::Error + Send + Sync + 'static,
544    {
545        let cache_key = pkg.into();
546
547        // Reject keys that could escape the cache root (GHSA-h672-p7h7-97v9).
548        let cache_segment = cache_key.to_path_segment()?;
549
550        let (_, writable_layers) = self.split_layers();
551
552        for layer in self.inner.layers.iter() {
553            let cache_path = layer.path.join(&cache_segment);
554
555            if cache_path.exists() {
556                match layer.try_validate(&cache_key).await {
557                    Ok(lock) => {
558                        return Ok(lock);
559                    }
560                    Err(PackageCacheLayerError::InvalidPackage) => {
561                        // Log and continue to the next layer
562                        tracing::warn!(
563                            "Invalid package in layer at path {:?}, trying next layer.",
564                            layer.path
565                        );
566                    }
567                    Err(PackageCacheLayerError::PackageNotFound) => {
568                        // Log and continue to the next layer
569                        tracing::debug!(
570                            "Package not found in layer at path {:?}, trying next layer.",
571                            layer.path
572                        );
573                    }
574                    Err(err) => return Err(err.into()),
575                }
576            }
577        }
578
579        // No matches in all layers, let's write to the first writable layer
580        tracing::debug!("no matches in all layers. writing to first writable layer");
581        if let Some(layer) = writable_layers.first() {
582            return match layer.validate_or_fetch(fetch, &cache_key, reporter).await {
583                Ok(cache_metadata) => Ok(cache_metadata),
584                Err(e) => Err(e.into()),
585            };
586        }
587
588        Err(PackageCacheError::NoWritableLayers)
589    }

615    /// Returns the directory that contains the specified package.
616    ///
617    /// This is a convenience wrapper around `get_or_fetch` which fetches the
618    /// package from the given path if the package could not be found in the
619    /// cache.
620    ///
621    /// If a [`PackageRecord`] is provided, the cache key is derived from the
622    /// record, which includes its hashes. This ensures that packages that
623    /// share the same name, version and build string but have different
624    /// content do not collide in the cache. Otherwise, the cache key is
625    /// derived from the package filename only.
626    pub async fn get_or_fetch_from_path(
627        &self,
628        path: &Path,
629        record: Option<&PackageRecord>,
630        reporter: Option<Arc<dyn CacheReporter>>,
631    ) -> Result<CacheMetadata, PackageCacheError> {
632        let path_buf = path.to_path_buf();
633        let mut cache_key: CacheKey = match record {
634            Some(record) => record.into(),
635            None => CondaArchiveIdentifier::try_from_path(&path_buf)
636                .unwrap()
637                .into(),
638        };
639        if self.cache_origin {
640            cache_key = cache_key.with_path(path);
641        }
642
643        self.get_or_fetch(
644            cache_key,
645            move |destination| {
646                let path_buf = path_buf.clone();
647                async move {
648                    rattler_package_streaming::tokio::fs::extract(&path_buf, &destination)
649                        .await
650                        .map(|_| ())
651                }
652            },
653            reporter,
654        )
655        .await
656    }

658    /// Returns the directory that contains the specified package.
659    ///
660    /// This is a convenience wrapper around `get_or_fetch` which fetches the
661    /// package from the given URL if the package could not be found in the
662    /// cache.
663    ///
664    /// This function assumes that the `client` is already configured with a
665    /// retry middleware that will retry any request that fails. This function
666    /// uses the passed in `retry_policy` if, after the request has been sent
667    /// and the response is successful, streaming of the package data fails
668    /// and the whole request must be retried.
669    ///
670    /// The optional `concurrent_requests_semaphore` limits the number of
671    /// simultaneous downloads. A permit is acquired only when a download is
672    /// actually started (i.e. the package is not already cached), and is held
673    /// for the duration of the download and extraction.
674    #[instrument(skip_all, fields(url=%url))]
675    pub async fn get_or_fetch_from_url_with_retry(
676        &self,
677        pkg: impl Into<CacheKey>,
678        url: Url,
679        client: LazyClient,
680        retry_policy: impl RetryPolicy + Send + 'static + Clone,
681        reporter: Option<Arc<dyn CacheReporter>>,
682        concurrent_requests_semaphore: Option<Arc<tokio::sync::Semaphore>>,
683    ) -> Result<CacheMetadata, PackageCacheError> {
684        let request_start = SystemTime::now();
685        // Convert into cache key
686        let mut cache_key = pkg.into();
687        if self.cache_origin {
688            cache_key = cache_key.with_url(url.clone());
689        }
690        // Sha256 of the expected package
691        let sha256 = cache_key.sha256();
692        let md5 = cache_key.md5();
693        let download_reporter = reporter.clone();
694        // Get or fetch the package, using the specified fetch function
695        self.get_or_fetch(cache_key, move |destination| {
696            let url = url.clone();
697            let client = client.clone();
698            let retry_policy = retry_policy.clone();
699            let download_reporter = download_reporter.clone();
700            let concurrent_requests_semaphore = concurrent_requests_semaphore.clone();
701            async move {
702                // Acquire a permit to limit the number of concurrent download
703                // and extraction operations. The permit is held until the
704                // closure returns and is released automatically when dropped.
705                let _permit = match concurrent_requests_semaphore {
706                    Some(semaphore) => Some(
707                        semaphore
708                            .acquire_owned()
709                            .await
710                            .expect("semaphore should not be closed"),
711                    ),
712                    None => None,
713                };
714
715                let mut current_try = 0;
716                // Retry until the retry policy says to stop
717                loop {
718                    current_try += 1;
719                    tracing::debug!("downloading {} to {}", &url, destination.display());
720                    // Extract the package
721                    let result = rattler_package_streaming::reqwest::tokio::extract(
722                        client.client().clone(),
723                        url.clone(),
724                        &destination,
725                        sha256,
726                        download_reporter.clone().map(|reporter| Arc::new(PassthroughReporter {
727                            reporter,
728                            index: Mutex::new(None),
729                        }) as Arc::<dyn DownloadReporter>),
730                    )
731                        .await;
732
733                    let err = match result {
734                        Ok(result) => {
735                            // HACK: Only check one hash. Sometimes it occurs that the server
736                            // reports the wrong md5 hash while the Sha256 hash is valid. We used to
737                            // error on this case. However, the Sha256 hash is already secure enough
738                            // that we can ignore this case.
739                            //
740                            // For context, conda itself only checks one hash.
741                            if let Some(sha256) = sha256 {
742                                if sha256 != result.sha256 {
743                                    // Delete the package if the hash does not match.
744                                    // Failure here is non-fatal: the TempDir guard will
745                                    // clean up on drop; log and continue so the retry
746                                    // loop can proceed rather than panicking.
747                                    if let Err(e) = tokio_fs::remove_dir_all(&destination).await {
748                                        tracing::warn!(
749                                            "failed to remove destination on sha256 mismatch \
750                                             (will be cleaned up on drop): {e}"
751                                        );
752                                    }
753                                    return Err(ExtractError::HashMismatch {
754                                        url: url.clone().redact().to_string(),
755                                        destination: destination.display().to_string(),
756                                        expected: hex::encode(sha256),
757                                        actual: hex::encode(result.sha256),
758                                        total_size: result.total_size,
759                                    });
760                                }
761                            }  else if let Some(md5) = md5
762                                && md5 != result.md5 {
763                                    // Delete the package if the hash does not match.
764                                    // Failure here is non-fatal: the TempDir guard will
765                                    // clean up on drop; log and continue so the retry
766                                    // loop can proceed rather than panicking.
767                                    if let Err(e) = tokio_fs::remove_dir_all(&destination).await {
768                                        tracing::warn!(
769                                            "failed to remove destination on md5 mismatch \
770                                             (will be cleaned up on drop): {e}"
771                                        );
772                                    }
773                                    return Err(ExtractError::HashMismatch {
774                                        url: url.clone().redact().to_string(),
775                                        destination: destination.display().to_string(),
776                                        expected: hex::encode(md5),
777                                        actual: hex::encode(result.md5),
778                                        total_size: result.total_size,
779                                    });
780                                }
781                            return Ok(());
782                        }
783                        Err(err) => err,
784                    };
785
786                    // Check if we should retry this error. We assume that the
787                    // user has middleware installed that handles connection
788                    // retries, but we still need to handle streaming failures
789                    // that occur after a successful response.
790
791                    if !err.should_retry() {
792                        return Err(err);
793                    }
794
795                    // Determine whether to retry based on the retry policy
796                    let execute_after = match retry_policy.should_retry(request_start, current_try) {
797                        RetryDecision::Retry { execute_after } => execute_after,
798                        RetryDecision::DoNotRetry => return Err(err),
799                    };
800                    let duration = execute_after.duration_since(SystemTime::now()).unwrap_or(Duration::ZERO);
801
802                    // Wait for a second to let the remote service restore itself. This increases the
803                    // chance of success.
804                    tracing::warn!(
805                        "failed to download and extract {} to {}: {}. Retry #{}, Sleeping {:?} until the next attempt...",
806                        &url,
807                        destination.display(),
808                        err,
809                        current_try,
810                        duration
811                    );
812                    tokio::time::sleep(duration).await;
813                }
814            }
815        }, reporter)
816            .await
817    }
818}

820/// Renames `from` to `to`. Plain `rename` on non-Windows; retries transient
821/// file-locking errors on Windows. See the Windows-only definition.
822#[cfg(not(windows))]
823async fn rename_with_retry(from: &Path, to: &Path) -> std::io::Result<()> {
824    tokio_fs::rename(from, to).await
825}
826
827/// Renames `from` to `to`, retrying transient Windows file-locking errors
828/// (`ERROR_ACCESS_DENIED`, `ERROR_SHARING_VIOLATION`) with exponential backoff.
829/// Antivirus scanners and the search indexer hold short-lived handles to
830/// freshly-extracted files, breaking `MoveFileEx`.
831/// See <https://github.com/prefix-dev/pixi/issues/5660>.
832///
833/// This deliberately uses `tokio::fs::rename` instead of the `fs_err` wrapper:
834/// `fs_err` wraps the OS error in a custom `io::Error` whose `raw_os_error()`
835/// returns `None` (and which exposes no `source()`), so the transient-error
836/// check below would never match and the rename would fail on the first
837/// attempt. The caller adds both paths to the error message, so no context is
838/// lost. See <https://github.com/prefix-dev/rattler-build/issues/2474>.
839#[cfg(windows)]
840async fn rename_with_retry(from: &Path, to: &Path) -> std::io::Result<()> {
841    use rattler_networking::retry_policies::ExponentialBackoff;
842
843    // Roughly 25s of total wait: AV scans of large freshly-extracted packages
844    // (e.g. python, openssl) can hold handles for well over the ~5s the
845    // previous 5x100ms..2s policy allowed.
846    let retry_policy = ExponentialBackoff::builder()
847        .retry_bounds(Duration::from_millis(100), Duration::from_secs(5))
848        .build_with_max_retries(10);
849    retry_io_on_transient(
850        || tokio::fs::rename(from, to),
851        is_transient_rename_error,
852        retry_policy,
853    )
854    .await
855}
856
857/// Calls `op` and retries on transient I/O errors using the given retry
858/// policy. The op is invoked again each iteration to obtain a fresh future.
859/// Non-transient errors and `RetryDecision::DoNotRetry` cause the loop to
860/// return the last error.
861#[cfg(any(windows, test))]
862async fn retry_io_on_transient<F, Fut, P, R>(
863    mut op: F,
864    is_transient: P,
865    retry_policy: R,
866) -> std::io::Result<()>
867where
868    F: FnMut() -> Fut,
869    Fut: Future<Output = std::io::Result<()>>,
870    P: Fn(&std::io::Error) -> bool,
871    R: RetryPolicy,
872{
873    let request_start = SystemTime::now();
874    let mut current_try: u32 = 0;
875    loop {
876        match op().await {
877            Ok(()) => return Ok(()),
878            Err(err) => {
879                if !is_transient(&err) {
880                    return Err(err);
881                }
882                let execute_after = match retry_policy.should_retry(request_start, current_try) {
883                    RetryDecision::Retry { execute_after } => execute_after,
884                    RetryDecision::DoNotRetry => return Err(err),
885                };
886                let duration = execute_after
887                    .duration_since(SystemTime::now())
888                    .unwrap_or(Duration::ZERO);
889                tracing::warn!(
890                    "transient I/O error '{}'; retry #{} in {:?}",
891                    err,
892                    current_try + 1,
893                    duration,
894                );
895                tokio::time::sleep(duration).await;
896                current_try += 1;
897            }
898        }
899    }
900}

905    // 5 = ERROR_ACCESS_DENIED, 32 = ERROR_SHARING_VIOLATION.

909/// Shared logic for validating a package.
910async fn validate_package_common<F, Fut, E>(
911    path: PathBuf,
912    known_valid_revision: Option<u64>,
913    given_sha: Option<&Sha256Hash>,
914    fetch: Option<F>,
915    reporter: Option<Arc<dyn CacheReporter>>,
916    validation_mode: ValidationMode,
917) -> Result<CacheMetadata, PackageCacheLayerError>
918where
919    F: Fn(PathBuf) -> Fut + Send,
920    Fut: Future<Output = Result<(), E>> + 'static,
921    E: Error + Send + Sync + 'static,
922{
923    // Open the cache metadata file to read/write revision and hash information.
924    // Concurrent access is coordinated via the global cache lock.
925    let lock_file_path = cache_lock::metadata_path(&path);
926
927    // Ensure the directory containing the lock-file exists.
928    if let Some(root_dir) = lock_file_path.parent() {
929        tokio_fs::create_dir_all(root_dir)
930            .map_err(|e| {
931                PackageCacheLayerError::LockError(
932                    format!("failed to create cache directory: '{}'", root_dir.display()),
933                    e,
934                )
935            })
936            .await?;
937    }
938
939    let mut metadata = CacheMetadataFile::acquire(&lock_file_path).await?;
940    let cache_revision = metadata.read_revision()?;
941    let locked_sha256 = metadata.read_sha256()?;
942
943    let hash_mismatch = cache_lock::sha256_mismatch(given_sha, locked_sha256.as_ref());
944
945    let cache_dir_exists = path.is_dir();
946    if cache_dir_exists && !hash_mismatch {
947        let path_inner = path.clone();
948
949        let reporter = reporter.as_deref().map(|r| (r, r.on_validate_start()));
950
951        // If we know the revision is already valid we can return immediately.
952        if known_valid_revision == Some(cache_revision) {
953            if let Some((reporter, index)) = reporter {
954                reporter.on_validate_complete(index);
955            }
956            return Ok(CacheMetadata {
957                revision: cache_revision,
958                sha256: locked_sha256,
959                path: path_inner,
960                index_json: None,
961                paths_json: None,
962            });
963        }
964
965        // Validate the package directory.
966        let validation_result = tokio::task::spawn_blocking(move || {
967            validate_package_directory(&path_inner, validation_mode)
968        })
969        .await;
970
971        if let Some((reporter, index)) = reporter {
972            reporter.on_validate_complete(index);
973        }
974
975        match validation_result {
976            Ok(Ok((index_json, paths_json))) => {
977                tracing::debug!("validation succeeded");
978                return Ok(CacheMetadata {
979                    revision: cache_revision,
980                    sha256: locked_sha256,
981                    path,
982                    index_json: Some(index_json),
983                    paths_json: Some(paths_json),
984                });
985            }
986            Ok(Err(e)) => {
987                tracing::warn!("validation for {path:?} failed: {e}");
988                if let Some(cause) = e.source() {
989                    tracing::debug!(
990                        "  Caused by: {}",
991                        std::iter::successors(Some(cause), |e| (*e).source())
992                            .format("\n  Caused by: ")
993                    );
994                }
995            }
996            Err(e) => {
997                if let Ok(panic) = e.try_into_panic() {
998                    std::panic::resume_unwind(panic)
999                }
1000            }
1001        }
1002    } else if !cache_dir_exists {
1003        tracing::debug!("cache directory does not exist");
1004    } else if hash_mismatch {
1005        tracing::warn!(
1006            "hash mismatch, wanted a package at location {} with hash {} but the cached package has hash {}, fetching package",
1007            path.display(),
1008            given_sha.map_or(String::from("<unknown>"), hex::encode),
1009            locked_sha256.map_or(String::from("<unknown>"), hex::encode)
1010        );
1011    }
1012
1013    // If the cache is stale, we need to fetch the package again.
1014    // Since we hold the global cache lock, we can safely update the metadata
1015    // and fetch the package without worrying about concurrent modifications.
1016    if let Some(ref fetch_fn) = fetch {
1017        // Write the new revision
1018        let new_revision = cache_revision + 1;
1019        metadata
1020            .write_revision_and_sha(new_revision, given_sha)
1021            .await?;
1022
1023        // Create a temporary directory in the same parent folder for atomic extraction.
1024        // Using the same parent ensures the rename operation is atomic (same filesystem).
1025        let parent_dir = path.parent().ok_or_else(|| {
1026            PackageCacheLayerError::OtherError(Box::new(std::io::Error::other(format!(
1027                "cache path '{}' has no parent directory",
1028                path.display()
1029            ))))
1030        })?;
1031
1032        let prefix = path.file_name().and_then(|n| n.to_str()).unwrap_or("pkg");
1033
1034        // Use tempfile::Builder to create a temp directory with automatic cleanup on failure
1035        let temp_dir = tempfile::Builder::new()
1036            .prefix(&format!(".{prefix}"))
1037            .tempdir_in(parent_dir)
1038            .map_err(|e| {
1039                PackageCacheLayerError::OtherError(Box::new(std::io::Error::other(format!(
1040                    "failed to create temp directory in '{}': {}",
1041                    parent_dir.display(),
1042                    e
1043                ))))
1044            })?;
1045
1046        // Fetch/extract the package to the temporary directory.
1047        let fetch_result = fetch_fn(temp_dir.path().to_path_buf()).await;
1048
1049        // Handle fetch result
1050        match fetch_result {
1051            Ok(()) => {
1052                // Take ownership of the temp directory path so it won't be auto-deleted
1053                let temp_path = temp_dir.keep();
1054
1055                // Remove any existing (potentially corrupted) directory at the final path
1056                if path.is_dir() {
1057                    tokio_fs::remove_dir_all(&path).await.map_err(|e| {
1058                        PackageCacheLayerError::OtherError(Box::new(std::io::Error::other(
1059                            format!(
1060                                "failed to remove existing cache directory '{}': {}",
1061                                path.display(),
1062                                e
1063                            ),
1064                        )))
1065                    })?;
1066                }
1067
1068                // Atomically rename the temp directory into the cache;
1069                // see `rename_with_retry` for the Windows retry rationale.
1070                rename_with_retry(&temp_path, &path).await.map_err(|e| {
1071                    PackageCacheLayerError::OtherError(Box::new(std::io::Error::other(format!(
1072                        "failed to rename temp directory '{}' to '{}': {}",
1073                        temp_path.display(),
1074                        path.display(),
1075                        e
1076                    ))))
1077                })?;
1078
1079                // After fetching, return the cache metadata with the new revision.
1080                // We don't need to re-validate since we just fetched it.
1081                Ok(CacheMetadata {
1082                    revision: new_revision,
1083                    sha256: given_sha.copied(),
1084                    path,
1085                    index_json: None,
1086                    paths_json: None,
1087                })
1088            }
1089            Err(e) => {
1090                // temp_dir is dropped here, automatically cleaning up the directory
1091                Err(PackageCacheLayerError::FetchError(Arc::new(e)))
1092            }
1093        }
1094    } else {
1095        Err(PackageCacheLayerError::InvalidPackage)
1096    }
1097}

1338    /// Drives `retry_io_on_transient` against a real `tokio_fs::rename` that
1339    /// fails because of an exclusive Win32 file-share lock, then succeeds once
1340    /// the op signals a background task to drop the handle. Synchronisation is
1341    /// done with `tokio::sync::Semaphore`, so the test does not depend on any
1342    /// wall-clock timing.
1343    ///
1344    /// POSIX `rename` ignores file-sharing flags, so the body is gated to
1345    /// Windows. The function itself always compiles; on non-Windows the
1346    /// `#[ignore]` attribute makes it surface as "skipped" in test reports.
1347    #[tokio::test]
1348    #[cfg_attr(
1349        not(windows),
1350        ignore = "Windows-specific: POSIX rename ignores file-sharing flags"
1351    )]
1352    async fn test_retry_io_on_transient_recovers_from_share_lock() {
1353        #[cfg(windows)]
1354        {
1355            use std::{
1356                os::windows::fs::OpenOptionsExt,
1357                sync::{
1358                    Arc,
1359                    atomic::{AtomicBool, Ordering},
1360                },
1361                time::Duration,
1362            };
1363
1364            use tokio::sync::Semaphore;
1365
1366            let dir = tempdir().unwrap();
1367            let from = dir.path().join("src");
1368            let to = dir.path().join("dst");
1369            std::fs::create_dir(&from).unwrap();
1370            let locked_path = from.join("locked.txt");
1371            std::fs::write(&locked_path, b"data").unwrap();
1372
1373            let handle = std::fs::OpenOptions::new()
1374                .read(true)
1375                .share_mode(0) // exclusive: blocks MoveFileEx on the parent dir.
1376                .open(&locked_path)
1377                .expect("open with exclusive share mode");
1378
1379            // Two semaphores choreograph the lock release without timing:
1380            // - `release_request` is granted by the op after the first failed
1381            //   rename, asking the background task to drop the handle.
1382            // - `release_done` is granted by the task once the handle is gone,
1383            //   so the op returns only after the file is actually unlocked.
1384            let release_request = Arc::new(Semaphore::new(0));
1385            let release_done = Arc::new(Semaphore::new(0));
1386
1387            let release_request_for_task = release_request.clone();
1388            let release_done_for_task = release_done.clone();
1389            tokio::spawn(async move {
1390                let _ = release_request_for_task.acquire().await;
1391                drop(handle);
1392                release_done_for_task.add_permits(1);
1393            });
1394
1395            let signaled = Arc::new(AtomicBool::new(false));
1396            let policy = ExponentialBackoffBuilder::default()
1397                .retry_bounds(Duration::ZERO, Duration::ZERO)
1398                .build_with_max_retries(5);
1399
1400            super::retry_io_on_transient(
1401                || {
1402                    let from = &from;
1403                    let to = &to;
1404                    let signaled = signaled.clone();
1405                    let release_request = release_request.clone();
1406                    let release_done = release_done.clone();
1407                    async move {
1408                        // Use the same rename op as `rename_with_retry`; the
1409                        // fs_err wrapper would strip `raw_os_error()` and make
1410                        // the transient check below never match.
1411                        let result = tokio::fs::rename(from, to).await;
1412                        if result.is_err() && !signaled.swap(true, Ordering::SeqCst) {
1413                            release_request.add_permits(1);
1414                            let _ = release_done.acquire().await;
1415                        }
1416                        result
1417                    }
1418                },
1419                super::is_transient_rename_error,
1420                policy,
1421            )
1422            .await
1423            .expect("rename should succeed once the share lock is released");
1424
1425            assert!(!from.exists());
1426            assert!(to.is_dir());
1427            assert_eq!(std::fs::read(to.join("locked.txt")).unwrap(), b"data");
1428        }
1429    }

1541    /// A helper middleware function that simulates a broken pipe error
1542    /// by returning a stream that errors after sending partial data.
1543    #[allow(clippy::type_complexity)]
1544    async fn fail_with_broken_pipe(
1545        State((count, bytes)): State<(Arc<Mutex<i32>>, Arc<Mutex<usize>>)>,
1546        req: Request<Body>,
1547        next: Next,
1548    ) -> Result<Response, StatusCode> {
1549        let count = {
1550            let mut count = count.lock().await;
1551            *count += 1;
1552            *count
1553        };
1554
1555        println!(
1556            "Running broken pipe middleware for request #{count} for {}",
1557            req.uri()
1558        );
1559        let response = next.run(req).await;
1560
1561        if count <= 2 {
1562            // Get the full response body
1563            let body = response.into_body();
1564            let mut body = body.into_data_stream();
1565            let mut buffer = Vec::new();
1566            while let Some(Ok(chunk)) = body.next().await {
1567                buffer.extend(chunk);
1568            }
1569
1570            let byte_count = *bytes.lock().await;
1571            // Create a stream that yields partial data then fails with broken pipe
1572            let partial_data: Bytes = buffer
1573                .into_iter()
1574                .take(byte_count)
1575                .collect::<Vec<u8>>()
1576                .into();
1577
1578            let stream = stream::unfold((false, partial_data), |(has_sent, data)| async move {
1579                if has_sent {
1580                    // Return broken pipe error on second iteration
1581                    return Some((
1582                        Err(std::io::Error::new(
1583                            std::io::ErrorKind::BrokenPipe,
1584                            "stream closed because of a broken pipe",
1585                        )),
1586                        (true, data),
1587                    ));
1588                }
1589                // Return the partial data first
1590                Some((Ok::<_, std::io::Error>(data.clone()), (true, data)))
1591            });
1592
1593            let body = Body::from_stream(stream);
1594            return Ok(Response::new(body));
1595        }
1596
1597        Ok(response)
1598    }

2462    // from the record is part of the cache key, so a stale entry is
2463    // re-extracted instead of being served for every later package.
2464    async fn test_get_or_fetch_from_path_record_replaces_stale_package() {
2465        let cpython_archive =
2466            get_test_data_dir().join("clobber/clobber-python-0.1.0-cpython.conda");
2467        let pypy_archive = get_test_data_dir().join("clobber/clobber-python-0.1.0-pypy.conda");
2468
2469        // Both records claim the same name-version-build, so they map to the
2470        // same cache directory, but each carries the sha256 of its archive.
2471        let record_for = |archive: &Path| {
2472            let mut record = PackageRecord::new(
2473                PackageName::new_unchecked("clobber-python"),
2474                "0.1.0".parse::<VersionWithSource>().unwrap(),
2475                "cpython".to_string(),
2476            );
2477            record.sha256 = Some(compute_file_digest::<Sha256>(archive).unwrap());
2478            record
2479        };
2480
2481        let packages_dir = tempdir().unwrap();
2482        let cache = PackageCache::new(packages_dir.path());
2483
2484        let cache_metadata = cache
2485            .get_or_fetch_from_path(&cpython_archive, Some(&record_for(&cpython_archive)), None)
2486            .await
2487            .unwrap();
2488        assert_eq!(cache_metadata.revision(), 1);
2489        assert_eq!(
2490            std::fs::read(cache_metadata.path().join("bin/python")).unwrap(),
2491            b"cpython\n"
2492        );
2493        drop(cache_metadata);
2494
2495        // Fetching a different package with the same name-version-build must
2496        // re-extract instead of serving the stale cpython content.
2497        let pypy_record = record_for(&pypy_archive);
2498        let cache_metadata = cache
2499            .get_or_fetch_from_path(&pypy_archive, Some(&pypy_record), None)
2500            .await
2501            .unwrap();
2502        assert_eq!(
2503            cache_metadata.revision(),
2504            2,
2505            "expected the stale cache entry to be re-extracted"
2506        );
2507        assert_eq!(cache_metadata.sha256, pypy_record.sha256);
2508        assert!(
2509            !cache_metadata.path().join("bin/python").exists(),
2510            "stale content from the previously cached package was served"
2511        );
2512
2513        // The cache metadata stores the 8-byte revision plus the 32-byte
2514        // sha256; without the sha the stale-entry check can never fire.
2515        let lock_contents = std::fs::read(
2516            packages_dir
2517                .path()
2518                .join("clobber-python-0.1.0-cpython.lock"),
2519        )
2520        .unwrap();
2521        assert_eq!(lock_contents.len(), 40);
2522        assert_eq!(&lock_contents[8..], pypy_record.sha256.unwrap().as_slice());
2523    }
