12/// Separates conda archives from each version of the wheel conversion format.
13#[derive(Debug, Hash, Clone, Copy, Eq, PartialEq)]
14pub(super) enum CacheNamespace {
15    Conda,
16    Wheel(u64),
17}
18
19impl CacheNamespace {
20    pub(super) fn is_wheel_directory(name: &str) -> bool {
21        name.strip_prefix("wheels-v").is_some_and(|version| {
22            !version.is_empty() && version.bytes().all(|byte| byte.is_ascii_digit())
23        })
24    }
25
26    pub(super) fn directory(self) -> PathBuf {
27        match self {
28            Self::Conda => PathBuf::new(),
29            Self::Wheel(version) => PathBuf::from(format!("wheels-v{version}")),
30        }
31    }
32}

34/// Provides a unique identifier for packages in the cache.
35/// TODO: This could not be unique over multiple subdir. How to handle?
36#[derive(Debug, Hash, Clone, Eq, PartialEq)]
37pub struct CacheKey {
38    pub(super) namespace: CacheNamespace,
39    pub(crate) name: String,
40    pub(crate) version: String,
41    pub(crate) build_string: String,
42    pub(crate) sha256: Option<Sha256Hash>,
43    pub(crate) md5: Option<Md5Hash>,
44    pub(crate) origin_hash: Option<String>,
45}
46
47impl CacheKey {
48    /// Identifies a wheel converted with the current converter version.
49    ///
50    /// Unlike `CacheKey::from(record)`, this selects the versioned wheel namespace.
51    /// Use it for wheel presence checks with extensionless download URLs.
52    pub fn from_wheel_record(record: &PackageRecord) -> Self {
53        Self::from(record).with_wheel_namespace()
54    }
55
56    pub(super) fn with_wheel_namespace(mut self) -> Self {
57        self.namespace = CacheNamespace::Wheel(CONVERSION_VERSION);
58        self
59    }
60
61    /// Adds a sha256 hash of the archive.
62    pub fn with_sha256(mut self, sha256: Sha256Hash) -> Self {
63        self.sha256 = Some(sha256);
64        self
65    }
66
67    /// Potentially adds a sha256 hash of the archive.
68    pub fn with_opt_sha256(mut self, sha256: Option<Sha256Hash>) -> Self {
69        self.sha256 = sha256;
70        self
71    }
72
73    /// Adds a md5 hash of the archive.
74    pub fn with_md5(mut self, md5: Md5Hash) -> Self {
75        self.md5 = Some(md5);
76        self
77    }
78
79    /// Potentially adds a md5 hash of the archive.
80    pub fn with_opt_md5(mut self, md5: Option<Md5Hash>) -> Self {
81        self.md5 = md5;
82        self
83    }
84
85    /// Adds a hash of the Url to the cache key
86    pub fn with_url(mut self, url: url::Url) -> Self {
87        let url_hash = compute_url_digest::<Sha256>(url);
88        self.origin_hash = Some(hex::encode(url_hash));
89        self
90    }
91
92    /// Adds a hash of the Path to the cache key
93    pub fn with_path(mut self, path: &Path) -> Self {
94        let path_hash = compute_bytes_digest::<Sha256>(path.as_os_str().as_encoded_bytes());
95        self.origin_hash = Some(hex::encode(path_hash));
96        self
97    }
98}

100impl CacheKey {
101    /// Renders the cache key as a directory name, rejecting metadata-derived
102    /// components that could escape the cache root (GHSA-h672-p7h7-97v9).
103    ///
104    /// Both the leaf and namespaced path go through this validation; there is no
105    /// `Display`/`to_string` that could bypass the check.
106    pub(crate) fn to_path_segment(&self) -> Result<String, InvalidPathComponentError> {
107        let segment = match &self.origin_hash {
108            Some(url_hash) => format!(
109                "{}-{}-{}-{}",
110                self.name, self.version, self.build_string, url_hash
111            ),
112            None => format!("{}-{}-{}", self.name, self.version, self.build_string),
113        };
114        ensure_safe_path_component(&segment)?;
115        Ok(segment)
116    }
117
118    /// Returns the validated package path relative to a cache layer.
119    pub(super) fn relative_path(&self) -> Result<PathBuf, InvalidPathComponentError> {
120        Ok(self.namespace.directory().join(self.to_path_segment()?))
121    }
122
123    /// Return the sha256 hash of the package if it is known.
124    pub fn sha256(&self) -> Option<Sha256Hash> {
125        self.sha256
126    }
127
128    /// Return the md5 hash of the package if it is known.
129    pub fn md5(&self) -> Option<Md5Hash> {
130        self.md5
131    }
132
133    /// Matches the parts of a [`MatchSpec`] represented by a cache key.
134    ///
135    /// A cache key contains name, version, build string and optional hashes.
136    /// Specifications constraining any other package metadata cannot be
137    /// evaluated reliably and therefore do not match.
138    pub(crate) fn matches_spec(&self, spec: &MatchSpec) -> bool {
139        if spec.build_number.is_some()
140            || spec.file_name.is_some()
141            || spec.extras.is_some()
142            || spec.flags.is_some()
143            || spec.channel.is_some()
144            || spec.subdir.is_some()
145            || spec.namespace.is_some()
146            || spec.url.is_some()
147            || spec.license.is_some()
148            || spec.license_family.is_some()
149            || spec.condition.is_some()
150            || spec.track_features.is_some()
151        {
152            return false;
153        }
154
155        let Ok(name) = self.name.parse::<PackageName>() else {
156            return false;
157        };
158        let Ok(version) = self.version.parse::<VersionWithSource>() else {
159            return false;
160        };
161        let mut record = PackageRecord::new(
162            name,
163            version,
164            BuildString::new_unchecked(&self.build_string),
165        );
166        record.sha256 = self.sha256;
167        record.md5 = self.md5;
168        spec.matches(&record)
169    }
170}
