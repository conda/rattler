9/// Provides a unique identifier for packages in the cache.
10/// TODO: This could not be unique over multiple subdir. How to handle?
11#[derive(Debug, Hash, Clone, Eq, PartialEq)]
12pub struct CacheKey {
13    pub(crate) name: String,
14    pub(crate) version: String,
15    pub(crate) build_string: String,
16    pub(crate) sha256: Option<Sha256Hash>,
17    pub(crate) md5: Option<Md5Hash>,
18    pub(crate) origin_hash: Option<String>,
19}

21impl CacheKey {

26    }

34    /// Adds a md5 hash of the archive.

36        self.md5 = Some(md5);
37        self
38    }
39
40    /// Potentially adds a md5 hash of the archive.
41    pub fn with_opt_md5(mut self, md5: Option<Md5Hash>) -> Self {
42        self.md5 = md5;
43        self
44    }
45
46    /// Adds a hash of the Url to the cache key
47    pub fn with_url(mut self, url: url::Url) -> Self {
48        let url_hash = compute_url_digest::<Sha256>(url);
49        self.origin_hash = Some(hex::encode(url_hash));
50        self
51    }
52
53    /// Adds a hash of the Path to the cache key
54    pub fn with_path(mut self, path: &Path) -> Self {
55        let path_hash = compute_bytes_digest::<Sha256>(path.as_os_str().as_encoded_bytes());
56        self.origin_hash = Some(hex::encode(path_hash));
57        self
58    }
59}

61impl CacheKey {
62    /// Renders the cache key as a directory name, rejecting metadata-derived
63    /// components that could escape the cache root (GHSA-h672-p7h7-97v9).
64    ///
65    /// This is intentionally the only way to turn a key into a path: there is no
66    /// `Display`/`to_string` that could bypass the check.
67    pub(crate) fn to_path_segment(&self) -> Result<String, InvalidPathComponentError> {
68        let segment = match &self.origin_hash {
69            Some(url_hash) => format!(
70                "{}-{}-{}-{}",
71                &self.name, &self.version, &self.build_string, url_hash
72            ),
73            None => format!("{}-{}-{}", &self.name, &self.version, &self.build_string),
74        };
75        ensure_safe_path_component(&segment)?;
76        Ok(segment)
77    }
78
79    /// Return the sha256 hash of the package if it is known.
80    pub fn sha256(&self) -> Option<Sha256Hash> {
81        self.sha256
82    }
83
84    /// Return the md5 hash of the package if it is known.
85    pub fn md5(&self) -> Option<Md5Hash> {
86        self.md5
87    }

89    /// Matches the parts of a [`MatchSpec`] represented by a cache key.
90    ///
91    /// A cache key contains name, version, build string and optional hashes.
92    /// Specifications constraining any other package metadata cannot be
93    /// evaluated reliably and therefore do not match.
94    pub(crate) fn matches_spec(&self, spec: &MatchSpec) -> bool {
95        if spec.build_number.is_some()
96            || spec.file_name.is_some()
97            || spec.extras.is_some()
98            || spec.flags.is_some()
99            || spec.channel.is_some()
100            || spec.subdir.is_some()
101            || spec.namespace.is_some()
102            || spec.url.is_some()
103            || spec.license.is_some()
104            || spec.license_family.is_some()
105            || spec.condition.is_some()
106            || spec.track_features.is_some()
107        {
108            return false;
109        }
110
111        let Ok(name) = self.name.parse::<PackageName>() else {
112            return false;
113        };
114        let Ok(version) = self.version.parse::<VersionWithSource>() else {
115            return false;
116        };
117        let mut record = PackageRecord::new(name, version, self.build_string.clone());
118        record.sha256 = self.sha256;
119        record.md5 = self.md5;
120        spec.matches(&record)
121    }
122}
