1//! Which virtual package names a set of package records can reference.
2//!
3//! A detector only needs to run when a solve could ask for one of its names.
4//! Scanning `depends`, `constrains`, and every `extra_depends` group of the
5//! records taking part in the solve collects these potential references.
6
7use std::collections::BTreeSet;
8
9use crate::{MatchSpec, PackageName, PackageRecord, ParseStrictness};
10
11/// The virtual package names, starting with two underscores, that any of
12/// `records` mentions in `depends`, `constrains`, or any `extra_depends` group.
13/// All optional groups contribute potential demand, regardless of activation.
14///
15/// Ordinary specs only have their exact package name parsed; the solver validates
16/// the remaining constraints. Qualified and bracket specs use lenient `MatchSpec`
17/// parsing so qualifiers and package-name patterns cannot become exact demands.
18pub fn referenced_virtual_packages<'a>(
19    records: impl IntoIterator<Item = &'a PackageRecord>,
20) -> BTreeSet<PackageName> {
21    let mut names = BTreeSet::new();
22    for record in records {
23        for spec in record
24            .depends
25            .iter()
26            .chain(&record.constrains)
27            .chain(record.extra_depends.values().flatten())
28        {
29            let spec = spec.trim_start();
30            let name = if spec.contains(':') || spec.contains('[') {
31                MatchSpec::from_str(spec, ParseStrictness::Lenient)
32                    .ok()
33                    .and_then(|spec| spec.name.into_exact())
34            } else if spec.starts_with("__") {
35                PackageName::from_matchspec_str(spec).ok()
36            } else {
37                None
38            };
39            if let Some(name) = name
40                && name.as_normalized().starts_with("__")
41            {
42                names.insert(name);
43            }
44        }
45    }
46    names
47}

49#[cfg(test)]
50mod tests {
51    use std::str::FromStr;
52
53    use crate::Version;
54
55    use super::*;
56
57    fn record(depends: &[&str], constrains: &[&str]) -> PackageRecord {
58        let mut record = PackageRecord::new(
59            PackageName::try_from("pkg").unwrap(),
60            Version::from_str("1").unwrap(),
61            "0".to_string(),
62        );
63        record.depends = depends.iter().map(ToString::to_string).collect();
64        record.constrains = constrains.iter().map(ToString::to_string).collect();
65        record
66    }
67
68    #[test]
69    fn collects_names_from_depends_and_constrains() {
70        let records = [
71            record(
72                &["python >=3.10", "__conda_forge_openmpi >=5.0,<6.0a0"],
73                &[],
74            ),
75            record(&["__cuda >=12"], &["__glibc >=2.28", "libfoo"]),
76            record(&["conda-forge::__conda_forge_mpich 4.*"], &[]),
77        ];
78        let names: Vec<String> = referenced_virtual_packages(&records)
79            .iter()
80            .map(|name| name.as_normalized().to_string())
81            .collect();
82        assert_eq!(
83            names,
84            [
85                "__conda_forge_mpich",
86                "__conda_forge_openmpi",
87                "__cuda",
88                "__glibc"
89            ]
90        );
91    }
92
93    #[test]
94    fn preserves_exact_names_and_ignores_patterns() {
95        let records = [record(
96            &[
97                "  __CUDA >=12",
98                "__cuda<13",
99                "conda-forge::__CUDA 12.*",
100                "__glibc[version='>=2.28']",
101                "__cuda[name=__other]",
102                "*[name=__other]",
103                "__cuda[ab]",
104                "__cuda* >=12",
105                "^__cuda.*$",
106                "python >=3.10",
107            ],
108            &["__GLIBC >=2.17", "__archspec=1=x86_64"],
109        )];
110        let names: Vec<_> = referenced_virtual_packages(&records)
111            .into_iter()
112            .map(|name| name.as_normalized().to_owned())
113            .collect();
114        assert_eq!(names, ["__archspec", "__cuda", "__glibc"]);
115    }

117    #[test]
118    fn collects_potential_names_from_all_optional_groups() {
119        let mut record = record(&["__cuda >=12"], &["__glibc >=2.28"]);
120        record.extra_depends.insert(
121            "gpu".to_string(),
122            vec!["__cuda >=11".to_string(), "__rocm >=6".to_string()],
123        );
124        record.extra_depends.insert(
125            "mpi".to_string(),
126            vec![
127                "__conda_forge_openmpi >=5".to_string(),
128                "__rocm >=5".to_string(),
129                "__glibc >=2.17".to_string(),
130            ],
131        );
132        record.extra_depends.insert("empty".to_string(), Vec::new());
133
134        let names: Vec<_> = referenced_virtual_packages([&record])
135            .into_iter()
136            .map(|name| name.as_normalized().to_owned())
137            .collect();
138        assert_eq!(
139            names,
140            ["__conda_forge_openmpi", "__cuda", "__glibc", "__rocm"]
141        );
142    }
