1//! Which virtual package names a set of package records can reference.
2//!
3//! A detector only needs to run when a solve could ask for one of its names.
4//! Scanning the `depends` and `constrains` of the records that take part in
5//! the solve tells which names that is.
6
7use std::collections::BTreeSet;
8
9use crate::{MatchSpec, PackageName, PackageRecord, ParseStrictness};
10
11/// The virtual package names, starting with two underscores, that any of
12/// `records` mentions in `depends` or `constrains`.
13///
14/// Ordinary specs only have their exact package name parsed; the solver validates
15/// the remaining constraints. Qualified and bracket specs use lenient `MatchSpec`
16/// parsing so qualifiers and package-name patterns cannot become exact demands.
17pub fn referenced_virtual_packages<'a>(
18    records: impl IntoIterator<Item = &'a PackageRecord>,
19) -> BTreeSet<PackageName> {
20    let mut names = BTreeSet::new();
21    for record in records {
22        for spec in record.depends.iter().chain(&record.constrains) {
23            let spec = spec.trim_start();
24            let name = if spec.contains(':') || spec.contains('[') {
25                MatchSpec::from_str(spec, ParseStrictness::Lenient)
26                    .ok()
27                    .and_then(|spec| spec.name.into_exact())
28            } else if spec.starts_with("__") {
29                PackageName::from_matchspec_str(spec).ok()
30            } else {
31                None
32            };
33            if let Some(name) = name
34                && name.as_normalized().starts_with("__")
35            {
36                names.insert(name);
37            }
38        }
39    }
40    names
41}

43#[cfg(test)]
44mod tests {
45    use std::str::FromStr;
46
47    use crate::Version;
48
49    use super::*;
50
51    fn record(depends: &[&str], constrains: &[&str]) -> PackageRecord {
52        let mut record = PackageRecord::new(
53            PackageName::try_from("pkg").unwrap(),
54            Version::from_str("1").unwrap(),
55            "0".to_string(),
56        );
57        record.depends = depends.iter().map(ToString::to_string).collect();
58        record.constrains = constrains.iter().map(ToString::to_string).collect();
59        record
60    }
61
62    #[test]
63    fn collects_names_from_depends_and_constrains() {
64        let records = [
65            record(
66                &["python >=3.10", "__conda_forge_openmpi >=5.0,<6.0a0"],
67                &[],
68            ),
69            record(&["__cuda >=12"], &["__glibc >=2.28", "libfoo"]),
70            record(&["conda-forge::__conda_forge_mpich 4.*"], &[]),
71        ];
72        let names: Vec<String> = referenced_virtual_packages(&records)
73            .iter()
74            .map(|name| name.as_normalized().to_string())
75            .collect();
76        assert_eq!(
77            names,
78            [
79                "__conda_forge_mpich",
80                "__conda_forge_openmpi",
81                "__cuda",
82                "__glibc"
83            ]
84        );
85    }
86
87    #[test]
88    fn preserves_exact_names_and_ignores_patterns() {
89        let records = [record(
90            &[
91                "  __CUDA >=12",
92                "__cuda<13",
93                "conda-forge::__CUDA 12.*",
94                "__glibc[version='>=2.28']",
95                "__cuda[name=__other]",
96                "*[name=__other]",
97                "__cuda[ab]",
98                "__cuda* >=12",
99                "^__cuda.*$",
100                "python >=3.10",
101            ],
102            &["__GLIBC >=2.17", "__archspec=1=x86_64"],
103        )];
104        let names: Vec<_> = referenced_virtual_packages(&records)
105            .into_iter()
106            .map(|name| name.as_normalized().to_owned())
107            .collect();
108        assert_eq!(names, ["__archspec", "__cuda", "__glibc"]);
109    }
