16/// A representation of the `paths.json` file found in package archives.

30fn serialize_sorted_paths<S>(paths: &[PathsEntry], serializer: S) -> Result<S::Ok, S::Error>
31where
32    S: Serializer,
33{
34    // Sort the paths by the relative_path attribute
35    let mut sorted_paths = paths.to_vec();
36    sorted_paths.sort_by(|a, b| a.relative_path.cmp(&b.relative_path));
37    sorted_paths.serialize(serializer)
38}

69    /// Constructs a new instance by looking at older (deprecated) files from a package directory.
70    ///
71    /// In older package archives the `paths.json` file does not exist. These packages contain the
72    /// information normally present in the `paths.json` file spread over different files in the
73    /// archive.
74    ///
75    /// This method takes parsed objects as input, to read the information from an extracted package
76    /// use [`Self::from_deprecated_package_directory`].
77    ///
78    /// - The `files` file contains a list of all files included in the package.
79    /// - The `has_prefix` file contains files that contain a "prefix".
80    /// - The `no_link` file contains files that should not be linked.
81    /// - The `no_softlink` file contains files that should not be soft-linked.
82    /// - The `path_type` is a function to determine which type of file a specific path is.
83    ///   Typically you would implement this with a function to check the filesystem.
84    pub fn from_deprecated<E>(
85        files: Files,
86        has_prefix: Option<HasPrefix>,
87        no_link: Option<NoLink>,
88        no_softlink: Option<NoSoftlink>,
89        path_type: impl Fn(&Path) -> Result<PathType, E>,
90    ) -> Result<Self, E> {
91        // Construct a HashSet of all paths that should not be linked.
92        let no_link: HashSet<PathBuf> = {
93            no_link
94                .into_iter()
95                .flat_map(|no_link| no_link.files.into_iter())
96                .chain(
97                    no_softlink
98                        .into_iter()
99                        .flat_map(|no_softlink| no_softlink.files.into_iter()),
100                )
101                .collect()
102        };
103
104        // Construct a mapping from path to prefix information
105        let has_prefix: HashMap<PathBuf, HasPrefixEntry> = has_prefix
106            .into_iter()
107            .flat_map(|has_prefix| has_prefix.files.into_iter())
108            .map(|entry| (entry.relative_path.clone(), entry))
109            .collect();
110
111        // Iterate over all files and create entries
112        Ok(Self {
113            paths: files
114                .files
115                .into_iter()
116                .map(|path| {
117                    let prefix = has_prefix.get(&path);
118                    let path_type = path_type(&path);
119
120                    match path_type {
121                        Ok(path_type) => Ok(PathsEntry {
122                            path_type,
123                            prefix_placeholder: prefix.map(|entry| PrefixPlaceholder {
124                                file_mode: entry.file_mode,
125                                placeholder: (*entry.prefix).to_owned(),
126                                experimental_offsets: None,
127                            }),
128                            no_link: no_link.contains(&path),
129                            sha256: None,
130                            size_in_bytes: None,
131                            relative_path: path,
132                        }),
133                        Err(e) => Err(e),
134                    }
135                })
136                .collect::<Result<_, _>>()?,
137            paths_version: 1,
138        })
139    }

141    /// Constructs a new instance by reading older (deprecated) files from a package directory.
142    ///
143    /// In older package archives the `paths.json` file does not exist. These packages contain the
144    /// information normally present in the `paths.json` file spread over different files in the
145    /// archive.
146    ///
147    /// This function reads the different files and tries to reconstruct a `paths.json` from it.
148    pub fn from_deprecated_package_directory(path: &Path) -> Result<Self, std::io::Error> {
149        let files = Files::from_package_directory(path)?;
150
151        let has_prefix = match HasPrefix::from_package_directory(path) {
152            Ok(has_prefix) => Some(has_prefix),
153            Err(e) if e.kind() == ErrorKind::NotFound => None,
154            Err(e) => return Err(e),
155        };
156        let no_link = match NoLink::from_package_directory(path) {
157            Ok(has_prefix) => Some(has_prefix),
158            Err(e) if e.kind() == ErrorKind::NotFound => None,
159            Err(e) => return Err(e),
160        };
161        let no_softlink = match NoSoftlink::from_package_directory(path) {
162            Ok(has_prefix) => Some(has_prefix),
163            Err(e) if e.kind() == ErrorKind::NotFound => None,
164            Err(e) => return Err(e),
165        };
166
167        Self::from_deprecated(files, has_prefix, no_link, no_softlink, |p| {
168            path.join(p).symlink_metadata().map(|metadata| {
169                if metadata.is_symlink() {
170                    PathType::SoftLink
171                } else if metadata.is_dir() {
172                    PathType::Directory
173                } else {
174                    PathType::HardLink
175                }
176            })
177        })
178    }

181/// Description off a placeholder text found in a file that must be replaced when installing the
182/// file into the prefix.
183#[derive(Clone, Debug, PartialEq, Eq, Hash)]
184pub struct PrefixPlaceholder {
185    /// The type of the file, either binary or text. Depending on the type of file either text
186    /// replacement is performed or `CString` replacement.
187    pub file_mode: FileMode,
188
189    /// The placeholder prefix used in the file. This is the path of the prefix when the package
190    /// was build.
191    pub placeholder: String,
192
193    /// The placeholder's occurrences in the file, as recorded by the producer in the `offsets`
194    /// and `shebang_length` keys proposed by the [draft CEP].
195    ///
196    /// `None` when the package records no offsets, or records them in a form that does not parse
197    /// as offset groups (such as the flat lists written by earlier drafts of the field); callers
198    /// locate the occurrences by searching the file contents. `Some(Err(_))` when the recorded
199    /// offsets violate the draft CEP; callers search the file contents as well and may report
200    /// the error. Only valid offsets recorded for [`Self::file_mode`] are serialized.
201    ///
202    /// **Experimental**: the Rust field is prefixed until
203    /// [conda/ceps#179](https://github.com/conda/ceps/pull/179) is finalized.
204    ///
205    /// [draft CEP]: https://github.com/conda/ceps/pull/179
206    pub experimental_offsets: Option<Result<PrefixOffsets, InvalidOffsetsError>>,
207}
208
209impl Serialize for PrefixPlaceholder {
210    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
211    where
212        S: Serializer,
213    {
214        // Offsets recorded for another file mode no longer describe the file.
215        let offsets = match &self.experimental_offsets {
216            Some(Ok(offsets)) if offsets.file_mode() == self.file_mode => Some(offsets),
217            Some(Ok(_) | Err(_)) | None => None,
218        };
219        SerializedPrefixPlaceholder {
220            file_mode: self.file_mode,
221            placeholder: &self.placeholder,
222            offsets: offsets.map(PrefixOffsets::groups),
223            shebang_length: offsets.and_then(PrefixOffsets::shebang_length),
224        }
225        .serialize(serializer)
226    }
227}

229impl<'de> Deserialize<'de> for PrefixPlaceholder {
230    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
231    where
232        D: serde::Deserializer<'de>,
233    {
234        let raw = RawPrefixPlaceholder::deserialize(deserializer)?;
235        let experimental_offsets = raw.offsets.map(|groups| {
236            let shebang_length = raw
237                .shebang_length
238                .map(|value| {
239                    value
240                        .as_u64()
241                        .and_then(|length| usize::try_from(length).ok())
242                        .ok_or(InvalidOffsetsError::MalformedShebangLength)
243                })
244                .transpose()?;
245            let groups = groups
246                .into_iter()
247                .map(RawOffsetGroup::into_offset_group)
248                .collect::<Result<Vec<_>, _>>()?;
249            PrefixOffsets::new(raw.file_mode, groups, shebang_length)
250        });
251        Ok(PrefixPlaceholder {
252            file_mode: raw.file_mode,
253            placeholder: raw.placeholder,
254            experimental_offsets,
255        })
256    }
257}

285/// An [`OffsetGroup`] as it appears in `paths.json`, before it is validated.
286#[derive(Deserialize)]
287struct RawOffsetGroup {
288    encoding: String,
289    ranges: OffsetRanges,
290    #[serde(flatten)]
291    other_members: BTreeMap<String, serde_json::Value>,
292}
293
294impl RawOffsetGroup {
295    /// Validates the group. A member other than `encoding` and `ranges` may change the meaning
296    /// of the group in a future revision of the draft CEP, so it invalidates the group.
297    fn into_offset_group(self) -> Result<OffsetGroup, InvalidOffsetsError> {
298        let encoding = self.encoding.parse::<OffsetEncoding>()?;
299        if !self.other_members.is_empty() {
300            return Err(InvalidOffsetsError::UnrecognizedMembers {
301                encoding,
302                members: self.other_members.into_keys().collect(),
303            });
304        }
305        OffsetGroup::new(encoding, self.ranges)
306    }
307}

348/// The encoding of one [`OffsetGroup`], from the closed set the [draft CEP] defines: the
349/// encodings replaced by existing installers.
350///
351/// [draft CEP]: https://github.com/conda/ceps/pull/179
352#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash, Serialize)]
353pub enum OffsetEncoding {
354    /// UTF-8 (`utf-8`).
355    #[serde(rename = "utf-8")]
356    Utf8,
357    /// UTF-16, little endian (`utf-16-le`).
358    #[serde(rename = "utf-16-le")]
359    Utf16Le,
360    /// UTF-16, big endian (`utf-16-be`).
361    #[serde(rename = "utf-16-be")]
362    Utf16Be,
363    /// UTF-32, little endian (`utf-32-le`).
364    #[serde(rename = "utf-32-le")]
365    Utf32Le,
366    /// UTF-32, big endian (`utf-32-be`).
367    #[serde(rename = "utf-32-be")]
368    Utf32Be,
369}
370
371impl OffsetEncoding {
372    /// Every encoding the draft CEP defines.
373    pub const DEFINED: [OffsetEncoding; 5] = [
374        OffsetEncoding::Utf8,
375        OffsetEncoding::Utf16Le,
376        OffsetEncoding::Utf16Be,
377        OffsetEncoding::Utf32Le,
378        OffsetEncoding::Utf32Be,
379    ];
380
381    /// The name of this encoding in `paths.json` (e.g. `utf-8`).
382    pub fn as_str(self) -> &'static str {
383        match self {
384            OffsetEncoding::Utf8 => "utf-8",
385            OffsetEncoding::Utf16Le => "utf-16-le",
386            OffsetEncoding::Utf16Be => "utf-16-be",
387            OffsetEncoding::Utf32Le => "utf-32-le",
388            OffsetEncoding::Utf32Be => "utf-32-be",
389        }
390    }
391
392    /// The size in bytes of one code unit of this encoding, which is also the size of the NUL
393    /// terminator of a c-string stored in it.
394    pub fn code_unit_size(self) -> usize {
395        match self {
396            OffsetEncoding::Utf8 => 1,
397            OffsetEncoding::Utf16Le | OffsetEncoding::Utf16Be => 2,
398            OffsetEncoding::Utf32Le | OffsetEncoding::Utf32Be => 4,
399        }
400    }
401
402    /// Encodes `text` with this encoding, without a byte order mark.
403    pub fn encode(self, text: &str) -> Vec<u8> {
404        let mut bytes = Vec::with_capacity(text.len() * self.code_unit_size());
405        match self {
406            OffsetEncoding::Utf8 => bytes.extend_from_slice(text.as_bytes()),
407            OffsetEncoding::Utf16Le => bytes.extend(text.encode_utf16().flat_map(u16::to_le_bytes)),
408            OffsetEncoding::Utf16Be => bytes.extend(text.encode_utf16().flat_map(u16::to_be_bytes)),
409            OffsetEncoding::Utf32Le => {
410                bytes.extend(text.chars().flat_map(|c| u32::from(c).to_le_bytes()));
411            }
412            OffsetEncoding::Utf32Be => {
413                bytes.extend(text.chars().flat_map(|c| u32::from(c).to_be_bytes()));
414            }
415        }
416        bytes
417    }
418}

423    }

437/// Where the placeholder occurs in a file under one encoding, as defined by the [draft CEP].
438///
439/// A group always lists at least one occurrence, and every c-string of a binary group lists at
440/// least one occurrence followed by its terminator.
441///
442/// [draft CEP]: https://github.com/conda/ceps/pull/179
443#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize)]
444pub struct OffsetGroup {
445    encoding: OffsetEncoding,
446    ranges: OffsetRanges,
447}
448
449impl OffsetGroup {
450    /// Creates a group of the occurrences recorded under `encoding`.
451    ///
452    /// Fails when `ranges` is empty or when a c-string of binary ranges does not list both an
453    /// occurrence and its terminator.
454    pub fn new(
455        encoding: OffsetEncoding,
456        ranges: OffsetRanges,
457    ) -> Result<Self, InvalidOffsetsError> {
458        if ranges.is_empty() {
459            return Err(InvalidOffsetsError::EmptyRanges(encoding));
460        }
461        if let OffsetRanges::Binary(cstrings) = &ranges
462            && cstrings.iter().any(|cstring| cstring.len() < 2)
463        {
464            return Err(InvalidOffsetsError::ShortCStringRanges(encoding));
465        }
466        Ok(OffsetGroup { encoding, ranges })
467    }
468
469    /// The encoding under which the occurrences were recorded.
470    pub fn encoding(&self) -> OffsetEncoding {
471        self.encoding
472    }
473
474    /// The byte offsets of the occurrences, never empty.
475    pub fn ranges(&self) -> &OffsetRanges {
476        &self.ranges
477    }
478}

480/// The byte offsets recorded in one [`OffsetGroup`].
481///
482/// The shape depends on the file mode:
483/// - **Text**: a flat list of byte positions (`[10, 45, 100]`).
484/// - **Binary**: grouped by c-string. Each inner array lists the prefix
485///   offsets followed by the position of the first byte of the NUL terminator
486///   (the encoding's zero code unit), or the file size when the final
487///   c-string is unterminated at end-of-file (`[[5, 39], [22, 30, 39]]`).
488///
489/// Occurrences inside the shebang region (the first
490/// [`PrefixOffsets::shebang_length`] bytes) are excluded; the installer
491/// transforms that region under its own shebang rules.
492#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, Hash)]
493#[serde(untagged)]
494pub enum OffsetRanges {
495    /// Text-mode ranges: flat list of byte positions where the placeholder
496    /// occurs under the group's encoding.
497    Text(Vec<usize>),
498    /// Binary-mode ranges: grouped by c-string. Each inner array contains
499    /// the prefix start positions followed by the NUL terminator position.
500    Binary(Vec<Vec<usize>>),
501}
502
503impl OffsetRanges {
504    /// Whether no positions are recorded at all.
505    pub fn is_empty(&self) -> bool {
506        match self {
507            OffsetRanges::Text(offsets) => offsets.is_empty(),
508            OffsetRanges::Binary(groups) => groups.is_empty(),
509        }
510    }
511}

513/// The offsets of the placeholder in one file, recorded per encoding as defined by the
514/// [draft CEP].
515///
516/// The groups have distinct encodings and the ranges shape of [`Self::file_mode`]. A list
517/// without groups only occurs for a text file whose occurrences all lie inside its shebang
518/// region, and [`Self::shebang_length`] is only recorded for text files. Whether the offsets
519/// match the file contents (ordering, bounds, the placeholder bytes being present) can only be
520/// checked against those contents, which the prefix replacement in `rattler` does. That check
521/// only reads the bytes the offsets point at, so the offsets are trusted to list every occurrence
522/// and the first terminator after each binary occurrence: an occurrence they leave out keeps the
523/// placeholder, and a terminator recorded past the real one moves the bytes in between.
524///
525/// [draft CEP]: https://github.com/conda/ceps/pull/179
526#[derive(Clone, Debug, PartialEq, Eq, Hash)]
527pub struct PrefixOffsets {
528    file_mode: FileMode,
529    groups: Vec<OffsetGroup>,
530    shebang_length: Option<usize>,
531}
532
533impl PrefixOffsets {
534    /// Creates the offsets recorded for a file with the given `file_mode`.
535    pub fn new(
536        file_mode: FileMode,
537        groups: Vec<OffsetGroup>,
538        shebang_length: Option<usize>,
539    ) -> Result<Self, InvalidOffsetsError> {
540        let has_shebang_length = shebang_length.is_some();
541        if has_shebang_length && file_mode == FileMode::Binary {
542            return Err(InvalidOffsetsError::ShebangLengthOnBinary);
543        }
544        if groups.is_empty() && !has_shebang_length {
545            return Err(InvalidOffsetsError::EmptyList);
546        }
547
548        for (index, group) in groups.iter().enumerate() {
549            if groups[..index]
550                .iter()
551                .any(|earlier| earlier.encoding == group.encoding)
552            {
553                return Err(InvalidOffsetsError::DuplicateEncoding(group.encoding));
554            }
555            let shape_matches = match (file_mode, &group.ranges) {
556                (FileMode::Text, OffsetRanges::Text(_))
557                | (FileMode::Binary, OffsetRanges::Binary(_)) => true,
558                (FileMode::Text, OffsetRanges::Binary(_))
559                | (FileMode::Binary, OffsetRanges::Text(_)) => false,
560            };
561            if !shape_matches {
562                return Err(InvalidOffsetsError::RangesShapeMismatch(group.encoding));
563            }
564        }
565
566        Ok(PrefixOffsets {
567            file_mode,
568            groups,
569            shebang_length,
570        })
571    }
572
573    /// The file mode the offsets were recorded for.
574    pub fn file_mode(&self) -> FileMode {
575        self.file_mode
576    }
577
578    /// The recorded groups, one per encoding under which the placeholder occurs outside the
579    /// shebang region.
580    pub fn groups(&self) -> &[OffsetGroup] {
581        &self.groups
582    }
583
584    /// The length in bytes of the file's shebang region: the first line including its
585    /// terminating newline, or the whole file size when the file contains no newline.
586    ///
587    /// Recorded exactly when the file is a text file that starts with the bytes `#!`, whether
588    /// or not the first line contains the placeholder.
589    pub fn shebang_length(&self) -> Option<usize> {
590        self.shebang_length
591    }
592}

594/// Why recorded offsets violate the [draft CEP].
595///
596/// Consumers that hit this locate the occurrences by searching the file contents instead.
597///
598/// [draft CEP]: https://github.com/conda/ceps/pull/179
599#[derive(Debug, Clone, PartialEq, Eq, Hash, thiserror::Error)]
600pub enum InvalidOffsetsError {
601    /// The offsets list is empty, which is only valid for a text file whose
602    /// occurrences all lie inside the shebang region.
603    #[error("the offsets list is empty, which is only valid for a text file with a shebang_length")]
604    EmptyList,
605
606    /// A group's encoding is not in the closed set defined by the draft CEP.
607    #[error("unrecognized encoding '{0}'")]
608    UnrecognizedEncoding(String),
609
610    /// A group has members beyond `encoding` and `ranges`.
611    #[error("the '{encoding}' group has unrecognized members: {}", members.join(", "))]
612    UnrecognizedMembers {
613        /// The group's encoding.
614        encoding: OffsetEncoding,
615        /// The names of the unrecognized members.
616        members: Vec<String>,
617    },
618
619    /// Two groups share the same encoding.
620    #[error("duplicate '{0}' groups")]
621    DuplicateEncoding(OffsetEncoding),
622
623    /// A group's ranges are empty.
624    #[error("the '{0}' group's ranges are empty")]
625    EmptyRanges(OffsetEncoding),
626
627    /// A group's ranges have the text shape for a binary file or the binary
628    /// shape for a text file.
629    #[error("the shape of the '{0}' group's ranges does not match the file mode")]
630    RangesShapeMismatch(OffsetEncoding),
631
632    /// A binary c-string lists fewer than two values, so it has no occurrence
633    /// or no terminator.
634    #[error("a c-string of the '{0}' group lists fewer than two values")]
635    ShortCStringRanges(OffsetEncoding),
636
637    /// `shebang_length` is recorded for an entry that is not a text file.
638    #[error("shebang_length is only valid for a text file")]
639    ShebangLengthOnBinary,
640
641    /// `shebang_length` is recorded but is not a byte length.
642    #[error("shebang_length is not a byte length")]
643    MalformedShebangLength,
644}

646/// Deserializes `offsets` leniently: a value that does not parse as a list of
647/// offset groups (for example the flat `[10, 45]` / `[[64, 96]]` lists
648/// written by earlier drafts of this field) yields `None` instead of failing
649/// the whole `paths.json`. The field is advisory; the search-based path
650/// handles the file correctly without it.
651fn deserialize_offset_groups<'de, D>(
652    deserializer: D,
653) -> Result<Option<Vec<RawOffsetGroup>>, D::Error>
654where
655    D: serde::Deserializer<'de>,
656{
657    let value = Option::<serde_json::Value>::deserialize(deserializer)?;
658    Ok(value.and_then(|value| serde_json::from_value(value).ok()))
659}
