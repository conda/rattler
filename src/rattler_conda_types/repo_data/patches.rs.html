14/// Represents a Conda repodata patch.
15///
16/// This struct contains information about a patch to a Conda repodata file,
17/// which is used to update the metadata for Conda packages. The patch contains
18/// information about changes to the repodata, such as removed (yanked) packages
19/// and updated package metadata.
20#[derive(Debug, Default, Clone)]
21pub struct RepoDataPatch {
22    /// The patches to apply for each subdir
23    pub subdirs: ahash::HashMap<String, PatchInstructions>,
24}
25
26impl RepoDataPatch {
27    /// Load repodata patches from an extracted repodata patches package
28    /// archive.
29    pub fn from_package(package: impl AsRef<Path>) -> io::Result<Self> {
30        let mut subdirs = ahash::HashMap::default();
31
32        // Iterate over all directories in the package
33        for entry in std::fs::read_dir(package)? {
34            let entry = entry?;
35            let file_name = entry.file_name();
36            let name = file_name.to_string_lossy();
37            let path = entry.path();
38            let patch_instructions_path = path.join("patch_instructions.json");
39            if patch_instructions_path.is_file() {
40                let contents = std::fs::read_to_string(&patch_instructions_path)?;
41                let instructions = serde_json::from_str(&contents)?;
42                subdirs.insert(name.to_string(), instructions);
43            }
44        }
45
46        Ok(Self { subdirs })
47    }
48}

50/// Contains items that overwrite metadata values stored for a single
51/// [`super::PackageRecord`].
52///
53/// Not all entries of a [`super::PackageRecord`] can be overwritten because
54/// that would cause difficult to solve inconsistencies. For instance, changing
55/// the sha256 hash of a file could break local caches because usually the data
56/// is cached by filename. With this struct we explicitly define which fields of
57/// a [`super::PackageRecord`] can be modified through repodata patches.
58#[serde_as]
59#[skip_serializing_none]
60#[derive(Debug, Deserialize, Serialize, Eq, PartialEq, Clone)]
61pub struct PackageRecordPatch {
62    /// Specification of packages this package depends on
63    pub depends: Option<Vec<String>>,
64
65    /// Additional constraints on packages. `constrains` are different from
66    /// `depends` in that packages specified in `depends` must be installed
67    /// next to this package, whereas packages specified in `constrains` are
68    /// not required to be installed, but if they are installed they must follow
69    /// these constraints.
70    pub constrains: Option<Vec<String>>,
71
72    /// Track features are nowadays only used to downweight packages (ie. give
73    /// them less priority). To that effect, the number of track features is
74    /// counted (number of commas) and the package is downweighted
75    /// by the number of `track_features`.
76    ///
77    /// This field is double wrapped to be able to express the value `null`
78    /// separate from it being missing from the JSON.
79    #[serde(
80        default,
81        skip_serializing_if = "Option::is_none",
82        with = "self::tracked_features"
83    )]
84    pub track_features: Option<Option<Vec<String>>>,
85
86    /// Features are a deprecated way to specify different feature sets for the
87    /// conda solver. This is not supported anymore and should not be used.
88    /// Instead, `mutex` packages should be used to specify
89    /// mutually exclusive features.
90    #[serde(
91        default,
92        skip_serializing_if = "Option::is_none",
93        with = "::serde_with::rust::double_option"
94    )]
95    pub features: Option<Option<String>>,
96
97    /// The specific license of the package
98    #[serde(
99        default,
100        skip_serializing_if = "Option::is_none",
101        with = "::serde_with::rust::double_option"
102    )]
103    pub license: Option<Option<String>>,
104
105    /// The license family
106    #[serde(
107        default,
108        skip_serializing_if = "Option::is_none",
109        with = "::serde_with::rust::double_option"
110    )]
111    pub license_family: Option<Option<String>>,
112
113    /// Package identifiers of packages that are equivalent to this package but
114    /// from other ecosystems.
115    /// See this CEP: <https://github.com/conda/ceps/pull/63>
116    ///
117    /// This field is double wrapped to be able to express the value `null`
118    /// separate from it being missing from the JSON.
119    #[serde(default, with = "::serde_with::rust::double_option")]
120    pub purls: Option<Option<BTreeSet<PackageUrl>>>,
121}

123mod tracked_features {
124    //! Helper module to serialize and deserialize the `track_features` field
125    //! and also allow it to be set to `null`.
126
127    use serde::{Deserializer, Serializer};
128    use serde_with::{DeserializeAs, SerializeAs};

130    use crate::utils::serde::Features;
131
132    /// Deserialize potentially non-existing optional value
133    pub fn deserialize<'de, D>(deserializer: D) -> Result<Option<Option<Vec<String>>>, D::Error>
134    where
135        D: Deserializer<'de>,
136    {
137        Option::<Features>::deserialize_as(deserializer).map(Some)
138    }
139
140    /// Serialize optional value
141    pub fn serialize<S>(
142        value: &Option<Option<Vec<String>>>,
143        serializer: S,
144    ) -> Result<S::Ok, S::Error>
145    where
146        S: Serializer,
147    {
148        match value {
149            None => serializer.serialize_unit(),
150            Some(v) => Option::<Features>::serialize_as(v, serializer),
151        }
152    }

218impl PackageRecord {
219    /// Apply a patch to a single package record
220    pub fn apply_patch(&mut self, patch: &PackageRecordPatch) {
221        if let Some(depends) = &patch.depends {
222            self.depends = depends.clone();
223        }
224        if let Some(constrains) = &patch.constrains {
225            self.constrains = constrains.clone();
226        }
227        if let Some(track_features) = &patch.track_features {
228            self.track_features = track_features.clone().unwrap_or_default();
229        }
230        if let Some(features) = &patch.features {
231            self.features = features.clone();
232        }
233        if let Some(license) = &patch.license {
234            self.license = license.clone();
235        }
236        if let Some(license_family) = &patch.license_family {
237            self.license_family = license_family.clone();
238        }
239        if let Some(package_urls) = &patch.purls {
240            self.purls = package_urls.clone();
241        }
242    }
243}

245/// Apply a patch to a repodata file
246/// Note that we currently do not handle `revoked` instructions
247pub fn apply_patches_impl(
248    packages: &mut IndexMap<DistArchiveIdentifier, PackageRecord, ahash::RandomState>,
249    conda_packages: &mut IndexMap<DistArchiveIdentifier, PackageRecord, ahash::RandomState>,
250    v3: &mut super::V3Packages,
251    removed: &mut ahash::HashSet<DistArchiveIdentifier>,
252    instructions: &PatchInstructions,
253) {
254    for (identifier, patch) in instructions.packages.iter() {
255        if let Some(record) = packages.get_mut(identifier) {
256            record.apply_patch(patch);
257        }
258
259        let conda_identifier = identifier.with_archive_type(CondaArchiveType::Conda.into());
260        if let Some(record) = conda_packages.get_mut(&conda_identifier) {
261            record.apply_patch(patch);
262        }
263    }
264
265    for (identifier, patch) in instructions.conda_packages.iter() {
266        if let Some(record) = conda_packages.get_mut(identifier) {
267            record.apply_patch(patch);
268        }
269    }
270
271    // Apply patches to v3 packages
272    for (identifier, patch) in instructions.v3.tar_bz2.iter() {
273        if let Some(record) = v3.tar_bz2.get_mut(identifier) {
274            record.apply_patch(patch);
275        }
276    }
277    for (identifier, patch) in instructions.v3.conda.iter() {
278        if let Some(record) = v3.conda.get_mut(identifier) {
279            record.apply_patch(patch);
280        }
281    }
282    for (identifier, patch) in instructions.v3.whl.iter() {
283        if let Some(record) = v3.whl.get_mut(identifier) {
284            record.package_record.apply_patch(patch);
285        }
286    }
287    for (extension, patch) in instructions.v3.extensions.iter() {
288        if patch.is_null() {
289            v3.extensions.remove(extension);
290        } else if let Some(bucket) = v3.extensions.get_mut(extension) {
291            apply_extension_patch(bucket, patch);
292        } else {
293            // `V3Extensions` rejects reserved names on construction, so an
294            // extension deserialized from patch instructions is safe to add.
295            let mut bucket = serde_json::Value::Null;
296            apply_extension_patch(&mut bucket, patch);
297            v3.extensions
298                .insert(extension.clone(), bucket)
299                .expect("v3 extension patch used a reserved bucket name");
300        }
301    }
302
303    // Mark packages as removed. Note: we only add them to the `removed` set
304    // and do NOT remove them from the package maps. The `removed` key signals
305    // that a package has been yanked, but its metadata should remain available.
306    for identifier in instructions.remove.iter() {
307        match identifier.archive_type {
308            DistArchiveType::Conda(CondaArchiveType::TarBz2) => {
309                removed.insert(identifier.clone());
310
311                // also mark equivalent .conda package as removed if it exists
312                let conda_identifier = identifier.with_archive_type(CondaArchiveType::Conda.into());
313                if conda_packages.contains_key(&conda_identifier) {
314                    removed.insert(conda_identifier);
315                }
316            }
317            DistArchiveType::Conda(CondaArchiveType::Conda) => {
318                removed.insert(identifier.clone());
319            }
320            DistArchiveType::Wheel(_) => {
321                // Wheel packages are stored in v3.whl; removal by
322                // DistArchiveIdentifier is not currently supported for v3.
323            }
324        }
325    }
326}

328/// Applies an unknown extension patch using JSON Merge Patch semantics.
329///
330/// Extension schemas are intentionally opaque, so recursively merging JSON
331/// objects preserves untouched data while letting patch instructions update
332/// individual fields. As specified by RFC 7396, an object patch treats a
333/// non-object or absent target as an empty object. Non-object values are
334/// replaced wholesale.
335fn apply_extension_patch(target: &mut serde_json::Value, patch: &serde_json::Value) {
336    if let Some(patch) = patch.as_object() {
337        if !target.is_object() {
338            *target = serde_json::Value::Object(serde_json::Map::new());
339        }
340
341        let target = target
342            .as_object_mut()
343            .expect("target was made into an object");
344        for (key, patch_value) in patch {
345            if patch_value.is_null() {
346                target.remove(key);
347            } else {
348                let target_value = target.entry(key.clone()).or_insert(serde_json::Value::Null);
349                apply_extension_patch(target_value, patch_value);
350            }
351        }
352    } else {
353        *target = patch.clone();
354    }
355}

417        let raw = serde_json::json!({
418            "packages": {},

420            "v3": {
421                "conda": {
422                    "demo-1.0-0": {
