14/// The sharded repodata holds a hashmap of package name -> shard (hash).
15/// This index file is stored under
16/// `<channel>/<subdir>/repodata_shards.msgpack.zst`
17#[serde_with::serde_as]
18#[derive(Debug, Clone, Serialize, Deserialize)]
19pub struct ShardedRepodata {
20    /// Additional information about the sharded subdirectory such as the base
21    /// url.
22    pub info: ShardedSubdirInfo,
23    /// The individual shards indexed by package name.
24    #[serde_as(as = "ahash::HashMap<_, SerializableHash<Sha256>>")]
25    pub shards: ahash::HashMap<String, Sha256Hash>,
26}

28/// Information about a sharded subdirectory that is stored inside the index
29/// file.
30#[serde_as]
31#[derive(Debug, Clone, Serialize, Deserialize)]
32pub struct ShardedSubdirInfo {
33    /// The name of the subdirectory
34    pub subdir: String,
35
36    /// The base url of the subdirectory. This is the location where the actual
37    /// packages are stored.
38    ///
39    /// This is used to construct the full url of the packages.
40    pub base_url: String,
41
42    /// The base url of the individual shards. This is the location where the
43    /// actual packages are stored.
44    ///
45    /// This is used to construct the full url of the shard.
46    pub shards_base_url: String,
47
48    /// The date at which this entry was created.
49    #[serde(skip_serializing_if = "Option::is_none")]
50    pub created_at: Option<Timestamp>,
51
52    /// Repodata revisions available through this sharded index.
53    ///
54    /// Serialized as a `vN`-keyed dictionary per the CEP draft
55    /// <https://github.com/conda/ceps/pull/146>.
56    #[serde_as(as = "IndexMap<DisplayFromStr, _>")]
57    #[serde(default, skip_serializing_if = "IndexMap::is_empty")]
58    pub repodata_revisions: RepodataRevisions,
59
60    /// Optional relationships to other channels as defined in
61    /// [CEP-42](https://github.com/conda/ceps/blob/main/cep-0042.md).
62    #[serde(default, skip_serializing_if = "ChannelRelations::is_none_or_empty")]
63    pub channel_relations: Option<ChannelRelations>,
64
65    /// The virtual package detectors the channel registers for this subdir,
66    /// with the same schema and semantics as the field in
67    /// [`ChannelInfo`](crate::ChannelInfo).
68    #[serde(
69        default,
70        skip_serializing_if = "Option::is_none",
71        deserialize_with = "crate::virtual_package_detector::deserialize_present"
72    )]
73    pub virtual_package_detectors: Option<DetectorRegistrationMetadata>,
74}

91    /// Shards are content-addressed (stored under the hash of their bytes), so
92    /// serialization must not depend on the insertion order of the underlying
93    /// maps and sets — otherwise every producer run writes a fresh shard file
94    /// and orphans the previous one.
95    #[test]
96    fn test_shard_serialization_is_insertion_order_independent() {
97        let entry = |n: u64| {
98            let key =
99                DistArchiveIdentifier::try_from_filename(&format!("multi-1.0.0-h_{n}.tar.bz2"))
100                    .unwrap();
101            let record = PackageRecord::new(
102                PackageName::new_unchecked("multi"),
103                Version::major(1),
104                format!("h_{n}"),
105            );
106            (key, record)
107        };
108
109        let shard_with_order = |order: &mut dyn Iterator<Item = u64>| {
110            let mut shard = Shard::default();
111            for n in order {
112                let (key, record) = entry(n);
113                shard.conda_packages.insert(key.clone(), record.clone());
114                shard.packages.insert(key.clone(), record);
115                shard.removed.insert(key);
116            }
117            rmp_serde::to_vec_named(&shard).unwrap()
118        };
119
120        let ascending = shard_with_order(&mut (0..10));
121        let descending = shard_with_order(&mut (0..10).rev());
122        assert_eq!(
123            ascending, descending,
124            "shard bytes must be independent of insertion order"
125        );
126    }

151    // See https://github.com/conda/ceps/blob/main/cep-0042.md
152    #[test]
153    fn test_sharded_subdir_info_channel_relations() {
154        // Deserialize a sharded index with channel_relations.
155        let raw = r#"{
156            "subdir": "linux-64",
157            "base_url": "./",
158            "shards_base_url": "./shards/",
159            "channel_relations": {
160                "base": "../conda-forge"
161            }
162        }"#;
163        let info: ShardedSubdirInfo = serde_json::from_str(raw).unwrap();
164        let relations = info.channel_relations.as_ref().unwrap();
165        assert_eq!(relations.base.as_deref(), Some("../conda-forge"));
166        assert_eq!(relations.overrides, None);
167
168        // `channel_relations` must be omitted when it is `None` and when all
169        // of its fields are unset.
170        for channel_relations in [None, Some(ChannelRelations::default())] {
171            let info = ShardedSubdirInfo {
172                subdir: "linux-64".to_string(),
173                base_url: "./".to_string(),
174                shards_base_url: "./shards/".to_string(),
175                created_at: None,
176                repodata_revisions: IndexMap::default(),
177                channel_relations,
178                virtual_package_detectors: None,
179            };
180            let json = serde_json::to_string(&info).unwrap();
181            assert!(!json.contains("channel_relations"));
182        }
183    }

216    #[test]
217    fn detector_metadata_preserves_raw_names_through_msgpack() {
218        let raw = serde_json::json!({
219            "info": {
220                "subdir": "linux-64",
221                "base_url": "./",
222                "shards_base_url": "./shards/",
223                "virtual_package_detectors": {"mpi-detect": ["__cuda", "CUDA"]},
224            },
225            "shards": {},
226        });
227        let encoded = rmp_serde::to_vec_named(&raw).unwrap();
228        let decoded: ShardedRepodata = rmp_serde::from_slice(&encoded).unwrap();
229        let parsed =
230            SubdirDetectorRegistrations::parse(decoded.info.virtual_package_detectors.as_ref())
231                .unwrap();
232        let combined = ChannelDetectorRegistrations::combine([&parsed]).unwrap();
233        assert_eq!(
234            combined.registrations()[0]
235                .virtual_packages
236                .iter()
237                .map(VirtualPackageName::as_normalized)
238                .collect::<Vec<_>>(),
239            ["__cuda"]
240        );
241        assert_eq!(combined.dropped_names()[0].name, "CUDA");
242    }

245/// An individual shard that contains repodata for a single package name.
246///
247/// Shards are content-addressed by the hash of their serialized bytes, so all
248/// maps and sets are sorted during serialization to keep the output
249/// deterministic regardless of insertion order.
250#[derive(Debug, Clone, Serialize, Deserialize, Default)]
251pub struct Shard {
252    /// The records for all `.tar.bz2` packages
253    #[serde(serialize_with = "sort_index_map_alphabetically")]
254    pub packages: IndexMap<DistArchiveIdentifier, PackageRecord, ahash::RandomState>,
255
256    /// The records for all `.conda` packages
257    #[serde(
258        rename = "packages.conda",
259        default,
260        serialize_with = "sort_index_map_alphabetically"
261    )]
262    pub conda_packages: IndexMap<DistArchiveIdentifier, PackageRecord, ahash::RandomState>,
263
264    /// Packages stored under the `v3` top-level key.
265    #[serde(default, skip_serializing_if = "V3Packages::is_empty")]
266    pub v3: V3Packages,
267
268    /// The file names of all removed for this shard
269    #[serde(default, serialize_with = "sort_set_alphabetically")]
270    pub removed: ahash::HashSet<DistArchiveIdentifier>,
271}
