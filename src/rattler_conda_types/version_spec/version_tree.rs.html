12/// A parser that parses version operators.
13fn parse_operator<'a, E: ParseError<&'a str>>(
14    input: &'a str,
15) -> Result<(&'a str, VersionOperators), nom::Err<E>> {
16    alt((
17        value(VersionOperators::Exact(EqualityOperator::Equals), tag("==")),
18        value(
19            VersionOperators::Exact(EqualityOperator::NotEquals),
20            tag("!="),
21        ),
22        value(
23            VersionOperators::StrictRange(StrictRangeOperator::StartsWith),
24            tag("="),
25        ),
26        value(
27            VersionOperators::Range(RangeOperator::GreaterEquals),
28            tag(">="),
29        ),
30        value(VersionOperators::Range(RangeOperator::Greater), tag(">")),
31        value(
32            VersionOperators::Range(RangeOperator::LessEquals),
33            tag("<="),
34        ),
35        value(VersionOperators::Range(RangeOperator::Less), tag("<")),
36        value(
37            VersionOperators::StrictRange(StrictRangeOperator::Compatible),
38            tag("~="),
39        ),
40    ))
41    .parse(input)
42}

44/// Recognizes the version epoch
45fn parse_version_epoch<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
46    input: &'a str,
47) -> Result<(&'a str, u32), nom::Err<E>> {
48    terminated(u32, tag("!")).parse(input)
49}

51/// A parser that recognizes a version
52pub(crate) fn recognize_version<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
53    allow_glob: bool,
54) -> impl FnMut(&'a str) -> IResult<&'a str, &'a str, E> {
55    /// Recognizes a single version component (`1`, `a`, `alpha`, `grub`)
56    fn recognize_version_component<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
57        allow_glob: bool,
58    ) -> impl FnMut(&'a str) -> IResult<&'a str, &'a str, E> {
59        move |input: &'a str| {
60            let ident = alpha1;
61            let num = digit1;
62            let glob = tag("*");
63            if allow_glob {
64                alt((ident, num, glob)).parse(input)
65            } else {
66                alt((ident, num)).parse(input)
67            }
68        }
69    }
70
71    /// Recognize one or more version components (`1.2.3`)
72    fn recognize_version_components<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
73        allow_glob: bool,
74    ) -> impl FnMut(&'a str) -> IResult<&'a str, &'a str, E> {
75        move |input: &'a str| {
76            // Parse the first component (required)
77            let (mut rest, _) = recognize_version_component::<E>(allow_glob)(input)?;
78            // Parse subsequent separator + component pairs without allocating
79            loop {
80                let after_sep = match opt(take_while::<_, _, E>(|c: char| {
81                    c == '.' || c == '-' || c == '_'
82                }))
83                .parse(rest)
84                {
85                    Ok((r, _)) => r,
86                    Err(_) => break,
87                };
88                match recognize_version_component::<E>(allow_glob)(after_sep) {
89                    Ok((r, _)) => rest = r,
90                    Err(_) => break,
91                }
92            }
93            // Versions can end in one or more underscores or dashes (e.g.
94            // `3.7b_`), those are part of the version. A trailing `.` stays
95            // unconsumed so globs like `3.*` keep working.
96            if let Ok((r, _)) = take_while1::<_, _, E>(|c: char| c == '_' || c == '-').parse(rest) {
97                rest = r;
98            }
99            let matched = &input[..input.len() - rest.len()];
100            Ok((rest, matched))
101        }
102    }
103
104    move |input: &'a str| {
105        recognize((
106            // Optional version epoch
107            opt(context("epoch", parse_version_epoch)),
108            // Version components
109            context("components", recognize_version_components(allow_glob)),
110            // Local version
111            opt(preceded(
112                tag("+"),
113                cut(context("local", recognize_version_components(allow_glob))),
114            )),
115        ))
116        .parse(input)
117    }
118}

120/// Recognize a version followed by a .* or *, or just a *
121pub(crate) fn recognize_version_with_star<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
122    input: &'a str,
123) -> Result<(&'a str, &'a str), nom::Err<E>> {
124    alt((
125        // A version with an optional * or .*.
126        terminated(
127            recognize_version(true),
128            take_while(|c: char| c == '.' || c == '*'),
129        ),
130        // Just a *
131        tag("*"),
132    ))
133    .parse(input)
134}

136/// A parser that recognized a constraint but does not actually parse it.
137pub(crate) fn recognize_constraint<'a, E: ParseError<&'a str> + ContextError<&'a str>>(
138    input: &'a str,
139) -> Result<(&'a str, &'a str), nom::Err<E>> {
140    alt((
141        // Any (* or *.*)
142        terminated(tag("*"), cut(opt(tag(".*")))),
143        // Regex
144        recognize(delimited(opt(tag("^")), not(tag("$")), tag("$"))),
145        // Version with optional operator followed by optional glob.
146        recognize(preceded(
147            opt(delimited(
148                opt(multispace0),
149                parse_operator,
150                opt(multispace0),
151            )),
152            cut(context("version", recognize_version_with_star)),
153        )),
154    ))
155    .parse(input)
156}

166    #[test]
167    fn test_parse_operator() {
168        type Err<'a> = nom::error::Error<&'a str>;
169
170        assert_eq!(
171            parse_operator::<Err<'_>>("=="),
172            Ok(("", VersionOperators::Exact(EqualityOperator::Equals)))
173        );
174        assert_eq!(
175            parse_operator::<Err<'_>>("!="),
176            Ok(("", VersionOperators::Exact(EqualityOperator::NotEquals)))
177        );
178        assert_eq!(
179            parse_operator::<Err<'_>>(">"),
180            Ok(("", VersionOperators::Range(RangeOperator::Greater)))
181        );
182        assert_eq!(
183            parse_operator::<Err<'_>>(">="),
184            Ok(("", VersionOperators::Range(RangeOperator::GreaterEquals)))
185        );
186        assert_eq!(
187            parse_operator::<Err<'_>>("<"),
188            Ok(("", VersionOperators::Range(RangeOperator::Less)))
189        );
190        assert_eq!(
191            parse_operator::<Err<'_>>("<="),
192            Ok(("", VersionOperators::Range(RangeOperator::LessEquals)))
193        );
194        assert_eq!(
195            parse_operator::<Err<'_>>("="),
196            Ok((
197                "",
198                VersionOperators::StrictRange(StrictRangeOperator::StartsWith)
199            ))
200        );
201        assert_eq!(
202            parse_operator::<Err<'_>>("~="),
203            Ok((
204                "",
205                VersionOperators::StrictRange(StrictRangeOperator::Compatible)
206            ))
207        );
208
209        // Anything else is an error
210        assert!(parse_operator::<Err<'_>>("").is_err());
211        assert!(parse_operator::<Err<'_>>("  >=").is_err());
212
213        // Only the operator is parsed
214        assert_eq!(
215            parse_operator::<Err<'_>>(">=3.8"),
216            Ok(("3.8", VersionOperators::Range(RangeOperator::GreaterEquals)))
217        );
218    }
