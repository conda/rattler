8/// An error occurred when accessing the authentication storage
9#[derive(thiserror::Error, Debug)]
10pub enum AuthenticationStorageError {
11    /// An error occurred when accessing the file storage
12    #[error(transparent)]
13    FileStorageError(#[from] crate::authentication_storage::backends::file::FileStorageError),
14    /// An error occurred when accessing the keyring storage
15    #[cfg(feature = "keyring")]
16    #[error(transparent)]
17    KeyringStorageError(
18        #[from] crate::authentication_storage::backends::keyring::KeyringAuthenticationStorageError,
19    ),
20    /// An error occurred when accessing the netrc storage
21    #[cfg(feature = "netrc-rs")]
22    #[error(transparent)]
23    NetRcStorageError(#[from] crate::authentication_storage::backends::netrc::NetRcStorageError),
24    /// An error occurred when accessing the memory storage
25    #[error(transparent)]
26    MemoryStorageError(#[from] crate::authentication_storage::backends::memory::MemoryStorageError),
27
28    /// Every configured backend refused to store the credentials. The
29    /// individual failures are logged as they happen; this is the summary the
30    /// caller sees.
31    #[error("none of the configured backends could store the credentials for '{host}': {backends}")]
32    StoreFailed {
33        /// The host the credentials were to be stored under.
34        host: String,
35        /// The names of the backends that were tried, in the order they were
36        /// consulted.
37        backends: String,
38    },
39
40    /// Every configured backend that holds the host refused to delete it.
41    #[error("none of the configured backends could delete the credentials for '{host}'")]
42    DeleteFailed {
43        /// The host whose credentials were to be deleted.
44        host: String,
45    },
46
47    /// A `(host, source)` pair was addressed, but no configured backend goes by
48    /// that name.
49    #[error("no configured backend is named '{backend}'")]
50    UnknownBackend {
51        /// The name that was looked for, as returned by
52        /// [`StorageBackend::name`].
53        backend: String,
54    },
55}

57/// A trait that defines the interface for authentication storage backends
58pub trait StorageBackend: std::fmt::Debug {
59    /// A short human-readable description identifying this backend (and any
60    /// relevant location, like a file path). Surfaced to users by the
61    /// `auth status` CLI so they can tell where each credential lives.
62    fn name(&self) -> String;
63
64    /// Store the given authentication information for the given host
65    fn store(
66        &self,
67        host: &str,
68        authentication: &Authentication,
69    ) -> Result<(), AuthenticationStorageError>;
70
71    /// Retrieve the authentication information for the given host
72    fn get(&self, host: &str) -> Result<Option<Authentication>, AuthenticationStorageError>;
73
74    /// List all authentication entries known to this backend.
75    ///
76    /// Some backends, such as platform keyrings, cannot enumerate arbitrary
77    /// legacy entries. They may return only entries that were stored with an
78    /// index maintained by this crate.
79    fn list(&self) -> Result<Vec<(String, Authentication)>, AuthenticationStorageError> {
80        Ok(Vec::new())
81    }
82
83    /// List the host keys this backend holds, *without* decrypting or fetching
84    /// stored secrets.
85    ///
86    /// The default implementation falls back to [`list`](Self::list), which is
87    /// expensive on backends like the macOS keychain where reading each
88    /// credential triggers a per-item ACL prompt. Backends that can enumerate
89    /// metadata cheaply should override this — that lets callers (e.g. an
90    /// interactive picker) show entries to the user without prompting for
91    /// every stored password.
92    fn list_keys(&self) -> Result<Vec<String>, AuthenticationStorageError> {
93        Ok(self.list()?.into_iter().map(|(host, _)| host).collect())
94    }
95
96    /// Delete the authentication information for the given host
97    fn delete(&self, host: &str) -> Result<(), AuthenticationStorageError>;
98}
