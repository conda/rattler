1//! Authentication methods for the conda ecosystem
2use std::str::FromStr;
3
4use serde::{Deserialize, Serialize};
5
6/// The different Authentication methods that are supported in the conda
7/// ecosystem
8#[derive(Clone, PartialEq, Eq, Serialize, Deserialize, Debug)]
9pub enum Authentication {
10    /// A bearer token is sent as a header of the form
11    /// `Authorization: Bearer {TOKEN}`
12    BearerToken(String),
13    /// A basic authentication token is sent as HTTP basic auth
14    BasicHTTP {
15        /// The username to use for basic auth
16        username: String,
17        /// The password to use for basic auth
18        password: String,
19    },
20    /// A conda token is sent in the URL as `/t/{TOKEN}/...`
21    CondaToken(String),
22    /// S3 credentials
23    S3Credentials {
24        /// The access key ID to use for S3 authentication
25        access_key_id: String,
26        /// The secret access key to use for S3 authentication
27        secret_access_key: String,
28        /// The session token to use for S3 authentication
29        session_token: Option<String>,
30    },
31    /// OAuth/OIDC credentials with automatic token refresh support
32    OAuth {
33        /// Optional audience requested at login, retained for refresh.
34        /// Absent in existing channel credentials.
35        #[serde(default, skip_serializing_if = "Option::is_none")]
36        audience: Option<String>,
37        /// The OAuth access token
38        access_token: String,
39        /// The OAuth refresh token (if available)
40        refresh_token: Option<String>,
41        /// Seconds since UNIX epoch when `access_token` expires
42        expires_at: Option<i64>,
43        /// Token endpoint URL (cached from OIDC discovery for refresh without
44        /// openidconnect)
45        token_endpoint: String,
46        /// Revocation endpoint URL (RFC 7009, for logout)
47        revocation_endpoint: Option<String>,
48        /// OAuth client ID
49        client_id: String,
50    },
51}
52
53/// An error that can occur when parsing an authentication string
54#[derive(Debug)]
55pub enum AuthenticationParseError {
56    /// The scheme is not valid
57    InvalidScheme,
58    /// The token could not be parsed
59    InvalidToken,
60}
61
62impl FromStr for Authentication {
63    type Err = AuthenticationParseError;
64
65    /// Parse an authentication string into an Authentication struct
66    fn from_str(s: &str) -> std::result::Result<Self, Self::Err> {
67        serde_json::from_str(s).map_err(|_err| AuthenticationParseError::InvalidToken)
68    }
69}
70
71impl Authentication {
72    /// Get the scheme of the authentication method
73    pub fn method(&self) -> &str {
74        match self {
75            Authentication::BearerToken(_) => "BearerToken",
76            Authentication::BasicHTTP { .. } => "BasicHTTP",
77            Authentication::CondaToken(_) => "CondaToken",
78            Authentication::S3Credentials { .. } => "S3",
79            Authentication::OAuth { .. } => "OAuth",
80        }
81    }
82}
