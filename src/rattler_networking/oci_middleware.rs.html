50/// Middleware to handle `oci://` URLs
51///
52/// Authentication follows the registry's `WWW-Authenticate` challenge: a
53/// `Bearer` challenge is exchanged for a scoped token at its realm, anything
54/// else sends stored credentials directly. Anonymous without
55/// [`with_authentication_storage`](Self::with_authentication_storage), which is
56/// enough for public registries.
57#[derive(Debug, Clone)]
58pub struct OciMiddleware {
59    /// Shared HTTP client reused across all OCI requests to avoid creating a
60    /// new connection pool on every token fetch or manifest pull.
61    client: LazyClient,
62
63    /// Credentials for private registries. Without a storage the middleware
64    /// only ever accesses registries anonymously.
65    auth_storage: Option<AuthenticationStorage>,
66
67    /// What [`OciMiddleware::registry_auth`] discovered about each registry
68    /// host so far, keyed by host.
69    registry_auth_cache: Arc<Mutex<HashMap<String, RegistryAuth>>>,
70}
71
72impl OciMiddleware {
73    /// Create a new [`OciMiddleware`] reusing the provided HTTP client.
74    pub fn new(client: impl Into<LazyClient>) -> Self {
75        Self {
76            client: client.into(),
77            auth_storage: None,
78            registry_auth_cache: Arc::new(Mutex::new(HashMap::new())),
79        }
80    }
81
82    /// Use `storage` to look up credentials for private registries.
83    #[must_use]
84    pub fn with_authentication_storage(mut self, storage: AuthenticationStorage) -> Self {
85        self.auth_storage = Some(storage);
86        self
87    }

89    /// Credentials stored for the registry, if any. [`AuthenticationStorage`]
90    /// resolves by host, so the `oci://` scheme is not a problem.
91    async fn stored_credentials(&self, url: &Url) -> Option<Authentication> {
92        match self
93            .auth_storage
94            .as_ref()?
95            .get_by_url_refreshed(url.clone())
96            .await
97        {
98            Ok((_, credentials)) => credentials,
99            Err(e) => {
100                tracing::warn!("OCI Mirror: could not look up credentials for {url}: {e}");
101                None
102            }
103        }
104    }
105
106    /// How `host` wants to be authenticated, probing it at most once per
107    /// process instead of once per artifact.
108    async fn registry_auth(&self, host: &str) -> RegistryAuth {
109        let cached = self
110            .registry_auth_cache
111            .lock()
112            .expect("OCI registry auth cache poisoned")
113            .get(host)
114            .cloned();
115        if let Some(cached) = cached {
116            return cached;
117        }
118
119        // Don't cache a failed probe in case it's a temporary error.
120        let Some(auth) = self.probe_registry_auth(host).await else {
121            return RegistryAuth::Direct;
122        };
123
124        // Concurrent first requests to the same host can both probe it. The
125        // answer is the same either way, so one duplicate request is cheaper
126        // than single-flight machinery.
127        self.registry_auth_cache
128            .lock()
129            .expect("OCI registry auth cache poisoned")
130            .insert(host.to_string(), auth.clone());
131
132        auth
133    }
134
135    /// Ask the registry how it wants to be authenticated with the OCI API
136    /// version check (`GET /v2/`).
137    ///
138    /// `None` when the registry did not answer, which is not the same as it
139    /// answering that it wants no negotiation.
140    async fn probe_registry_auth(&self, host: &str) -> Option<RegistryAuth> {
141        let Ok(url) = format!("https://{host}/v2/").parse::<Url>() else {
142            return Some(RegistryAuth::Direct);
143        };
144
145        match self.client.client().get(url.clone()).send().await {
146            Ok(response) => {
147                let status = response.status();
148                let challenges = parse_challenges(response.headers());
149                let result = registry_auth_from_probe(status, &challenges);
150                if result.is_none() {
151                    // Do not let a transient response permanently poison the
152                    // host-wide auth cache as `Direct`.
153                    tracing::debug!(
154                        "OCI Mirror: auth probe {url} returned unusable status {status} or challenge"
155                    );
156                }
157                result
158            }
159            Err(e) => {
160                // The artifact request that follows reports a better error than
161                // we could here, so a failed probe must not be fatal.
162                tracing::debug!("OCI Mirror: could not probe {url} for its auth challenge: {e}");
163                None
164            }
165        }
166    }
167
168    /// The `Authorization` header to send for `oci_url`, honouring the
169    /// registry's challenge.
170    async fn authorization_header(
171        &self,
172        oci_url: &OCIUrl,
173        action: OciAction,
174    ) -> Result<Option<HeaderValue>, OciMiddlewareError> {
175        let credentials = self.stored_credentials(&oci_url.url).await;
176
177        // A stored bearer token already *is* a registry token, so there is
178        // nothing to exchange it for.
179        if matches!(
180            credentials,
181            Some(Authentication::BearerToken(_) | Authentication::OAuth { .. })
182        ) {
183            return Ok(credentials_header(credentials.as_ref()));
184        }
185
186        match self.registry_auth(&oci_url.host).await {
187            RegistryAuth::TokenExchange { realm, service } => {
188                let token_url = token_url(&realm, service.as_deref(), &oci_url.path, action);
189                let token = get_token(&self.client, &token_url, credentials.as_ref()).await?;
190                let mut header = HeaderValue::from_str(&format!("Bearer {token}"))?;
191                header.set_sensitive(true);
192                Ok(Some(header))
193            }
194            RegistryAuth::Direct => Ok(credentials_header(credentials.as_ref())),
195        }
196    }
197
198    /// Resolve a challenge returned by a concrete manifest or blob request.
199    /// Unlike the `/v2/` probe, this is authoritative for the requested
200    /// repository and therefore replaces the cached host-wide answer.
201    async fn authorization_from_challenges(
202        &self,
203        oci_url: &OCIUrl,
204        challenges: &[Challenge],
205    ) -> Result<Option<HeaderValue>, OciMiddlewareError> {
206        let auth = registry_auth_from_challenges(challenges);
207        self.registry_auth_cache
208            .lock()
209            .expect("OCI registry auth cache poisoned")
210            .insert(oci_url.host.clone(), auth.clone());
211
212        let credentials = self.stored_credentials(&oci_url.url).await;
213        match auth {
214            RegistryAuth::TokenExchange { realm, service } => {
215                let token_url =
216                    token_url(&realm, service.as_deref(), &oci_url.path, OciAction::Pull);
217                let token = get_token(&self.client, &token_url, credentials.as_ref()).await?;
218                let mut header = HeaderValue::from_str(&format!("Bearer {token}"))?;
219                header.set_sensitive(true);
220                Ok(Some(header))
221            }
222            RegistryAuth::Direct => Ok(credentials_header(credentials.as_ref())),
223        }
224    }
225
226    /// Turn an `oci://` request into a request for the registry blob that holds
227    /// the artifact.
228    async fn rewrite_to_blob_request(
229        &self,
230        oci_url: &OCIUrl,
231        req: &mut Request,
232    ) -> Result<(), OciMiddlewareError> {
233        let authorization = self.authorization_header(oci_url, OciAction::Pull).await?;
234        match oci_url
235            .set_blob_url(&self.client, req, authorization.as_ref())
236            .await
237        {
238            Err(OciMiddlewareError::AuthenticationRequired(challenges)) => {
239                let authorization = self
240                    .authorization_from_challenges(oci_url, &challenges)
241                    .await?;
242                oci_url
243                    .set_blob_url(&self.client, req, authorization.as_ref())
244                    .await
245            }
246            result => result,
247        }
248    }
249}

251/// The authentication a registry asks for in its `GET /v2/` response.
252#[derive(Clone, Debug, PartialEq, Eq)]
253enum RegistryAuth {
254    /// A `Bearer` challenge: exchange credentials for a scoped registry token
255    /// at the challenge's realm (`https://ghcr.io/token` for ghcr.io).
256    TokenExchange { realm: Url, service: Option<String> },
257
258    /// Everything else: a `Basic` challenge (Amazon ECR), no challenge at all,
259    /// or a scheme we don't implement. There is nothing to negotiate, so send
260    /// the stored credentials as they are — or nothing, and let the registry
261    /// answer with its own error.
262    Direct,
263}
264
265/// Interpret the result of the best-effort `GET /v2/` probe.
266///
267/// Unexpected statuses and a `401` without a parseable challenge are not
268/// cached: both can be transient and neither tells us how the registry wants
269/// the concrete repository request authenticated.
270fn registry_auth_from_probe(status: StatusCode, challenges: &[Challenge]) -> Option<RegistryAuth> {
271    if status == StatusCode::UNAUTHORIZED {
272        return (!challenges.is_empty()).then(|| registry_auth_from_challenges(challenges));
273    }
274    status
275        .is_success()
276        .then(|| registry_auth_from_challenges(challenges))
277}
278
279/// Pick the authentication flow from a registry's challenges.
280///
281/// A `Bearer` challenge whose `realm` is missing, unparsable, or not HTTPS
282/// degrades to [`RegistryAuth::Direct`] rather than erroring: a registry we
283/// cannot negotiate with may still accept stored credentials directly.
284fn registry_auth_from_challenges(challenges: &[Challenge]) -> RegistryAuth {
285    for challenge in challenges {
286        if !challenge.scheme.eq_ignore_ascii_case("bearer") {
287            continue;
288        }
289        let Some(realm) = challenge
290            .params
291            .get("realm")
292            .and_then(|realm| Url::parse(realm).ok())
293            .filter(|realm| realm.scheme() == "https")
294        else {
295            // Basic credentials are forwarded to this endpoint during token
296            // exchange. Never allow a challenge to downgrade them to cleartext.
297            tracing::debug!("OCI Mirror: ignoring Bearer challenge without a usable HTTPS realm");
298            continue;
299        };
300        return RegistryAuth::TokenExchange {
301            realm,
302            service: challenge.params.get("service").cloned(),
303        };
304    }
305    RegistryAuth::Direct
306}

308/// The Docker-style token exchange URL: the challenge's realm, plus the service
309/// it named and the scope for this artifact.
310fn token_url(realm: &Url, service: Option<&str>, path: &str, action: OciAction) -> Url {
311    let mut url = realm.clone();
312    let mut query = url.query_pairs_mut();
313    if let Some(service) = service {
314        query.append_pair("service", service);
315    }
316    query.append_pair("scope", &format!("repository:{path}:{action}"));
317    drop(query);
318    url
319}

321/// The `Authorization` header for stored credentials. `None` when there are
322/// none, or when they are not something an HTTP header can carry: conda tokens
323/// live in the URL and S3 credentials sign the request instead.
324fn credentials_header(credentials: Option<&Authentication>) -> Option<HeaderValue> {
325    let value = match credentials? {
326        Authentication::BasicHTTP { username, password } => {
327            format!(
328                "Basic {}",
329                BASE64_STANDARD.encode(format!("{username}:{password}"))
330            )
331        }
332        Authentication::BearerToken(token) => format!("Bearer {token}"),
333        Authentication::OAuth { access_token, .. } => format!("Bearer {access_token}"),
334        Authentication::CondaToken(_) | Authentication::S3Credentials { .. } => return None,
335    };
336
337    // Never log the value itself, not even in the error case.
338    let Ok(mut header) = HeaderValue::from_str(&value) else {
339        tracing::warn!(
340            "OCI Mirror: stored credentials are not a valid header value, continuing without them"
341        );
342        return None;
343    };
344    header.set_sensitive(true);
345    Some(header)
346}

373// [oci://ghcr.io/channel-mirrors/conda-forge]/[osx-arm64/xtensor]
374async fn get_token(
375    client: &LazyClient,
376    token_url: &Url,
377    credentials: Option<&Authentication>,
378) -> Result<String, OciMiddlewareError> {
379    let mut request = client.client().get(token_url.clone());
380
381    // Like Docker, present stored credentials to the token endpoint: an
382    // anonymous exchange only ever yields a token for public repositories.
383    if let Some(header) = credentials_header(credentials) {
384        request = request.header(AUTHORIZATION, header);
385    }
386
387    let response = request.send().await?;
388
389    match response.error_for_status() {
390        Ok(response) => {
391            let token = response.json::<OCIToken>().await?;
392            Ok(token.token)
393        }
394        Err(e) => {
395            tracing::error!("OCI Mirror: failed to get token with URL: {}", token_url);
396            Err(OciMiddlewareError::Reqwest(e))
397        }
398    }
399}

410/// OCI registry tags are not allowed to contain `+`, `!`, or `=`, so we need to
411/// replace them with something else (reverse of `version_build_tag`)
412#[allow(dead_code)]
413fn reverse_version_build_tag(tag: &str) -> String {
414    tag.replace("__p__", "+")
415        .replace("__e__", "!")
416        .replace("__eq__", "=")
417}

419/// OCI registry tags are not allowed to contain `+`, `!`, or `=`, so we need to
420/// replace them with something else
421fn version_build_tag(tag: &str) -> String {
422    tag.replace('+', "__p__")
423        .replace('!', "__e__")
424        .replace('=', "__eq__")
425}

440    pub fn new(url: &Url) -> Result<Self, OciMiddlewareError> {
441        // get filename (last segment of path)
442        let filename = url
443            .path_segments()
444            .and_then(|mut s| s.next_back())
445            .ok_or_else(|| {
446                OciMiddlewareError::InvalidUrl(url.clone(), "URL has no path segments")
447            })?;
448
449        let mut res = OCIUrl {
450            url: url.clone(),
451            tag: "latest".to_string(),
452            media_type: "".to_string(),
453            host: url.host_str().unwrap_or("").to_string(),
454            path: url.path().trim_start_matches('/').to_string(),
455        };
456
457        let mut computed_filename = filename.to_string();
458
459        // We reimplement some archive name splitting logic from rattler here
460        // because we don't want to introduce cyclic dependencies
461        if let Some(archive_name) = filename.strip_suffix(".conda") {
462            let parts = archive_name.rsplitn(3, '-').collect::<Vec<&str>>();
463            match parts.as_slice() {
464                [build, version, name] => {
465                    computed_filename = name.to_string();
466                    res.tag = version_build_tag(&format!("{version}-{build}"));
467                    res.media_type = "application/vnd.conda.package.v2".to_string();
468                }
469                _ => {
470                    return Err(OciMiddlewareError::InvalidUrl(
471                        url.clone(),
472                        "package filename must have the form name-version-build.conda",
473                    ));
474                }
475            }
476        } else if let Some(archive_name) = filename.strip_suffix(".tar.bz2") {
477            let parts = archive_name.rsplitn(3, '-').collect::<Vec<&str>>();
478            match parts.as_slice() {
479                [build, version, name] => {
480                    computed_filename = name.to_string();
481                    res.tag = version_build_tag(&format!("{version}-{build}"));
482                    res.media_type = "application/vnd.conda.package.v1".to_string();
483                }
484                _ => {
485                    return Err(OciMiddlewareError::InvalidUrl(
486                        url.clone(),
487                        "package filename must have the form name-version-build.tar.bz2",
488                    ));
489                }
490            }
491        } else if filename.starts_with("repodata.json") {
492            computed_filename = "repodata.json".to_string();
493            if filename == "repodata.json" {
494                res.media_type = "application/vnd.conda.repodata.v1+json".to_string();
495            } else if filename.ends_with(".gz") {
496                res.media_type = "application/vnd.conda.repodata.v1+json+gzip".to_string();
497            } else if filename.ends_with(".bz2") {
498                res.media_type = "application/vnd.conda.repodata.v1+json+bz2".to_string();
499            } else if filename.ends_with(".zst") {
500                res.media_type = "application/vnd.conda.repodata.v1+json+zst".to_string();
501            }
502        }
503
504        // OCI image names cannot start with `_`, so we prefix it with `zzz`
505        if computed_filename.starts_with('_') {
506            computed_filename = format!("zzz{computed_filename}");
507        }
508
509        res.url = url.join(&computed_filename)?;
510        res.path = res.url.path().trim_start_matches('/').to_string();
511        Ok(res)
512    }

514    /// Point `req` at the blob holding this artifact, authenticated with
515    /// `authorization` (which the registry's challenge decided upon).
516    pub async fn set_blob_url(
517        &self,
518        client: &LazyClient,
519        req: &mut Request,
520        authorization: Option<&HeaderValue>,
521    ) -> Result<(), OciMiddlewareError> {
522        if let Some(header) = authorization {
523            req.headers_mut().insert(AUTHORIZATION, header.clone());
524        }
525
526        // if we know the hash, we can pull the artifact directly
527        // if we don't, we need to pull the manifest and then pull the artifact
528        if let Some(expected_sha_hash) = req
529            .headers()
530            .get("X-Expected-Sha256")
531            .and_then(|s| s.to_str().ok())
532        {
533            *req.url_mut() = self.blob_url(&format!("sha256:{expected_sha_hash}"))?;
534        } else {
535            // get the tag from the URL retrieve the manifest
536            let manifest_url = self.manifest_url()?; // TODO: handle error
537
538            let mut manifest_request = client
539                .client()
540                .get(manifest_url)
541                .header(ACCEPT, "application/vnd.oci.image.manifest.v1+json");
542            if let Some(header) = authorization {
543                manifest_request = manifest_request.header(AUTHORIZATION, header.clone());
544            }
545
546            let manifest = manifest_request.send().await?;
547            if manifest.status() == StatusCode::UNAUTHORIZED {
548                let challenges = parse_challenges(manifest.headers());
549                if !challenges.is_empty() {
550                    return Err(OciMiddlewareError::AuthenticationRequired(challenges));
551                }
552            }
553
554            if !manifest.status().is_success() {
555                return Err(OciMiddlewareError::ManifestRequestFailed(manifest.status()));
556            }
557            let manifest: Manifest = manifest.json().await?;
558
559            let layer = if let Some(layer) = manifest
560                .layers
561                .iter()
562                .find(|l| l.media_type == self.media_type)
563            {
564                layer
565            } else {
566                return Err(OciMiddlewareError::LayerNotFound);
567            };
568
569            *req.url_mut() = self.blob_url(&layer.digest)?;
570        }
571
572        Ok(())
573    }
574}

596#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
597#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
598impl Middleware for OciMiddleware {
599    async fn handle(
600        &self,
601        mut req: Request,
602        extensions: &mut Extensions,
603        next: Next<'_>,
604    ) -> reqwest_middleware::Result<Response> {
605        // if the URL is not an OCI URL, we don't need to do anything
606        if req.url().scheme() != "oci" {
607            return next.run(req, extensions).await;
608        }
609
610        let oci_url = match OCIUrl::new(req.url()) {
611            Ok(url) => url,
612            Err(e) => return Err(reqwest_middleware::Error::Middleware(e.into())),
613        };
614        let res = self.rewrite_to_blob_request(&oci_url, &mut req).await;
615
616        match res {
617            Ok(_) => {
618                // Keep one copy so a repository-specific challenge can be
619                // answered and replayed exactly once.
620                let Some(mut retry_req) = req.try_clone() else {
621                    return next.run(req, extensions).await;
622                };
623                let response = next.clone().run(req, extensions).await?;
624                if response.status() != StatusCode::UNAUTHORIZED {
625                    return Ok(response);
626                }
627
628                let challenges = parse_challenges(response.headers());
629                if challenges.is_empty() {
630                    return Ok(response);
631                }
632                let authorization = self
633                    .authorization_from_challenges(&oci_url, &challenges)
634                    .await
635                    .map_err(|e| reqwest_middleware::Error::Middleware(e.into()))?;
636                let Some(authorization) = authorization else {
637                    return Ok(response);
638                };
639                retry_req.headers_mut().insert(AUTHORIZATION, authorization);
640                next.run(retry_req, extensions).await
641            }
642            Err(e) => match e {
643                // Return clean 404s rather than erroring as callers safely handle them.
644                OciMiddlewareError::LayerNotFound => {
645                    return Ok(create_404_response(
646                        req.url(),
647                        "No layer available for media type",
648                    ));
649                }
650                OciMiddlewareError::ManifestRequestFailed(StatusCode::NOT_FOUND) => {
651                    return Ok(create_404_response(req.url(), "Manifest not found"));
652                }
653                _ => {
654                    return Err(reqwest_middleware::Error::Middleware(e.into()));
655                }
656            },
657        }
658    }

671    /// The challenges of a registry that answers `GET /v2/` with `header`.
672    fn challenges(header: &str) -> Vec<Challenge> {
673        let mut headers = http::HeaderMap::new();
674        headers.insert(
675            http::header::WWW_AUTHENTICATE,
676            http::HeaderValue::from_str(header).unwrap(),
677        );
678        parse_challenges(&headers)
679    }

681    /// ghcr.io's challenge must keep producing exactly the token URL the
682    /// middleware used to hardcode.
683    #[test]
684    fn bearer_challenge_builds_scoped_token_url() {
685        let auth = registry_auth_from_challenges(&challenges(
686            r#"Bearer realm="https://ghcr.io/token",service="ghcr.io""#,
687        ));
688        let RegistryAuth::TokenExchange { realm, service } = auth else {
689            panic!("a Bearer challenge with a realm must yield a token exchange, got {auth:?}");
690        };
691
692        let url = token_url(
693            &realm,
694            service.as_deref(),
695            "channel-mirrors/conda-forge/noarch/xtensor",
696            OciAction::Pull,
697        );
698
699        assert_eq!(url.path(), "/token");
700        // `scope` is percent-encoded here but not in the old hand-built URL;
701        // registries decode the query either way.
702        let query: Vec<_> = url.query_pairs().collect();
703        assert_eq!(
704            query,
705            vec![
706                ("service".into(), "ghcr.io".into()),
707                (
708                    "scope".into(),
709                    "repository:channel-mirrors/conda-forge/noarch/xtensor:pull".into()
710                ),
711            ]
712        );
713    }

770    fn unusable_challenges_degrade_to_direct() {

773            r#"Bearer realm="not a url""#,
