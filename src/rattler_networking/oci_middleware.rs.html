50/// Middleware to handle `oci://` URLs
51///
52/// Authentication follows the registry's `WWW-Authenticate` challenge: a
53/// `Bearer` challenge is exchanged for a scoped token at its realm, anything
54/// else sends stored credentials directly. Anonymous without
55/// [`with_authentication_storage`](Self::with_authentication_storage), which is
56/// enough for public registries.
57#[derive(Debug, Clone)]
58pub struct OciMiddleware {
59    /// Shared HTTP client reused across all OCI requests to avoid creating a
60    /// new connection pool on every token fetch or manifest pull.
61    client: LazyClient,
62
63    /// Credentials for private registries. Without a storage the middleware
64    /// only ever accesses registries anonymously.
65    auth_storage: Option<AuthenticationStorage>,
66
67    /// What [`OciMiddleware::registry_auth`] discovered about each registry
68    /// host so far, keyed by host.
69    registry_auth_cache: Arc<Mutex<HashMap<String, RegistryAuth>>>,
70}
71
72impl OciMiddleware {
73    /// Create a new [`OciMiddleware`] reusing the provided HTTP client.
74    pub fn new(client: impl Into<LazyClient>) -> Self {
75        Self {
76            client: client.into(),
77            auth_storage: None,
78            registry_auth_cache: Arc::new(Mutex::new(HashMap::new())),
79        }
80    }
81
82    /// Use `storage` to look up credentials for private registries.
83    #[must_use]
84    pub fn with_authentication_storage(mut self, storage: AuthenticationStorage) -> Self {
85        self.auth_storage = Some(storage);
86        self
87    }
88
89    /// Credentials stored for the registry, if any. [`AuthenticationStorage`]
90    /// resolves by host, so the `oci://` scheme is not a problem.
91    async fn stored_credentials(&self, url: &Url) -> Option<Authentication> {
92        match self
93            .auth_storage
94            .as_ref()?
95            .get_by_url_refreshed(url.clone())
96            .await
97        {
98            Ok((_, credentials)) => credentials,
99            Err(e) => {
100                tracing::warn!("OCI Mirror: could not look up credentials for {url}: {e}");
101                None
102            }
103        }
104    }
105
106    /// How `host` wants to be authenticated, probing it at most once per
107    /// process instead of once per artifact.
108    async fn registry_auth(&self, host: &str) -> RegistryAuth {
109        let cached = self
110            .registry_auth_cache
111            .lock()
112            .expect("OCI registry auth cache poisoned")
113            .get(host)
114            .cloned();
115        if let Some(cached) = cached {
116            return cached;
117        }
118
119        // Don't cache a failed probe in case it's a temporary error.
120        let Some(auth) = self.probe_registry_auth(host).await else {
121            return RegistryAuth::Direct;
122        };
123
124        // Concurrent first requests to the same host can both probe it. The
125        // answer is the same either way, so one duplicate request is cheaper
126        // than single-flight machinery.
127        self.registry_auth_cache
128            .lock()
129            .expect("OCI registry auth cache poisoned")
130            .insert(host.to_string(), auth.clone());
131
132        auth
133    }
134
135    /// Ask the registry how it wants to be authenticated with the OCI API
136    /// version check (`GET /v2/`).
137    ///
138    /// `None` when the registry did not answer, which is not the same as it
139    /// answering that it wants no negotiation.
140    async fn probe_registry_auth(&self, host: &str) -> Option<RegistryAuth> {
141        let Ok(url) = format!("https://{host}/v2/").parse::<Url>() else {
142            return Some(RegistryAuth::Direct);
143        };
144
145        match self.client.client().get(url.clone()).send().await {
146            Ok(response) => {
147                let status = response.status();
148                let challenges = parse_challenges(response.headers());
149                let result = registry_auth_from_probe(status, &challenges);
150                if result.is_none() {
151                    // Do not let a transient response permanently poison the
152                    // host-wide auth cache as `Direct`.
153                    tracing::debug!(
154                        "OCI Mirror: auth probe {url} returned unusable status {status} or challenge"
155                    );
156                }
157                result
158            }
159            Err(e) => {
160                // The artifact request that follows reports a better error than
161                // we could here, so a failed probe must not be fatal.
162                tracing::debug!("OCI Mirror: could not probe {url} for its auth challenge: {e}");
163                None
164            }
165        }
166    }
167
168    /// The `Authorization` header to send for `oci_url`, honouring the
169    /// registry's challenge.
170    async fn authorization_header(
171        &self,
172        oci_url: &OCIUrl,
173        action: OciAction,
174    ) -> Result<Option<HeaderValue>, OciMiddlewareError> {
175        let credentials = self.stored_credentials(&oci_url.url).await;
176
177        // A stored bearer token already *is* a registry token, so there is
178        // nothing to exchange it for.
179        if matches!(
180            credentials,
181            Some(Authentication::BearerToken(_) | Authentication::OAuth { .. })
182        ) {
183            return Ok(credentials_header(credentials.as_ref()));
184        }
185
186        match self.registry_auth(&oci_url.host).await {
187            RegistryAuth::TokenExchange { realm, service } => {
188                let token_url = token_url(&realm, service.as_deref(), &oci_url.path, action);
189                let token = get_token(&self.client, &token_url, credentials.as_ref()).await?;
190                let mut header = HeaderValue::from_str(&format!("Bearer {token}"))?;
191                header.set_sensitive(true);
192                Ok(Some(header))
193            }
194            RegistryAuth::Direct => Ok(credentials_header(credentials.as_ref())),
195        }
196    }
197
198    /// Resolve a challenge returned by a concrete manifest or blob request.
199    /// Unlike the `/v2/` probe, this is authoritative for the requested
200    /// repository and therefore replaces the cached host-wide answer.
201    async fn authorization_from_challenges(
202        &self,
203        oci_url: &OCIUrl,
204        challenges: &[Challenge],
205    ) -> Result<Option<HeaderValue>, OciMiddlewareError> {
206        let auth = registry_auth_from_challenges(challenges);
207        self.registry_auth_cache
208            .lock()
209            .expect("OCI registry auth cache poisoned")
210            .insert(oci_url.host.clone(), auth.clone());
211
212        let credentials = self.stored_credentials(&oci_url.url).await;
213        match auth {
214            RegistryAuth::TokenExchange { realm, service } => {
215                let token_url =
216                    token_url(&realm, service.as_deref(), &oci_url.path, OciAction::Pull);
217                let token = get_token(&self.client, &token_url, credentials.as_ref()).await?;
218                let mut header = HeaderValue::from_str(&format!("Bearer {token}"))?;
219                header.set_sensitive(true);
220                Ok(Some(header))
221            }
222            RegistryAuth::Direct => Ok(credentials_header(credentials.as_ref())),
223        }
224    }
225
226    /// Turn an `oci://` request into a request for the registry blob that holds
227    /// the artifact.
228    async fn rewrite_to_blob_request(
229        &self,
230        oci_url: &OCIUrl,
231        req: &mut Request,
232        expected_sha256: Option<&str>,
233    ) -> Result<(), OciMiddlewareError> {
234        let authorization = self.authorization_header(oci_url, OciAction::Pull).await?;
235        match oci_url
236            .set_blob_url(&self.client, req, authorization.as_ref(), expected_sha256)
237            .await
238        {
239            Err(OciMiddlewareError::AuthenticationRequired(challenges)) => {
240                let authorization = self
241                    .authorization_from_challenges(oci_url, &challenges)
242                    .await?;
243                oci_url
244                    .set_blob_url(&self.client, req, authorization.as_ref(), expected_sha256)
245                    .await
246            }
247            result => result,
248        }
249    }
250
251    /// Send `req` downstream, keeping one copy so a repository-specific
252    /// `WWW-Authenticate` challenge can be answered and the request replayed
253    /// exactly once.
254    async fn run_with_challenge_retry(
255        &self,
256        oci_url: &OCIUrl,
257        req: Request,
258        extensions: &mut Extensions,
259        next: &Next<'_>,
260    ) -> reqwest_middleware::Result<Response> {
261        let Some(mut retry_req) = req.try_clone() else {
262            return next.clone().run(req, extensions).await;
263        };
264        let response = next.clone().run(req, extensions).await?;
265        if response.status() != StatusCode::UNAUTHORIZED {
266            return Ok(response);
267        }
268
269        let challenges = parse_challenges(response.headers());
270        if challenges.is_empty() {
271            return Ok(response);
272        }
273        let Some(authorization) = self
274            .authorization_from_challenges(oci_url, &challenges)
275            .await
276            .map_err(|e| reqwest_middleware::Error::Middleware(e.into()))?
277        else {
278            return Ok(response);
279        };
280        retry_req.headers_mut().insert(AUTHORIZATION, authorization);
281        next.clone().run(retry_req, extensions).await
282    }
283}

285/// The authentication a registry asks for in its `GET /v2/` response.
286#[derive(Clone, Debug, PartialEq, Eq)]
287enum RegistryAuth {
288    /// A `Bearer` challenge: exchange credentials for a scoped registry token
289    /// at the challenge's realm (`https://ghcr.io/token` for ghcr.io).
290    TokenExchange { realm: Url, service: Option<String> },
291
292    /// Everything else: a `Basic` challenge (Amazon ECR), no challenge at all,
293    /// or a scheme we don't implement. There is nothing to negotiate, so send
294    /// the stored credentials as they are — or nothing, and let the registry
295    /// answer with its own error.
296    Direct,
297}
298
299/// Interpret the result of the best-effort `GET /v2/` probe.
300///
301/// Unexpected statuses and a `401` without a parseable challenge are not
302/// cached: both can be transient and neither tells us how the registry wants
303/// the concrete repository request authenticated.
304fn registry_auth_from_probe(status: StatusCode, challenges: &[Challenge]) -> Option<RegistryAuth> {
305    if status == StatusCode::UNAUTHORIZED {
306        return (!challenges.is_empty()).then(|| registry_auth_from_challenges(challenges));
307    }
308    status
309        .is_success()
310        .then(|| registry_auth_from_challenges(challenges))
311}
312
313/// Pick the authentication flow from a registry's challenges.
314///
315/// A `Bearer` challenge whose `realm` is missing, unparsable, or not HTTPS
316/// degrades to [`RegistryAuth::Direct`] rather than erroring: a registry we
317/// cannot negotiate with may still accept stored credentials directly.
318fn registry_auth_from_challenges(challenges: &[Challenge]) -> RegistryAuth {
319    for challenge in challenges {
320        if !challenge.scheme.eq_ignore_ascii_case("bearer") {
321            continue;
322        }
323        let Some(realm) = challenge
324            .params
325            .get("realm")
326            .and_then(|realm| Url::parse(realm).ok())
327            .filter(|realm| realm.scheme() == "https")
328        else {
329            // Basic credentials are forwarded to this endpoint during token
330            // exchange. Never allow a challenge to downgrade them to cleartext.
331            tracing::debug!("OCI Mirror: ignoring Bearer challenge without a usable HTTPS realm");
332            continue;
333        };
334        return RegistryAuth::TokenExchange {
335            realm,
336            service: challenge.params.get("service").cloned(),
337        };
338    }
339    RegistryAuth::Direct
340}

342/// The Docker-style token exchange URL: the challenge's realm, plus the service
343/// it named and the scope for this artifact.
344fn token_url(realm: &Url, service: Option<&str>, path: &str, action: OciAction) -> Url {
345    let mut url = realm.clone();
346    let mut query = url.query_pairs_mut();
347    if let Some(service) = service {
348        query.append_pair("service", service);
349    }
350    query.append_pair("scope", &format!("repository:{path}:{action}"));
351    drop(query);
352    url
353}

355/// The `Authorization` header for stored credentials. `None` when there are
356/// none, or when they are not something an HTTP header can carry: conda tokens
357/// live in the URL and S3 credentials sign the request instead.
358fn credentials_header(credentials: Option<&Authentication>) -> Option<HeaderValue> {
359    let value = match credentials? {
360        Authentication::BasicHTTP { username, password } => {
361            format!(
362                "Basic {}",
363                BASE64_STANDARD.encode(format!("{username}:{password}"))
364            )
365        }
366        Authentication::BearerToken(token) => format!("Bearer {token}"),
367        Authentication::OAuth { access_token, .. } => format!("Bearer {access_token}"),
368        Authentication::CondaToken(_) | Authentication::S3Credentials { .. } => return None,
369    };
370
371    // Never log the value itself, not even in the error case.
372    let Ok(mut header) = HeaderValue::from_str(&value) else {
373        tracing::warn!(
374            "OCI Mirror: stored credentials are not a valid header value, continuing without them"
375        );
376        return None;
377    };
378    header.set_sensitive(true);
379    Some(header)
380}

407// [oci://ghcr.io/channel-mirrors/conda-forge]/[osx-arm64/xtensor]
408async fn get_token(
409    client: &LazyClient,
410    token_url: &Url,
411    credentials: Option<&Authentication>,
412) -> Result<String, OciMiddlewareError> {
413    let mut request = client.client().get(token_url.clone());
414
415    // Like Docker, present stored credentials to the token endpoint: an
416    // anonymous exchange only ever yields a token for public repositories.
417    if let Some(header) = credentials_header(credentials) {
418        request = request.header(AUTHORIZATION, header);
419    }
420
421    let response = request.send().await?;
422
423    match response.error_for_status() {
424        Ok(response) => {
425            let token = response.json::<OCIToken>().await?;
426            Ok(token.token)
427        }
428        Err(e) => {
429            tracing::error!("OCI Mirror: failed to get token with URL: {}", token_url);
430            Err(OciMiddlewareError::Reqwest(e))
431        }
432    }
433}

436struct OCIUrl {
437    url: Url,
438    host: String,
439    path: String,
440    tag: String,

444/// OCI registry tags are not allowed to contain `+`, `!`, or `=`, so we need to
445/// replace them with something else (reverse of `version_build_tag`)
446#[allow(dead_code)]
447fn reverse_version_build_tag(tag: &str) -> String {
448    tag.replace("__p__", "+")
449        .replace("__e__", "!")
450        .replace("__eq__", "=")
451}
452
453/// OCI registry tags are not allowed to contain `+`, `!`, or `=`, so we need to
454/// replace them with something else
455fn version_build_tag(tag: &str) -> String {
456    tag.replace('+', "__p__")
457        .replace('!', "__e__")
458        .replace('=', "__eq__")
459}

461impl OCIUrl {
462    pub fn manifest_url(&self) -> Result<Url, ParseError> {
463        format!(
464            "https://{}/v2/{}/manifests/{}",
465            self.host, self.path, self.tag
466        )
467        .parse()
468    }
469
470    pub fn blob_url(&self, sha256: &str) -> Result<Url, ParseError> {
471        format!("https://{}/v2/{}/blobs/{}", self.host, self.path, sha256).parse()
472    }
473
474    pub fn new(url: &Url) -> Result<Self, OciMiddlewareError> {
475        // get filename (last segment of path)
476        let filename = url
477            .path_segments()
478            .and_then(|mut s| s.next_back())
479            .ok_or_else(|| {
480                OciMiddlewareError::InvalidUrl(url.clone(), "URL has no path segments")
481            })?;
482
483        let mut res = OCIUrl {
484            url: url.clone(),
485            tag: "latest".to_string(),
486            media_type: "".to_string(),
487            host: url.host_str().unwrap_or("").to_string(),
488            path: url.path().trim_start_matches('/').to_string(),
489        };
490
491        let mut computed_filename = filename.to_string();
492
493        // We reimplement some archive name splitting logic from rattler here
494        // because we don't want to introduce cyclic dependencies
495        if let Some(archive_name) = filename.strip_suffix(".conda") {
496            let parts = archive_name.rsplitn(3, '-').collect::<Vec<&str>>();
497            match parts.as_slice() {
498                [build, version, name] => {
499                    computed_filename = name.to_string();
500                    res.tag = version_build_tag(&format!("{version}-{build}"));
501                    res.media_type = "application/vnd.conda.package.v2".to_string();
502                }
503                _ => {
504                    return Err(OciMiddlewareError::InvalidUrl(
505                        url.clone(),
506                        "package filename must have the form name-version-build.conda",
507                    ));
508                }
509            }
510        } else if let Some(archive_name) = filename.strip_suffix(".tar.bz2") {
511            let parts = archive_name.rsplitn(3, '-').collect::<Vec<&str>>();
512            match parts.as_slice() {
513                [build, version, name] => {
514                    computed_filename = name.to_string();
515                    res.tag = version_build_tag(&format!("{version}-{build}"));
516                    res.media_type = "application/vnd.conda.package.v1".to_string();
517                }
518                _ => {
519                    return Err(OciMiddlewareError::InvalidUrl(
520                        url.clone(),
521                        "package filename must have the form name-version-build.tar.bz2",
522                    ));
523                }
524            }
525        } else if filename.starts_with("repodata.json") {
526            computed_filename = "repodata.json".to_string();
527            if filename == "repodata.json" {
528                res.media_type = "application/vnd.conda.repodata.v1+json".to_string();
529            } else if filename.ends_with(".gz") {
530                res.media_type = "application/vnd.conda.repodata.v1+json+gzip".to_string();
531            } else if filename.ends_with(".bz2") {
532                res.media_type = "application/vnd.conda.repodata.v1+json+bz2".to_string();
533            } else if filename.ends_with(".zst") {
534                res.media_type = "application/vnd.conda.repodata.v1+json+zst".to_string();
535            }
536        }
537
538        // OCI image names cannot start with `_`, so we prefix it with `zzz`
539        if computed_filename.starts_with('_') {
540            computed_filename = format!("zzz{computed_filename}");
541        }
542
543        res.url = url.join(&computed_filename)?;
544        res.path = res.url.path().trim_start_matches('/').to_string();
545        Ok(res)
546    }
547
548    /// Point `req` at the blob holding this artifact, authenticated with
549    /// `authorization` (which the registry's challenge decided upon).
550    ///
551    /// With `expected_sha256` the blob is addressed directly; without it the
552    /// manifest is pulled first to learn the digest.
553    pub async fn set_blob_url(
554        &self,
555        client: &LazyClient,
556        req: &mut Request,
557        authorization: Option<&HeaderValue>,
558        expected_sha256: Option<&str>,
559    ) -> Result<(), OciMiddlewareError> {
560        if let Some(header) = authorization {
561            req.headers_mut().insert(AUTHORIZATION, header.clone());
562        }
563
564        // if we know the hash, we can pull the artifact directly
565        // if we don't, we need to pull the manifest and then pull the artifact
566        if let Some(expected_sha_hash) = expected_sha256 {
567            *req.url_mut() = self.blob_url(&format!("sha256:{expected_sha_hash}"))?;
568        } else {
569            // get the tag from the URL retrieve the manifest
570            let manifest_url = self.manifest_url()?; // TODO: handle error
571
572            let mut manifest_request = client
573                .client()
574                .get(manifest_url)
575                .header(ACCEPT, "application/vnd.oci.image.manifest.v1+json");
576            if let Some(header) = authorization {
577                manifest_request = manifest_request.header(AUTHORIZATION, header.clone());
578            }
579
580            let manifest = manifest_request.send().await?;
581            if manifest.status() == StatusCode::UNAUTHORIZED {
582                let challenges = parse_challenges(manifest.headers());
583                if !challenges.is_empty() {
584                    return Err(OciMiddlewareError::AuthenticationRequired(challenges));
585                }
586            }
587
588            if !manifest.status().is_success() {
589                return Err(OciMiddlewareError::ManifestRequestFailed(manifest.status()));
590            }
591            let manifest: Manifest = manifest.json().await?;
592
593            let layer = if let Some(layer) = manifest
594                .layers
595                .iter()
596                .find(|l| l.media_type == self.media_type)
597            {
598                layer
599            } else {
600                return Err(OciMiddlewareError::LayerNotFound);
601            };
602
603            *req.url_mut() = self.blob_url(&layer.digest)?;
604        }
605
606        Ok(())
607    }
608}

630#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
631#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
632impl Middleware for OciMiddleware {
633    async fn handle(
634        &self,
635        mut req: Request,
636        extensions: &mut Extensions,
637        next: Next<'_>,
638    ) -> reqwest_middleware::Result<Response> {
639        // if the URL is not an OCI URL, we don't need to do anything
640        if req.url().scheme() != "oci" {
641            return next.run(req, extensions).await;
642        }
643
644        let oci_url = match OCIUrl::new(req.url()) {
645            Ok(url) => url,
646            Err(e) => return Err(reqwest_middleware::Error::Middleware(e.into())),
647        };
648
649        let expected_sha256 = req
650            .headers()
651            .get("X-Expected-Sha256")
652            .and_then(|s| s.to_str().ok())
653            .map(ToString::to_string);
654
655        if let Err(e) = self
656            .rewrite_to_blob_request(&oci_url, &mut req, expected_sha256.as_deref())
657            .await
658        {
659            return lookup_error_to_response(e, req.url());
660        }
661
662        let fallback_req = if expected_sha256.is_some() {
663            req.try_clone()
664        } else {
665            None
666        };
667        let response = self
668            .run_with_challenge_retry(&oci_url, req, extensions, &next)
669            .await?;
670
671        // Pull-through caches (e.g. Amazon ECR) only import an artifact once
672        // its manifest is pulled, so a digest-addressed blob can 404 while the
673        // manifest route still works. Retry through the manifest once.
674        let Some(mut fallback_req) = fallback_req else {
675            return Ok(response);
676        };
677        if response.status() != StatusCode::NOT_FOUND {
678            return Ok(response);
679        }
680        if let Err(e) = self
681            .rewrite_to_blob_request(&oci_url, &mut fallback_req, None)
682            .await
683        {
684            return lookup_error_to_response(e, fallback_req.url());
685        }
686        self.run_with_challenge_retry(&oci_url, fallback_req, extensions, &next)
687            .await
688    }
689}
690
691/// Determine whether a lookup error is a safe 404 or a fatal error.
692fn lookup_error_to_response(
693    error: OciMiddlewareError,
694    url: &Url,
695) -> reqwest_middleware::Result<Response> {
696    match error {
697        OciMiddlewareError::LayerNotFound => Ok(create_404_response(
698            url,
699            "No layer available for media type",
700        )),
701        OciMiddlewareError::ManifestRequestFailed(StatusCode::NOT_FOUND) => {
702            Ok(create_404_response(url, "Manifest not found"))
703        }
704        _ => Err(reqwest_middleware::Error::Middleware(error.into())),
705    }
706}
