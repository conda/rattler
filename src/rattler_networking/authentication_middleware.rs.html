19/// `reqwest` middleware to authenticate requests
20#[derive(Clone)]
21pub struct AuthenticationMiddleware {
22    auth_storage: AuthenticationStorage,
23    credential_sources: HashMap<url::Origin, url::Host<String>>,
24}

26#[cfg_attr(not(target_arch = "wasm32"), async_trait::async_trait)]
27#[cfg_attr(target_arch = "wasm32", async_trait::async_trait(?Send))]
28impl Middleware for AuthenticationMiddleware {
29    async fn handle(
30        &self,
31        req: Request,
32        extensions: &mut http::Extensions,
33        next: Next<'_>,
34    ) -> reqwest_middleware::Result<Response> {
35        // If an `Authorization` header is already present, don't authenticate
36        if req.headers().get(reqwest::header::AUTHORIZATION).is_some() {
37            return next.run(req, extensions).await;
38        }
39
40        let url = req.url().clone();
41        let explicit_source = !self.credential_sources.is_empty();
42        let selected = if explicit_source {
43            let Some(host) = self.credential_sources.get(&url.origin()) else {
44                return next.run(req, extensions).await;
45            };
46            let key = host.to_string();
47            self.auth_storage
48                .get(&key)
49                .map(|auth| (url, auth.map(|auth| (key, auth))))
50                .map_err(|_error| ())
51        } else {
52            self.auth_storage
53                .get_by_url_with_host(url)
54                .map_err(|_error| ())
55        };
56        match selected {
57            Err(_) => {
58                // Forward error to caller (invalid URL)
59                next.run(req, extensions).await
60            }
61            Ok((url, auth_with_key)) => {
62                let auth = match auth_with_key {
63                    Some((matched_key, auth)) => {
64                        let refresh_result = oauth_refresh::maybe_refresh_oauth(
65                            &self.auth_storage,
66                            auth,
67                            &matched_key,
68                        )
69                        .await;
70                        if let Some(failure) = refresh_result.failure() {
71                            tracing::warn!(
72                                "OAuth refresh for '{matched_key}' did not produce fresh credentials: {failure}"
73                            );
74                        }
75                        refresh_result.into_authentication()
76                    }
77                    None => None,
78                };
79
80                // Refresh may have picked up a different credential kind.
81                if explicit_source
82                    && auth.as_ref().is_some_and(|auth| {
83                        !matches!(
84                            auth,
85                            Authentication::OAuth { .. } | Authentication::BearerToken(_)
86                        )
87                    })
88                {
89                    tracing::warn!(
90                        "Selected credential is not an OAuth or bearer token; not sending it to the configured origin"
91                    );
92                    return next.run(req, extensions).await;
93                }
94
95                let url = Self::authenticate_url(url, &auth);
96
97                let mut req = req;
98                *req.url_mut() = url;
99
100                let req = Self::authenticate_request(req, &auth).await?;
101                next.run(req, extensions).await
102            }
103        }
104    }
105}

107impl AuthenticationMiddleware {
108    /// Create a new authentication middleware with the given authentication
109    /// storage
110    pub fn from_auth_storage(auth_storage: AuthenticationStorage) -> Self {
111        Self {
112            auth_storage,
113            credential_sources: HashMap::new(),
114        }
115    }
116
117    /// Create a new authentication middleware with the default authentication
118    /// storage
119    pub fn from_env_and_defaults() -> Result<Self, AuthenticationStorageError> {
120        Ok(Self {
121            auth_storage: AuthenticationStorage::from_env_and_defaults()?,
122            credential_sources: HashMap::new(),
123        })
124    }
125
126    /// Reuse `source_host`'s OAuth or bearer credential for `trusted_origins`,
127    /// without wildcard fallback. Refresh updates the source credential.
128    ///
129    /// Calls add mappings; repeated origins replace their source. Empty input
130    /// changes nothing. Once configured, unmapped origins and missing or
131    /// unsupported credentials remain anonymous.
132    ///
133    /// Use on a dedicated API client, not stacked with channel authentication.
134    /// Disable redirects on the underlying client for credential-bearing requests.
135    /// Audience-token refresh is supported on native targets only.
136    pub fn with_credentials_from(
137        mut self,
138        source_host: url::Host<String>,
139        trusted_origins: impl IntoIterator<Item = url::Origin>,
140    ) -> Self {
141        for origin in trusted_origins {
142            self.credential_sources.insert(origin, source_host.clone());
143        }
144        self
145    }
146
147    /// Authenticate the given URL with the given authentication information
148    fn authenticate_url(url: Url, auth: &Option<Authentication>) -> Url {
149        if let Some(credentials) = auth {
150            match credentials {
151                Authentication::CondaToken(token) => {
152                    let path = url.path();
153
154                    let mut new_path = String::new();
155                    new_path.push_str(format!("/t/{token}").as_str());
156                    new_path.push_str(path);
157
158                    let mut url = url.clone();
159                    url.set_path(&new_path);
160                    url
161                }
162                _ => url,
163            }
164        } else {
165            url
166        }
167    }
168
169    /// Authenticate the given request with the given authentication information
170    async fn authenticate_request(
171        mut req: reqwest::Request,
172        auth: &Option<Authentication>,
173    ) -> reqwest_middleware::Result<reqwest::Request> {
174        if let Some(credentials) = auth {
175            match credentials {
176                Authentication::BearerToken(token) => {
177                    let bearer_auth = format!("Bearer {token}");
178
179                    let mut header_value = reqwest::header::HeaderValue::from_str(&bearer_auth)
180                        .map_err(reqwest_middleware::Error::middleware)?;
181                    header_value.set_sensitive(true);
182
183                    req.headers_mut()
184                        .insert(reqwest::header::AUTHORIZATION, header_value);
185                    Ok(req)
186                }
187                Authentication::BasicHTTP { username, password } => {
188                    let basic_auth = format!("{username}:{password}");
189                    let basic_auth = BASE64_STANDARD.encode(basic_auth);
190                    let basic_auth = format!("Basic {basic_auth}");
191
192                    let mut header_value = reqwest::header::HeaderValue::from_str(&basic_auth)
193                        .expect("base64 can always be converted to a header value");
194                    header_value.set_sensitive(true);
195                    req.headers_mut()
196                        .insert(reqwest::header::AUTHORIZATION, header_value);
197                    Ok(req)
198                }
199                Authentication::OAuth { access_token, .. } => {
200                    let bearer_auth = format!("Bearer {access_token}");
201
202                    let mut header_value = reqwest::header::HeaderValue::from_str(&bearer_auth)
203                        .map_err(reqwest_middleware::Error::middleware)?;
204                    header_value.set_sensitive(true);
205
206                    req.headers_mut()
207                        .insert(reqwest::header::AUTHORIZATION, header_value);
208                    Ok(req)
209                }
210                Authentication::CondaToken(_) | Authentication::S3Credentials { .. } => Ok(req),
211            }
212        } else {
213            Ok(req)
214        }
215    }
216}

218/// Returns the default auth storage directory used by rattler.
219/// Would be placed in $HOME/.rattler, except when there is no home then it will
220/// be put in '/rattler/'
221pub fn default_auth_store_fallback_directory() -> &'static Path {
222    static FALLBACK_AUTH_DIR: OnceLock<PathBuf> = OnceLock::new();
223    FALLBACK_AUTH_DIR.get_or_init(|| {
224        #[cfg(feature = "dirs")]
225        return dirs::home_dir()
226            .map_or_else(|| {
227                tracing::warn!("using '/rattler' to store fallback authentication credentials because the home directory could not be found");
228                // This can only happen if the dirs lib can't find a home directory this is very unlikely.
229                PathBuf::from("/rattler/")
230            }, |home| home.join(".rattler/"));
231        #[cfg(not(feature = "dirs"))]
232        {
233            PathBuf::from("/rattler/")
234        }
235    })
236}

261    // Requests are only authenticated when executed, so we need to capture and
262    // cancel the request
263    struct CaptureAbortMiddleware {
264        pub captured_tx: tokio::sync::mpsc::Sender<reqwest::Request>,
265    }
266
267    #[async_trait::async_trait]
268    impl Middleware for CaptureAbortMiddleware {
269        async fn handle(
270            &self,
271            req: Request,
272            _: &mut http::Extensions,
273            _: Next<'_>,
274        ) -> reqwest_middleware::Result<Response> {
275            self.captured_tx
276                .send(req)
277                .await
278                .expect("failed to capture request");
279            Err(reqwest_middleware::Error::middleware(
280                std::io::Error::other("captured request, aborting"),
281            ))
282        }
283    }

285    fn make_client_harness(
286        middleware: AuthenticationMiddleware,
287    ) -> (
288        reqwest_middleware::ClientWithMiddleware,
289        tokio::sync::mpsc::Receiver<reqwest::Request>,
290    ) {
291        let (captured_tx, captured_rx) = tokio::sync::mpsc::channel(1);
292        let client = reqwest_middleware::ClientBuilder::new(reqwest::Client::default())
293            .with(middleware)
294            .with_arc(Arc::new(CaptureAbortMiddleware { captured_tx }))
295            .build();
296
297        (client, captured_rx)
298    }

520    async fn concurrent_oauth_refresh_is_coalesced_by_authentication_middleware() {
521        #[derive(Clone)]
522        struct TestState {
523            audience: Option<&'static str>,
524            refresh_count: Arc<AtomicUsize>,
525            seen_authorization: Arc<Mutex<Vec<Option<String>>>>,
526        }
527
528        async fn token(
529            State(state): State<TestState>,
530            Form(form): Form<HashMap<String, String>>,
531        ) -> (StatusCode, Json<serde_json::Value>) {
532            assert_eq!(form.get("audience").map(String::as_str), state.audience);
533            state.refresh_count.fetch_add(1, Ordering::SeqCst);
534            (
535                StatusCode::OK,
536                Json(json!({
537                    "access_token": "fresh-access-token",
538                    "refresh_token": "rotated-refresh-token",
539                    "expires_in": 3600,
540                    "token_type": "Bearer",
541                })),
542            )
543        }
544
545        async fn repo(State(state): State<TestState>, headers: HeaderMap) -> &'static str {
546            let authorization = headers
547                .get(reqwest::header::AUTHORIZATION)
548                .and_then(|value| value.to_str().ok())
549                .map(ToOwned::to_owned);
550            state.seen_authorization.lock().unwrap().push(authorization);
551            "ok"
552        }
553
554        for (audience, source_host) in [
555            (None, None),
556            (Some("https://audit.example"), Some("issuer.example")),
557        ] {
558            let state = TestState {
559                audience,
560                refresh_count: Arc::new(AtomicUsize::new(0)),
561                seen_authorization: Arc::new(Mutex::new(Vec::new())),
562            };
563            let router = Router::new()
564                .route("/token", post(token))
565                .route("/repo", post(repo))
566                .with_state(state.clone());
567            let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
568            let addr = listener.local_addr().unwrap();
569            tokio::spawn(async move { axum::serve(listener, router).await.unwrap() });
570
571            let host = source_host.unwrap_or("127.0.0.1").to_owned();
572            let mut storage = AuthenticationStorage::empty();
573            storage.add_backend(Arc::new(MemoryStorage::new()));
574            storage
575                .store(
576                    &host,
577                    &Authentication::OAuth {
578                        audience: audience.map(str::to_owned),
579                        access_token: "expired-access-token".to_string(),
580                        refresh_token: Some("refresh-token".to_string()),
581                        expires_at: Some(0),
582                        token_endpoint: format!("http://{addr}/token"),
583                        revocation_endpoint: None,
584                        client_id: "client-id".to_string(),
585                    },
586                )
587                .unwrap();
588
589            let repo_url = format!("http://{addr}/repo");
590            let second_listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
591            let second_url = format!("http://{}/repo", second_listener.local_addr().unwrap());
592            let second_router = Router::new()
593                .route("/repo", post(repo))
594                .with_state(state.clone());
595            tokio::spawn(async move { axum::serve(second_listener, second_router).await.unwrap() });
596            let mut middleware = AuthenticationMiddleware::from_auth_storage(storage.clone());
597            if let Some(source_host) = source_host {
598                let host = url::Host::parse(source_host).unwrap();
599                middleware = middleware.with_credentials_from(
600                    host,
601                    [
602                        Url::parse(&repo_url).unwrap().origin(),
603                        Url::parse(&second_url).unwrap().origin(),
604                    ],
605                );
606            }
607            let client = reqwest_middleware::ClientBuilder::new(
608                reqwest::Client::builder()
609                    .redirect(reqwest::redirect::Policy::none())
610                    .build()
611                    .unwrap(),
612            )
613            .with(middleware)
614            .build();
615
616            let responses = join_all((0..8).map(|index| {
617                let url = if source_host.is_some() && index % 2 == 1 {
618                    &second_url
619                } else {
620                    &repo_url
621                };
622                client.post(url).send()
623            }))
624            .await;
625            for response in responses {
626                assert_eq!(response.unwrap().status(), StatusCode::OK);
627            }
628
629            assert_eq!(state.refresh_count.load(Ordering::SeqCst), 1);
630            assert!(
631                matches!(storage.get(&host).unwrap(), Some(Authentication::OAuth { audience: saved, refresh_token: Some(refresh), .. }) if saved.as_deref() == audience && refresh == "rotated-refresh-token")
632            );
633            let seen_authorization = state.seen_authorization.lock().unwrap();
634            assert_eq!(seen_authorization.len(), 8);
635            assert!(
636                seen_authorization
637                    .iter()
638                    .all(|auth| { auth.as_deref() == Some("Bearer fresh-access-token") })
639            );
640        }
641    }

685                    audience: None,

687                    refresh_token: Some("refresh-token".to_string()),
688                    expires_at: Some(0),
689                    token_endpoint: format!("http://{addr}/token"),
690                    revocation_endpoint: None,
691                    client_id: "client-id".to_string(),
692                },
693            )
694            .unwrap();
695

697            .with(AuthenticationMiddleware::from_auth_storage(storage))

882    async fn reused_host_credential_is_sent_only_to_trusted_origin() {
883        const API: &str = "https://api.basilisk.example";
884        let api_origin = Url::parse(API).unwrap().origin();
885        let mut storage = AuthenticationStorage::empty();
886        storage.add_backend(Arc::new(MemoryStorage::new()));
887        let grant = Authentication::OAuth {
888            audience: Some(API.into()),
889            access_token: "fixture.host.token".into(),
890            refresh_token: None,
891            expires_at: Some(i64::MAX),
892            token_endpoint: "https://issuer.example/token".into(),
893            revocation_endpoint: None,
894            client_id: "rattler".into(),
895        };
896        storage.store("issuer.example", &grant).unwrap();
897        storage
898            .store(
899                "*.basilisk.example",
900                &Authentication::BearerToken("fixture-wildcard".into()),
901            )
902            .unwrap();
903
904        let header = |request: &Request| {
905            request
906                .headers()
907                .get("authorization")
908                .map(|v| v.to_str().unwrap().to_owned())
909        };
910
911        // Default host and wildcard lookup.
912        let (http, mut captured) =
913            make_client_harness(AuthenticationMiddleware::from_auth_storage(storage.clone()));
914        let _ = http.post("https://issuer.example/channel").send().await;
915        assert_eq!(
916            header(&captured.recv().await.unwrap()).as_deref(),
917            Some("Bearer fixture.host.token")
918        );
919        let _ = http.post(API).send().await;
920        assert_eq!(
921            header(&captured.recv().await.unwrap()).as_deref(),
922            Some("Bearer fixture-wildcard")
923        );
924
925        // Explicit origin mapping.
926        let mapped = AuthenticationMiddleware::from_auth_storage(storage.clone())
927            .with_credentials_from(
928                url::Host::parse("issuer.example").unwrap(),
929                [api_origin.clone()],
930            );
931        for (url, expected) in [
932            (format!("{API}/v1/audit"), Some("Bearer fixture.host.token")),
933            ("https://issuer.example/channel".to_string(), None),
934            ("https://api.basilisk.example:444/".to_string(), None),
935            ("http://api.basilisk.example/".to_string(), None),
936        ] {
937            let (http, mut captured) = make_client_harness(mapped.clone());
938            let _ = http.post(&url).send().await;
939            assert_eq!(
940                header(&captured.recv().await.unwrap()).as_deref(),
941                expected,
942                "{url}"
943            );
944        }
945
946        storage
947            .store(
948                "conda.example",
949                &Authentication::CondaToken("fixture-conda".into()),
950            )
951            .unwrap();
952        let (http, mut captured) = make_client_harness(
953            AuthenticationMiddleware::from_auth_storage(storage.clone()).with_credentials_from(
954                url::Host::parse("conda.example").unwrap(),
955                [api_origin.clone()],
956            ),
957        );
958        let _ = http.post(format!("{API}/v1/audit")).send().await;
959        let request = captured.recv().await.unwrap();
960        assert_eq!(header(&request), None);
961        assert_eq!(request.url().path(), "/v1/audit");
962
963        let (http, mut captured) = make_client_harness(
964            AuthenticationMiddleware::from_auth_storage(storage)
965                .with_credentials_from(url::Host::parse("missing.example").unwrap(), [api_origin]),
966        );
967        let _ = http.post(API).send().await;
968        assert_eq!(header(&captured.recv().await.unwrap()), None);
969    }
