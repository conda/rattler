1//! Convert the proxy settings of the shared rattler configuration into
2//! [`reqwest::Proxy`] instances.
3
4use rattler_config::config::proxy::ProxyConfig;
5use reqwest::NoProxy;
6
7/// Build the [`reqwest::Proxy`] instances described by a [`ProxyConfig`],
8/// honoring its `non-proxy-hosts` exclusion list.
9///
10/// Returns an empty vector when no proxy is configured. Note that
11/// [`ProxyConfig::default`] is empty; combine with
12/// [`ProxyConfig::from_env`] first if the environment variables should be
13/// taken into account (reqwest also applies proxy environment variables
14/// itself unless [`reqwest::ClientBuilder::no_proxy`] is used).
15pub fn proxies_from_config(config: &ProxyConfig) -> Result<Vec<reqwest::Proxy>, reqwest::Error> {
16    let mut proxies = Vec::new();
17    if config.is_default() {
18        return Ok(proxies);
19    }
20
21    let no_proxy = (!config.non_proxy_hosts.is_empty())
22        .then(|| NoProxy::from_string(&config.non_proxy_hosts.join(",")));
23
24    match (&config.http, &config.https) {
25        (Some(http), Some(https)) if http == https => {
26            proxies.push(reqwest::Proxy::all(http.clone())?.no_proxy(no_proxy.flatten()));
27        }
28        (http, https) => {
29            if let Some(http) = http {
30                proxies
31                    .push(reqwest::Proxy::http(http.clone())?.no_proxy(no_proxy.clone().flatten()));
32            }
33            if let Some(https) = https {
34                proxies.push(reqwest::Proxy::https(https.clone())?.no_proxy(no_proxy.flatten()));
35            }
36        }
37    }
38
39    Ok(proxies)
40}

42#[cfg(test)]
43mod tests {
44    use super::*;
45
46    #[test]
47    fn empty_config_yields_no_proxies() {
48        let proxies = proxies_from_config(&ProxyConfig::default()).unwrap();
49        assert!(proxies.is_empty());
50    }
51
52    #[test]
53    fn same_url_for_http_and_https_yields_single_proxy() {
54        let url = url::Url::parse("http://proxy.example.com:8080").unwrap();
55        let config = ProxyConfig {
56            http: Some(url.clone()),
57            https: Some(url),
58            non_proxy_hosts: vec!["localhost".to_string()],
59        };
60        let proxies = proxies_from_config(&config).unwrap();
61        assert_eq!(proxies.len(), 1);
62    }
63
64    #[test]
65    fn distinct_urls_yield_two_proxies() {
66        let config = ProxyConfig {
67            http: Some(url::Url::parse("http://proxy.example.com:8080").unwrap()),
68            https: Some(url::Url::parse("http://secure-proxy.example.com:8080").unwrap()),
69            non_proxy_hosts: vec![],
70        };
71        let proxies = proxies_from_config(&config).unwrap();
72        assert_eq!(proxies.len(), 2);
73    }
74}
