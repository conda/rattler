1//! Exchange of a CI provider's OIDC ID token for a bearer token.
2//!
3//! 1. Ask `ambient-id` for an OIDC ID token with the configured `audience`
4//!    claim (`None` outside supported CI providers).
5//! 2. Exchange it at the server (see [`ExchangeProtocol`]) for a
6//!    short-lived bearer token.
7//!
8//! Used by `rattler auth login --workload-identity` and, through
9//! [`crate::trusted_publishing`], for trusted publishing to prefix.dev.
10

12use reqwest_middleware::ClientWithMiddleware;
13use serde::{Deserialize, Serialize};
14use thiserror::Error;
15use url::Url;

19/// Default path of the prefix.dev mint endpoint.
20pub(crate) const DEFAULT_MINT_PATH: &str = "/api/oidc/mint_token";
21
22/// Well-known path of OAuth 2.0 Authorization Server Metadata (RFC 8414 §3).
23const AUTHORIZATION_SERVER_METADATA_PATH: &str = "/.well-known/oauth-authorization-server";
24
25/// The RFC 8693 token exchange grant type.
26const TOKEN_EXCHANGE_GRANT_TYPE: &str = "urn:ietf:params:oauth:grant-type:token-exchange";
27
28/// How the CI provider's OIDC ID token is exchanged for a bearer token.
29#[derive(Debug, Clone, PartialEq, Eq)]
30pub enum ExchangeProtocol {
31    /// OAuth 2.0 Token Exchange (RFC 8693) at `endpoint`.
32    TokenExchange {
33        /// The token endpoint, joined onto the server URL with [`Url::join`]:
34        /// either an absolute URL or a path starting with `/`. RFC 8693
35        /// doesn't define a path, so it must be configured.
36        endpoint: String,
37    },
38    /// The prefix.dev convention: `POST {"token": "<id token>"}` to `path`;
39    /// the response body is the bearer token. This is prefix.dev's own API,
40    /// not based on any standard.
41    ///
42    /// `path` is joined onto the server URL with [`Url::join`]; it must
43    /// start with `/` or it would resolve relative to the URL's path.
44    /// [`crate::trusted_publishing::TrustedPublishingFlow::new`] normalizes
45    /// a missing leading slash.
46    PrefixMint {
47        /// Path on the server where the ID token is exchanged.
48        path: String,
49    },
50}
51
52/// Which audience to request the ID token for and how to exchange it.
53///
54/// On GitLab CI the runner must populate the OIDC ID token under an env
55/// var that `ambient-id` derives from [`audience`](Self::audience)
56/// (uppercased, non-alphanumerics to `_`, suffixed `_ID_TOKEN`; audience
57/// `prefix.dev` resolves to `PREFIX_DEV_ID_TOKEN`). Set it via the
58/// `id_tokens` block in `.gitlab-ci.yml`.
59#[derive(Debug, Clone)]
60pub struct OidcExchangeOptions {
61    /// The `aud` claim requested in the OIDC ID token. The server validates
62    /// this against its trust configuration before issuing a token.
63    pub audience: String,
64    /// How the ID token is exchanged for a bearer token.
65    pub exchange: ExchangeProtocol,
66}
67
68impl OidcExchangeOptions {
69    /// An RFC 8693 token exchange at `endpoint` (see
70    /// [`ExchangeProtocol::TokenExchange`]).
71    pub fn token_exchange(audience: impl Into<String>, endpoint: impl Into<String>) -> Self {
72        Self {
73            audience: audience.into(),
74            exchange: ExchangeProtocol::TokenExchange {
75                endpoint: endpoint.into(),
76            },
77        }
78    }
79
80    /// prefix.dev's mint endpoint on `server`. prefix.dev deployments
81    /// (`prefix.dev` and `*.prefix.dev`) validate ID tokens against the
82    /// shared audience `prefix.dev`; other servers following the prefix.dev
83    /// convention use their host name, scoping each ID token to the server
84    /// it is sent to. Falls back to `prefix.dev` when `server` has no host.
85    ///
86    /// Does not validate scheme or host; callers handling ambient CI
87    /// credentials must enforce `https` and an allow-list themselves.
88    pub fn prefix_dev(server: &Url) -> Self {
89        let audience = match server.host_str() {
90            Some(host) if !is_prefix_dev_host(host) => host.to_string(),
91            _ => "prefix.dev".to_string(),
92        };
93        Self {
94            audience,
95            exchange: ExchangeProtocol::PrefixMint {
96                path: DEFAULT_MINT_PATH.to_string(),
97            },
98        }
99    }
100}
101
102/// Returns `true` for `prefix.dev` and any true subdomain (`*.prefix.dev`).
103/// Lookalikes (`evil-prefix.dev`, `prefix.dev.evil.com`) and trailing-dot
104/// hosts (`beta.prefix.dev.`, preserved as-is by [`Url`]) fail closed.
105pub(crate) fn is_prefix_dev_host(host: &str) -> bool {
106    host == "prefix.dev" || host.ends_with(".prefix.dev")
107}

109/// The fields of OAuth 2.0 Authorization Server Metadata (RFC 8414 §2) we use.
110#[derive(Deserialize)]
111struct AuthorizationServerMetadata {
112    issuer: String,
113    token_endpoint: Option<String>,
114    grant_types_supported: Option<Vec<String>>,
115}
116
117/// The metadata URL for `issuer`: the well-known path is inserted between
118/// the host and the issuer's path, if any (RFC 8414 §3.1).
119fn authorization_server_metadata_url(issuer: &Url) -> Url {
120    let mut url = issuer.clone();
121    let issuer_path = issuer.path().trim_end_matches('/');
122    url.set_path(&format!(
123        "{AUTHORIZATION_SERVER_METADATA_PATH}{issuer_path}"
124    ));
125    url.set_query(None);
126    url.set_fragment(None);
127    url
128}
129
130/// Looks up the token endpoint of `issuer` in its OAuth 2.0 Authorization
131/// Server Metadata (RFC 8414).
132///
133/// Returns `None` when the server publishes no metadata, or when the
134/// metadata can't be used: it's for a different issuer (which RFC 8414 §3.3
135/// forbids using), has no `token_endpoint`, or lists `grant_types_supported`
136/// without the token exchange grant.
137pub async fn discover_token_endpoint(
138    client: &ClientWithMiddleware,
139    issuer: &Url,
140) -> Result<Option<Url>, OidcExchangeError> {
141    let metadata_url = authorization_server_metadata_url(issuer);
142    let response = client
143        .get(metadata_url.clone())
144        .send()
145        .await
146        .map_err(|err| OidcExchangeError::ReqwestMiddleware(metadata_url.clone(), err))?;
147    if !response.status().is_success() {
148        tracing::debug!(
149            "No authorization server metadata at {metadata_url} (HTTP {})",
150            response.status()
151        );
152        return Ok(None);
153    }
154    let body = response
155        .bytes()
156        .await
157        .map_err(|err| OidcExchangeError::Reqwest(metadata_url.clone(), err))?;
158    let Ok(metadata) = serde_json::from_slice::<AuthorizationServerMetadata>(&body) else {
159        tracing::debug!("Invalid authorization server metadata at {metadata_url}");
160        return Ok(None);
161    };
162
163    // RFC 8414 §3.3: the `issuer` in the metadata must be identical to the
164    // issuer the metadata was requested for, or it must not be used.
165    // Compared without a trailing slash, which `Url` always adds to an
166    // empty path.
167    if metadata.issuer.trim_end_matches('/') != issuer.as_str().trim_end_matches('/') {
168        tracing::warn!(
169            "Ignoring authorization server metadata at {metadata_url}: it is for issuer `{}`",
170            metadata.issuer
171        );
172        return Ok(None);
173    }
174    // RFC 8414 §2: when `grant_types_supported` is omitted, the default
175    // doesn't include token exchange. We still try the token endpoint then,
176    // since many servers don't list every grant they support.
177    if let Some(grant_types) = &metadata.grant_types_supported
178        && !grant_types.iter().any(|g| g == TOKEN_EXCHANGE_GRANT_TYPE)
179    {
180        tracing::debug!("{metadata_url} doesn't list the token exchange grant type");
181        return Ok(None);
182    }
183    Ok(metadata
184        .token_endpoint
185        .and_then(|endpoint| Url::parse(&endpoint).ok()))
186}

188/// Errors that can occur while exchanging an OIDC ID token.
189#[derive(Debug, Error)]
190pub enum OidcExchangeError {
191    /// Failed to parse a URL.
192    #[error(transparent)]
193    Url(#[from] url::ParseError),
194    /// HTTP request failed at the reqwest layer.
195    #[error("Failed to fetch: `{0}`")]
196    Reqwest(Url, #[source] reqwest::Error),
197    /// HTTP request failed at the reqwest-middleware layer.
198    #[error("Failed to fetch: `{0}`")]
199    ReqwestMiddleware(Url, #[source] reqwest_middleware::Error),
200    /// The prefix.dev mint endpoint returned an error.
201    #[error(
202        "Server returned error code {0} from the mint endpoint, is trusted publishing correctly configured?\nResponse: {1}"
203    )]
204    MintToken(StatusCode, String),
205    /// The token exchange endpoint returned an error.
206    #[error(
207        "Server returned error code {0} from the OIDC token exchange, is the server's OIDC trust configuration (issuer, audience) correct?\nResponse: {1}"
208    )]
209    TokenExchange(StatusCode, String),
210    /// The token exchange succeeded but the response had no usable
211    /// `access_token`.
212    #[error("The OIDC token exchange response did not contain an `access_token`")]
213    MissingAccessToken,
214    /// Retrieving the OIDC ID token from the CI provider failed.
215    #[error("Failed to retrieve an OIDC ID token from the CI provider")]
216    OidcToken(#[from] ambient_id::Error),
217}
218
219/// Returns the short-lived token to use against `server_url`, or `None` when
220/// `ambient-id` reports no usable CI provider.
221///
222/// Delegates OIDC ID-token retrieval to `ambient-id`; this function owns the
223/// exchange with `server_url`.
224pub async fn get_token(
225    client: &ClientWithMiddleware,
226    server_url: &Url,
227    options: &OidcExchangeOptions,
228) -> Result<Option<BearerToken>, OidcExchangeError> {
229    let detector = ambient_id::Detector::new_with_client(client.clone());
230    let Some(oidc_token) = detector.detect(&options.audience).await? else {
231        return Ok(None);
232    };
233
234    let token = match &options.exchange {
235        ExchangeProtocol::TokenExchange { endpoint } => {
236            token_exchange(oidc_token.reveal(), server_url, endpoint, client).await?
237        }
238        ExchangeProtocol::PrefixMint { path } => {
239            prefix_mint(oidc_token.reveal(), server_url, path, client).await?
240        }
241    };
242
243    tracing::info!("Exchanged the OIDC token from the CI provider for a bearer token");
244
245    Ok(Some(token))
246}

248/// The body sent to the prefix.dev mint endpoint.
249#[derive(Serialize)]
250struct MintTokenRequest<'a> {
251    token: &'a str,
252}
253
254async fn prefix_mint(
255    oidc_token: &str,
256    server_url: &Url,
257    mint_path: &str,
258    client: &ClientWithMiddleware,
259) -> Result<BearerToken, OidcExchangeError> {
260    let mint_token_url = server_url.join(mint_path)?;
261    tracing::info!("Querying the trusted publishing token from {mint_token_url}");
262
263    let response = client
264        .post(mint_token_url.clone())
265        .json(&MintTokenRequest { token: oidc_token })
266        .send()
267        .await
268        .map_err(|err| OidcExchangeError::ReqwestMiddleware(mint_token_url.clone(), err))?;
269
270    let status = response.status();
271    let body = response
272        .bytes()
273        .await
274        .map_err(|err| OidcExchangeError::Reqwest(mint_token_url.clone(), err))?;
275
276    if status.is_success() {
277        Ok(BearerToken::new(String::from_utf8_lossy(&body).to_string()))
278    } else {
279        Err(OidcExchangeError::MintToken(
280            status,
281            String::from_utf8_lossy(&body).to_string(),
282        ))
283    }
284}

286/// The part of the RFC 8693 token exchange response we use.
287///
288/// RFC 8693 §2.2.1 also requires `issued_token_type` and `token_type`. We
289/// only need `access_token`, so we don't fail if a server leaves them out.
290#[derive(Deserialize)]
291struct TokenExchangeResponse {
292    access_token: Option<String>,
293}
294
295async fn token_exchange(
296    oidc_token: &str,
297    server_url: &Url,
298    endpoint: &str,
299    client: &ClientWithMiddleware,
300) -> Result<BearerToken, OidcExchangeError> {
301    let exchange_url = server_url.join(endpoint)?;
302    tracing::info!("Exchanging the OIDC token at {exchange_url}");
303
304    // RFC 8693 §2.1: a form-encoded request. No client authentication: the
305    // ID token identifies the caller, and the RFC leaves client
306    // authentication to the server.
307    let body = url::form_urlencoded::Serializer::new(String::new())
308        .append_pair("grant_type", TOKEN_EXCHANGE_GRANT_TYPE)
309        .append_pair(
310            "subject_token_type",
311            "urn:ietf:params:oauth:token-type:id_token",
312        )
313        .append_pair("subject_token", oidc_token)
314        .finish();
315    let request = client
316        .post(exchange_url.clone())
317        .header(CONTENT_TYPE, "application/x-www-form-urlencoded")
318        .body(body);
319    let response = request
320        .send()
321        .await
322        .map_err(|err| OidcExchangeError::ReqwestMiddleware(exchange_url.clone(), err))?;
323
324    let status = response.status();
325    let body = response
326        .bytes()
327        .await
328        .map_err(|err| OidcExchangeError::Reqwest(exchange_url.clone(), err))?;
329
330    if !status.is_success() {
331        return Err(OidcExchangeError::TokenExchange(
332            status,
333            String::from_utf8_lossy(&body).to_string(),
334        ));
335    }
336
337    // Don't echo the body on a malformed success response: it may contain
338    // the issued token.
339    let token = serde_json::from_slice::<TokenExchangeResponse>(&body)
340        .ok()
341        .and_then(|response| response.access_token)
342        .filter(|token| !token.is_empty())
343        .ok_or(OidcExchangeError::MissingAccessToken)?;
344    Ok(BearerToken::new(token))
345}

363    /// A captured request: its `content-type` and raw body.
364    type Captured = Arc<Mutex<Option<(String, String)>>>;
365
366    /// Serves a token endpoint at `path` that captures the request and
367    /// answers with `response`.
368    async fn token_server(
369        path: &'static str,
370        response: (StatusCode, &'static str),
371    ) -> (Url, Captured) {
372        let seen: Captured = Arc::default();
373        let seen_in_handler = seen.clone();
374        let router = axum::Router::new().route(
375            path,
376            post(move |headers: HeaderMap, body: Bytes| {
377                let seen = seen_in_handler.clone();
378                async move {
379                    let content_type = headers
380                        .get("content-type")
381                        .and_then(|v| v.to_str().ok())
382                        .unwrap_or_default()
383                        .to_string();
384                    *seen.lock().unwrap() =
385                        Some((content_type, String::from_utf8_lossy(&body).to_string()));
386                    response
387                }
388            }),
389        );
390        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
391        let addr = listener.local_addr().unwrap();
392        tokio::spawn(async move { axum::serve(listener, router).await.unwrap() });
393        (Url::parse(&format!("http://{addr}")).unwrap(), seen)
394    }

480    /// Serves `metadata` as the RFC 8414 metadata of the issuer at the
481    /// server's root; `{issuer}` in it is replaced with the server's URL.
482    async fn metadata_server(metadata: &'static str) -> Url {
483        use axum::routing::get;
484
485        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
486        let addr = listener.local_addr().unwrap();
487        let metadata = metadata.replace("{issuer}", &format!("http://{addr}"));
488        let router = axum::Router::new().route(
489            "/.well-known/oauth-authorization-server",
490            get(move || async move { metadata }),
491        );
492        tokio::spawn(async move { axum::serve(listener, router).await.unwrap() });
493        Url::parse(&format!("http://{addr}")).unwrap()
494    }

569    #[test]
570    fn prefix_dev_audience() {
571        let audience =
572            |url: &str| OidcExchangeOptions::prefix_dev(&Url::parse(url).unwrap()).audience;
573
574        // prefix.dev deployments share the audience "prefix.dev"
575        assert_eq!(audience("https://prefix.dev"), "prefix.dev");
576        assert_eq!(
577            audience("https://beta.prefix.dev/some-channel"),
578            "prefix.dev"
579        );
580        // other hosts use their normalized host name
581        assert_eq!(
582            audience("https://conda.example.com/channel"),
583            "conda.example.com"
584        );
585        assert_eq!(
586            audience("https://Conda.EXAMPLE.com:443/channel"),
587            "conda.example.com"
588        );
589        // lookalike hosts are not part of the family
590        assert_eq!(
591            audience("https://evil-prefix.dev/channel"),
592            "evil-prefix.dev"
593        );
594        // no host falls back to prefix.dev
595        assert_eq!(audience("data:text/plain,hello"), "prefix.dev");
596
597        assert_eq!(
598            OidcExchangeOptions::prefix_dev(&Url::parse("https://prefix.dev").unwrap()).exchange,
599            ExchangeProtocol::PrefixMint {
600                path: "/api/oidc/mint_token".to_string()
