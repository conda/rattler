18impl LockFile {
19    /// Serialize validated CEP-37 v1 YAML without modifying this model.
20    ///
21    /// Packages and target platforms are sorted; maps use lexical key order.
22    /// Channel priority and source order are preserved. Checksums are lowercase,
23    /// absent checksums are omitted, and channel entries are always objects.
24    pub fn to_yaml(&self) -> Result<String, Error> {
25        self.validate()?;
26        let packages = sorted_packages(self);
27        let mut platforms: Vec<_> = self.metadata.platforms.iter().map(String::as_str).collect();
28        platforms.sort_unstable();
29        let document = CanonicalLock {
30            version: 1,
31            metadata: CanonicalMetadata::new(&self.metadata, &platforms),
32            package: PackageList(&packages),
33        };
34        serde_saphyr::to_string(&document).map_err(|error| {
35            Error::new(
36                NodePath::root(),
37                ErrorKind::Serialize {
38                    message: error.to_string(),
39                },
40            )
41        })
42    }
43
44    /// Validate and serialize before opening the destination, so invalid models
45    /// never truncate an existing lockfile. Writing is not an atomic replacement.
46    ///
47    /// ```no_run
48    /// # use rattler_conda_lock::LockFile;
49    /// LockFile::default().to_path("conda-lock.yml")?;
50    /// # Ok::<(), rattler_conda_lock::Error>(())
51    /// ```
52    pub fn to_path(&self, path: impl AsRef<Path>) -> Result<(), Error> {
53        let yaml = self.to_yaml()?;
54        let path = path.as_ref();
55        std::fs::write(path, yaml).map_err(|error| {
56            Error::new(
57                NodePath::root(),
58                ErrorKind::Write {
59                    path: path.to_owned(),
60                    error: Arc::new(error),
61                },
62            )
63        })
64    }

66    /// Compute package-derived SHA256 hashes for each declared target platform.
67    ///
68    /// **These are not conda-lock's input-spec hashes.** Existing metadata hashes
69    /// are neither validated nor included. This method does not validate the
70    /// model.
71    ///
72    /// The preimage is the UTF-8 JSON that Python's
73    /// `json.dumps(packages, sort_keys=True)` produces for the platform's
74    /// packages, so an empty platform hashes `[]`. Each package object holds
75    /// `build` (omitted if absent), `category`, `dependencies`, `hash`,
76    /// `manager`, `name`, `optional`, `platform`, `source` (omitted if absent),
77    /// `url` and `version`. Hash fields are `md5` then `sha256`, omitted if
78    /// absent and ASCII-lowercased; source fields are `type` (always `url`) then
79    /// `url`. Every other string, including names, URLs and dependency
80    /// constraints, is hashed unchanged.
81    ///
82    /// Packages sort by `(platform, manager, name, category)` (`conda` before
83    /// `pip`). Ties, possible only for invalid duplicate identities, sort by the
84    /// model's field order: name, version, manager, platform, dependencies, URL,
85    /// hashes (MD5 then SHA256), source URL, build, category, optional. Strings
86    /// and maps compare lexically, absent options precede present ones, and false
87    /// precedes true. Thus input package order never affects the result.
88    ///
89    /// ```
90    /// use rattler_conda_lock::LockFile;
91    /// let mut lock = LockFile::default();
92    /// lock.metadata.platforms.push("linux-64".into());
93    /// lock.metadata.content_hash = lock.compute_content_hashes();
94    /// lock.validate()?;
95    /// assert_eq!(lock.metadata.content_hash["linux-64"],
96    ///     "4f53cda18c2baa0c0354bb5f9a3ecbe5ed12ab4d8e11ba873c2f11161202b945");
97    /// # Ok::<(), rattler_conda_lock::Error>(())
98    /// ```
99    pub fn compute_content_hashes(&self) -> BTreeMap<String, String> {
100        let packages = sorted_packages(self);
101        self.metadata
102            .platforms
103            .iter()
104            .map(|platform| {
105                let start = packages
106                    .partition_point(|package| package.platform.as_str() < platform.as_str());
107                let end = packages
108                    .partition_point(|package| package.platform.as_str() <= platform.as_str());
109                let mut serializer = serde_json::Serializer::with_formatter(
110                    HashingWriter::<_, Sha256>::new(std::io::sink()),
111                    PythonFormatter::default(),
112                );
113                // Only strings, booleans, maps and arrays reach an infallible
114                // sink; no user-provided serializer is called.
115                PackageList(&packages[start..end])
116                    .serialize(&mut serializer)
117                    .expect("canonical package JSON is infallible");
118                let (_, digest) = serializer.into_inner().finalize();
119                (platform.clone(), hex::encode(digest))
120            })
121            .collect()
122    }
123}

125fn sorted_packages(lock: &LockFile) -> Vec<&Package> {
126    let mut packages: Vec<_> = lock.package.iter().collect();
127    packages.sort_unstable_by(|a, b| a.identity().cmp(&b.identity()).then_with(|| a.cmp(b)));
128    packages
129}

169struct ContentHashes<'a>(&'a BTreeMap<String, String>);
170
171impl Serialize for ContentHashes<'_> {
172    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
173        let mut map = serializer.serialize_map(Some(self.0.len()))?;
174        for (platform, hash) in self.0 {
175            map.serialize_entry(platform, &lowercase(hash))?;
176        }
177        map.end()
178    }
179}
180
181struct InputHashes<'a>(&'a BTreeMap<String, Hashes>);
182
183impl Serialize for InputHashes<'_> {
184    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
185        let mut map = serializer.serialize_map(Some(self.0.len()))?;
186        for (path, hashes) in self.0 {
187            map.serialize_entry(path, &CanonicalHashes::new(hashes))?;
188        }
189        map.end()
190    }
191}

193struct PackageList<'a>(&'a [&'a Package]);
194
195impl Serialize for PackageList<'_> {
196    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
197        let mut sequence = serializer.serialize_seq(Some(self.0.len()))?;
198        for package in self.0 {
199            sequence.serialize_element(&CanonicalPackage::new(package))?;
200        }
201        sequence.end()
202    }
203}

240#[derive(Serialize)]
241struct CanonicalHashes<'a> {
242    #[serde(skip_serializing_if = "Option::is_none")]
243    md5: Option<Cow<'a, str>>,
244    #[serde(skip_serializing_if = "Option::is_none")]
245    sha256: Option<Cow<'a, str>>,
246}
247
248impl<'a> CanonicalHashes<'a> {
249    fn new(hashes: &'a Hashes) -> Self {
250        Self {
251            md5: hashes.md5.as_deref().map(lowercase),
252            sha256: hashes.sha256.as_deref().map(lowercase),
253        }
254    }
255}
256
257fn lowercase(value: &str) -> Cow<'_, str> {
258    if value.bytes().any(|byte| byte.is_ascii_uppercase()) {
259        Cow::Owned(value.to_ascii_lowercase())
260    } else {
261        Cow::Borrowed(value)
262    }
263}

291    #[test]
292    fn hashes_ignore_provenance_but_cover_package_semantics() {
293        let mut lock = lock();
294        let original = lock.compute_content_hashes();
295        lock.metadata.sources.push("environment.yml".into());
296        lock.metadata
297            .content_hash
298            .insert("linux-64".into(), "not yet a digest".into());
299        assert_eq!(original, lock.compute_content_hashes());
300        lock.package[0].optional = true;
301        let changed = lock.compute_content_hashes();
302        assert_ne!(original["linux-64"], changed["linux-64"]);
303        assert_eq!(original["win-64"], changed["win-64"]);
304    }
305
306    #[test]
307    fn order_and_checksum_case_do_not_change_output() {
308        let mut lock = lock();
309        let mut other = lock.package[0].clone();
310        other.name = "another".into();
311        lock.package.push(other);
312        lock.metadata.content_hash = lock.compute_content_hashes();
313        let yaml = lock.to_yaml().unwrap();
314        let hashes = lock.compute_content_hashes();
315        lock.package.reverse();
316        lock.metadata.platforms.reverse();
317        for package in &mut lock.package {
318            package.hash.sha256 = package
319                .hash
320                .sha256
321                .take()
322                .map(|value| value.to_ascii_lowercase());
323        }
324        assert_eq!(yaml, lock.to_yaml().unwrap());
325        assert_eq!(hashes, lock.compute_content_hashes());
326        assert!(!yaml.contains("md5:"));
327        let parsed = yaml.parse::<LockFile>().unwrap();
328        assert_eq!(parsed.to_yaml().unwrap(), yaml);
329        assert_eq!(
330            parsed.package[0].hash.sha256.as_deref(),
331            Some("abababababababababababababababababababababababababababababababab")
332        );
333    }
