8/// A path to a node in the lock-file model, such as `package[3].hash.sha256`.
9///
10/// Paths are built from [`NodePath::root`] with [`NodePath::field`] and
11/// [`NodePath::index`] so that diagnostics and the source map agree on one
12/// spelling. Field names are appended verbatim, including names that contain a
13/// `.`; [`crate::Document`] resolves those by exact lookup before shortening.
14#[derive(Clone, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
15pub struct NodePath(String);
16
17impl NodePath {
18    /// The path of the document itself.
19    pub fn root() -> Self {
20        Self(String::new())
21    }
22
23    /// The path of a child field, or of a mapping entry with key `name`.
24    #[must_use]
25    pub fn field(&self, name: &str) -> Self {
26        let mut path = String::with_capacity(self.0.len() + name.len() + 1);
27        path.push_str(&self.0);
28        if !self.0.is_empty() {
29            path.push('.');
30        }
31        path.push_str(name);
32        Self(path)
33    }
34
35    /// The path of the `index`th element of this sequence.
36    #[must_use]
37    pub fn index(&self, index: usize) -> Self {
38        Self(format!("{}[{index}]", self.0))
39    }
40
41    /// The dotted spelling of this path. Empty for the document itself.
42    pub fn as_str(&self) -> &str {
43        &self.0
44    }
45
46    /// Whether this path denotes the whole document.
47    pub fn is_root(&self) -> bool {
48        self.0.is_empty()
49    }
50}
51
52impl fmt::Display for NodePath {
53    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
54        f.write_str(&self.0)
55    }
56}
57
58impl fmt::Debug for NodePath {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        fmt::Debug::fmt(&self.0, f)
61    }
62}

64impl From<&str> for NodePath {
65    fn from(value: &str) -> Self {
66        Self(value.to_owned())
67    }
68}

71    fn from(value: String) -> Self {
72        Self(value)
73    }
74}

77    fn as_ref(&self) -> &str {
78        &self.0
79    }
80}

85    }

94/// The YAML shape a node was required to have.
95#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
96pub enum ValueKind {
97    /// A YAML mapping.
98    Mapping,
99    /// A YAML sequence.
100    Sequence,
101    /// A scalar readable as a string.
102    String,
103    /// A `true`/`false` scalar.
104    Boolean,
105}
106
107impl fmt::Display for ValueKind {
108    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
109        f.write_str(match self {
110            Self::Mapping => "a mapping",
111            Self::Sequence => "a sequence",
112            Self::String => "a string",
113            Self::Boolean => "a boolean",
114        })
115    }
116}

118/// What went wrong, independent of where it went wrong.
119///
120/// Match on this instead of on rendered messages: the `Display` text is
121/// documentation, the variants are the contract.
122#[derive(Clone, Debug, thiserror::Error)]
123#[non_exhaustive]
124pub enum ErrorKind {
125    /// The input is not well-formed YAML. The message comes from the parser.
126    #[error("{message}")]
127    Syntax {
128        /// The parser's description of the syntax error.
129        message: String,
130    },
131
132    /// The input holds more than one YAML document.
133    #[error("a lock file must contain exactly one YAML document")]
134    MultipleDocuments,
135
136    /// A mapping defines the same key twice.
137    #[error("duplicate mapping key {key:?}")]
138    DuplicateKey {
139        /// The repeated key.
140        key: String,
141    },
142
143    /// A node has a shape the schema does not allow here.
144    #[error("expected {0}")]
145    UnexpectedType(ValueKind),
146
147    /// A required field is absent.
148    #[error("missing required field {field:?}")]
149    MissingField {
150        /// The absent field.
151        field: &'static str,
152    },
153
154    /// A field is present that this schema version does not define. Unknown
155    /// fields are rejected because they may change what a lock file installs.
156    #[error("unknown field {field:?}")]
157    UnknownField {
158        /// The unexpected field.
159        field: String,
160    },
161
162    /// The `version` field names a schema this crate cannot read.
163    #[error("unsupported lock-file version; expected integer 1")]
164    UnsupportedVersion,
165
166    /// A package names an installer other than `conda` or `pip`.
167    #[error("unknown package manager {found:?}; expected \"conda\" or \"pip\"")]
168    UnknownManager {
169        /// The manager as written.
170        found: String,
171    },
172
173    /// A package source uses a type other than `url`.
174    #[error("unsupported package source type {found:?}; expected \"url\"")]
175    UnsupportedSourceType {
176        /// The source type as written.
177        found: String,
178    },
179
180    /// A declared target platform is not a CEP-26 subdir.
181    #[error("expected a CEP-26 target subdir (os-arch), excluding noarch")]
182    InvalidPlatform,
183
184    /// The same target platform is declared twice.
185    #[error("duplicate target platform")]
186    DuplicatePlatform,
187
188    /// A declared target platform has no `content_hash` entry.
189    #[error("missing content hash for target platform {platform}")]
190    MissingContentHash {
191        /// The target platform without a hash.
192        platform: String,
193    },
194
195    /// A `content_hash` entry names a platform that is not a declared target.
196    #[error("content hash refers to an undeclared target platform")]
197    UndeclaredContentHashPlatform,
198
199    /// A channel has an empty URL or name.
200    #[error("channel URL or name must not be empty")]
201    EmptyChannel,
202
203    /// A source is absolute, a URL, or otherwise not relative to the lock file.
204    #[error("source paths must be nonempty and relative to the lock file")]
205    InvalidSourcePath,
206
207    /// The same source path is declared twice.
208    #[error("duplicate source path")]
209    DuplicateSource,
210
211    /// `created_at` is not a whole-second UTC timestamp.
212    #[error("expected a UTC timestamp in YYYY-MM-DDTHH:MM:SSZ form")]
213    InvalidTimestamp,
214
215    /// An `inputs_metadata` entry names a source that is not declared.
216    #[error("input hash refers to an undeclared source")]
217    UndeclaredInputSource,
218
219    /// A declared source has no `inputs_metadata` entry, although the lock file
220    /// records hashes for its other sources.
221    #[error("missing input hashes for source {source_path}")]
222    MissingInputHashes {
223        /// The declared source that has no hashes.
224        source_path: String,
225    },
226
227    /// A package or dependency name is invalid for its installer.
228    #[error("invalid {manager} package name")]
229    InvalidPackageName {
230        /// The installer whose naming rules apply.
231        manager: Manager,
232    },
233
234    /// A resolved version is invalid for its installer.
235    #[error("invalid resolved {manager} package version")]
236    InvalidVersion {
237        /// The installer whose version syntax applies.
238        manager: Manager,
239    },
240
241    /// A package targets a platform that is not declared in the metadata.
242    #[error("package target is not declared in metadata.platforms")]
243    UndeclaredPackagePlatform,
244
245    /// A package has an empty install category.
246    #[error("package category must not be empty")]
247    EmptyCategory,
248
249    /// Two packages share one install identity, so the file does not describe a
250    /// single installable set.
251    #[error("duplicate (name, manager, platform, category) package identity")]
252    DuplicatePackageIdentity,
253
254    /// A URL is relative, empty, or contains whitespace.
255    #[error("expected an absolute URL without whitespace")]
256    InvalidUrl,
257
258    /// A checksum is not a hexadecimal digest of the expected length.
259    #[error("expected exactly {length} hexadecimal digits")]
260    InvalidDigest {
261        /// The digest length the algorithm requires.
262        length: usize,
263    },
264
265    /// A direct-source package records something other than a revision where an
266    /// artifact would record a digest.
267    #[error("expected up to 64 hexadecimal digits")]
268    InvalidRevision,
269
270    /// A conda build string contains characters conda does not allow.
271    #[error("invalid conda build string")]
272    InvalidBuildString,
273
274    /// A dependency constraint is invalid for its installer.
275    #[error("invalid {manager} dependency constraint")]
276    InvalidDependencyConstraint {
277        /// The installer whose constraint syntax applies.
278        manager: Manager,
279    },
280
281    /// The model could not be emitted as YAML.
282    #[error("could not serialize the lock file: {message}")]
283    Serialize {
284        /// The emitter's description of the failure.
285        message: String,
286    },
287
288    /// A lock file could not be read.
289    #[error("could not read {}: {error}", path.display())]
290    Read {
291        /// The path that was read.
292        path: PathBuf,
293        /// The underlying I/O error.
294        error: Arc<std::io::Error>,
295    },
296
297    /// A lock file could not be written.
298    #[error("could not write {}: {error}", path.display())]
299    Write {
300        /// The path that was written.
301        path: PathBuf,
302        /// The underlying I/O error.
303        error: Arc<std::io::Error>,
304    },
305}

307/// A primary or related location in a diagnostic.
308#[derive(Clone, Debug)]
309pub struct Label {
310    pub(crate) path: NodePath,
311    pub(crate) message: Option<&'static str>,
312    pub(crate) span: Option<Range<usize>>,
313}
314
315impl Label {
316    /// The model path this label points at.
317    pub fn path(&self) -> &NodePath {
318        &self.path
319    }
320
321    /// Why this location matters. Absent on the primary label, whose
322    /// explanation is the diagnostic's [`ErrorKind`].
323    pub fn message(&self) -> Option<&'static str> {
324        self.message
325    }
326
327    /// The UTF-8 byte range in the original source, if known.
328    pub fn span(&self) -> Option<Range<usize>> {
329        self.span.clone()
330    }
331}

333/// The locations a diagnostic applies to: a primary model path, any related
334/// paths, and the source text they resolve against once a document has been
335/// consulted.
336///
337/// This is the part of a diagnostic that does not depend on what went wrong,
338/// so conversion layers with their own error kind — such as
339/// `rattler_lock::conda_lock::CondaLockError` — can embed it directly instead
340/// of parameterizing over the kind; see [`crate::Document::locate`].
341#[derive(Clone, Debug)]
342pub struct Labels {
343    pub(crate) labels: Vec<Label>,
344    pub(crate) source: Option<Arc<str>>,
345    pub(crate) name: Option<Arc<str>>,
346    #[cfg(feature = "miette")]
347    pub(crate) diagnostic_source: Option<Arc<miette::NamedSource<Arc<str>>>>,
348}
349
350impl Labels {
351    /// Start from a primary model path, without source information.
352    pub fn new(path: impl Into<NodePath>) -> Self {
353        Self {
354            labels: vec![Label {
355                path: path.into(),
356                message: None,
357                span: None,
358            }],
359            source: None,
360            name: None,
361            #[cfg(feature = "miette")]
362            diagnostic_source: None,
363        }
364    }
365
366    /// Add a related model path, for example the first of two duplicate entries.
367    #[must_use]
368    pub fn with_related_path(mut self, path: impl Into<NodePath>, message: &'static str) -> Self {
369        self.labels.push(Label {
370            path: path.into(),
371            message: Some(message),
372            span: None,
373        });
374        self
375    }
376
377    /// The primary model path. The root path denotes the whole document.
378    pub fn path(&self) -> &NodePath {
379        &self.labels[0].path
380    }
381
382    /// Primary and related labels, the primary one first.
383    pub fn labels(&self) -> &[Label] {
384        &self.labels
385    }
386
387    /// The span of the primary label, if source context is available.
388    pub fn span(&self) -> Option<Range<usize>> {
389        self.labels[0].span.clone()
390    }
391
392    /// Original YAML text, if these labels have document context.
393    pub fn source_text(&self) -> Option<&str> {
394        self.source.as_deref()
395    }
396
397    /// Original file name, if the document was read from a path.
398    pub fn source_name(&self) -> Option<&str> {
399        self.name.as_deref()
400    }
401
402    /// Write the `file: path: ` prefix that precedes a diagnostic message, so
403    /// that every diagnostic carrying these labels renders the same way.
404    pub fn write_prefix(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
405        if let Some(name) = self.source_name() {
406            write!(f, "{name}: ")?;
407        }
408        if !self.path().is_root() {
409            write!(f, "{}: ", self.path())?;
410        }
411        Ok(())
412    }
413
414    #[cfg(feature = "miette")]
415    pub(crate) fn source_code(&self) -> Option<&dyn miette::SourceCode> {
416        self.diagnostic_source
417            .as_deref()
418            .map(|source| source as &dyn miette::SourceCode)
419    }
420
421    /// The labels as miette spans, with `primary` explaining the primary label.
422    #[cfg(feature = "miette")]
423    pub(crate) fn spans(
424        &self,
425        primary: &dyn fmt::Display,
426    ) -> Box<dyn Iterator<Item = miette::LabeledSpan> + '_> {
427        let primary = primary.to_string();
428        Box::new(self.labels.iter().filter_map(move |label| {
429            let message = label.message.map_or_else(|| primary.clone(), str::to_owned);
430            label
431                .span
432                .clone()
433                .map(|span| miette::LabeledSpan::new_with_span(Some(message), span))
434        }))
435    }
436
437    /// Render these labels as a [`Report`] under the given message, for a
438    /// caller whose own error kind does not implement `miette::Diagnostic`.
439    pub fn report(&self, message: impl fmt::Display) -> Report {
440        Report {
441            message: message.to_string(),
442            labels: self.clone(),
443        }
444    }
445
446    pub(crate) fn with_primary_span(mut self, span: Option<Range<usize>>) -> Self {
447        self.labels[0].span = span;
448        self
449    }
450
451    pub(crate) fn with_related_span(mut self, span: Option<Range<usize>>) -> Self {
452        if let Some(label) = self.labels.last_mut() {
453            label.span = span;
454        }
455        self
456    }
457
458    pub(crate) fn attach_source(&mut self, source: Arc<str>, name: Option<Arc<str>>) {
459        #[cfg(feature = "miette")]
460        {
461            self.diagnostic_source = Some(Arc::new(miette::NamedSource::new(
462                name.as_deref().unwrap_or("conda-lock.yml"),
463                source.clone(),
464            )));
465        }
466        self.source = Some(source);
467        self.name = name;
468    }
469}

471/// A validation or parse failure: an [`ErrorKind`] plus the locations it
472/// applies to.
473///
474/// Errors raised from a model carry model paths only. Parsing a
475/// [`crate::Document`], or passing a model error through
476/// [`crate::Document::locate`], adds the source text and byte spans.
477#[derive(Clone, Debug)]
478pub struct Error {
479    pub(crate) kind: ErrorKind,
480    pub(crate) labels: Labels,
481}
482
483impl Error {
484    /// Construct an error at a model path, without source information.
485    pub fn new(path: impl Into<NodePath>, kind: ErrorKind) -> Self {
486        Self {
487            kind,
488            labels: Labels::new(path),
489        }
490    }
491
492    /// Add a related model path, for example the first of two duplicate entries.
493    #[must_use]
494    pub fn with_related_path(mut self, path: impl Into<NodePath>, message: &'static str) -> Self {
495        self.labels = self.labels.with_related_path(path, message);
496        self
497    }
498
499    /// What went wrong.
500    pub fn kind(&self) -> &ErrorKind {
501        &self.kind
502    }
503
504    /// Split into the classification and its locations, for conversion layers
505    /// that reclassify this error under their own kind while keeping its
506    /// labels; see `rattler_lock::conda_lock::CondaLockError`.
507    pub fn into_parts(self) -> (ErrorKind, Labels) {
508        (self.kind, self.labels)
509    }
510
511    /// The primary model path. The root path denotes the whole document.
512    pub fn path(&self) -> &NodePath {
513        self.labels.path()
514    }
515
516    /// Primary and related labels, the primary one first.
517    pub fn labels(&self) -> &[Label] {
518        self.labels.labels()
519    }
520
521    /// The span of the primary label, if source context is available.
522    pub fn span(&self) -> Option<Range<usize>> {
523        self.labels.span()
524    }
525
526    /// Original YAML text, if this diagnostic has document context.
527    pub fn source_text(&self) -> Option<&str> {
528        self.labels.source_text()
529    }
530
531    /// Original file name, if the document was read from a path.
532    pub fn source_name(&self) -> Option<&str> {
533        self.labels.source_name()
534    }
535
536    /// Render this error as a [`Report`], the way it renders through
537    /// `miette::Diagnostic` for the `miette` feature, without requiring that
538    /// feature at the call site.
539    pub fn report(&self) -> Report {
540        self.labels.report(&self.kind)
541    }
542
543    pub(crate) fn with_primary_span(mut self, span: Option<Range<usize>>) -> Self {
544        self.labels = self.labels.with_primary_span(span);
545        self
546    }
547
548    pub(crate) fn with_related_span(mut self, span: Option<Range<usize>>) -> Self {
549        self.labels = self.labels.with_related_span(span);
550        self
551    }
552
553    pub(crate) fn attach_source(&mut self, source: Arc<str>, name: Option<Arc<str>>) {
554        self.labels.attach_source(source, name);
555    }
556}
557
558impl fmt::Display for Error {
559    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
560        self.labels.write_prefix(f)?;
561        write!(f, "{}", self.kind)
562    }
563}
564
565impl std::error::Error for Error {}

567#[cfg(feature = "miette")]
568impl miette::Diagnostic for Error {
569    fn source_code(&self) -> Option<&dyn miette::SourceCode> {
570        self.labels.source_code()
571    }
572
573    fn labels(&self) -> Option<Box<dyn Iterator<Item = miette::LabeledSpan> + '_>> {
574        Some(self.labels.spans(&self.kind))

578/// A message plus [`Labels`], for callers whose own error kind does not
579/// implement `miette::Diagnostic` — for example `rattler_lock`, which has no
580/// `miette` dependency of its own and asks [`Labels::report`] or
581/// [`Error::report`] for one instead.
582#[derive(Clone, Debug)]
583pub struct Report {
584    message: String,
585    labels: Labels,
586}
