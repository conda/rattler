16/// An immutable lock file together with its original YAML source locations.
17///
18/// Borrow the model with [`Self::lock_file`] for source-aware conversions. To edit
19/// it, consume the document with [`Self::into_lock_file`]; source locations cannot
20/// then accidentally refer to a changed model.
21#[derive(Clone, Debug)]
22pub struct Document {
23    lock_file: LockFile,
24    source: Arc<str>,
25    name: Option<Arc<str>>,
26    spans: SourceMap,
27}

29impl Document {
30    /// Parse and validate a single CEP-37 v1 YAML document.
31    ///
32    /// Every error this returns already carries the source text and the span of
33    /// the offending node.
34    ///
35    /// ```
36    /// use rattler_conda_lock::{Document, Manager};
37    ///
38    /// let source = "\
39    /// version: 1
40    /// metadata:
41    ///   content_hash:
42    ///     linux-64: 8b7df143d91c716ecfa5fc1730022f6b421b05cedee8fd52b1fc65a96030ad52
43    ///   channels:
44    ///     - url: conda-forge
45    ///       used_env_vars: []
46    ///   platforms: [linux-64]
47    ///   sources: [environment.yml]
48    /// package:
49    ///   - name: python
50    ///     version: '3.13.1'
51    ///     manager: conda
52    ///     platform: linux-64
53    ///     dependencies:
54    ///       libgcc: '>=13'
55    ///     url: https://conda.anaconda.org/conda-forge/linux-64/python-3.13.1-h9e4cc4f_0.conda
56    ///     hash:
57    ///       sha256: 1b98e5a2b1f5b3e2a6b44f5a5f42c9b1e2a3f5a6b7c8d9e0f1a2b3c4d5e6f708
58    ///     category: main
59    ///     optional: false
60    /// ";
61    /// let document = Document::parse(source)?;
62    ///
63    /// let package = &document.lock_file().package[0];
64    /// assert_eq!(package.manager, Manager::Conda);
65    /// assert_eq!(package.dependencies["libgcc"], ">=13");
66    ///
67    /// // The version keeps the spelling of the file, not a YAML float.
68    /// assert_eq!(package.version, "3.13.1");
69    ///
70    /// // Every node can be located in the original text.
71    /// let span = document.span("package[0].url").unwrap();
72    /// assert!(source[span].ends_with("python-3.13.1-h9e4cc4f_0.conda"));
73    /// # Ok::<(), rattler_conda_lock::Error>(())
74    /// ```
75    pub fn parse(source: &str) -> Result<Self, Error> {
76        Self::parse_named(Arc::from(source), None)
77    }

79    /// Read UTF-8 YAML, retaining the file name in subsequent diagnostics.
80    pub fn from_path(path: impl AsRef<Path>) -> Result<Self, Error> {
81        let path = path.as_ref();
82        let name: Arc<str> = Arc::from(path.to_string_lossy().as_ref());
83        let source = std::fs::read_to_string(path).map_err(|error| {
84            let mut error = Error::new(
85                NodePath::root(),
86                ErrorKind::Read {
87                    path: path.to_owned(),
88                    error: Arc::new(error),
89                },
90            );
91            error.labels.name = Some(name.clone());
92            error
93        })?;
94        Self::parse_named(Arc::from(source), Some(name))
95    }
96
97    fn parse_named(source: Arc<str>, name: Option<Arc<str>>) -> Result<Self, Error> {
98        let (lock_file, spans) = crate::parse::parse(&source).map_err(|mut error| {
99            error.attach_source(source.clone(), name.clone());
100            error
101        })?;
102        let document = Self {
103            lock_file,
104            source,
105            name,
106            spans,
107        };
108        document.lock_file.validate().map_err(|mut error| {
109            document.locate(&mut error.labels);
110            error
111        })?;
112        Ok(document)
113    }

142    /// Fills in *where* a diagnostic is, by looking up its model paths in this
143    /// document.
144    ///
145    /// A diagnostic built from a model says what is wrong and which field it is
146    /// about — `package[2].url` — but not where that field is in the file. This
147    /// looks each of its paths up in the source map and attaches the byte span
148    /// it found, together with this document's text and file name, so the error
149    /// can be printed against the offending line.
150    ///
151    /// Use it for diagnostics you raise yourself about a document you parsed.
152    /// Anything this crate returns for a `Document`, and anything
153    /// `rattler_lock::LockFile::from_conda_lock_document` returns, is located
154    /// already.
155    ///
156    /// A label that already has a span keeps it, and a path with no recorded
157    /// span falls back to its nearest recorded ancestor, so an error about a
158    /// missing field still points at the mapping that lacks it.
159    ///
160    /// ```
161    /// use rattler_conda_lock::{Document, Labels};
162    ///
163    /// # let source = "metadata:\n  content_hash: {}\n  channels: []\n  platforms: []\n  sources: [environment.yml]\npackage: []\n";
164    /// let document = Document::parse(source)?;
165    ///
166    /// let mut labels = Labels::new("metadata.sources[0]");
167    /// assert!(labels.span().is_none());
168    ///
169    /// document.locate(&mut labels);
170    /// assert_eq!(&source[labels.span().unwrap()], "environment.yml");
171    /// # Ok::<(), rattler_conda_lock::Error>(())
172    /// ```
173    pub fn locate(&self, labels: &mut Labels) {
174        locate_labels(labels, &self.spans);
175        labels.attach_source(self.source.clone(), self.name.clone());
176    }
177}

179pub(crate) fn locate_labels(labels: &mut Labels, spans: &SourceMap) {
180    let mut definitions = Vec::new();
181    for label in &mut labels.labels {
182        if label.span.is_some() {
183            continue;
184        }
185        let mut path = label.path.as_str();
186        let found = loop {
187            if let Some(span) = spans.get(path) {
188                if span.referenced.is_some() {
189                    break Some((span, path));
190                }
191                if let Some(parent) = span.parent.as_ref() {
192                    path = parent.as_str();
193                    continue;
194                }
195            }
196            if path.is_empty() {
197                break None;
198            }
199            // Exact lookup above handles mapping keys containing dots. Only a
200            // missing path is shortened, selecting the longest recorded parent.
201            path = match path.rfind(['.', '[']) {
202                Some(index) => &path[..index],
203                None => "",
204            };
205        };
206        if let Some((span, resolved)) = found {
207            // A block collection spans from its first child, so its own value
208            // span is empty; the mapping key that introduced it is the node a
209            // reader recognizes. Falling back to the document as a whole points
210            // at nothing, so a path the document does not contain at all — a
211            // caller's own option, say — keeps no span.
212            let resolved_document = resolved.is_empty() && !label.path.is_root();
213            label.span = span
214                .referenced
215                .clone()
216                .filter(|range| !range.is_empty())
217                .or_else(|| span.key.clone())
218                .or_else(|| {
219                    (!resolved_document)
220                        .then(|| span.referenced.clone())
221                        .flatten()
222                });
223            if label.span.is_some() && span.defined.is_some() && span.defined != span.referenced {
224                definitions.push(Label {
225                    path: label.path.clone(),
226                    message: Some("value defined here (referenced through a YAML alias)"),
227                    span: span.defined.clone(),
228                });
229            }
230        }
231    }
232    labels.labels.extend(definitions);
233}
