1//! Discovery and retrieval of attestation sidecars.
2//!
3//! Under [CEP 50](https://conda.org/learn/ceps/cep-0050) a channel serves the
4//! attestations of a package at `<package_url>.sigs` (mutable) and at
5//! `<package_url>.sigs.<sha256>` (immutable, content-addressed). The hash is
6//! advertised in the repodata through the `attestations_sha256` field of the
7//! package record. Clients discover sidecars exclusively through that field
8//! and always fetch the content-addressed URL.
9
10use futures::StreamExt;
11use rattler_conda_types::RepoDataRecord;
12use rattler_digest::{Sha256, Sha256Hash};
13use reqwest_middleware::ClientWithMiddleware;
14use sigstore_types::Bundle;
15use tokio::io::AsyncReadExt;
16use url::Url;
17
18use crate::error::{SigstoreError, SigstoreResult};

20/// The suffix that is appended to a package filename to obtain the attestation
21/// sidecar filename.
22pub const SIDECAR_SUFFIX: &str = ".sigs";
23
24/// The default maximum size of a sidecar in bytes. Sigstore bundles are in the
25/// order of 10 KiB each, so this leaves room for a generous number of
26/// attestations per package while bounding what a malicious server can make a
27/// client buffer.
28pub const DEFAULT_MAX_SIDECAR_SIZE: u64 = 4 * 1024 * 1024;
29
30/// The attestation sidecar of a package: a hash-verified JSON array of Sigstore
31/// bundles.
32#[derive(Debug, Clone)]
33pub struct AttestationSidecar {
34    /// The content-addressed URL the sidecar was retrieved from.
35    pub url: Url,
36
37    /// The SHA256 of the sidecar bytes. This is guaranteed to match the hash
38    /// that was advertised in the repodata.
39    pub sha256: Sha256Hash,
40
41    /// The bundles contained in the sidecar, in the order they appear.
42    pub bundles: Vec<Bundle>,
43}
44
45/// Returns the mutable sidecar URL for a package.
46///
47/// This appends `.sigs` to the package URL's path while preserving its query
48/// string. It is useful when no repodata record is available to advertise the
49/// hash of an immutable, content-addressed sidecar.
50pub fn mutable_sidecar_url(package_url: &Url) -> SigstoreResult<Url> {
51    let mut url = package_url.clone();
52    url.path_segments()
53        .and_then(|mut segments| segments.next_back().map(str::to_owned))
54        .filter(|filename| !filename.is_empty())
55        .ok_or_else(|| SigstoreError::InvalidSidecarUrl(package_url.clone()))?;
56    url.set_path(&format!("{}{SIDECAR_SUFFIX}", url.path()));
57    Ok(url)
58}
59
60/// Returns the content-addressed sidecar URL for a package.
61///
62/// The URL is derived by appending `.sigs.<sha256>` to the last path segment
63/// of the package URL. Any query string (e.g. an authentication token) is
64/// preserved.
65pub fn sidecar_url(package_url: &Url, attestations_sha256: &Sha256Hash) -> SigstoreResult<Url> {
66    let mut url = package_url.clone();
67    url.path_segments()
68        .and_then(|mut segments| segments.next_back().map(str::to_owned))
69        .filter(|filename| !filename.is_empty())
70        .ok_or_else(|| SigstoreError::InvalidSidecarUrl(package_url.clone()))?;
71    let sidecar_path = format!(
72        "{}{SIDECAR_SUFFIX}.{}",
73        url.path(),
74        hex::encode(attestations_sha256)
75    );
76    // set_path preserves existing percent escapes, unlike pushing a segment.
77    url.set_path(&sidecar_path);
78    Ok(url)
79}
80
81/// Returns the content-addressed sidecar URL for a record, or `None` if the
82/// record does not advertise attestations.
83pub fn sidecar_url_for_record(record: &RepoDataRecord) -> SigstoreResult<Option<Url>> {
84    record
85        .package_record
86        .attestations_sha256
87        .as_ref()
88        .map(|hash| sidecar_url(&record.url, hash))
89        .transpose()
90}

92/// Parses sidecar bytes after checking them against the advertised hash.
93///
94/// The bytes must hash to `expected_sha256` and decode as a non-empty JSON
95/// array of Sigstore bundles.
96pub fn parse_sidecar(
97    url: &Url,
98    bytes: &[u8],
99    expected_sha256: &Sha256Hash,
100) -> SigstoreResult<AttestationSidecar> {
101    let actual = rattler_digest::compute_bytes_digest::<Sha256>(bytes);
102    if &actual != expected_sha256 {
103        return Err(SigstoreError::SidecarHashMismatch {
104            url: url.clone(),
105            expected: hex::encode(expected_sha256).into_boxed_str(),
106            actual: hex::encode(actual).into_boxed_str(),
107        });
108    }
109
110    let bundles = parse_bundles(url, bytes)?;
111
112    Ok(AttestationSidecar {
113        url: url.clone(),
114        sha256: actual,
115        bundles,
116    })
117}
118
119/// Parses a non-empty JSON array of Sigstore bundles.
120///
121/// Unlike [`parse_sidecar`], this does not require or validate a sidecar hash.
122/// It is intended for explicitly supplied attestations and mutable `.sigs`
123/// URLs, where no repodata record advertises a content hash.
124pub fn parse_bundles(url: &Url, bytes: &[u8]) -> SigstoreResult<Vec<Bundle>> {
125    let raw_bundles: Vec<serde_json::Value> =
126        serde_json::from_slice(bytes).map_err(|err| SigstoreError::MalformedSidecar {
127            url: url.clone(),
128            message: format!("expected a JSON array of Sigstore bundles: {err}"),
129        })?;
130    if raw_bundles.is_empty() {
131        return Err(SigstoreError::MalformedSidecar {
132            url: url.clone(),
133            message: "the array of Sigstore bundles is empty".to_string(),
134        });
135    }
136
137    raw_bundles
138        .into_iter()
139        .enumerate()
140        .map(|(index, raw)| {
141            serde_json::from_value::<Bundle>(raw).map_err(|err| SigstoreError::MalformedSidecar {
142                url: url.clone(),
143                message: format!("bundle {index} is not a valid Sigstore bundle: {err}"),
144            })
145        })
146        .collect()
147}

149/// Fetches a bounded sidecar from an explicit URL and parses its bundles.
150///
151/// This accepts both remote URLs and local `file://` URLs. It does not check a
152/// sidecar content hash; callers still verify every bundle against the package
153/// artifact itself.
154pub async fn fetch_bundles(
155    client: &ClientWithMiddleware,
156    url: &Url,
157    max_size: u64,
158) -> SigstoreResult<Vec<Bundle>> {
159    let bytes = fetch_bounded(client, url, max_size).await?;
160    parse_bundles(url, &bytes)
161}
162
163/// Fetches and validates the attestation sidecar advertised by a record.
164///
165/// Returns `Ok(None)` if the record does not advertise attestations. The body
166/// is streamed and rejected as soon as it grows beyond `max_size` bytes. The
167/// received bytes are checked against the advertised hash before parsing.
168///
169/// Per the CEP, an unavailable, oversized, malformed or hash-mismatched
170/// sidecar is a retrieval failure and is reported as an error.
171pub async fn fetch_sidecar(
172    client: &ClientWithMiddleware,
173    record: &RepoDataRecord,
174    max_size: u64,
175) -> SigstoreResult<Option<AttestationSidecar>> {
176    let Some(expected_sha256) = record.package_record.attestations_sha256.as_ref() else {
177        return Ok(None);
178    };
179    let url = sidecar_url(&record.url, expected_sha256)?;
180    let bytes = fetch_bounded(client, &url, max_size).await?;
181    parse_sidecar(&url, &bytes, expected_sha256).map(Some)
182}
183
184/// Downloads `url` into memory, failing once more than `max_size` bytes have
185/// been received.
186async fn fetch_bounded(
187    client: &ClientWithMiddleware,
188    url: &Url,
189    max_size: u64,
190) -> SigstoreResult<Vec<u8>> {
191    if url.scheme() == "file" {
192        return fetch_bounded_file(url, max_size).await;
193    }
194
195    // The error messages name the redacted sidecar URL, so the URL is
196    // stripped from reqwest errors rather than redacted: a middleware may have
197    // rewritten the request URL into one carrying a secret that redaction does
198    // not recognize, such as an S3 presigned URL.
199    let response =
200        client
201            .get(url.clone())
202            .send()
203            .await
204            .map_err(|source| SigstoreError::FetchSidecar {
205                url: url.clone(),
206                source: source.without_url(),
207            })?;
208
209    let status = response.status();
210    if !status.is_success() {
211        return Err(SigstoreError::SidecarHttpStatus {
212            url: url.clone(),
213            status,
214        });
215    }
216
217    if response.content_length().is_some_and(|len| len > max_size) {
218        return Err(SigstoreError::SidecarTooLarge {
219            url: url.clone(),
220            max_size,
221        });
222    }
223
224    let mut bytes = Vec::new();
225    let mut stream = response.bytes_stream();
226    while let Some(chunk) = stream.next().await {
227        let chunk = chunk.map_err(|source| SigstoreError::ReadSidecar {
228            url: url.clone(),
229            source: source.without_url(),
230        })?;
231        if (bytes.len() + chunk.len()) as u64 > max_size {
232            return Err(SigstoreError::SidecarTooLarge {
233                url: url.clone(),
234                max_size,
235            });
236        }
237        bytes.extend_from_slice(&chunk);
238    }
239    Ok(bytes)
240}

242/// Reads a local sidecar while retaining the same memory bound as HTTP
243/// downloads. Reading one byte beyond the limit lets us distinguish a file at
244/// the limit from an oversized file without buffering the rest of it.
245async fn fetch_bounded_file(url: &Url, max_size: u64) -> SigstoreResult<Vec<u8>> {
246    let path = url
247        .to_file_path()
248        .map_err(|()| SigstoreError::InvalidSidecarFileUrl(url.clone()))?;
249    let file =
250        tokio::fs::File::open(path)
251            .await
252            .map_err(|source| SigstoreError::ReadLocalSidecar {
253                url: url.clone(),
254                source,
255            })?;
256    let mut bytes = Vec::new();
257    file.take(max_size.saturating_add(1))
258        .read_to_end(&mut bytes)
259        .await
260        .map_err(|source| SigstoreError::ReadLocalSidecar {
261            url: url.clone(),
262            source,
263        })?;
264    if bytes.len() as u64 > max_size {
265        return Err(SigstoreError::SidecarTooLarge {
266            url: url.clone(),
267            max_size,
268        });
269    }
270    Ok(bytes)
271}

277    fn hash_of(bytes: &[u8]) -> Sha256Hash {
278        rattler_digest::compute_bytes_digest::<Sha256>(bytes)

281    #[test]
282    fn sidecar_url_appends_suffix_and_hash() {
283        let hash = hash_of(b"sidecar");
284        let package_url =
285            Url::parse("https://prefix.dev/channel/linux-64/foo-1.0-h1_0.conda").unwrap();
286        let url = sidecar_url(&package_url, &hash).unwrap();
287        assert_eq!(
288            url.as_str(),
289            format!(
290                "https://prefix.dev/channel/linux-64/foo-1.0-h1_0.conda.sigs.{}",
291                hex::encode(hash)
292            )
293        );
294    }
295
296    #[test]
297    fn sidecar_url_preserves_query() {
298        let hash = hash_of(b"sidecar");
299        let package_url = Url::parse(
300            "https://example.com/t/secret/channel/noarch/foo-1.0-h1_0.tar.bz2?token=abc",
301        )
302        .unwrap();
303        let url = sidecar_url(&package_url, &hash).unwrap();
304        assert!(
305            url.path()
306                .ends_with(&format!(".tar.bz2.sigs.{}", hex::encode(hash)))
307        );
308        assert_eq!(url.query(), Some("token=abc"));
309    }

311    #[test]
312    fn mutable_sidecar_url_appends_suffix_and_preserves_query() {
313        let package_url =

331    #[test]

333        let url = Url::parse("https://example.com/foo.conda.sigs.abc").unwrap();
334        let err = parse_sidecar(&url, b"[]", &hash_of(b"other")).unwrap_err();

341    #[test]
342    fn parse_sidecar_rejects_malformed_content() {
343        let url = Url::parse("https://example.com/foo.conda.sigs.abc").unwrap();
344        for bytes in [&b"not json"[..], b"{}", b"[]", b"[{\"mediaType\": 1}]"] {
345            let err = parse_sidecar(&url, bytes, &hash_of(bytes)).unwrap_err();
346            assert!(
347                matches!(err, SigstoreError::MalformedSidecar { .. }),
348                "{bytes:?}: {err}"
349            );
350        }
351    }
