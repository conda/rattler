1//! Discovery and retrieval of attestation sidecars.
2//!
3//! The conda CEP on distribution of Sigstore attestations serves the
4//! attestations of a package at `<package_url>.sigs` (mutable) and at
5//! `<package_url>.sigs.<sha256>` (immutable, content-addressed). The hash is
6//! advertised in the repodata through the `attestations_sha256` field of the
7//! package record. Clients discover sidecars exclusively through that field
8//! and always fetch the content-addressed URL.
9
10use futures::StreamExt;
11use rattler_conda_types::RepoDataRecord;
12use rattler_digest::{Sha256, Sha256Hash};
13use rattler_redaction::Redact;
14use reqwest_middleware::ClientWithMiddleware;
15use sigstore_types::Bundle;
16use tokio::io::AsyncReadExt;
17use url::Url;

19use crate::error::{SigstoreError, SigstoreResult};

21/// The suffix that is appended to a package filename to obtain the attestation
22/// sidecar filename.
23pub const SIDECAR_SUFFIX: &str = ".sigs";
24
25/// The default maximum size of a sidecar in bytes. Sigstore bundles are in the
26/// order of 10 KiB each, so this leaves room for a generous number of
27/// attestations per package while bounding what a malicious server can make a
28/// client buffer.
29pub const DEFAULT_MAX_SIDECAR_SIZE: u64 = 4 * 1024 * 1024;
30
31/// The attestation sidecar of a package: a hash-verified JSON array of Sigstore
32/// bundles.
33#[derive(Debug, Clone)]
34pub struct AttestationSidecar {
35    /// The content-addressed URL the sidecar was retrieved from.
36    pub url: Url,
37
38    /// The SHA256 of the sidecar bytes. This is guaranteed to match the hash
39    /// that was advertised in the repodata.
40    pub sha256: Sha256Hash,
41
42    /// The bundles contained in the sidecar, in the order they appear.
43    pub bundles: Vec<Bundle>,
44}
45
46/// Returns the mutable sidecar URL for a package.
47///
48/// This appends `.sigs` to the package URL's path while preserving its query
49/// string. It is useful when no repodata record is available to advertise the
50/// hash of an immutable, content-addressed sidecar.
51pub fn mutable_sidecar_url(package_url: &Url) -> SigstoreResult<Url> {
52    let mut url = package_url.clone();
53    url.path_segments()
54        .and_then(|mut segments| segments.next_back().map(str::to_owned))
55        .filter(|filename| !filename.is_empty())
56        .ok_or_else(|| SigstoreError::InvalidSidecarUrl(package_url.clone()))?;
57    url.set_path(&format!("{}{SIDECAR_SUFFIX}", url.path()));
58    Ok(url)
59}
60
61/// Returns the content-addressed sidecar URL for a package.
62///
63/// The URL is derived by appending `.sigs.<sha256>` to the last path segment
64/// of the package URL. Any query string (e.g. an authentication token) is
65/// preserved.
66pub fn sidecar_url(package_url: &Url, attestations_sha256: &Sha256Hash) -> SigstoreResult<Url> {
67    let mut url = package_url.clone();
68    url.path_segments()
69        .and_then(|mut segments| segments.next_back().map(str::to_owned))
70        .filter(|filename| !filename.is_empty())
71        .ok_or_else(|| SigstoreError::InvalidSidecarUrl(package_url.clone()))?;
72    let sidecar_path = format!(
73        "{}{SIDECAR_SUFFIX}.{}",
74        url.path(),
75        hex::encode(attestations_sha256)
76    );
77    // set_path preserves existing percent escapes, unlike pushing a segment.
78    url.set_path(&sidecar_path);
79    Ok(url)
80}
81
82/// Returns the content-addressed sidecar URL for a record, or `None` if the
83/// record does not advertise attestations.
84pub fn sidecar_url_for_record(record: &RepoDataRecord) -> SigstoreResult<Option<Url>> {
85    record
86        .package_record
87        .attestations_sha256
88        .as_ref()
89        .map(|hash| sidecar_url(&record.url, hash))
90        .transpose()
91}

93/// Parses sidecar bytes after checking them against the advertised hash.
94///
95/// The bytes must hash to `expected_sha256` and decode as a non-empty JSON
96/// array of Sigstore bundles.
97pub fn parse_sidecar(
98    url: &Url,
99    bytes: &[u8],
100    expected_sha256: &Sha256Hash,
101) -> SigstoreResult<AttestationSidecar> {
102    let actual = rattler_digest::compute_bytes_digest::<Sha256>(bytes);
103    if &actual != expected_sha256 {
104        return Err(SigstoreError::SidecarHashMismatch {
105            url: url.clone(),
106            expected: hex::encode(expected_sha256).into_boxed_str(),
107            actual: hex::encode(actual).into_boxed_str(),
108        });
109    }
110
111    let bundles = parse_bundles(url, bytes)?;
112
113    Ok(AttestationSidecar {
114        url: url.clone(),
115        sha256: actual,
116        bundles,
117    })
118}
119
120/// Parses a non-empty JSON array of Sigstore bundles.
121///
122/// Unlike [`parse_sidecar`], this does not require or validate a sidecar hash.
123/// It is intended for explicitly supplied attestations and mutable `.sigs`
124/// URLs, where no repodata record advertises a content hash.
125pub fn parse_bundles(url: &Url, bytes: &[u8]) -> SigstoreResult<Vec<Bundle>> {
126    let raw_bundles: Vec<serde_json::Value> =
127        serde_json::from_slice(bytes).map_err(|err| SigstoreError::MalformedSidecar {
128            url: url.clone(),
129            message: format!("expected a JSON array of Sigstore bundles: {err}"),
130        })?;
131    if raw_bundles.is_empty() {
132        return Err(SigstoreError::MalformedSidecar {
133            url: url.clone(),
134            message: "the array of Sigstore bundles is empty".to_string(),
135        });
136    }
137
138    raw_bundles
139        .into_iter()
140        .enumerate()
141        .map(|(index, raw)| {
142            serde_json::from_value::<Bundle>(raw).map_err(|err| SigstoreError::MalformedSidecar {
143                url: url.clone(),
144                message: format!("bundle {index} is not a valid Sigstore bundle: {err}"),
145            })
146        })
147        .collect()
148}

150/// Fetches a bounded sidecar from an explicit URL and parses its bundles.
151///
152/// This accepts both remote URLs and local `file://` URLs. It does not check a
153/// sidecar content hash; callers still verify every bundle against the package
154/// artifact itself.
155pub async fn fetch_bundles(
156    client: &ClientWithMiddleware,
157    url: &Url,
158    max_size: u64,
159) -> SigstoreResult<Vec<Bundle>> {
160    let bytes = fetch_bounded(client, url, max_size).await?;
161    parse_bundles(url, &bytes)
162}
163
164/// Fetches and validates the attestation sidecar advertised by a record.
165///
166/// Returns `Ok(None)` if the record does not advertise attestations. The body
167/// is streamed and rejected as soon as it grows beyond `max_size` bytes. The
168/// received bytes are checked against the advertised hash before parsing.
169///
170/// Per the CEP, an unavailable, oversized, malformed or hash-mismatched
171/// sidecar is a retrieval failure and is reported as an error.
172pub async fn fetch_sidecar(
173    client: &ClientWithMiddleware,
174    record: &RepoDataRecord,
175    max_size: u64,
176) -> SigstoreResult<Option<AttestationSidecar>> {
177    let Some(expected_sha256) = record.package_record.attestations_sha256.as_ref() else {
178        return Ok(None);
179    };
180    let url = sidecar_url(&record.url, expected_sha256)?;
181    let bytes = fetch_bounded(client, &url, max_size).await?;
182    parse_sidecar(&url, &bytes, expected_sha256).map(Some)
183}

185/// Downloads `url` into memory, failing once more than `max_size` bytes have
186/// been received.
187async fn fetch_bounded(
188    client: &ClientWithMiddleware,
189    url: &Url,
190    max_size: u64,
191) -> SigstoreResult<Vec<u8>> {
192    if url.scheme() == "file" {
193        return fetch_bounded_file(url, max_size).await;
194    }
195
196    let response =
197        client
198            .get(url.clone())
199            .send()
200            .await
201            .map_err(|source| SigstoreError::FetchSidecar {
202                url: url.clone(),
203                source: source.redact(),
204            })?;
205
206    let status = response.status();
207    if !status.is_success() {
208        return Err(SigstoreError::SidecarHttpStatus {
209            url: url.clone(),
210            status,
211        });
212    }
213
214    if response.content_length().is_some_and(|len| len > max_size) {
215        return Err(SigstoreError::SidecarTooLarge {
216            url: url.clone(),
217            max_size,
218        });
219    }
220
221    let mut bytes = Vec::new();
222    let mut stream = response.bytes_stream();
223    while let Some(chunk) = stream.next().await {
224        let chunk = chunk.map_err(|source| SigstoreError::ReadSidecar {
225            url: url.clone(),
226            source: source.redact(),
227        })?;
228        if (bytes.len() + chunk.len()) as u64 > max_size {
229            return Err(SigstoreError::SidecarTooLarge {
230                url: url.clone(),
231                max_size,
232            });
233        }
234        bytes.extend_from_slice(&chunk);
235    }
236    Ok(bytes)
237}

239/// Reads a local sidecar while retaining the same memory bound as HTTP
240/// downloads. Reading one byte beyond the limit lets us distinguish a file at
241/// the limit from an oversized file without buffering the rest of it.
242async fn fetch_bounded_file(url: &Url, max_size: u64) -> SigstoreResult<Vec<u8>> {
243    let path = url
244        .to_file_path()
245        .map_err(|()| SigstoreError::InvalidSidecarFileUrl(url.clone()))?;
246    let file =
247        tokio::fs::File::open(path)
248            .await
249            .map_err(|source| SigstoreError::ReadLocalSidecar {
250                url: url.clone(),
251                source,
252            })?;
253    let mut bytes = Vec::new();
254    file.take(max_size.saturating_add(1))
255        .read_to_end(&mut bytes)
256        .await
257        .map_err(|source| SigstoreError::ReadLocalSidecar {
258            url: url.clone(),
259            source,
260        })?;
261    if bytes.len() as u64 > max_size {
262        return Err(SigstoreError::SidecarTooLarge {
263            url: url.clone(),
264            max_size,
265        });
266    }
267    Ok(bytes)
268}

274    fn hash_of(bytes: &[u8]) -> Sha256Hash {
275        rattler_digest::compute_bytes_digest::<Sha256>(bytes)

278    #[test]
279    fn sidecar_url_appends_suffix_and_hash() {
280        let hash = hash_of(b"sidecar");
281        let package_url =
282            Url::parse("https://prefix.dev/channel/linux-64/foo-1.0-h1_0.conda").unwrap();
283        let url = sidecar_url(&package_url, &hash).unwrap();
284        assert_eq!(
285            url.as_str(),
286            format!(
287                "https://prefix.dev/channel/linux-64/foo-1.0-h1_0.conda.sigs.{}",
288                hex::encode(hash)
289            )
290        );
291    }
292
293    #[test]
294    fn sidecar_url_preserves_query() {
295        let hash = hash_of(b"sidecar");
296        let package_url = Url::parse(
297            "https://example.com/t/secret/channel/noarch/foo-1.0-h1_0.tar.bz2?token=abc",
298        )
299        .unwrap();
300        let url = sidecar_url(&package_url, &hash).unwrap();
301        assert!(
302            url.path()
303                .ends_with(&format!(".tar.bz2.sigs.{}", hex::encode(hash)))
304        );
305        assert_eq!(url.query(), Some("token=abc"));
306    }

308    #[test]
309    fn mutable_sidecar_url_appends_suffix_and_preserves_query() {
310        let package_url =
311            Url::parse("https://example.com/channel/noarch/foo-1%21post1-0.conda?token=abc")
312                .unwrap();

314        assert_eq!(url.path(), "/channel/noarch/foo-1%21post1-0.conda.sigs");
315        assert_eq!(url.query(), Some("token=abc"));

328    #[test]

330        let url = Url::parse("https://example.com/foo.conda.sigs.abc").unwrap();
331        let err = parse_sidecar(&url, b"[]", &hash_of(b"other")).unwrap_err();
332        assert!(
333            matches!(err, SigstoreError::SidecarHashMismatch { .. }),

338    #[test]
339    fn parse_sidecar_rejects_malformed_content() {
340        let url = Url::parse("https://example.com/foo.conda.sigs.abc").unwrap();
341        for bytes in [&b"not json"[..], b"{}", b"[]", b"[{\"mediaType\": 1}]"] {
342            let err = parse_sidecar(&url, bytes, &hash_of(bytes)).unwrap_err();
343            assert!(
344                matches!(err, SigstoreError::MalformedSidecar { .. }),
345                "{bytes:?}: {err}"
346            );
347        }
