1//! Error types for attestation discovery, retrieval and verification.
2
3use rattler_redaction::Redact;
4use url::Url;
5
6/// Errors that can occur while fetching or verifying Sigstore attestations for
7/// a conda package.
8#[derive(Debug, thiserror::Error)]
9pub enum SigstoreError {
10    /// The package record does not carry an `attestations_sha256` field, so no
11    /// attestations are advertised for it.
12    #[error("no attestations are advertised for {0}")]
13    NoAttestationsAdvertised(String),
14
15    /// The package record does not carry a `sha256`, which is required to bind
16    /// an attestation to the package without re-downloading it.
17    #[error("{0} has no sha256 in its record, cannot bind attestations to it")]
18    MissingPackageSha256(String),
19
20    /// The sidecar URL could not be derived from the package URL.
21    #[error("cannot derive an attestation sidecar URL from {}", .0.clone().redact())]
22    InvalidSidecarUrl(Url),
23
24    /// A `file://` sidecar URL could not be converted to a local path.
25    #[error("cannot convert attestation sidecar URL {} to a local path", .0.clone().redact())]
26    InvalidSidecarFileUrl(Url),
27
28    /// The HTTP request for the sidecar failed.
29    #[error("failed to fetch attestation sidecar from {}", .url.clone().redact())]
30    FetchSidecar {
31        /// The sidecar URL.
32        url: Url,
33        /// The underlying error.
34        #[source]
35        source: reqwest_middleware::Error,
36    },
37
38    /// The server answered the sidecar request with a non-success status.
39    #[error("attestation sidecar at {} returned HTTP {status}", .url.clone().redact())]
40    SidecarHttpStatus {
41        /// The sidecar URL.
42        url: Url,
43        /// The HTTP status code.
44        status: reqwest::StatusCode,
45    },
46
47    /// Reading the response body failed.
48    #[error("failed to read attestation sidecar from {}", .url.clone().redact())]
49    ReadSidecar {
50        /// The sidecar URL.
51        url: Url,
52        /// The underlying error.
53        #[source]
54        source: reqwest::Error,
55    },
56
57    /// Opening or reading a local sidecar failed.
58    #[error("failed to read local attestation sidecar from {}", .url.clone().redact())]
59    ReadLocalSidecar {
60        /// The sidecar URL.
61        url: Url,
62        /// The underlying I/O error.
63        #[source]
64        source: std::io::Error,
65    },
66
67    /// The sidecar is larger than the configured maximum.
68    #[error("attestation sidecar at {} exceeds the maximum size of {max_size} bytes", .url.clone().redact())]
69    SidecarTooLarge {
70        /// The sidecar URL.
71        url: Url,
72        /// The configured maximum size in bytes.
73        max_size: u64,
74    },
75
76    /// The SHA256 of the sidecar bytes does not match the hash advertised in
77    /// the repodata.
78    #[error(
79        "attestation sidecar at {} does not match the advertised hash (expected {expected}, got {actual})", .url.clone().redact()
80    )]
81    SidecarHashMismatch {
82        /// The sidecar URL.
83        url: Url,
84        /// The hash advertised in the repodata, hex encoded.
85        expected: Box<str>,
86        /// The hash of the bytes that were received, hex encoded.
87        actual: Box<str>,
88    },
89
90    /// The sidecar is not a non-empty JSON array of Sigstore bundles.
91    #[error("attestation sidecar at {} is malformed: {message}", .url.clone().redact())]
92    MalformedSidecar {
93        /// The sidecar URL.
94        url: Url,
95        /// A description of the problem.
96        message: String,
97    },
98
99    /// The Sigstore trusted root could not be loaded.
100    #[error("failed to load the Sigstore trusted root: {0}")]
101    TrustedRoot(String),
102
103    /// None of the attestations in the sidecar passed verification.
104    #[error("no attestation for {package} passed verification:{}", format_reasons(.reasons))]
105    VerificationFailed {
106        /// The package filename.
107        package: String,
108        /// One reason per rejected bundle.
109        reasons: Vec<String>,
110    },
111}
112
113fn format_reasons(reasons: &[String]) -> String {
114    reasons
115        .iter()
116        .map(|reason| format!("\n  - {reason}"))
117        .collect()
118}
119
120/// A convenience alias for results in this crate.
121pub type SigstoreResult<T> = Result<T, SigstoreError>;
