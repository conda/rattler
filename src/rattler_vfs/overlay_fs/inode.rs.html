1//! Inode bookkeeping for the writable overlay.
2//!
3//! The lower layer (read-only VFS) owns inodes `1..UPPER_INODE_BASE`. Upper
4//! layer (overlay) entries get inodes `UPPER_INODE_BASE..u64::MAX` assigned
5//! lazily by [`UpperInodeMap`]. [`ResolvedIno`] wraps the result of "given
6//! this kernel-visible inode, which layer is it actually in right now?",
7//! since rename and copy-on-write can promote a lower inode into the upper
8//! layer dynamically.
9//!
10//! Extracted from `overlay_fs.rs` to keep the inode allocator and the COW /
11//! `VfsOps` machinery in separate modules. Pure data + bookkeeping; no I/O.
12
13use std::{
14    collections::HashMap,
15    path::{Path, PathBuf},
16    sync::atomic::{AtomicU64, Ordering},
17};
18
19/// Inodes at or above this value belong to the upper (overlay) layer.
20/// Anything below is owned by the lower read-only VFS.
21pub(crate) const UPPER_INODE_BASE: u64 = u64::MAX / 2;
22
23/// Bidirectional inode ↔ path mapping for upper-layer entries.
24pub(crate) struct UpperInodeMap {
25    path_to_ino: HashMap<PathBuf, u64>,
26    ino_to_path: HashMap<u64, PathBuf>,
27    next_ino: AtomicU64,
28}
29
30impl UpperInodeMap {
31    pub(crate) fn new() -> Self {
32        Self {
33            path_to_ino: HashMap::new(),
34            ino_to_path: HashMap::new(),
35            next_ino: AtomicU64::new(UPPER_INODE_BASE),
36        }
37    }
38
39    pub(crate) fn get_or_assign(&mut self, virtual_path: PathBuf) -> u64 {
40        if let Some(&ino) = self.path_to_ino.get(&virtual_path) {
41            return ino;
42        }
43        let ino = self.next_ino.fetch_add(1, Ordering::Relaxed);
44        self.ino_to_path.insert(ino, virtual_path.clone());
45        self.path_to_ino.insert(virtual_path, ino);
46        ino
47    }
48
49    pub(crate) fn path_for(&self, ino: u64) -> Option<&PathBuf> {
50        self.ino_to_path.get(&ino)
51    }
52
53    /// Reverse lookup. Used by tests; production code asks the upper map
54    /// via `path_for` after promotion.
55    #[cfg(test)]
56    pub(crate) fn ino_for_path(&self, virtual_path: &Path) -> Option<u64> {
57        self.path_to_ino.get(virtual_path).copied()
58    }
59
60    /// Remap an existing inode from `old_path` to `new_path`.
61    /// The kernel expects the source inode to remain valid after rename,
62    /// just pointing at the new path.
63    pub(crate) fn rename_path(&mut self, old_path: &Path, new_path: PathBuf) {
64        // Clean up any existing inode at the destination (overwrite case)
65        if let Some(old_dst_ino) = self.path_to_ino.remove(&new_path) {
66            self.ino_to_path.remove(&old_dst_ino);
67        }
68        // Remap source inode to destination path
69        if let Some(ino) = self.path_to_ino.remove(old_path) {
70            self.ino_to_path.insert(ino, new_path.clone());
71            self.path_to_ino.insert(new_path, ino);
72        }
73    }
74}
75
76/// Result of resolving a kernel-visible inode to its current backing layer.
77pub(crate) enum ResolvedIno {
78    /// File is in the upper (overlay) layer at this virtual path.
79    Upper(PathBuf),
80    /// File is in the lower (read-only) layer with this inode and virtual path.
81    Lower(u64, PathBuf),
82}
