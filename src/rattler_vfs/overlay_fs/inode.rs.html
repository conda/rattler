1//! Inode bookkeeping for the writable overlay.
2//!
3//! The lower layer (read-only VFS) owns inodes `1..UPPER_INODE_BASE`. Upper
4//! layer (overlay) entries get inodes `UPPER_INODE_BASE..u64::MAX` assigned
5//! lazily by [`UpperInodeMap`]. [`ResolvedIno`] wraps the result of "given
6//! this kernel-visible inode, which layer is it actually in right now?",
7//! since rename and copy-on-write can promote a lower inode into the upper
8//! layer dynamically.
9//!
10//! Extracted from `overlay_fs.rs` to keep the inode allocator and the COW /
11//! `VfsOps` machinery in separate modules. Pure data + bookkeeping; no I/O.
12
13use std::{
14    collections::HashMap,
15    path::{Path, PathBuf},
16};
17
18/// Inodes at or above this value belong to the upper (overlay) layer.
19/// Anything below is owned by the lower read-only VFS.
20pub(crate) const UPPER_INODE_BASE: u64 = u64::MAX / 2;
21
22/// Bidirectional inode ↔ path mapping for upper-layer entries.
23pub(crate) struct UpperInodeMap {
24    path_to_ino: HashMap<PathBuf, u64>,
25    /// Reverse map, indexed by `ino - UPPER_INODE_BASE`. Upper inodes are handed
26    /// out sequentially from `UPPER_INODE_BASE`, so a `Vec` is denser and faster
27    /// than a `HashMap`. A slot is `None` once a rename orphans its inode (the
28    /// overwritten destination — see [`Self::rename_path`]).
29    ino_to_path: Vec<Option<PathBuf>>,
30}
31
32impl UpperInodeMap {
33    pub(crate) fn new() -> Self {
34        Self {
35            path_to_ino: HashMap::new(),
36            ino_to_path: Vec::new(),
37        }
38    }
39
40    pub(crate) fn get_or_assign(&mut self, virtual_path: PathBuf) -> u64 {
41        if let Some(&ino) = self.path_to_ino.get(&virtual_path) {
42            return ino;
43        }
44        // Next sequential inode = base + number already assigned.
45        let ino = UPPER_INODE_BASE + self.ino_to_path.len() as u64;
46        self.ino_to_path.push(Some(virtual_path.clone()));
47        self.path_to_ino.insert(virtual_path, ino);
48        ino
49    }
50
51    pub(crate) fn path_for(&self, ino: u64) -> Option<&PathBuf> {
52        let idx = ino.checked_sub(UPPER_INODE_BASE)? as usize;
53        self.ino_to_path.get(idx)?.as_ref()
54    }
55
56    /// Reverse lookup. Used by tests; production code asks the upper map
57    /// via `path_for` after promotion.
58    #[cfg(test)]
59    pub(crate) fn ino_for_path(&self, virtual_path: &Path) -> Option<u64> {
60        self.path_to_ino.get(virtual_path).copied()
61    }
62
63    /// Remap an existing inode from `old_path` to `new_path`.
64    /// The kernel expects the source inode to remain valid after rename,
65    /// just pointing at the new path.
66    pub(crate) fn rename_path(&mut self, old_path: &Path, new_path: PathBuf) {
67        // Clean up any existing inode at the destination (overwrite case):
68        // orphan its slot rather than shifting the Vec, so other inodes keep
69        // their indices.
70        if let Some(old_dst_ino) = self.path_to_ino.remove(&new_path)
71            && let Some(slot) = self.slot_mut(old_dst_ino)
72        {
73            *slot = None;
74        }
75        // Remap source inode to destination path
76        if let Some(ino) = self.path_to_ino.remove(old_path) {
77            if let Some(slot) = self.slot_mut(ino) {
78                *slot = Some(new_path.clone());
79            }
80            self.path_to_ino.insert(new_path, ino);
81        }
82    }
83
84    /// Mutable access to the reverse-map slot for an upper inode, if in range.
85    fn slot_mut(&mut self, ino: u64) -> Option<&mut Option<PathBuf>> {
86        let idx = ino.checked_sub(UPPER_INODE_BASE)? as usize;
87        self.ino_to_path.get_mut(idx)
88    }
89}
90
91/// Result of resolving a kernel-visible inode to its current backing layer.
92pub(crate) enum ResolvedIno {
93    /// File is in the upper (overlay) layer at this virtual path.
94    Upper(PathBuf),
95    /// File is in the lower (read-only) layer with this inode and virtual path.
96    Lower(u64, PathBuf),
97}
