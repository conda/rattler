1//! `PrjFileNameCompare`-compatible file-name ordering for `ProjFS` enumeration.
2//!
3//! Windows Projected File System (`ProjFS`) requires the entries returned from a
4//! directory-enumeration callback to be sorted in the exact order the
5//! filesystem itself uses — the order defined by the Win32 `PrjFileNameCompare`
6//! API. If a provider returns entries in any other order (for example plain
7//! Rust `str`/UTF-8 byte order), `ProjFS`'s streaming enumeration can skip or
8//! miss-report entries: it walks the provider's list assuming it is already in
9//! `PrjFileNameCompare` order and stops matching as soon as that assumption is
10//! violated. See issue #2581.
11//!
12//! `PrjFileNameCompare` performs a **case-insensitive, locale-independent
13//! (ordinal) comparison of UTF-16 code units**, mirroring
14//! `RtlCompareUnicodeString(.., CaseInSensitive = TRUE)` — the same ordering
15//! NTFS uses for its directory index. This differs from Rust's `str::cmp`
16//! (UTF-8 byte order) in two important ways:
17//!
18//! 1. **Case-insensitivity.** `str::cmp` orders every uppercase ASCII letter
19//!    before every lowercase one (`'Z'` = 0x5A < `'a'` = 0x61), so it sorts
20//!    `"Zebra" < "apple"`. `PrjFileNameCompare` upper-cases both operands
21//!    first, so it sorts `"apple" < "Zebra"`.
22//! 2. **Comparison unit.** The comparison is over upper-cased UTF-16 code
23//!    units, not raw UTF-8 bytes. For the Basic Multilingual Plane, UTF-16
24//!    code-unit order equals Unicode scalar order equals UTF-8 byte order, so
25//!    this only diverges for supplementary-plane characters (encoded as
26//!    surrogate pairs, code units 0xD800–0xDFFF, which sort *before* the
27//!    0xE000–0xFFFF range in UTF-16 but *after* it in scalar/UTF-8 order).
28//!
29//! Because `ProjFS` is Windows-only, this comparator only ever *feeds* `ProjFS`
30//! on Windows. It is nevertheless implemented as a platform-independent
31//! pure-Rust function (no `windows` dependency, not `cfg(windows)`-gated) for
32//! two reasons: it keeps the enumeration sort as a single, testable source of
33//! truth that runs identically everywhere, and it lets the unit tests below run
34//! on any host (including the macOS/Linux CI that cannot compile the `ProjFS`
35//! adapter). A `#[cfg(windows)]` test asserts the pure-Rust ordering agrees
36//! with the real `PrjFileNameCompare` on a Windows host.
37//!
38//! ## Known divergences from the real `PrjFileNameCompare`
39//!
40//! The upper-casing here uses Rust's Unicode simple (1:1) uppercase mapping,
41//! restricted to mappings that stay within a single BMP code unit. The real
42//! API uses Windows' internal NLS/`$UpCase` table, which:
43//!
44//! * can differ from the Unicode mapping for a small number of code points and
45//!   may vary slightly between Windows versions;
46//! * never *expands* a character during upper-casing (e.g. `'ß'` U+00DF stays
47//!   `'ß'` rather than becoming `"SS"`). We match this by only applying
48//!   mappings that produce exactly one BMP code unit and leaving everything
49//!   else unchanged.
50//!
51//! For the ASCII-dominated filename space of conda packages (letters, digits,
52//! `.`, `-`, `_`, `+`, `~`) the two orderings are identical.
53
54use std::cmp::Ordering;
55
56/// Upper-case a single UTF-16 code unit the way `PrjFileNameCompare` /
57/// `RtlUpcaseUnicodeChar` conceptually does: a 1:1 map that never expands.
58///
59/// * ASCII `a`–`z` map to `A`–`Z` (fast path, and exact).
60/// * Surrogate-range units (0xD800–0xDFFF) are returned unchanged — upper-casing
61///   is applied per code unit and surrogate halves have no case.
62/// * Other BMP units use Rust's Unicode simple uppercase, but only when it
63///   yields exactly one code unit that still fits in the BMP; otherwise the
64///   unit is left unchanged (matching NTFS, which never expands during upcase).
65fn upcase_u16(unit: u16) -> u16 {
66    if unit < 0x80 {
67        // ASCII fast path.
68        if (u16::from(b'a')..=u16::from(b'z')).contains(&unit) {
69            return unit - 32;
70        }
71        return unit;
72    }
73    if (0xD800..=0xDFFF).contains(&unit) {
74        // Lone surrogate code unit: no case mapping.
75        return unit;
76    }
77    if let Some(c) = char::from_u32(u32::from(unit)) {
78        let mut upper = c.to_uppercase();
79        match (upper.next(), upper.next()) {
80            // Exactly one resulting char, and it still fits in a single UTF-16
81            // code unit (i.e. it is itself in the BMP). This rejects both
82            // expansions (e.g. 'ß' -> "SS") and any mapping that would leave
83            // the BMP, matching the non-expanding NTFS/ProjFS upcase.
84            (Some(u), None) if u as u32 <= 0xFFFF => return u as u16,
85            _ => return unit,
86        }
87    }
88    unit
89}

91/// Compare two file names using `PrjFileNameCompare` semantics: a
92/// case-insensitive, ordinal comparison over upper-cased UTF-16 code units.
93///
94/// This is the ordering `ProjFS` requires for directory-enumeration results.
95/// See the [module docs](self) for the precise semantics and the known
96/// divergences from the real Win32 API.
97pub fn prj_file_name_compare(a: &str, b: &str) -> Ordering {
98    let mut a_units = a.encode_utf16();
99    let mut b_units = b.encode_utf16();
100    loop {
101        match (a_units.next(), b_units.next()) {
102            (Some(x), Some(y)) => {
103                let ord = upcase_u16(x).cmp(&upcase_u16(y));
104                if ord != Ordering::Equal {
105                    return ord;
106                }
107            }
108            // The shorter name sorts first once it runs out of code units.
109            (Some(_), None) => return Ordering::Greater,
110            (None, Some(_)) => return Ordering::Less,
111            (None, None) => return Ordering::Equal,
112        }
113    }
114}

116#[cfg(test)]
117mod tests {

119    use std::cmp::Ordering;

121    #[test]
122    fn case_insensitive_equality() {
123        assert_eq!(prj_file_name_compare("README", "readme"), Ordering::Equal);
124        assert_eq!(prj_file_name_compare("ReadMe", "rEADmE"), Ordering::Equal);
125        assert_eq!(prj_file_name_compare("ABC", "abc"), Ordering::Equal);
126        // Case-insensitive, so these are Equal even though str::cmp would not be.
127        assert_ne!("README".cmp("readme"), Ordering::Equal);
128    }
129
130    #[test]
131    fn case_insensitive_ordering_differs_from_str_cmp() {
132        // str::cmp puts uppercase before lowercase: 'B' (0x42) < 'a' (0x61),
133        // so "Banana" < "apple" in byte order.
134        assert_eq!("Banana".cmp("apple"), Ordering::Less);
135        // PrjFileNameCompare upcases first: "APPLE" < "BANANA".
136        assert_eq!(prj_file_name_compare("apple", "Banana"), Ordering::Less);
137        assert_eq!(prj_file_name_compare("Banana", "apple"), Ordering::Greater);
138
139        // Classic case: "Zebra" vs "apple".
140        assert_eq!("Zebra".cmp("apple"), Ordering::Less); // 'Z' < 'a'
141        assert_eq!(prj_file_name_compare("Zebra", "apple"), Ordering::Greater);
142    }
143
144    #[test]
145    fn underscore_vs_letters_diverges_from_str_cmp() {
146        // '_' is 0x5F, which sits *between* uppercase 'Z' (0x5A) and lowercase
147        // 'a' (0x61). This makes '_' vs a letter order flip under upcasing.
148        //
149        // str::cmp: '_' (0x5F) < 'a' (0x61)  => "_foo" < "afoo"
150        assert_eq!("_foo".cmp("afoo"), Ordering::Less);
151        // Prj: 'a' upcases to 'A' (0x41) < '_' (0x5F) => "afoo" < "_foo"
152        assert_eq!(prj_file_name_compare("afoo", "_foo"), Ordering::Less);
153        assert_eq!(prj_file_name_compare("_foo", "afoo"), Ordering::Greater);
154
155        // But against an *uppercase* letter str::cmp already agrees with Prj,
156        // since 'A' (0x41) < '_' (0x5F).
157        assert_eq!("Afoo".cmp("_foo"), Ordering::Less);
158        assert_eq!(prj_file_name_compare("Afoo", "_foo"), Ordering::Less);
159    }

161    #[test]
162    fn separator_punctuation_is_ordinal() {
163        // No special-casing of separators: '-' (0x2D) < '.' (0x2E) < '_' (0x5F).
164        assert_eq!(prj_file_name_compare("a-b", "a.b"), Ordering::Less);
165        assert_eq!(prj_file_name_compare("a.b", "a_b"), Ordering::Less);
166        assert_eq!(prj_file_name_compare("a-b", "a_b"), Ordering::Less);
167        // '.' (0x2E) < '0' (0x30) < '+' ... check '+' (0x2B) < '-' (0x2D).
168        assert_eq!(prj_file_name_compare("a+b", "a-b"), Ordering::Less);
169        // Digits come before uppercase letters: '9' (0x39) < 'A' (0x41).
170        assert_eq!(prj_file_name_compare("9x", "ax"), Ordering::Less);
171    }

173    #[test]
174    fn digits_are_lexicographic_not_numeric() {
175        // No natural/numeric sort: "file10" sorts before "file2" because

183    fn prefix_sorts_before_longer() {
184        assert_eq!(prj_file_name_compare("abc", "abcd"), Ordering::Less);
185        assert_eq!(prj_file_name_compare("abcd", "abc"), Ordering::Greater);
186        assert_eq!(prj_file_name_compare("", "a"), Ordering::Less);
187        assert_eq!(prj_file_name_compare("", ""), Ordering::Equal);

189        assert_eq!(prj_file_name_compare("ABC", "abcd"), Ordering::Less);

192    #[test]
193    fn full_enumeration_sort_differs_from_str_cmp() {
194        // A directory listing resembling a conda env, in scrambled order.
195        let mut prj_sorted = vec![
196            "Scripts",
197            "activate",
198            "_conda",
199            "Library",
200            "conda.exe",
201            "LICENSE",
202            "README.md",
203            "bin",
204            "etc",
205        ];
206        let mut str_sorted = prj_sorted.clone();
207
208        prj_sorted.sort_by(|a, b| prj_file_name_compare(a, b));
209        str_sorted.sort();
210
211        // PrjFileNameCompare (case-insensitive): '_' (0x5F) is greater than any
212        // letter once letters are upcased, so "_conda" sorts last.
213        assert_eq!(
214            prj_sorted,
215            vec![
216                "activate",
217                "bin",
218                "conda.exe",
219                "etc",
220                "Library",
221                "LICENSE",
222                "README.md",
223                "Scripts",
224                "_conda",
225            ]
226        );
227
228        // Plain str::cmp is byte order: all capitalized names come first, and
229        // "_conda" lands among/after them (0x5F > uppercase, < lowercase).
230        assert_ne!(prj_sorted, str_sorted);
231    }

233    #[test]
234    fn non_ascii_bmp_case_folding() {
235        // Latin-1 accented letters fold case (1:1 within the BMP).
236        assert_eq!(prj_file_name_compare("café", "CAFÉ"), Ordering::Equal);
237        assert_eq!(prj_file_name_compare("Ünïcödé", "ünïcödé"), Ordering::Equal);
238        // 'ß' (U+00DF) has no single-unit uppercase, so it is left unchanged
239        // (we do NOT expand to "SS"); it therefore only equals itself.
240        assert_eq!(prj_file_name_compare("straße", "straße"), Ordering::Equal);
241        assert_ne!(prj_file_name_compare("straße", "STRASSE"), Ordering::Equal);
242    }

244    /// On a real Windows host, assert our pure-Rust ordering agrees with the
245    /// authoritative Win32 `PrjFileNameCompare` for a battery of names. This is
246    /// the check that guarantees the enumeration sort matches what ProjFS
247    /// expects; it cannot run on non-Windows CI.
248    #[cfg(windows)]
249    #[test]
250    fn agrees_with_win32_prj_file_name_compare() {
251        use std::os::windows::ffi::OsStrExt;
252        use windows::Win32::Storage::ProjectedFileSystem::PrjFileNameCompare;
253        use windows::core::PCWSTR;
254
255        fn wide(s: &str) -> Vec<u16> {
256            std::ffi::OsStr::new(s)
257                .encode_wide()
258                .chain(Some(0))
259                .collect()
260        }
261        fn win32(a: &str, b: &str) -> Ordering {
262            let aw = wide(a);
263            let bw = wide(b);
264            let r = unsafe { PrjFileNameCompare(PCWSTR(aw.as_ptr()), PCWSTR(bw.as_ptr())) };
265            r.cmp(&0)
266        }
267
268        let names = [
269            "activate",
270            "Activate",
271            "ACTIVATE",
272            "_conda",
273            "conda.exe",
274            "Library",
275            "LICENSE",
276            "README.md",
277            "Scripts",
278            "bin",
279            "etc",
280            "a-b",
281            "a.b",
282            "a_b",
283            "a+b",
284            "file2",
285            "file10",
286            "café",
287            "CAFÉ",
288            "straße",
289            "python3.11",
290            "Python3.11",
291            "",
292        ];
293        for &a in &names {
294            for &b in &names {
295                assert_eq!(
296                    prj_file_name_compare(a, b),
297                    win32(a, b),
298                    "divergence for {a:?} vs {b:?}"
299                );
300            }
301        }
302    }
