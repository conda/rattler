13/// Error returned by [`VfsOps`] operations.
14///
15/// A small, transport-agnostic error enum rather than a bare `errno` `i32`, so
16/// the VFS layer names its failure modes explicitly. Transport adapters convert
17/// it to their own wire error (NFS `nfsstat3`, FUSE `Errno`, `ProjFS` `HRESULT`)
18/// — usually via [`VfsError::errno`] / `From<VfsError> for i32`.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum VfsError {
21    /// No such file or directory (`ENOENT`).
22    NotFound,
23    /// Permission denied (`EACCES`).
24    PermissionDenied,
25    /// Not a directory (`ENOTDIR`).
26    NotADirectory,
27    /// Is a directory (`EISDIR`).
28    IsADirectory,
29    /// Read-only filesystem (`EROFS`).
30    ReadOnly,
31    /// File exists (`EEXIST`).
32    AlreadyExists,
33    /// Directory not empty (`ENOTEMPTY`).
34    NotEmpty,
35    /// No space left on device (`ENOSPC`).
36    NoSpace,
37    /// Invalid argument (`EINVAL`) — e.g. an unsafe wire filename.
38    InvalidArgument,
39    /// Operation not permitted (`EPERM`).
40    NotPermitted,
41    /// Catch-all I/O error (`EIO`).
42    Io,
43}
44
45impl VfsError {
46    /// The POSIX `errno` this error maps to.
47    pub fn errno(self) -> i32 {
48        match self {
49            Self::NotFound => libc::ENOENT,
50            Self::PermissionDenied => libc::EACCES,
51            Self::NotADirectory => libc::ENOTDIR,
52            Self::IsADirectory => libc::EISDIR,
53            Self::ReadOnly => libc::EROFS,
54            Self::AlreadyExists => libc::EEXIST,
55            Self::NotEmpty => libc::ENOTEMPTY,
56            Self::NoSpace => libc::ENOSPC,
57            Self::InvalidArgument => libc::EINVAL,
58            Self::NotPermitted => libc::EPERM,
59            Self::Io => libc::EIO,
60        }
61    }
62
63    /// Map a POSIX `errno` back to a [`VfsError`], collapsing anything unknown
64    /// to [`VfsError::Io`].
65    pub fn from_errno(errno: i32) -> Self {
66        match errno {
67            libc::ENOENT => Self::NotFound,
68            libc::EACCES => Self::PermissionDenied,
69            libc::ENOTDIR => Self::NotADirectory,
70            libc::EISDIR => Self::IsADirectory,
71            libc::EROFS => Self::ReadOnly,
72            libc::EEXIST => Self::AlreadyExists,
73            libc::ENOTEMPTY => Self::NotEmpty,
74            libc::ENOSPC => Self::NoSpace,
75            libc::EINVAL => Self::InvalidArgument,
76            libc::EPERM => Self::NotPermitted,
77            _ => Self::Io,
78        }
79    }
80}
81
82impl From<VfsError> for i32 {
83    fn from(e: VfsError) -> Self {
84        e.errno()
85    }
86}
87
88impl std::fmt::Display for VfsError {
89    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
90        write!(f, "{self:?} (errno {})", self.errno())
91    }
92}

99/// An opaque write-handle token returned by [`VfsOps::open_write`] /
100/// [`VfsOps::create`] and passed back to [`VfsOps::write`],
101/// [`VfsOps::read_handle`], and [`VfsOps::release_write`].
102///
103/// A newtype rather than a bare `u64` so a file handle can't be silently
104/// swapped for an inode (both would otherwise be `u64`) — the NFS adapter in
105/// particular keeps an inode→handle map where mixing the two would be a subtle
106/// bug. Inodes are deliberately left as `u64`: they're used as array indices,
107/// map keys, and in the overlay's `UPPER_INODE_BASE` partition arithmetic,
108/// where a newtype would add noise without preventing a real error.
109#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
110pub struct Fh(pub u64);
111
112impl Fh {
113    /// The underlying handle value.
114    pub fn get(self) -> u64 {
115        self.0
116    }
117}
118
119impl From<u64> for Fh {
120    fn from(v: u64) -> Self {
121        Self(v)
122    }
123}
124
125impl From<Fh> for u64 {
126    fn from(fh: Fh) -> Self {
127        fh.0
128    }
129}
130
131impl std::fmt::Display for Fh {
132    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
133        write!(f, "{}", self.0)
134    }
135}

145/// File attributes — transport-agnostic equivalent of `fuser::FileAttr`.
146#[derive(Debug, Clone)]
147pub struct FileAttr {
148    pub ino: u64,
149    pub size: u64,
150    pub blocks: u64,
151    pub atime: SystemTime,
152    pub mtime: SystemTime,
153    pub ctime: SystemTime,
154    pub kind: FileKind,
155    pub perm: u16,
156    pub nlink: u32,
157    pub uid: u32,
158    pub gid: u32,
159}
160
161impl FileAttr {
162    /// Build a `FileAttr` from OS filesystem metadata, abstracting away
163    /// platform differences (blocks, ctime, permissions, uid/gid).
164    pub fn from_metadata(metadata: &std::fs::Metadata, ino: u64) -> Self {
165        #[cfg(unix)]
166        let (blocks, ctime, perm, uid, gid) = {
167            use std::os::unix::fs::{MetadataExt, PermissionsExt};
168            use std::time::Duration;
169            (
170                metadata.blocks(),
171                UNIX_EPOCH + Duration::new(metadata.ctime() as u64, metadata.ctime_nsec() as u32),
172                (metadata.permissions().mode() & 0o777) as u16,
173                metadata.uid(),
174                metadata.gid(),
175            )
176        };
177        #[cfg(not(unix))]
178        let (blocks, ctime, perm, uid, gid) = {
179            (
180                (metadata.len() + 511) / 512,
181                metadata.modified().unwrap_or(UNIX_EPOCH),
182                if metadata.is_dir() {
183                    0o755u16
184                } else {
185                    0o644u16
186                },
187                0u32,
188                0u32,
189            )
190        };
191
192        let kind = if metadata.is_dir() {
193            FileKind::Directory
194        } else if metadata.is_symlink() {
195            FileKind::Symlink
196        } else {
197            FileKind::RegularFile
198        };
199
200        FileAttr {
201            ino,
202            size: metadata.len(),
203            blocks,
204            atime: metadata.accessed().unwrap_or(UNIX_EPOCH),
205            mtime: metadata.modified().unwrap_or(UNIX_EPOCH),
206            ctime,
207            kind,
208            perm,
209            nlink: 1,
210            uid,
211            gid,
212        }
213    }
214}

216/// Get the current user's UID and GID. Returns `(0, 0)` on non-Unix platforms.
217pub fn current_uid_gid() -> (u32, u32) {
218    #[cfg(unix)]
219    {
220        unsafe { (libc::getuid(), libc::getgid()) }
221    }
222    #[cfg(not(unix))]
223    {
224        (0, 0)
225    }
226}

228/// Set Unix file permissions cross-platform. On Windows, maps the write bits
229/// to the read-only flag.
230pub fn set_file_permissions(path: &Path, mode: u32) -> std::io::Result<()> {
231    #[cfg(unix)]
232    {
233        use std::os::unix::fs::PermissionsExt;
234        std::fs::set_permissions(path, std::fs::Permissions::from_mode(mode))
235    }
236    #[cfg(not(unix))]
237    {
238        let _ = mode;
239        let readonly = mode & 0o222 == 0;
240        let mut perms = std::fs::metadata(path)?.permissions();
241        perms.set_readonly(readonly);
242        std::fs::set_permissions(path, perms)
243    }
244}

268/// Transport-agnostic filesystem operations.
269///
270/// Read operations are required. Write operations default to `EROFS` (read-only
271/// filesystem), allowing read-only implementations to skip them.
272pub trait VfsOps: Send + Sync + 'static {
273    fn lookup(&self, parent: u64, name: &OsStr) -> VfsResult<FileAttr>;
274    fn getattr(&self, ino: u64) -> VfsResult<FileAttr>;
275    fn readlink(&self, ino: u64) -> VfsResult<PathBuf>;
276
277    /// Read bytes from a file at the given offset. The VFS handles prefix
278    /// replacement, codesign, and passthrough transparently.
279    fn read(&self, ino: u64, offset: u64, size: u32) -> VfsResult<Vec<u8>>;
280
281    /// Hint about how the adapter should serve this file's content.
282    fn content_source(&self, ino: u64) -> VfsResult<ContentSource>;
283
284    fn readdir(&self, ino: u64, offset: u64) -> VfsResult<Vec<DirEntry>>;
285
286    /// Resolve an inode to its virtual path (relative to root).
287    /// Used by the overlay to map lower inodes to paths for whiteout checks.
288    fn ino_to_path(&self, ino: u64) -> VfsResult<PathBuf> {
289        let _ = ino;
290        Err(VfsError::NotFound)
291    }
292
293    // Write operations — default to read-only (`EROFS`) for read-only impls.
294
295    /// Open a file for writing. Returns a write handle.
296    fn open_write(&self, _ino: u64) -> VfsResult<Fh> {
297        Err(VfsError::ReadOnly)
298    }
299    /// Read from a write handle (for files currently open for writing).
300    fn read_handle(&self, _fh: Fh, _offset: u64, _size: u32) -> VfsResult<Vec<u8>> {
301        Err(VfsError::ReadOnly)
302    }
303    fn write(&self, _fh: Fh, _offset: u64, _data: &[u8]) -> VfsResult<u32> {
304        Err(VfsError::ReadOnly)
305    }
306    fn release_write(&self, _fh: Fh) {}
307
308    fn create(&self, _parent: u64, _name: &OsStr, _mode: u32) -> VfsResult<(FileAttr, Fh)> {
309        Err(VfsError::ReadOnly)
310    }
311    fn unlink(&self, _parent: u64, _name: &OsStr) -> VfsResult<()> {
312        Err(VfsError::ReadOnly)
313    }
314    fn mkdir(&self, _parent: u64, _name: &OsStr, _mode: u32) -> VfsResult<FileAttr> {
315        Err(VfsError::ReadOnly)
316    }
317    fn rmdir(&self, _parent: u64, _name: &OsStr) -> VfsResult<()> {
318        Err(VfsError::ReadOnly)
319    }
320    fn rename(
321        &self,
322        _parent: u64,
323        _name: &OsStr,
324        _newparent: u64,
325        _newname: &OsStr,
326        _flags: u32,
327    ) -> VfsResult<()> {
328        Err(VfsError::ReadOnly)
329    }
330    fn setattr(&self, _ino: u64, _size: Option<u64>, _mode: Option<u32>) -> VfsResult<FileAttr> {
331        Err(VfsError::ReadOnly)
332    }
