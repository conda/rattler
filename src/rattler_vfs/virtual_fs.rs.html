22/// Compare a directory entry's name against a lookup name. Case-sensitive on
23/// Unix; case-insensitive on Windows, where NTFS/ProjFS resolve paths
24/// case-insensitively (a lookup for `Lib\\Foo` must find the stored `lib/foo`).
25/// Conda paths are ASCII, so ASCII case folding is sufficient.
26fn names_match(entry_name: &OsStr, lookup_name: &OsStr) -> bool {
27    #[cfg(windows)]
28    {
29        entry_name
30            .to_string_lossy()
31            .eq_ignore_ascii_case(&lookup_name.to_string_lossy())
32    }
33    #[cfg(not(windows))]
34    {
35        entry_name == lookup_name
36    }
37}

39/// Target-prefix string spliced into prefix-placeholder files, mirroring
40/// `rattler::install::link`. On Windows targets backslashes become forward
41/// slashes (as the installer does, so patched text files stay validly escaped);
42/// elsewhere the mount point is used verbatim. Branches on `platform` (not
43/// `cfg!`) so it's testable on any host.
44pub(crate) fn target_prefix_for_platform(mount_point: &Path, platform: Subdir) -> String {
45    let raw = mount_point.to_string_lossy();
46    if platform.is_windows() {
47        raw.replace('\\', "/")
48    } else {
49        raw.into_owned()
50    }
51}

53/// Bounded cache keyed by inode, evicting entries to stay within `max_entries`.
54///
55/// The eviction policy is chosen at construction via `touch_on_get`:
56/// - `false` (FIFO): insertion order is preserved and a `get` never reorders.
57///   Used for the codesign cache, where each entry is a whole re-signed binary
58///   and an unbounded map would pin the entire binary set in memory.
59/// - `true` (LRU): a `get` marks the entry most-recently-used, so the entry a
60///   caller is actively hitting is never the one evicted. Used for the mmap
61///   cache, so streaming one file back-to-back never evicts its own mapping.
62///
63/// In both cases re-inserting an existing key updates the value in place
64/// without growing the cache or creating a duplicate order entry, and an
65/// evicted entry is simply recomputed on the next access. For the mmap cache
66/// the value is an [`Arc<Mmap>`], which keeps the file mapped by the kernel on
67/// its own — a cached entry holds no Rust `File` and leaks no descriptor, so
68/// the bound only caps virtual-address use for very large environments.
69///
70/// Every entry carries a logical timestamp (set on insert, and refreshed by
71/// `get` under LRU), so a hit is `O(1)`; only an insert that overflows the
72/// bound pays an `O(n)` scan for the oldest entry. The mmap cache is hit on
73/// every read, so hits must not scan the cache.
74struct BoundedCache<V> {
75    map: HashMap<u64, (V, u64)>,
76    tick: u64,
77    max_entries: usize,
78    touch_on_get: bool,
79}
80
81impl<V> BoundedCache<V> {
82    fn new(max_entries: usize, touch_on_get: bool) -> Self {
83        Self {
84            map: HashMap::new(),
85            tick: 0,
86            max_entries,
87            touch_on_get,
88        }
89    }
90
91    fn next_tick(&mut self) -> u64 {
92        self.tick += 1;
93        self.tick
94    }
95
96    fn get(&mut self, ino: u64) -> Option<&V> {
97        let tick = if self.touch_on_get {
98            Some(self.next_tick())
99        } else {
100            None
101        };
102        let (value, stamp) = self.map.get_mut(&ino)?;
103        if let Some(tick) = tick {
104            *stamp = tick;
105        }
106        Some(value)
107    }
108
109    fn clear(&mut self) {
110        self.map.clear();
111    }
112
113    fn insert(&mut self, ino: u64, value: V) {
114        // Re-inserting an existing key updates the value in place and keeps
115        // its position (FIFO) or marks it as most recently used (LRU).
116        if let Some((existing, stamp)) = self.map.get_mut(&ino) {
117            *existing = value;
118            if self.touch_on_get {
119                self.tick += 1;
120                *stamp = self.tick;
121            }
122            return;
123        }
124        let tick = self.next_tick();
125        self.map.insert(ino, (value, tick));
126        if self.map.len() > self.max_entries {
127            // Evict the oldest eighth in one go rather than just the single
128            // oldest entry: finding the oldest is a scan of the whole cache,
129            // and a workload streaming more files than the cache holds would
130            // otherwise pay that scan on every insert.
131            let evict = self.map.len() - self.max_entries + self.max_entries / 8;
132            let mut stamps: Vec<(u64, u64)> = self
133                .map
134                .iter()
135                .map(|(&ino, &(_, stamp))| (stamp, ino))
136                .collect();
137            stamps.select_nth_unstable(evict - 1);
138            for &(_, ino) in &stamps[..evict] {
139                self.map.remove(&ino);
140            }
141        }
142    }
143}

145/// Read the first `n` bytes of a file (fewer when the file is shorter).
146///
147/// Used to load just the shebang region during plan construction; `n` comes
148/// from the recorded `shebang_length`, so it is bounded by `take` rather than
149/// pre-allocated in case the metadata is nonsense.
150fn read_leading_bytes(path: &Path, n: usize) -> std::io::Result<Vec<u8>> {
151    let mut buf = Vec::new();
152    File::open(path)?.take(n as u64).read_to_end(&mut buf)?;
153    Ok(buf)
154}

156/// How many backing files [`VirtualFS`] keeps open: a small share of the
157/// process's descriptor limit (macOS defaults to a soft limit of 256), and no
158/// more than 256.
159fn file_cache_capacity() -> usize {
160    #[cfg(unix)]
161    {
162        let mut limit = libc::rlimit {
163            rlim_cur: 0,
164            rlim_max: 0,
165        };
166        // SAFETY: `getrlimit` only writes to the struct we pass.
167        if unsafe { libc::getrlimit(libc::RLIMIT_NOFILE, &mut limit) } == 0 {
168            let soft = usize::try_from(limit.rlim_cur).unwrap_or(usize::MAX);
169            return (soft / 8).clamp(8, 256);
170        }
171        32
172    }
173    #[cfg(not(unix))]
174    {
175        256
176    }
177}

209    /// (exactly as the installer does) and the remaining occurrences are body

217pub struct VirtualFS {
218    metadata: Vec<MetadataNode>,
219    #[cfg_attr(not(target_os = "macos"), allow(dead_code))]
220    platform: Subdir,
221    uid: u32,
222    gid: u32,
223    /// Pre-computed replacement plans for files with prefix placeholders.
224    /// Keyed by inode. Populated eagerly at construction from paths.json
225    /// offsets or by scanning the source file.
226    offset_cache: HashMap<u64, ReplacementPlan>,
227    /// Cache for fully materialized + codesigned binary content (macOS only).
228    /// Keyed by inode. Only populated for binary-mode prefix files that need
229    /// ad-hoc re-signing, since codesign requires the full file.
230    #[cfg_attr(not(target_os = "macos"), allow(dead_code))]
231    codesign_cache: Mutex<BoundedCache<Vec<u8>>>,
232    /// Cache of memory-mapped source files, keyed by inode, so streaming a
233    /// prefix-replaced file doesn't re-`open` + re-`mmap` it on every window.
234    mmap_cache: Mutex<BoundedCache<Arc<Mmap>>>,
235    /// Open backing files for plain (untransformed) reads by inode, which NFS
236    /// and FUSE without open handles issue. Read with `pread` rather than
237    /// memory-mapped: mapping and unmapping a file per read stream costs page
238    /// faults, and every unmap in a multi-threaded process interrupts all
239    /// CPUs to flush their TLBs. Bounded well below the usual descriptor
240    /// limit.
241    file_cache: Mutex<BoundedCache<Arc<File>>>,
242    /// Name → child index for directories with many children, indexed by the
243    /// directory's metadata index and built on the first lookup in that
244    /// directory (a run touches few of them). Small directories are searched
245    /// linearly. Without this, every lookup in a large directory (`lib/`,
246    /// `site-packages/`) is a linear scan, and lookups are the most frequent
247    /// operation while e.g. Python resolves imports.
248    child_maps: Vec<OnceLock<HashMap<OsString, usize>>>,
249    /// Lazily computed attributes per metadata index. The tree and the package
250    /// cache files backing it are immutable while mounted, so the `stat` of the
251    /// backing file only has to happen once per node.
252    attr_cache: Vec<OnceLock<FileAttr>>,
253    /// The target prefix spliced into placeholder files (see
254    /// [`target_prefix_for_platform`]), computed once instead of per read.
255    target_prefix: String,
256}

292    pub(crate) fn build(
293        mut metadata: Vec<MetadataNode>,
294        mount_point: &Path,
295        platform: Subdir,
296    ) -> Self {
297        let target_prefix = target_prefix_for_platform(mount_point, platform);
298
299        // Placeholder locations come from the package index (see
300        // `crate::package_index`) when the tree was built from one; take them
301        // out of the nodes, they are turned into plans below.
302        // Only the (few) files with a placeholder need a plan.
303        let work: Vec<(usize, Option<ScanResult>)> = metadata
304            .iter_mut()
305            .enumerate()
306            .filter_map(|(i, node)| {
307                let file = node.as_file_mut()?;
308                file.prefix_placeholder.as_ref()?;
309                Some((i, file.scan.take()))
310            })
311            .collect();
312        let needs_scanning = work.iter().any(|(_, scan)| scan.is_none());
313
314        // Compute replacement plans and text-mode file sizes. Without an index
315        // the placeholder is located now, which may mean reading and scanning
316        // the whole file; do that in parallel so mount time is bounded by the
317        // largest file rather than their sum. With every location known from
318        // the index a plan costs at most a short read, not worth the thread
319        // pool.
320        let plan_file = |(i, scan): (usize, Option<ScanResult>)| {
321            let file = metadata[i].as_file()?;
322            let placeholder = file.prefix_placeholder.as_ref()?;
323            let ino = (i + 1) as u64;
324
325            // Resolve the on-disk cache path, preferring cache_prefix_path
326            // (set for noarch Python files where virtual path differs from cache path).
327            let cache_path = {
328                let p = (*file.cache_base_path).to_path_buf();
329                let prefix = match &file.cache_prefix_path {
330                    Some(cp) => cp.as_path(),
331                    None => &metadata[file.parent].as_directory().unwrap().prefix_path,
332                };
333                p.join(prefix).join(&file.file_name)
334            };
335
336            let scan = match scan {
337                Some(scan) => scan,
338                None => {
339                    crate::package_index::locate_placeholder(&cache_path, placeholder, platform)?
340                }
341            };
342
343            match scan {
344                ScanResult::Binary(cstrings) => {
345                    Some((ino, ReplacementPlan::Binary(cstrings), None))
346                }
347                ScanResult::Text {
348                    region_end,
349                    occurrences,
350                } => {
351                    let source_len = match &file.stat {
352                        Some(stat) => stat.len,
353                        None => match fs::symlink_metadata(&cache_path) {
354                            Ok(m) => m.len(),
355                            Err(e) => {
356                                tracing::warn!(
357                                    "failed to stat {} for offset computation: {}",
358                                    cache_path.display(),
359                                    e
360                                );
361                                return None;
362                            }
363                        },
364                    };
365                    // The shebang region is transformed as a whole, so it is
366                    // the one part of the file that has to be read here.
367                    let region = if region_end > 0 {
368                        match read_leading_bytes(&cache_path, region_end) {
369                            Ok(region) if region.len() == region_end => region,
370                            Ok(_) | Err(_) => {
371                                tracing::warn!(
372                                    "failed to read the shebang of {}",
373                                    cache_path.display()
374                                );
375                                return None;
376                            }
377                        }
378                    } else {
379                        Vec::new()
380                    };
381                    let plan = crate::prefix_replacement::TextPlan::from_scan(
382                        &region,
383                        occurrences,
384                        &placeholder.placeholder,
385                        &target_prefix,
386                        &platform,
387                    );
388                    // Post-replacement size: the transformed shebang region plus
389                    // the unchanged body length plus the per-occurrence delta.
390                    let new_size = plan.output_len(source_len as usize) as u64;
391                    Some((ino, ReplacementPlan::Text(plan), Some(new_size)))
392                }
393            }
394        };
395        let planned: Vec<(u64, ReplacementPlan, Option<u64>)> = if needs_scanning {
396            work.into_par_iter().filter_map(plan_file).collect()
397        } else {
398            work.into_iter().filter_map(plan_file).collect()
399        };
400
401        // Apply the parallel results sequentially: insert each plan and record
402        // the transformed text-file sizes back into the metadata tree.
403        let mut offset_cache = HashMap::with_capacity(planned.len());
404        for (ino, plan, computed_size) in planned {
405            if let Some(size) = computed_size {
406                metadata[(ino - 1) as usize]
407                    .as_file_mut()
408                    .unwrap()
409                    .computed_size = Some(size);
410            }
411            offset_cache.insert(ino, plan);
412        }
413
414        let child_maps = (0..metadata.len()).map(|_| OnceLock::new()).collect();
415        let attr_cache = (0..metadata.len()).map(|_| OnceLock::new()).collect();
416        let (uid, gid) = current_uid_gid();
417
418        VirtualFS {
419            metadata,
420            platform,
421            uid,
422            gid,
423            offset_cache,
424            codesign_cache: Mutex::new(BoundedCache::new(16, false)),
425            mmap_cache: Mutex::new(BoundedCache::new(1024, true)),
426            file_cache: Mutex::new(BoundedCache::new(file_cache_capacity(), true)),
427            child_maps,
428            attr_cache,
429            target_prefix,
430        }
431    }

433    /// The metadata index of the child of directory `parent_index` named
434    /// `name`, if any.
435    fn find_child(&self, parent_index: usize, name: &OsStr) -> Result<Option<usize>, VfsError> {
436        let Some(parent_directory) = self.metadata[parent_index].as_directory() else {
437            return Err(VfsError::NotADirectory);
438        };
439        if parent_directory.children.len() > CHILD_MAP_THRESHOLD {
440            let map = self.child_maps[parent_index].get_or_init(|| {
441                let mut map = HashMap::with_capacity(parent_directory.children.len());
442                for &child in &parent_directory.children {
443                    // Keep the first match, like the linear scan does.
444                    map.entry(child_key(self.metadata[child].file_name()).into_owned())
445                        .or_insert(child);
446                }
447                map
448            });
449            return Ok(map.get::<OsStr>(&child_key(name)).copied());
450        }
451        Ok(parent_directory
452            .children
453            .iter()
454            .copied()
455            .find(|&child| names_match(self.metadata[child].file_name(), name)))
456    }

506    fn compute_attr(&self, index: usize) -> FileAttr {
507        let ino = (index + 1) as u64;
508        match &self.metadata[index] {
509            MetadataNode::Directory(_) => self.make_attr(ino, 0, FileKind::Directory, 0o755),
510            MetadataNode::File(file) => {
511                if let Some(ref content) = file.virtual_content {
512                    return self.make_attr(ino, content.len() as u64, FileKind::RegularFile, 0o775);
513                }
514
515                // Attributes recorded in the package index, else a `stat`.
516                let attr = match &file.stat {
517                    Some(stat) => Ok(FileAttr {
518                        ino,
519                        size: stat.len,
520                        blocks: stat.blocks,
521                        atime: stat.mtime,
522                        mtime: stat.mtime,
523                        ctime: stat.ctime,
524                        kind: stat.kind,
525                        perm: stat.perm,
526                        nlink: 1,
527                        uid: stat.uid,
528                        gid: stat.gid,
529                    }),
530                    None => fs::symlink_metadata(self._getpath(file))
531                        .map(|metadata| FileAttr::from_metadata(&metadata, ino)),
532                };
533                match attr {
534                    Ok(mut attr) => {
535                        // Override size if prefix replacement changes the file length
536                        if let Some(computed) = file.computed_size {
537                            attr.size = computed;
538                        }
539                        attr
540                    }
541                    Err(e) => {
542                        tracing::warn!("failed to stat {}: {}", self._getpath(file).display(), e);
543                        self.make_attr(ino, 0, FileKind::RegularFile, 0o644)
544                    }
545                }
546            }
547        }
548    }

577    pub(crate) fn do_content_source(&self, ino: u64) -> Result<ContentSource, VfsError> {
578        let index = self.validate_ino(ino)?;
579
580        let Some(current_file) = self.metadata[index].as_file() else {
581            return Err(VfsError::NotFound); // directories don't have readable content
582        };
583
584        if current_file.path_type == PathType::SoftLink {
585            return Err(VfsError::NotFound); // symlinks don't have readable content
586        }
587
588        if current_file.virtual_content.is_some() {
589            return Ok(ContentSource::Virtual);
590        }
591
592        if current_file.prefix_placeholder.is_some() {
593            return Ok(ContentSource::Transformed);
594        }
595
596        let path = self._getpath(current_file);
597        Ok(ContentSource::Direct(path))
598    }

600    /// Memory-map the source file for `ino`, reusing a cached mapping when one
601    /// exists. On a miss the file is mapped once and the mapping cached; a
602    /// concurrent miss on the same inode simply maps twice and keeps whichever
603    /// mapping the cache records — both are valid views of an immutable cache
604    /// file.
605    fn mmap_for(&self, ino: u64, path: &Path) -> Result<Arc<Mmap>, VfsError> {
606        if let Some(mmap) = self.mmap_cache.lock().unwrap().get(ino).cloned() {
607            return Ok(mmap);
608        }
609        let file = File::open(path).map_err(|e| {
610            tracing::warn!("failed to open {}: {}", path.display(), e);
611            VfsError::Io
612        })?;
613        let mmap = Arc::new(unsafe { Mmap::map(&file) }.map_err(|e| {
614            tracing::warn!("failed to memory map {}: {}", path.display(), e);
615            VfsError::Io
616        })?);
617        self.mmap_cache.lock().unwrap().insert(ino, mmap.clone());
618        Ok(mmap)
619    }

621    /// The open backing file of `ino`, reusing a cached descriptor.
622    fn file_for(&self, ino: u64, node: &FileNode) -> Result<Arc<File>, VfsError> {
623        if let Some(file) = self.file_cache.lock().unwrap().get(ino).cloned() {
624            return Ok(file);
625        }
626        let path = self._getpath(node);
627        let file = match File::open(&path) {
628            // Out of descriptors (the process limit is shared with whatever
629            // else the host process does): give back the cached ones and try
630            // once more before failing the read.
631            Err(e) if is_out_of_descriptors(&e) => {
632                self.file_cache.lock().unwrap().clear();
633                File::open(&path)
634            }
635            result => result,
636        };
637        let file = Arc::new(file.map_err(|e| {
638            tracing::warn!("failed to open {}: {}", path.display(), e);
639            VfsError::Io
640        })?);
641        self.file_cache.lock().unwrap().insert(ino, file.clone());
642        Ok(file)
643    }

645    pub(crate) fn do_read(&self, ino: u64, offset: u64, size: u32) -> Result<Vec<u8>, VfsError> {
646        let index = self.validate_ino(ino)?;
647
648        let Some(current_file) = self.metadata[index].as_file() else {
649            return Ok(vec![]); // directories
650        };
651
652        if current_file.path_type == PathType::SoftLink {
653            return Ok(vec![]); // symlinks
654        }
655
656        // Virtual files (e.g. entry points) are served directly from memory
657        if let Some(ref content) = current_file.virtual_content {
658            let start = (offset as usize).min(content.len());
659            let end = (start + size as usize).min(content.len());
660            return Ok(content[start..end].to_vec());
661        }
662
663        // Files without prefix replacement: read the source bytes from a
664        // cached open file. Transports that address reads by inode (NFS, FUSE
665        // without opens) issue many reads per file; re-opening the file for
666        // each one costs an `open` + `close` per request.
667        if current_file.prefix_placeholder.is_none() {
668            let file = self.file_for(ino, current_file)?;
669            let mut buf = vec![0u8; size as usize];
670            let mut filled = 0;
671            while filled < buf.len() {
672                match read_at(&file, &mut buf[filled..], offset + filled as u64) {
673                    Ok(0) => break,
674                    Ok(n) => filled += n,
675                    Err(e) if e.kind() == std::io::ErrorKind::Interrupted => {}
676                    Err(e) => {
677                        tracing::warn!(
678                            "failed to read {}: {}",
679                            self._getpath(current_file).display(),
680                            e
681                        );
682                        return Err(VfsError::Io);
683                    }
684                }
685            }
686            buf.truncate(filled);
687            return Ok(buf);
688        }
689
690        let path = self._getpath(current_file);
691
692        // Has prefix placeholder — use ranged replacement
693        let placeholder = current_file.prefix_placeholder.as_ref().unwrap();
694
695        let mmap = self.mmap_for(ino, &path)?;
696
697        // Must match the prefix used to build the plan in `with_platform`.
698        let prefixes = crate::prefix_replacement::EncodedPrefixes::new(
699            &placeholder.placeholder,
700            &self.target_prefix,
701        );
702
703        let start = offset as usize;
704        // `offset` is client-supplied and unclamped here; the ranged reads
705        // clamp `end` to the transformed length, but the addition itself must
706        // not wrap for a pathological offset.
707        let end = start.saturating_add(size as usize);
708
709        let Some(plan) = self.offset_cache.get(&ino) else {
710            // No plan — serve source bytes directly
711            let s = start.min(mmap.len());
712            let e = (s + size as usize).min(mmap.len());
713            return Ok(mmap[s..e].to_vec());
714        };
715
716        match plan {
717            ReplacementPlan::Binary(cstrings) => {
718                // macOS binaries need codesign after prefix replacement.
719                // Codesign rehashes every page so it can't be done as a ranged
720                // operation. Materialize + resign once, cache for subsequent reads.
721                // The codesign module is compiled only on macOS — other targets
722                // fall through to `binary_ranged_read` directly. With no
723                // occurrences to replace the bytes are served verbatim and the
724                // original signature stays valid, so the codesign path is
725                // skipped too.
726                #[cfg(target_os = "macos")]
727                if self.platform.is_osx() && !cstrings.is_empty() {
728                    // Fast path: serve from cache
729                    if let Some(cached) = self.codesign_cache.lock().unwrap().get(ino) {
730                        let s = start.min(cached.len());
731                        let e = (s + size as usize).min(cached.len());
732                        return Ok(cached[s..e].to_vec());
733                    }
734
735                    // Slow path: materialize, resign, cache. The whole-file
736                    // ranged read applies the same plan as every other read,
737                    // in every encoding the plan holds.
738                    let mut output = crate::prefix_replacement::binary_ranged_read(
739                        &mmap,
740                        &prefixes,
741                        cstrings,
742                        0,
743                        mmap.len(),
744                    );
745
746                    if let Err(e) = crate::codesign::adhoc_resign(&mut output) {
747                        tracing::warn!("ad-hoc re-signing failed for {}: {}", path.display(), e);
748                    }
749
750                    let s = start.min(output.len());
751                    let e = (s + size as usize).min(output.len());
752                    let result = output[s..e].to_vec();
753                    self.codesign_cache.lock().unwrap().insert(ino, output);
754                    return Ok(result);
755                }
756
757                Ok(crate::prefix_replacement::binary_ranged_read(
758                    &mmap, &prefixes, cstrings, start, end,
759                ))
760            }
761            ReplacementPlan::Text(text_plan) => Ok(crate::prefix_replacement::text_ranged_read(
762                &mmap, &prefixes, text_plan, start, end,
763            )),
764        }
765    }

767    pub(crate) fn do_readdir(&self, ino: u64, offset: u64) -> Result<Vec<DirEntry>, VfsError> {
768        let index = self.validate_ino(ino)?;
769
770        let Some(current_directory) = self.metadata[index].as_directory() else {
771            return Err(VfsError::NotADirectory);
772        };
773
774        let mut entries = Vec::new();
775
776        if offset == 0 {
777            entries.push(DirEntry {
778                ino: (current_directory.parent + 1) as u64,
779                kind: FileKind::Directory,
780                name: OsString::from(".."),
781            });
782        }
783        if offset <= 1 {
784            entries.push(DirEntry {
785                ino,
786                kind: FileKind::Directory,
787                name: OsString::from("."),
788            });
789        }
790
791        for child_index in current_directory
792            .children
793            .iter()
794            .skip(offset.saturating_sub(2) as usize)
795        {
796            let child = &self.metadata[*child_index];
797            let kind = match child {
798                MetadataNode::Directory(_) => FileKind::Directory,
799                MetadataNode::File(f) => {
800                    if f.path_type == PathType::SoftLink {
801                        FileKind::Symlink
802                    } else {
803                        FileKind::RegularFile
804                    }
805                }
806            };
807            entries.push(DirEntry {
808                ino: (child_index + 1) as u64,
809                kind,
810                name: child.file_name().to_owned(),
811            });
812        }
813
814        Ok(entries)
815    }

838    fn ino_to_path(&self, ino: u64) -> Result<PathBuf, VfsError> {
839        let index = self.validate_ino(ino)?;
840        let entry = &self.metadata[index];
841
842        // Root directory → empty path
843        if index == 0 {
844            return Ok(PathBuf::new());
845        }
846
847        match entry {
848            MetadataNode::Directory(dir) => {
849                // prefix_path is like "./lib/python3.14" — strip the "./" prefix
850                let p = dir
851                    .prefix_path
852                    .strip_prefix("./")
853                    .unwrap_or(&dir.prefix_path);
854                Ok(p.to_path_buf())
855            }
856            MetadataNode::File(file) => {
857                let parent = &self.metadata[file.parent];
858                let parent_dir = parent.as_directory().ok_or(VfsError::NotFound)?;
859                let parent_path = parent_dir
860                    .prefix_path
861                    .strip_prefix("./")
862                    .unwrap_or(&parent_dir.prefix_path);
863                Ok(parent_path.join(&file.file_name))
864            }
865        }
866    }
